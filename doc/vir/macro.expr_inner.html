macro_rules! expr_inner {

    (@expr_one; ( $($cond:tt)+ ) ? ( $($e_then:tt)+ ) : ( $($e_else:tt)+ )) => { ... };
    (@expr_one; ( $($lhs:tt)+ ) && ( $($rhs:tt)+ )) => { ... };
    (@expr_one; ( $($lhs:tt)+ ) || ( $($rhs:tt)+ )) => { ... };
    (@expr_one; ( $($lhs:tt)+ ) == > ( $($rhs:tt)+ )) => { ... };
    (@expr_one; ( $($lhs:tt)+ ) <= ( $($rhs:tt)+ )) => { ... };
    (@expr_one; ( $($lhs:tt)+ ) < ( $($rhs:tt)+ )) => { ... };
    (@expr_one; ( $($lhs:tt)+ ) + ( $($rhs:tt)+ )) => { ... };

    (@expr_one; domain( $($inner:tt)+ )) => { ... };
    (@expr_one; | $inner:tt |) => { ... };
    (@expr_one; old( $($inner:tt)+ )) => { ... };
    (@expr_one; null) => { ... };
    (@expr_one; true) => { ... };
    (@expr_one; false) => { ... };
    (@expr_one; result: $t:tt) => { ... };
    (@expr_one; forall $($tokens:tt)+) => { ... };

    (@expr_one; $($tokens:tt)*) => { ... };

    (@forall_qvars($qvars:ident); , [$outer_decl:expr] $($tokens:tt)*) => { ... };
    (@forall_qvars($qvars:ident); , $qvar:ident : $qtype:tt $($tokens:tt)* ) => { ... };
    (@forall_qvars($qvars:ident); $($tokens:tt)*) => { ... };
