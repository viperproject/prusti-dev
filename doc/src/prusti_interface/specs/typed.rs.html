55    pub(crate) fn defid_for_export(
56        &self,
57    ) -> (
58        // Specs
59        Vec<DefId>,
60        // Pure fns
61        Vec<DefId>,
62        // Predicates
63        Vec<DefId>,
64    ) {
65        let mut specs = Vec::new();
66        let mut pure_fns = Vec::new();
67        let mut predicates = Vec::new();
68        for (def_id, spec_graph) in &self.proc_specs {
69            let all_specs = std::iter::once(&spec_graph.base_spec)
70                .chain(spec_graph.specs_with_constraints.values());
71            for spec in all_specs {
72                if let Some(pres) = spec.pres.extract_with_selective_replacement() {
73                    specs.extend(pres);
74                }
75                if let Some(posts) = spec.posts.extract_with_selective_replacement() {
76                    specs.extend(posts);
77                }
78                if let Some(Some(term)) = spec.terminates.extract_with_selective_replacement() {
79                    specs.push(term.to_def_id());
80                }
81                if let Some(pledges) = spec.pledges.extract_with_selective_replacement() {
82                    specs.extend(pledges.iter().filter_map(|pledge| pledge.lhs));
83                    specs.extend(pledges.iter().map(|pledge| pledge.rhs));
84                }
85                // Trusted functions are opaque and need no body; `extern_spec`
86                // targets are trusted in the same way (their `#[trusted]` is
87                // mandatory). Every other pure function needs its MIR loaded
88                // for pure encoding and cross-crate export.
89                let is_trusted = spec.trusted.extract_inherit().expect("Expected trusted");
90                if spec.kind.is_pure().expect("Expected pure") && !is_trusted {
91                    pure_fns.push(*def_id)
92                }
93                if let Some(ProcedureSpecificationKind::Predicate(Some(def_id))) =
94                    spec.kind.extract_with_selective_replacement()
95                {
96                    predicates.push(*def_id);
97                }
98            }
99        }
100        for spec in self.type_specs.values() {
101            if let Some(invariants) = spec.invariant.extract_with_selective_replacement() {
102                specs.extend(invariants);
103            }
104        }
105        (specs, pure_fns, predicates)
106    }

108    pub(crate) fn import_external(
109        &mut self,
110        proc_specs: FxHashMap<DefId, SpecGraph<ProcedureSpecification>>,
111        type_specs: FxHashMap<DefId, TypeSpecification>,
112        env: &Environment,
113    ) {
114        let duplicate_error = |item_id, spec_id_a: DefId, spec_id_b: DefId| {
115            PrustiError::incorrect(
116                format!(
117                    "duplicate specification for `{}` from crate `{}` and `{}`",
118                    env.name.get_item_name(item_id),
119                    env.name.crate_name(spec_id_a.krate),
120                    env.name.crate_name(spec_id_b.krate),
121                ),
122                crate::specs::MultiSpan::from_spans(vec![
123                    env.query.get_def_span(spec_id_a),
124                    env.query.get_def_span(spec_id_b),
125                ]),
126            )
127            .emit(&env.diagnostic)
128        };
129        for (k, v) in proc_specs {
130            if let Some(other) = self.proc_specs.insert(k, v) {
131                let v = self.proc_specs.get(&k).unwrap();
132                duplicate_error(k, other.base_spec.source, v.base_spec.source);
133            }
134        }
135        for (k, v) in type_specs {
136            if let Some(other) = self.type_specs.insert(k, v) {
137                let v = self.type_specs.get(&k).unwrap();
138                duplicate_error(k, other.source, v.source);
139            }
140        }
141    }

143    pub fn all_values_debug(&self, hide_uuids: bool) -> Vec<String> {
144        let loop_specs: Vec<_> = self
145            .loop_specs
146            .values()
147            .map(|spec| format!("{spec:?}"))
148            .collect();
149        let proc_specs: Vec<_> = self
150            .proc_specs
151            .values()
152            .map(|spec| format!("{:?}", spec.base_spec))
153            .collect();
154        let type_specs: Vec<_> = self
155            .type_specs
156            .values()
157            .map(|spec| format!("{spec:?}"))
158            .collect();
159        let asserts: Vec<_> = self
160            .prusti_assertions
161            .values()
162            .map(|spec| format!("{spec:?}"))
163            .collect();
164        let assumptions: Vec<_> = self
165            .prusti_assumptions
166            .values()
167            .map(|spec| format!("{spec:?}"))
168            .collect();
169        let refutations: Vec<_> = self
170            .prusti_refutations
171            .values()
172            .map(|spec| format!("{spec:?}"))
173            .collect();
174        let mut values = Vec::new();
175        values.extend(loop_specs);
176        values.extend(proc_specs);
177        values.extend(type_specs);
178        values.extend(asserts);
179        values.extend(assumptions);
180        values.extend(refutations);
181        if hide_uuids {
182            let uuid =
183                Regex::new("[a-z0-9]{8}-[a-z0-9]{4}-[a-z0-9]{4}-[a-z0-9]{4}-[a-z0-9]{12}").unwrap();
184            let num_uuid = Regex::new("[a-z0-9]{32}").unwrap();
185            let mut replaced_values: Vec<String> = vec![];
186            for item in values {
187                let item = num_uuid.replace_all(&item, "$(NUM_UUID)");
188                let item = uuid.replace_all(&item, "$(UUID)");
189                replaced_values.push(String::from(item));
190            }
191            values = replaced_values;
192        }
193        // We sort in this strange way so that the output is
194        // determinstic enough to be used in tests.
195        values.sort_by_key(|v| (v.len(), v.to_string()));
196        values
197    }

327/// The base container to store a contract of a procedure.
328/// A contract can be divided into multiple specifications:

338}

349/// Provides various methods to construct a [SpecGraph] for a [ProcedureSpecification].
350/// In particular, these methods ensure that the resulting [SpecGraph]'s specs are *consistent*,
351/// which means:
352/// - The postconditions of any constrained spec always contain the postconditions of the base spec
353/// - The [ProcedureSpecificationKind] for the base spec and constrained specs is always the same
354/// - The `trusted` flag for the base spec and constrained specs is always the same
355/// - The [Pledge]s for the base spec and constrained specs are always the same
356///
357/// Note: Unlike postconditions, the preconditions are not copied amongst the base spec
358/// and constrained specs.
359///
360/// # A note about behavioral subtyping
361/// For a [SpecGraph] to be sound, we require that the constrained specs are valid to the base spec
362/// w.r.t. behavioral subtyping rules.
363/// We at least require that for some base spec B and any constrained spec C:
364/// - (a) The postconditions of C are at least as strong as the postconditions of B
365/// - (b) The preconditions of C are at least as weak as the preconditions of B
366/// - *...more..*
367///
368/// The consistency guarantees mentioned above satisfy (a) by construction but do not guarantee (b).
369///
370/// **Important**: There is no automatic check that guarantees the validity of a [SpecGraph].
371///
372/// # Example: Pre- and postconditions
373/// ```ignore
374/// trait MarkerTrait {}
375///
376/// trait SomeTrait {
377///     #[requires(x > 0)]
378///     #[ensures(x > 0)]
379///     #[refine_spec(where T: MarkerTrait, [
380///         requires(x > 10),
381///         ensures(x > 10),
382///     ])]
383///     fn foo<T>(&self, x: i32) -> i32;
384/// }
385///
386/// struct SomeStruct;
387/// #[refine_trait_spec]
388/// impl SomeTrait for SomeStruct {
389///     #[requires(x >= 0)]
390///     #[ensures(x > 10)]
391///     #[refine_spec(where T: MarkerTrait, [
392///         requires(x >= -5),
393///         ensures(x > 20),
394///     ])]
395///     fn foo<T>(&self, x: i32) -> i32 {
396///         42
397///     }
398/// }
399/// ```
400/// Let `B_T` be the base spec of `SomeTrait`, `B_S` be the base spec of `SomeStruct`,
401/// `C_T` be the constrained spec of `SomeTrait` and `C_S` be the constrained spec of `SomeStruct`.
402/// - The computed [SpecGraph] of `SomeTrait` will be:
403///     - `pres(B_T) ≡ x > 0`
404///     - `posts(B_T) ≡ x > 0`
405///     - `pres(C_T) ≡ x > 10`
406///     - `posts(C_T) ≡ posts(B_T) && x > 10 ≡ x > 0 && x > 10`
407/// - The computed [SpecGraph] of `SomeStruct` will be:
408///     - `pres(B_S) ≡ x >= 0`
409///     - `posts(B_S) ≡ x > 10`
410///     - `pres(C_S) ≡ x >= -5`
411///     - `posts(C_S) ≡ posts(B_S) && x > 20 ≡ x > 10 && x > 20`
412///
413/// When using `SomeStruct::foo`, we resolve to either `B_S` or `C_S`.
414/// ```ignore
415/// impl MarkerTrait for i32 {}
416/// fn main() {
417///     let s = SomeStruct;
418///     let r = s.foo::<i32>(100); // i32 implements MarkerTrait -> C_S is active
419///     let r = s.foo::<u32>(100); // u32 does not implement MarkerTrait -> B_S is active
420/// }
421/// ```
422impl SpecGraph<ProcedureSpecification> {
423    /// Attaches the precondition `pre` to this [SpecGraph].
424    ///
425    /// If this precondition has a constraint it will be attached to the corresponding
426    /// constrained spec, otherwise just to the base spec.
427    pub fn add_precondition<'tcx>(&mut self, pre: LocalDefId, env: &Environment<'tcx>) {
428        match self.get_constraint(pre, env) {
429            None => {
430                self.base_spec.pres.push(pre.to_def_id());
431                // Preconditions are explicitly not copied (as opposed to postconditions)
432                // This would always violate behavioral subtyping rules
433            }
434            Some(constraint) => {
435                self.get_constrained_spec_mut(constraint)
436                    .pres
437                    .push(pre.to_def_id());
438            }
439        }
440    }
441
442    /// Attaches the postcondition `post` to this [SpecGraph].
443    ///
444    /// If this postcondition has a constraint it will be attached to the corresponding
445    /// constrained spec **and** the base spec, otherwise just to the base spec.
446    pub fn add_postcondition<'tcx>(&mut self, post: LocalDefId, env: &Environment<'tcx>) {
447        match self.get_constraint(post, env) {
448            None => {
449                self.base_spec.posts.push(post.to_def_id());
450                self.specs_with_constraints
451                    .values_mut()
452                    .for_each(|s| s.posts.push(post.to_def_id()));
453            }
454            Some(obligation) => {
455                self.get_constrained_spec_mut(obligation)
456                    .posts
457                    .push(post.to_def_id());
458            }
459        }
460    }
461
462    pub fn add_purity<'tcx>(&mut self, purity: LocalDefId, env: &Environment<'tcx>) {
463        match self.get_constraint(purity, env) {
464            None => {
465                unreachable!(
466                    "separate purity defs are only used for type-conditional spec refinements"
467                )
468            }
469            Some(constraint) => {
470                let constrained_spec = self.get_constrained_spec_mut(constraint);
471                constrained_spec.kind =
472                    SpecificationItem::Inherent(ProcedureSpecificationKind::Pure);
473                // need to store this as well, since without pres or posts we couldn't find any def id with the trait bounds
474                constrained_spec.purity.set(Some(purity.to_def_id()));
475            }
476        }
477    }
478
479    /// Attaches the `pledge` to the base spec and all constrained specs.
480    pub fn add_pledge(&mut self, pledge: Pledge) {
481        self.base_spec.pledges.push(pledge);
482        self.specs_with_constraints
483            .values_mut()
484            .for_each(|s| s.pledges.push(pledge));
485    }
486
487    /// Sets the trusted flag for the base spec and all constrained specs.
488    pub fn set_trusted(&mut self, trusted: bool) {
489        self.base_spec.trusted.set(trusted);
490        self.specs_with_constraints
491            .values_mut()
492            .for_each(|s| s.trusted.set(trusted));
493    }
494
495    /// Sets the termination flag for the base spec and all constrained specs.
496    pub fn set_terminates(&mut self, terminates: LocalDefId) {
497        self.base_spec.terminates.set(Some(terminates));
498        self.specs_with_constraints
499            .values_mut()
500            .for_each(|s| s.terminates.set(Some(terminates)));
501    }
502
503    /// Sets the [ProcedureSpecificationKind] for the base spec and all constrained specs.
504    pub fn set_kind(&mut self, kind: ProcedureSpecificationKind) {
505        self.base_spec.kind.set(kind);
506        self.specs_with_constraints
507            .values_mut()
508            .for_each(|s| s.kind.set(kind));
509    }
510
511    /// Sets the [ExternSpecKind] for the base spec and all constrained specs.
512    pub fn set_extern_spec(&mut self, extern_spec: ExternSpecKind) {
513        self.base_spec.extern_spec = Some(extern_spec);
514        self.specs_with_constraints
515            .values_mut()
516            .for_each(|s| s.extern_spec = Some(extern_spec));
517    }
518
519    /// Lazily gets/creates a constrained spec.
520    /// If the constrained spec does not yet exist, the base spec serves as a template for
521    /// the newly created constrained spec.
522    fn get_constrained_spec_mut(
523        &mut self,
524        constraint: SpecConstraintKind,
525    ) -> &mut ProcedureSpecification {
526        self.specs_with_constraints
527            .entry(constraint)
528            .or_insert_with(|| self.base_spec.clone())
529    }
530
531    /// Gets the constraint of a spec function `spec`.
532    ///
533    /// Multiple constraints are currently not supported.
534    fn get_constraint<'tcx>(
535        &self,
536        spec: LocalDefId,
537        env: &Environment<'tcx>,
538    ) -> Option<SpecConstraintKind> {
539        let attrs = env.query.get_local_attributes(spec);
540        if has_trait_bounds_type_cond_spec(attrs) {
541            return Some(SpecConstraintKind::ResolveGenericParamTraitBounds);
542        }
543        None
544    }
545}

564/// A specification, such as preconditions or a `#[pure]` annotation.
565/// Contains information about the refinement of these specifications.
566#[derive(Debug, Clone, Copy, PartialEq, Eq, TyEncodable, TyDecodable)]
567pub enum SpecificationItem<T> {
568    /// Represents an empty specification, i.e. when the user has not defined the property
569    Empty,
570
571    /// Represents a specification typed by the user
572    Inherent(T),
573
574    /// Represents a specification which was inherited from somewhere (e.g. from a trait method)
575    Inherited(T),
576
577    /// Represents an item which was refined
578    /// This happens for example if the user annotates a trait method and its implementation method
579    /// with a precondition. This variant then holds the trait's precondition and the impl's precondition.
580    Refined(T, T),
581}
582
583impl<T> SpecificationItem<T> {
584    pub fn is_empty(&self) -> bool {
585        matches!(self, SpecificationItem::Empty)
586    }
587
588    /// Applies `f` to each contained value, keeping the variant.
589    pub fn map<U>(&self, mut f: impl FnMut(&T) -> U) -> SpecificationItem<U> {
590        match self {
591            SpecificationItem::Empty => SpecificationItem::Empty,
592            SpecificationItem::Inherent(val) => SpecificationItem::Inherent(f(val)),
593            SpecificationItem::Inherited(val) => SpecificationItem::Inherited(f(val)),
594            SpecificationItem::Refined(from, to) => SpecificationItem::Refined(f(from), f(to)),
595        }
596    }
597
598    /// Returns the contained value of this item
599    fn get(&self) -> Option<(Option<&T>, &T)> {
600        // TODO(tymap): this API is not good: it must be possible to tell that
601        //   the result was inherited, as we will e.g. need to rebase our substs
602        //   to the trait
603        match self {
604            SpecificationItem::Empty => None,
605            SpecificationItem::Inherited(val) => Some((None, val)),
606            SpecificationItem::Inherent(val) => Some((None, val)),
607            SpecificationItem::Refined(from, to) => Some((Some(from), to)),
608        }
609    }
610
611    /// Extracts the refined value out of this item by applying the provided strategy
612    pub fn extract_with_strategy<'a, R, S: FnOnce((Option<&'a T>, &'a T)) -> R>(
613        &'a self,
614        strategy: S,
615    ) -> Option<R> {
616        self.get().map(strategy)
617    }

619    /// Uses alternative C as discussed in
620    /// https://ethz.ch/content/dam/ethz/special-interest/infk/chair-program-method/pm/documents/Education/Theses/Matthias_Erdin_MA_report.pdf
621    /// pp 19-23
622    pub fn extract_with_selective_replacement(&self) -> Option<&T> {
623        self.extract_with_strategy(|(_, refined)| refined)
624    }
625
626    #[track_caller]
627    pub fn expect_empty_or_inherent(&self) -> Option<&T> {
628        match self {
629            SpecificationItem::Empty => None,
630            SpecificationItem::Inherent(item) => Some(item),
631            _ => unreachable!(),
632        }
633    }
634
635    #[track_caller]
636    pub fn expect_inherent(&self) -> &T {
637        match self {
638            SpecificationItem::Inherent(item) => item,
639            _ => unreachable!(),
640        }
641    }
642
643    #[track_caller]
644    pub fn expect_inherited(&self) -> &T {
645        match self {
646            SpecificationItem::Inherited(item) => item,
647            _ => unreachable!(),
648        }
649    }
650
651    #[track_caller]
652    pub fn expect_refined(&self) -> (&T, &T) {
653        match self {
654            SpecificationItem::Refined(a, b) => (a, b),
655            _ => unreachable!(),
656        }
657    }

660impl<T: Clone> SpecificationItem<T> {
661    pub fn set(&mut self, new_value: T) {
662        match self {
663            SpecificationItem::Empty => *self = SpecificationItem::Inherent(new_value),
664            SpecificationItem::Inherent(val) => *val = new_value,
665            SpecificationItem::Refined(_, values) => *values = new_value,
666            SpecificationItem::Inherited(inherited) => {
667                *self = SpecificationItem::Refined(inherited.clone(), new_value)
668            }
669        }
670    }
671}

673impl SpecificationItem<bool> {
674    pub fn extract_inherit(&self) -> Option<bool> {
675        self.extract_with_strategy(|(refined_from, refined)| {
676            *(refined_from.unwrap_or(&false)) || *refined
677        })
678    }

681#[derive(Debug)]
682pub enum ProcedureSpecificationKindError {
683    /// Occurs whenever the relation between [ProcedureSpecificationKind]s is violated.
684    /// The relation is: predicates ⊂ pure functions ⊂ impure functions
685    /// However, we can not refine a non-predicate to a predicate, since predicates are not callable.
686    /// This validation is not automatically performed during refinement. It needs to be invoked
687    /// manually with [SpecificationItem<ProcedureSpecificationKind>::validate]
688    InvalidSpecKindRefinement(ProcedureSpecificationKind, ProcedureSpecificationKind),
689}
690
691impl SpecificationItem<ProcedureSpecificationKind> {
692    pub fn is_pure(&self) -> Result<bool, ProcedureSpecificationKindError> {
693        self.validate()?;
694
695        Ok(matches!(
696            self.extract_with_selective_replacement(),
697            Some(ProcedureSpecificationKind::Pure) | Some(ProcedureSpecificationKind::Predicate(_))
698        ))
699    }
700
701    pub fn is_impure(&self) -> Result<bool, ProcedureSpecificationKindError> {
702        self.validate()?;
703
704        Ok(match self.extract_with_selective_replacement() {
705            Some(refined) => matches!(refined, ProcedureSpecificationKind::Impure),
706            _ => true,
707        })
708    }
709
710    pub fn get_predicate_body(&self) -> Result<Option<&DefId>, ProcedureSpecificationKindError> {
711        self.validate()?;
712
713        Ok(match self.extract_with_selective_replacement() {
714            Some(ProcedureSpecificationKind::Predicate(pred_id)) => pred_id.as_ref(),
715            _ => None,
716        })
717    }
718
719    /// Ensures that refined [ProcedureSpecificationKind]s are valid.
720    /// See [ProcedureSpecificationKindError] for detailed error descriptions.
721    pub fn validate(&self) -> Result<(), ProcedureSpecificationKindError> {
722        use ProcedureSpecificationKind::*;
723        if let SpecificationItem::Refined(base, refined) = self {
724            match (base, refined) {
725                (Impure, Impure) | (Impure, Pure) | (Pure, Pure) | (Predicate(_), Predicate(_)) => {
726                    Ok(())
727                }
728                _ => Err(ProcedureSpecificationKindError::InvalidSpecKindRefinement(
729                    *base, *refined,
730                )),
731            }
732        } else {
733            Ok(())
734        }
735    }
736}

738impl<T> SpecificationItem<Vec<T>> {
739    pub fn new(vec: Vec<T>) -> Self {
740        if vec.is_empty() {
741            SpecificationItem::Empty
742        } else {
743            SpecificationItem::Inherent(vec)
744        }
745    }
746
747    pub fn extract_with_selective_replacement_iter(&self) -> Box<dyn Iterator<Item = &T> + '_> {
748        if let Some(items) = self.extract_with_selective_replacement() {
749            return Box::new(items.iter());
750        }
751        Box::new(std::iter::empty())
752    }
753}

755impl<T: Clone> SpecificationItem<Vec<T>> {
756    pub fn push(&mut self, value: T) {
757        match self {
758            SpecificationItem::Empty => *self = SpecificationItem::Inherent(vec![value]),
759            SpecificationItem::Inherent(values) => values.push(value),
760            SpecificationItem::Refined(_, values) => values.push(value),
761            SpecificationItem::Inherited(inherited) => {
762                *self = SpecificationItem::Refined(inherited.clone(), vec![value])
763            }
764        }
765    }

768pub trait Refinable {
769    fn refine(self, other: &Self) -> Self
770    where
771        Self: Sized;
772}
773
774impl<T: Debug + Clone + PartialEq> Refinable for SpecificationItem<T> {
775    #[tracing::instrument(level = "trace", ret)]
776    fn refine(self, other: &Self) -> Self
777    where
778        Self: Sized,
779    {
780        use SpecificationItem::*;
781        let other_val = match other {
782            Empty => unreachable!("Trait spec should never be empty"),
783            Inherent(val) => val,
784            Inherited(val) => val,
785            Refined(_, _) => panic!("Can not refine with a refined item"),
786        };
787
788        match self {
789            Empty => Inherited(other_val.clone()),
790            Inherent(val) | Inherited(val) => Refined(other_val.clone(), val),
791            Refined(from, val) if &from == other_val => Refined(from, val),
792            Refined(_, _) => panic!("Can not refine this refined item"),
793        }
794    }
795}
796
797impl Refinable for ProcedureSpecification {
798    fn refine(self, other: &Self) -> Self {
799        // Unspecified trait specs should be treated as a default value instead of nothing
800        // See issue-1022
801        type SpecVec<T> = SpecificationItem<Vec<T>>;
802        static EMPTYL: SpecVec<DefId> = SpecificationItem::Inherent(vec![]);
803        static EMPTYP: SpecVec<Pledge> = SpecificationItem::Inherent(vec![]);
804        fn replace_empty<'a, T>(empty: &'a SpecVec<T>, spec: &'a SpecVec<T>) -> &'a SpecVec<T> {
805            match spec {
806                SpecificationItem::Empty => empty,
807                other => other,
808            }
809        }
810        ProcedureSpecification {
811            source: self.source,
812            // Record the extern spec backing this spec's derivation, preferring
813            // our own. It must be inherited from the trait: spec items inherited
814            // from an `extern_spec` trait are closures inside the generated stub
815            // and need the extern-spec kind to encode, and foreign impls of such
816            // traits are trusted based on it (see `spec_is_trusted`). It does
817            // NOT mean the function itself is an extern-spec target.
818            extern_spec: self.extern_spec.or(other.extern_spec),
819            pres: self.pres.refine(replace_empty(&EMPTYL, &other.pres)),
820            posts: self.posts.refine(replace_empty(&EMPTYL, &other.posts)),
821            pledges: self.pledges.refine(replace_empty(&EMPTYP, &other.pledges)),
822            kind: self.kind.refine(&other.kind),
823            // `trusted` is deliberately not inherited: it only applies to the
824            // item it is written on. An impl is verified against the inherited
825            // contract unless the impl itself is marked `#[trusted]`, wherever
826            // that impl is verified. The one exception is an impl that cannot
827            // be annotated at all -- a foreign impl of an `extern_spec` trait
828            // -- which `spec_is_trusted` keeps opaque via `extern_spec`.
829            trusted: self.trusted,
830            terminates: self.terminates.refine(&other.terminates),
831            purity: self.purity.refine(&other.purity),
832        }
833    }
834}

836/// Identical to `prusti_specs::ExternSpecKind` or `ExternSpecDeclaration` but
837/// implements required traits.
838#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash, TyEncodable, TyDecodable)]
839pub enum ExternSpecKind {
840    InherentImpl,
841    TraitImpl,
842    Trait,
843    Method,
844}
845
846impl From<&ExternSpecDeclaration> for ExternSpecKind {
847    fn from(kind: &ExternSpecDeclaration) -> Self {
848        match kind {
849            ExternSpecDeclaration::Inherent(..) => ExternSpecKind::InherentImpl,
850            ExternSpecDeclaration::TraitImpl(..) => ExternSpecKind::TraitImpl,
851            ExternSpecDeclaration::Trait(..) => ExternSpecKind::Trait,
852            ExternSpecDeclaration::Method(..) => ExternSpecKind::Method,
853        }
854    }
855}
