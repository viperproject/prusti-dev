13/// Defines the context for which we perform refinement.
14/// It can be thought of as the variants of [SpecQuery] for which we can perform refinement.
15#[derive(Debug)]
16struct RefinementContext<'qry, 'tcx> {
17    impl_query: &'qry SpecQuery<'tcx>,
18    trait_query: SpecQuery<'tcx>,
19}
20
21impl<'qry, 'tcx> RefinementContext<'qry, 'tcx> {
22    /// Tries to create a refinement context.
23    /// Returns None if refinement is not needed
24    fn try_from(tcx: ty::TyCtxt<'tcx>, query: &'qry SpecQuery<'tcx>) -> Option<Self> {
25        match query {
26            SpecQuery::FunctionCallEncoding(FunctionCallEncodingQuery {
27                called_def_id: def_id,
28                call_substs: substs,
29                ..
30            })
31            | SpecQuery::FunctionDefEncoding(def_id, substs)
32            | SpecQuery::GetProcKind(def_id, substs) => {
33                let (trait_def_id, trait_substs) = find_trait_method_substs(tcx, *def_id, substs)?;
34                let trait_query = query.adapt_to(trait_def_id, trait_substs);
35                Some(RefinementContext {
36                    impl_query: query,
37                    trait_query,
38                })
39            }
40            // All other queries do not need refinement
41            _ => None,
42        }
43    }
44}

46#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
47pub struct FunctionCallEncodingQuery<'tcx> {
48    pub called_def_id: DefId,
49    pub caller_def_id: DefId,
50    pub call_substs: &'tcx [GenericArg<'tcx>],
51}
52
53#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
54pub enum SpecQuery<'tcx> {
55    FunctionDefEncoding(DefId, &'tcx [GenericArg<'tcx>]),
56    FunctionCallEncoding(FunctionCallEncodingQuery<'tcx>),
57    /// For determining the [ProcedureSpecificationKind] of a procedure, e.g.
58    /// for a check whether the function is pure or impure
59    GetProcKind(DefId, &'tcx [GenericArg<'tcx>]),
60    FetchSpan(DefId),
61}
62
63impl<'tcx> SpecQuery<'tcx> {
64    pub fn referred_def_id(&self) -> DefId {
65        match self {
66            SpecQuery::FunctionDefEncoding(def_id, _)
67            | SpecQuery::FunctionCallEncoding(FunctionCallEncodingQuery {
68                called_def_id: def_id,
69                ..
70            })
71            | SpecQuery::GetProcKind(def_id, _)
72            | SpecQuery::FetchSpan(def_id) => *def_id,
73        }
74    }
75
76    pub fn adapt_to(&self, new_def_id: DefId, new_substs: &'tcx [GenericArg<'tcx>]) -> Self {
77        use SpecQuery::*;
78        match self {
79            FunctionDefEncoding(_, _) => FunctionDefEncoding(new_def_id, new_substs),
80            FunctionCallEncoding(FunctionCallEncodingQuery { caller_def_id, .. }) => {
81                FunctionCallEncoding(FunctionCallEncodingQuery {
82                    called_def_id: new_def_id,
83                    caller_def_id: *caller_def_id,
84                    call_substs: new_substs,
85                })
86            }
87            GetProcKind(_, _) => GetProcKind(new_def_id, new_substs),
88            FetchSpan(_) => FetchSpan(new_def_id),
89        }
90    }
91}

97    /// The query can resolve to different [ProcedureSpecification]s due to type-conditional spec refinements.

140    pub fn get_and_refine_proc_spec<'a, 'env: 'a>(
141        &'a mut self,
142        tcx: ty::TyCtxt<'tcx>,
143        query: SpecQuery<'tcx>,
144    ) -> Option<&'a ProcedureSpecification> {
145        if self.is_refined(&query) {
146            return self.get_proc_spec(&query);
147        }
148
149        match RefinementContext::try_from(tcx, &query) {
150            Some(context) => {
151                let refined =
152                    self.perform_proc_spec_refinement(context.impl_query, &context.trait_query);
153                assert!(
154                    refined.is_some(),
155                    "Could not perform refinement for {query:?}"
156                );
157                refined
158            }
159            _ => self.get_proc_spec(&query),
160        }
161    }
162
163    #[tracing::instrument(level = "debug", skip(self))]
164    fn perform_proc_spec_refinement<'a>(
165        &'a mut self,
166        impl_query: &SpecQuery<'tcx>,
167        trait_query: &SpecQuery<'tcx>,
168    ) -> Option<&'a ProcedureSpecification> {
169        let trait_spec = self
170            .get_proc_spec(trait_query)
171            .cloned()
172            .unwrap_or_else(|| ProcedureSpecification::empty(trait_query.referred_def_id()));
173
174        // A trait impl that does not carry a `#[refine_trait_spec]` has no
175        // annotations (enforced during collection) and hence no spec entry:
176        // it inherits the trait spec wholesale. An impl that does refine has a
177        // spec entry, which refines the trait spec below.
178        let impl_spec = self.get_proc_spec(impl_query).cloned().unwrap_or_else(|| {
179            ProcedureSpecification::empty_inheriting(impl_query.referred_def_id())
180        });
181        let refined = impl_spec.refine(&trait_spec);
182
183        self.refined_specs.insert(*impl_query, refined);
184        self.get_proc_spec(impl_query)
185    }
186
187    fn get_proc_spec<'a>(&'a self, query: &SpecQuery<'tcx>) -> Option<&'a ProcedureSpecification> {
188        self.refined_specs.get(query).or_else(|| {
189            self.user_typed_specs
190                .get_proc_spec(&query.referred_def_id())
191                .map(|spec| &spec.base_spec)
192        })
193    }
194
195    fn is_refined(&self, query: &SpecQuery<'tcx>) -> bool {
196        self.refined_specs.contains_key(query)
197    }
198}

200/// If the given `impl_method_def_id` is an implementation of a trait
201/// method, return the `DefId` of that trait method as well as an adapted
202/// version of the callsite `impl_method_substs` substitutions.
203pub fn find_trait_method_substs<'tcx>(
204    tcx: ty::TyCtxt<'tcx>,
205    impl_method_def_id: ProcedureDefId, // what are we calling?
206    impl_method_substs: &'tcx [GenericArg<'tcx>], // what are the substs on the call?
207) -> Option<(ProcedureDefId, &'tcx [GenericArg<'tcx>])> {
208    let impl_def_id = tcx.impl_of_assoc(impl_method_def_id)?;
209    let trait_ref = tcx.impl_opt_trait_ref(impl_def_id)?.skip_binder();
210
211    // At this point, we know that the given method:
212    // - belongs to an impl block and
213    // - the impl block implements a trait.
214    // For the `get_assoc_item` call, we therefore `unwrap`, as not finding
215    // the associated item would be a (compiler) internal error.
216    let trait_def_id = trait_ref.def_id;
217    let trait_method_def_id = get_assoc_item(tcx, trait_def_id, impl_method_def_id)
218        .unwrap()
219        .def_id;
220
221    // sanity check: have we been given the correct number of substs?
222    let identity_impl_method = ty::List::identity_for_item(tcx, impl_method_def_id);
223    assert_eq!(identity_impl_method.len(), impl_method_substs.len());
224
225    // Given:
226    // ```
227    // trait Trait<Tp> {
228    //     fn f<Tx, Ty, Tz>();
229    // }
230    // struct Struct<Ex, Ey> { ... }
231    // impl<A, B, C> Trait<A> for Struct<B, C> {
232    //     fn f<X, Y, Z>() { ... }
233    // }
234    // ```
235    //
236    // The various substs look like this:
237    // - identity for Trait:           `[Self, Tp]`
238    // - identity for Trait::f:        `[Self, Tp, Tx, Ty, Tz]`
239    // - substs of the impl trait ref: `[Struct<B, C>, A]`
240    // - identity for the impl:        `[A, B, C]`
241    // - identity for Struct::f:       `[A, B, C, X, Y, Z]`
242    //
243    // What we need is a substs suitable for a call to Trait::f, which is in
244    // this case `[Struct<B, C>, A, X, Y, Z]`. More generally, it is the
245    // concatenation of the trait ref substs with the identity of the impl
246    // method after skipping the identity of the impl.
247    //
248    // We also need to subst the prefix (`[Struct<B, C>, A]` in the example
249    // above) with call substs, so that we get the trait's type parameters
250    // more precisely.
251    let call_trait_substs =
252        ty::EarlyBinder::bind(tcx, trait_ref.args).instantiate(tcx, impl_method_substs);
253    let impl_substs = ty::List::identity_for_item(tcx, impl_def_id);
254    let trait_method_substs = tcx.mk_args_from_iter(
255        call_trait_substs
256            .skip_normalization()
257            .iter()
258            .chain(impl_method_substs.iter().copied().skip(impl_substs.len())),
259    );
260
261    // sanity check: do we now have the correct number of substs?
262    let identity_trait_method = ty::List::identity_for_item(tcx, trait_method_def_id);
263    assert_eq!(trait_method_substs.len(), identity_trait_method.len());
264
265    Some((trait_method_def_id, trait_method_substs))
266}

268pub fn get_assoc_item<'tcx>(
269    tcx: ty::TyCtxt<'tcx>,
270    trait_id: DefId,
271    item_id: DefId,
272) -> Option<ty::AssocItem> {
273    // FIXME: Probably we should use https://doc.rust-lang.org/nightly/nightly-rustc/rustc_middle/ty/struct.AssociatedItems.html#method.find_by_name_and_namespace
274    // instead.
275    tcx.associated_items(trait_id)
276        .filter_by_name_unhygienic(tcx.item_name(item_id))
277        .next()
278        .cloned()
