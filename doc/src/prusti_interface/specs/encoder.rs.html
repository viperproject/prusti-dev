43impl<'a, 'tcx> DefSpecsEncoder<'a, 'tcx> {
44    pub fn serialize<T: for<'b> Encodable<DefSpecsEncoder<'b, 'tcx>>>(
45        tcx: TyCtxt<'tcx>,
46        path: &Path,
47        meta: T,
48    ) -> Result<(), io::Error> {
49        let _ = std::fs::File::create(path)?;
50        std::fs::create_dir_all(path.parent().unwrap())?;
51
52        let hygiene_context = HygieneEncodeContext::default();
53
54        let mut opaque = opaque::FileEncoder::new(path)?;
55        opaque.emit_raw_bytes(SPECS_MAGIC);
56        opaque.emit_raw_bytes(&SPECS_FORMAT_VERSION.to_le_bytes());
57        // Will be filled with the position of the allocation index after
58        // encoding everything (same as the crate root position in rustc's
59        // metadata encoder).
60        opaque.emit_raw_bytes(&0u64.to_le_bytes());
61
62        let mut encoder = DefSpecsEncoder {
63            tcx,
64            opaque,
65            type_shorthands: Default::default(),
66            predicate_shorthands: Default::default(),
67            interpret_allocs: Default::default(),
68            hygiene_context: &hygiene_context,
69            symbol_table: Default::default(),
70            byte_symbol_table: Default::default(),
71        };
72
73        meta.encode(&mut encoder);
74        let alloc_index_pos = encoder.encode_interpret_alloc_index();
75        encoder.finish(alloc_index_pos).map_err(|e| e.1)
76    }
77
78    /// Encodes the allocations referenced by the already encoded data,
79    /// followed by an index with the position of each one (indexed like
80    /// `interpret_allocs`). Returns the position of that index.
81    /// Encoding an allocation can reference further allocations, so we loop
82    /// until no new ones show up (same as rustc's metadata encoder).
83    fn encode_interpret_alloc_index(&mut self) -> usize {
84        let tcx = self.tcx;
85        let mut alloc_index = Vec::new();
86        let mut n = 0;
87        while n < self.interpret_allocs.len() {
88            let new_n = self.interpret_allocs.len();
89            for idx in n..new_n {
90                let id = self.interpret_allocs[idx];
91                alloc_index.push(self.position() as u64);
92                interpret::specialized_encode_alloc_id(self, tcx, id);
93            }
94            n = new_n;
95        }
96        let alloc_index_pos = self.position();
97        alloc_index.encode(self);
98        alloc_index_pos
99    }
100
101    pub fn finish(mut self, alloc_index_pos: usize) -> Result<(), (PathBuf, Error)> {
102        self.opaque.finish()?;
103        encode_alloc_index_position(self.opaque.file(), alloc_index_pos)
104            .map_err(|err| (self.opaque.path().to_path_buf(), err))
105    }
106}

108// Same as `encode_root_position` in rustc's metadata encoder
109fn encode_alloc_index_position(mut file: &File, pos: usize) -> Result<(), Error> {
110    // We will return to this position after writing the index position.
111    let pos_before_seek = file.stream_position()?;
112
113    let header = SPECS_MAGIC.len() + size_of_val(&SPECS_FORMAT_VERSION);
114    file.seek(io::SeekFrom::Start(header as u64))?;
115    file.write_all(&(pos as u64).to_le_bytes())?;
116
117    // Return to the position where we were before writing the index position.
118    file.seek(io::SeekFrom::Start(pos_before_seek))?;
119    Ok(())
120}

122// Taken from rustc:
123// https://doc.rust-lang.org/nightly/nightly-rustc/rustc_metadata/rmeta/encoder/macro.encoder_methods.html
124macro_rules! encoder_methods {
125    ($($name:ident($ty:ty);)*) => {
126        $(fn $name(&mut self, value: $ty) -> () {
127            self.opaque.$name(value)
128        })*
129    }

154impl<'a, 'tcx> SpanEncoder for DefSpecsEncoder<'a, 'tcx> {
155    fn encode_span(&mut self, span: Span) {
156        let sm = self.tcx.sess.source_map();
157        let local_crate_stable_id = self.tcx.stable_crate_id(LOCAL_CRATE);
158        for bp in [span.lo(), span.hi()] {
159            let sf = sm.lookup_source_file(bp);
160
161            let ssfi =
162                StableSourceFileId::from_filename_for_export(&sf.name, local_crate_stable_id);
163            ssfi.encode(self);
164            // Not sure if this is the most stable way to encode a BytePos. If it fails
165            // try finding a function in `SourceMap` or `SourceFile` instead. E.g. the
166            // `bytepos_to_file_charpos` fn which returns `CharPos` (though there is
167            // currently no fn mapping back to `BytePos` for decode)
168            (bp - sf.start_pos).encode(self);
169        }
170    }
171    fn encode_symbol(&mut self, sym: Symbol) {
172        // if symbol preinterned, emit tag and symbol index
173        if Symbol::is_predefined(sym.as_u32()) {
174            self.opaque.emit_u8(SYMBOL_PREINTERNED);
175            self.opaque.emit_u32(sym.as_u32());
176        } else {
177            // otherwise write it as string or as offset to it
178            match self.symbol_table.entry(sym) {
179                Entry::Vacant(o) => {
180                    self.opaque.emit_u8(SYMBOL_STR);
181                    let pos = self.opaque.position();
182                    o.insert(pos);
183                    self.emit_str(sym.as_str());
184                }
185                Entry::Occupied(o) => {
186                    let x = *o.get();
187                    self.emit_u8(SYMBOL_OFFSET);
188                    self.emit_usize(x);
189                }
190            }
191        }
192    }
193
194    fn encode_expn_id(&mut self, eid: ExpnId) {
195        self.hygiene_context.schedule_expn_data_for_encoding(eid);
196        eid.krate.encode(self);
197        eid.local_id.as_u32().encode(self);
198    }
199    fn encode_syntax_context(&mut self, ctx: SyntaxContext) {
200        raw_encode_syntax_context(ctx, self.hygiene_context, self);
201    }
202    fn encode_crate_num(&mut self, cnum: CrateNum) {
203        self.tcx.stable_crate_id(cnum).encode(self)
204    }
205    fn encode_def_index(&mut self, _: DefIndex) {
206        panic!("encoding `DefIndex` without context");
207    }
208    fn encode_def_id(&mut self, id: DefId) {
209        self.tcx.def_path_hash(id).encode(self)
210    }
211
212    fn encode_byte_symbol(&mut self, byte_sym: ByteSymbol) {
213        // if symbol preinterned, emit tag and symbol index
214        if Symbol::is_predefined(byte_sym.as_u32()) {
215            self.opaque.emit_u8(BYTE_SYMBOL_PREINTERNED);
216            self.opaque.emit_u32(byte_sym.as_u32());
217        } else {
218            // otherwise write it as string or as offset to it
219            match self.byte_symbol_table.entry(byte_sym) {
220                Entry::Vacant(o) => {
221                    self.opaque.emit_u8(BYTE_SYMBOL_STR);
222                    let pos = self.opaque.position();
223                    o.insert(pos);
224                    self.emit_byte_str(byte_sym.as_byte_str());
225                }
226                Entry::Occupied(o) => {
227                    let x = *o.get();
228                    self.emit_u8(BYTE_SYMBOL_OFFSET);
229                    self.emit_usize(x);
230                }
231            }
232        }
233    }
234}
