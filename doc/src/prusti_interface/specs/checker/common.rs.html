17/// An `instravisit::Visitor` like trait which visits expressions in non-spec code.
18///
19/// Call the [wrap_as_visitor()] method to convert this type to an `instravisit::Visitor`
20pub trait NonSpecExprVisitor<'tcx> {
21    fn tcx(&self) -> TyCtxt<'tcx>;
22
23    /// Delegate method for `intravisit::Visitor::visit_expr`.
24    /// Note: This is just a delegate method, no call to `intravisit::walk_expr` needed.
25    fn visit_expr(&mut self, ex: &'tcx hir::Expr<'tcx>);
26
27    /// Wraps this delegate into a type which implements `intravisit::Visitor`, calling the
28    /// [visit_expr] delegate method.
29    fn wrap_as_visitor(self) -> NonSpecExprVisitorWrapper<'tcx, Self>
30    where
31        Self: Sized,
32    {
33        NonSpecExprVisitorWrapper {
34            wrapped: self,
35            _marker: std::marker::PhantomData,
36        }
37    }
38}
39
40/// A newtype wrapper for a [NonSpecExprVisitor] implementing type which implements the `intravisit::Visitor` trait
41pub struct NonSpecExprVisitorWrapper<'tcx, T: NonSpecExprVisitor<'tcx>> {
42    pub wrapped: T,
43    _marker: std::marker::PhantomData<&'tcx T>,
44}
45
46/// An implementation for `intravisit::Visitor` for [NonSpecExprVisitor] implementing types
47impl<'tcx, T: NonSpecExprVisitor<'tcx>> Visitor<'tcx> for NonSpecExprVisitorWrapper<'tcx, T> {
48    type NestedFilter = prusti_rustc_interface::middle::hir::nested_filter::All;
49
50    fn maybe_tcx(&mut self) -> Self::MaybeTyCtxt {
51        self.wrapped.tcx()
52    }
53
54    fn visit_item(&mut self, i: &'tcx hir::Item<'tcx>) {
55        // restrict to "interesting" sub-nodes to visit, i.e. anything that
56        // could be or contain call (or other usage of predicate) expressions
57        use hir::ItemKind::*;
58
59        match i.kind {
60            Static(..) | Fn { .. } | Mod(..) | Impl { .. } => {
61                intravisit::walk_item(self, i);
62            }
63            _ => {
64                // don't recurse into e.g. struct decls, type aliases, consts etc.
65            }
66        }
67    }
68
69    fn visit_expr(&mut self, ex: &'tcx hir::Expr<'tcx>) {
70        self.wrapped.visit_expr(ex);
71        intravisit::walk_expr(self, ex);
72    }
73
74    fn visit_fn(
75        &mut self,
76        fk: intravisit::FnKind<'tcx>,
77        fd: &'tcx hir::FnDecl<'tcx>,
78        b: hir::BodyId,
79        _s: Span,
80        local_id: LocalDefId,
81    ) {
82        // Stop checking inside `prusti::spec_only` functions
83        let tcx = self.wrapped.tcx();
84        let attrs = tcx.hir_attrs(tcx.local_def_id_to_hir_id(local_id));
85        if has_spec_only_attr(attrs) {
86            return;
87        }
88
89        intravisit::walk_fn(self, fk, fd, b, local_id);
90    }
91}
