9/// Checks for mismatched version issues between `prusti` and `prusti-contracts`/`prusti-specs`
10pub struct MismatchedVersionsChecker;
11
12impl<'tcx> SpecCheckerStrategy<'tcx> for MismatchedVersionsChecker {
13    #[tracing::instrument(
14        name = "MismatchedVersionsChecker::check",
15        level = "debug",
16        skip(self, env)
17    )]
18    fn check(&self, env: &Environment<'tcx>) -> Vec<PrustiError> {
19        let mut check_version = CheckVersionVisitor {
20            env_query: env.query,
21            errors: Vec::new(),
22        };
23        env.query.tcx().hir_walk_attributes(&mut check_version);
24
25        if let Some(min_prusti_version) = config::min_prusti_version() {
26            if env.compare_prusti_version(&min_prusti_version).is_lt() {
27                check_version.errors.push(PrustiError::incorrect(
28                    format!(
29                        "Running Prusti version '{}' when the minimum version required by the \
30                        `min_prusti_version` flag of the crate being compiled is '{min_prusti_version}'! \
31                        Please update Prusti to version '{min_prusti_version}' or newer.",
32                        env.get_prusti_version(),
33                    ),
34                    MultiSpan::new(),
35                ));
36            }
37        }
38        check_version.errors
39    }
40}
41
42/// Finds all `#[prusti::specs_version = "..."]` attributes and compares them
43/// to the version it was compiled with, throwing an error if it was compiled
44/// with a higher version.
45struct CheckVersionVisitor<'tcx> {
46    env_query: EnvQuery<'tcx>,
47    errors: Vec<PrustiError>,
48}
49
50impl<'tcx> hir::intravisit::Visitor<'tcx> for CheckVersionVisitor<'tcx> {
51    type NestedFilter = prusti_rustc_interface::middle::hir::nested_filter::All;
52
53    fn maybe_tcx(&mut self) -> Self::MaybeTyCtxt {
54        self.env_query.tcx()
55    }
56
57    fn visit_attribute(&mut self, attr: &'tcx hir::Attribute) {
58        // Only report one such error here:
59        if !self.errors.is_empty() {
60            return;
61        }
62        if let Some(used_specs_version) = utils::read_specs_version_attr(attr) {
63            // If 'compiled with version' > 'version used now for proc-macros' then complain (of `prusti-specs`)
64            if Environment::compare_to_curr_specs_version(&used_specs_version).is_gt() {
65                let downgrade_msg = config::min_prusti_version()
66                    .map(|prusti_ver| format!(" Alternatively you could downgrade your Prusti version to '{prusti_ver}'."))
67                    .unwrap_or_default();
68                self.errors.push(PrustiError::incorrect(
69                    format!(
70                        "The version of `prusti-specs` used is '{used_specs_version}', however your Prusti executable was \
71                        built with '{0}'! Please use `prusti-specs` version '{0}' or newer.{downgrade_msg}",
72                        Environment::get_specs_version(),
73                    ),
74                    MultiSpan::from_span(attr.span()),
75                ));
76            }
77        }
78    }
79}
