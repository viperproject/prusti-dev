42/// Facade to the Rust compiler.
43pub struct Environment<'tcx> {
44    prusti_version: &'static str,
45    pub body: EnvBody<'tcx>,
46    pub diagnostic: EnvDiagnostic<'tcx>,
47    pub name: EnvName<'tcx>,
48    pub query: EnvQuery<'tcx>,
49}
50
51impl<'tcx> Environment<'tcx> {
52    /// Builds an environment given a compiler state.
53    pub fn new(tcx: TyCtxt<'tcx>, prusti_version: &'static str) -> Self {
54        Environment {
55            prusti_version,
56            body: EnvBody::new(tcx),
57            diagnostic: EnvDiagnostic::new(tcx),
58            name: EnvName::new(tcx),
59            query: EnvQuery::new(tcx),
60        }
61    }
62
63    /// Returns the typing context
64    pub fn tcx(&self) -> TyCtxt<'tcx> {
65        self.query.tcx()
66    }
67
68    /// Get ids of Rust procedures that are annotated with a Prusti specification
69    #[tracing::instrument(level = "debug", skip(self))]
70    pub fn get_annotated_procedures_and_types(&self) -> (Vec<ProcedureDefId>, Vec<ty::Ty<'tcx>>) {
71        let mut visitor = CollectPrustiSpecVisitor::new(self);
72        visitor.visit_all_item_likes();
73
74        let mut cl_visitor = CollectClosureDefsVisitor::new(self);
75        self.query
76            .tcx()
77            .hir_visit_all_item_likes_in_crate(&mut cl_visitor);
78
79        let (mut procedures, types) = visitor.into_result();
80        procedures.extend(cl_visitor.get_closure_defs());
81        (procedures, types)
82    }
83
84    /// Compare the current version of the `prusti` crate to the given other version
85    pub fn compare_prusti_version(&self, other: &str) -> std::cmp::Ordering {
86        version_compare::compare(self.prusti_version, other)
87            .unwrap_or_else(|_| {
88                panic!(
89                    "Failed to compare Prusti version '{}' with other version '{other}'!",
90                    self.prusti_version
91                )
92            })
93            .ord()
94            .unwrap()
95    }
96
97    /// Get the current version of the `prusti` crate
98    pub fn get_prusti_version(&self) -> &'static str {
99        self.prusti_version
100    }
101
102    /// Compare to the current version of the `prusti-specs` crate which we've been compiled with
103    pub fn compare_to_curr_specs_version(other: &str) -> std::cmp::Ordering {
104        let compiled_specs_version = prusti_specs::SPECS_VERSION;
105        version_compare::compare(compiled_specs_version, other)
106            .unwrap_or_else(|_| {
107                panic!(
108                    "Failed to compare `prusti-specs` version '{}' with found version '{other}'!",
109                    prusti_specs::SPECS_VERSION
110                )
111            })
112            .ord()
113            .unwrap()
114    }
115
116    /// Get the current version of the `prusti` crate
117    pub fn get_specs_version() -> &'static str {
118        prusti_specs::SPECS_VERSION
119    }
120}
