154    /// Computes the signature of the function with subst applied.
155    pub fn get_fn_sig(
156        self,
157        def_id: impl IntoParam<ProcedureDefId>,
158        substs: GenericArgsRef<'tcx>,
159    ) -> ty::PolyFnSig<'tcx> {
160        let def_id = def_id.into_param();
161        let sig = if self.tcx.is_closure_like(def_id) {
162            ty::EarlyBinder::bind(self.tcx, substs.as_closure().sig())
163        } else {
164            self.tcx.fn_sig(def_id)
165        };
166        sig.instantiate(self.tcx, substs).skip_normalization()
167    }

186    /// If the given `impl_method_def_id` is an implementation of a trait
187    /// method, return the `DefId` of that trait method as well as an adapted
188    /// version of the callsite `impl_method_substs` substitutions.
189    pub fn find_trait_method_substs(
190        self,
191        impl_method_def_id: impl IntoParam<ProcedureDefId>, // what are we calling?
192        impl_method_substs: GenericArgsRef<'tcx>,           // what are the substs on the call?
193    ) -> Option<(ProcedureDefId, GenericArgsRef<'tcx>)> {
194        let impl_method_def_id = impl_method_def_id.into_param();
195        let impl_def_id = self.tcx.impl_of_assoc(impl_method_def_id)?;
196        let trait_ref = self.tcx.impl_opt_trait_ref(impl_def_id)?.skip_binder();
197
198        // At this point, we know that the given method:
199        // - belongs to an impl block and
200        // - the impl block implements a trait.
201        // For the `get_assoc_item` call, we therefore `unwrap`, as not finding
202        // the associated item would be a (compiler) internal error.
203        let trait_def_id = trait_ref.def_id;
204        let trait_method_def_id = self
205            .get_assoc_item(trait_def_id, impl_method_def_id)
206            .unwrap()
207            .def_id;
208
209        // sanity check: have we been given the correct number of substs?
210        let identity_impl_method = self.identity_substs(impl_method_def_id);
211        assert_eq!(identity_impl_method.len(), impl_method_substs.len());
212
213        // Given:
214        // ```
215        // trait Trait<Tp> {
216        //     fn f<Tx, Ty, Tz>();
217        // }
218        // struct Struct<Ex, Ey> { ... }
219        // impl<A, B, C> Trait<A> for Struct<B, C> {
220        //     fn f<X, Y, Z>() { ... }
221        // }
222        // ```
223        //
224        // The various substs look like this:
225        // - identity for Trait:           `[Self, Tp]`
226        // - identity for Trait::f:        `[Self, Tp, Tx, Ty, Tz]`
227        // - substs of the impl trait ref: `[Struct<B, C>, A]`
228        // - identity for the impl:        `[A, B, C]`
229        // - identity for Struct::f:       `[A, B, C, X, Y, Z]`
230        //
231        // What we need is a substs suitable for a call to Trait::f, which is in
232        // this case `[Struct<B, C>, A, X, Y, Z]`. More generally, it is the
233        // concatenation of the trait ref substs with the identity of the impl
234        // method after skipping the identity of the impl.
235        //
236        // We also need to subst the prefix (`[Struct<B, C>, A]` in the example
237        // above) with call substs, so that we get the trait's type parameters
238        // more precisely. We can do this directly with `impl_method_substs`
239        // because they contain the substs for the `impl` block as a prefix.
240        let call_trait_substs = ty::EarlyBinder::bind(self.tcx, trait_ref.args)
241            .instantiate(self.tcx, impl_method_substs);
242        let impl_substs = self.identity_substs(impl_def_id);
243        let trait_method_substs = self.tcx.mk_args_from_iter(
244            call_trait_substs
245                .skip_normalization()
246                .iter()
247                .chain(impl_method_substs.iter().skip(impl_substs.len())),
248        );
249
250        // sanity check: do we now have the correct number of substs?
251        let identity_trait_method = self.identity_substs(trait_method_def_id);
252        assert_eq!(trait_method_substs.len(), identity_trait_method.len());
253
254        Some((trait_method_def_id, trait_method_substs))
255    }

257    /// Given some trait item `proc_def_id` which is called with `substs`
258    /// (under the caller's `typing_env`), returns the `DefId` of the trait
259    /// `impl` block that the call selects. Returns `None` if `proc_def_id` is
260    /// no trait item or selection does not yield a user-defined impl (e.g.
261    /// the `Self` type is generic).
262    #[tracing::instrument(level = "debug", skip(self, typing_env))]
263    pub fn find_trait_impl_of_method_call(
264        self,
265        typing_env: ty::TypingEnv<'tcx>,
266        proc_def_id: impl IntoParam<ProcedureDefId> + Debug,
267        substs: GenericArgsRef<'tcx>,
268    ) -> Option<DefId> {
269        let proc_def_id = proc_def_id.into_param();
270        let trait_id = self.get_trait_of_assoc(proc_def_id)?;
271        debug!(
272            "Fetching implementations of method '{:?}' defined in trait '{}' with substs '{:?}'",
273            proc_def_id,
274            self.tcx.def_path_str(trait_id),
275            substs
276        );
277        // Selection runs in a fresh inference context: region variables from
278        // the caller's context (e.g. borrowck'd MIR) do not affect which impl
279        // matches and must be erased, while type/const inference variables
280        // would be dangling indices - a caller bug.
281        assert!(
282            !substs.has_non_region_infer(),
283            "trait selection on substs with inference variables: {substs:?}"
284        );
285        let substs = self.tcx.erase_and_anonymize_regions(substs);
286        let infcx = self.infer_ctxt();
287        let mut sc = SelectionContext::new(&infcx);
288        let trait_ref = ty::TraitRef::new(self.tcx, trait_id, substs);
289        let obligation = Obligation::new(
290            self.tcx,
291            ObligationCause::dummy(),
292            typing_env.param_env,
293            TraitPredicate {
294                trait_ref,
295                polarity: PredicatePolarity::Positive,
296            },
297        );
298        let result = sc.select(&obligation);
299        match result {
300            Ok(Some(ImplSource::UserDefined(data))) => Some(data.impl_def_id),
301            _ => None,
302        }
303    }

305    /// Given some procedure `proc_def_id` which is called, this method returns the actual method which will be executed when `proc_def_id` is defined on a trait.
306    /// Returns `None` if this method can not be found or the provided `proc_def_id` is no trait item. Returns `proc_def_id` if a default implementation is called.
307    #[tracing::instrument(level = "debug", skip(self, caller_def_id))]
308    pub fn find_impl_of_trait_method_call(
309        self,
310        caller_def_id: impl IntoParam<ProcedureDefId>, // where are we calling from?
311        proc_def_id: impl IntoParam<ProcedureDefId> + Debug,
312        substs: GenericArgsRef<'tcx>,
313    ) -> Option<ProcedureDefId> {
314        let proc_def_id = proc_def_id.into_param();
315        let typing_env = ty::TypingEnv::post_analysis(self.tcx, caller_def_id.into_param());
316        let impl_def_id = self.find_trait_impl_of_method_call(typing_env, proc_def_id, substs)?;
317        for item in self.tcx.associated_items(impl_def_id).in_definition_order() {
318            if let Some(id) = item.trait_item_def_id() {
319                if id == proc_def_id {
320                    return Some(item.def_id);
321                }
322            }
323        }
324        if self
325            .tcx
326            .associated_item(proc_def_id)
327            .defaultness(self.tcx)
328            .has_value()
329        {
330            return Some(proc_def_id);
331        }
332        unreachable!()
333    }

335    /// Given the `called_def_id` of a function, called from `caller_def_id`,
336    /// and some substitutions `call_substs` applied for that call, returns
337    /// `true` if the call is to a function in the specified crate
338    /// `crate_name`. A trait method call is attributed to the crate of the
339    /// `impl` it relies on (e.g. `PartialOrd::le` on `Int` belongs to
340    /// `prusti_contracts`, even though the default `le` body lives in
341    /// `core`); other calls, and trait calls whose impl is unknown (e.g. a
342    /// generic `Self`), to the crate of the called definition itself.
343    pub fn is_function_in_crate(
344        self,
345        caller_def_id: impl IntoParam<ProcedureDefId>, // where are we calling from?
346        called_def_id: impl IntoParam<ProcedureDefId>, // what are we calling?
347        call_substs: GenericArgsRef<'tcx>,
348        crate_name: &'tcx str,
349    ) -> bool {
350        let called_def_id = called_def_id.into_param();
351        let typing_env = ty::TypingEnv::post_analysis(self.tcx, caller_def_id.into_param());
352        let def_id = self
353            .find_trait_impl_of_method_call(typing_env, called_def_id, call_substs)
354            .unwrap_or(called_def_id);
355        self.tcx.crate_name(def_id.krate).as_str() == crate_name
356    }

364    /// Given a call to `called_def_id` from within `caller_def_id`, returns
365    /// the `DefId` that will actually be called if known (i.e. if a trait
366    /// method call actually resolves to a concrete implementation), as well as
367    /// the correct substitutions for that call. If a method is not resolved,
368    /// returns the original `called_def_id` and `call_substs`.
369    pub fn resolve_method_call(
370        self,
371        caller_def_id: impl IntoParam<ProcedureDefId>, // where are we calling from?
372        called_def_id: impl IntoParam<ProcedureDefId>, // what are we calling?
373        call_substs: GenericArgsRef<'tcx>,
374    ) -> (ProcedureDefId, GenericArgsRef<'tcx>) {
375        let called_def_id = called_def_id.into_param();
376
377        (|| {
378            // trait resolution does not depend on lifetimes, and in fact fails in the presence of uninferred regions
379            let clean_substs = self.tcx.erase_and_anonymize_regions(call_substs);
380
381            let typing_env = ty::TypingEnv::post_analysis(self.tcx, caller_def_id.into_param());
382            let instance = self
383                .tcx
384                .resolve_instance_raw(typing_env.as_query_input((called_def_id, clean_substs)))
385                .ok()??;
386            let resolved_def_id = instance.def_id();
387            let resolved_substs = if resolved_def_id == called_def_id {
388                // if no trait resolution occurred, we can keep the non-erased substs
389                call_substs
390            } else {
391                instance.args
392            };
393
394            Some((resolved_def_id, resolved_substs))
395        })()
396        .unwrap_or((called_def_id, call_substs))
397    }

399    /// Checks whether `ty` is copy.
400    /// The type is wrapped into a `Binder` to handle regions correctly.
401    /// The `param_env` should be passed as a `ProcedureDefId` which is
402    /// then used to calculate the param env; i.e. the set of
403    /// where-clauses that are in scope at this particular point.
404    pub fn type_is_copy(
405        self,
406        ty: ty::Binder<'tcx, ty::Ty<'tcx>>,
407        param_env: impl IntoParamTcx<'tcx, ParamEnv<'tcx>>,
408    ) -> bool {
409        let param_env = param_env.into_param(self.tcx);
410        let typing_env = ty::TypingEnv::new(param_env, ty::TypingMode::PostAnalysis);
411        // Normalize the type to account for associated types
412        let ty = self.resolve_assoc_types(ty, param_env).skip_binder();
413        let ty = self.tcx.erase_and_anonymize_regions(ty);
414        self.tcx.is_copy_raw(typing_env.as_query_input(ty))
415    }

430    /// Checks whether the given type implements the trait with the given DefId.
431    /// Accounts for generic params on the trait by the given `trait_substs`.
432    /// The `param_env` should be passed as a `ProcedureDefId` which is
433    /// then used to calculate the param env; i.e. the set of
434    /// where-clauses that are in scope at this particular point.
435    pub fn type_implements_trait_with_trait_substs(
436        self,
437        trait_def_id: DefId,
438        trait_substs: impl IntoIterator<Item = impl Into<ty::GenericArg<'tcx>>>,
439        param_env: impl IntoParamTcx<'tcx, ParamEnv<'tcx>>,
440    ) -> bool {
441        assert!(self.tcx.is_trait(trait_def_id));
442        let infcx = self.infer_ctxt();
443        let mut type_freshener = TypeFreshener::new(&infcx);
444        infcx
445            .type_implements_trait(
446                trait_def_id,
447                // If `ty` has any inference variables (e.g. a region variable), then using it with
448                // the freshly-created `InferCtxt` (i.e. `tcx.infer_ctxt().enter(..)`) will cause
449                // a panic, since those inference variables don't exist in the new `InferCtxt`.
450                // See: https://rust-lang.zulipchat.com/#narrow/stream/182449-t-compiler.2Fhelp/topic/.E2.9C.94.20Panic.20in.20is_copy_modulo_regions
451                trait_substs
452                    .into_iter()
453                    .map(|ty| ty.into().fold_with(&mut type_freshener)),
454                param_env.into_param(self.tcx),
455            )
456            .must_apply_considering_regions()
457    }

465    /// Evaluates the provided [ty::Predicate].
466    /// Returns true if the predicate is fulfilled.
467    /// Returns false if the predicate is not fulfilled or it could not be evaluated.
468    /// The `param_env` should be passed as a `ProcedureDefId` which is
469    /// then used to calculate the param env; i.e. the set of
470    /// where-clauses that are in scope at this particular point.
471    #[tracing::instrument(level = "debug", skip(self))]
472    pub fn evaluate_predicate(
473        self,
474        predicate: ty::Predicate<'tcx>,
475        param_env: impl IntoParamTcx<'tcx, ParamEnv<'tcx>> + Debug,
476    ) -> bool {
477        let obligation = Obligation::new(
478            self.tcx,
479            ObligationCause::dummy(),
480            param_env.into_param(self.tcx),
481            predicate,
482        );
483
484        self.infer_ctxt()
485            .predicate_must_hold_considering_regions(&obligation)
486    }

488    /// Normalizes associated types in foldable types,
489    /// i.e. this resolves projection types ([ty::TyKind::Projection]s)
490    /// **Important:** Regions while be erased during this process
491    /// The `param_env` should be passed as a `ProcedureDefId` which is
492    /// then used to calculate the param env; i.e. the set of
493    /// where-clauses that are in scope at this particular point.
494    #[tracing::instrument(level = "debug", skip(self))]
495    pub fn resolve_assoc_types<T: ty::TypeFoldable<TyCtxt<'tcx>> + Debug + Copy>(
496        self,
497        normalizable: T,
498        param_env: impl IntoParamTcx<'tcx, ParamEnv<'tcx>> + Debug,
499    ) -> T {
500        let param_env = param_env.into_param(self.tcx);
501        let typing_env = ty::TypingEnv::new(param_env, ty::TypingMode::PostAnalysis);
502        let norm_res = self
503            .tcx
504            .try_normalize_erasing_regions(typing_env, Unnormalized::new(normalizable));
505
506        match norm_res {
507            Ok(normalized) => {
508                debug!("Normalized {normalizable:?}: {normalized:?}");
509                normalized
510            }
511            Err(err) => {
512                debug!("Error while resolving associated types for {normalizable:?}: {err:?}");
513                normalizable
514            }
515        }
516    }
