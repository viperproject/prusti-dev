1//! A module that invokes the verifier `prusti-viper`
2
3use log::{debug, warn};
4use prusti_interface::{data::VerificationTask, environment::Environment, specs::typed};
5use prusti_utils::{config, report::user};
6
7#[tracing::instrument(name = "prusti::verify", level = "debug", skip(env))]
8pub fn verify<'tcx>(
9    env: Environment<'tcx>,
10    def_spec: typed::DefSpecificationMap,
11    verification_task: VerificationTask<'tcx>,
12) {
13    if env.diagnostic.has_errors() {
14        warn!("The compiler reported an error, so the program will not be verified.");
15    } else {
16        debug!("Verification task: {:?}", verification_task);
17        user::message(format!(
18            "{}erification of {} items...",
19            if verification_task.selective {
20                "Selective v"
21            } else {
22                "V"
23            },
24            verification_task.procedures.len()
25        ));
26
27        if config::print_collected_verification_items() {
28            println!(
29                "Collected verification items {}:",
30                verification_task.procedures.len()
31            );
32            for procedure in &verification_task.procedures {
33                println!(
34                    "procedure: {} at {:?}",
35                    env.name.get_item_def_path(*procedure),
36                    env.query.get_def_span(procedure)
37                );
38            }
39        }
40
41        // encode the crate to a RequestWithContext
42        // TODO: push RequestWithContext through (replace VerificationRequest
43        //   which is constructed further inside `prusti_server`)
44        let request = prusti_encoder::test_entrypoint(
45            env.tcx(),
46            env.body,
47            def_spec,
48            if verification_task.selective {
49                Some(verification_task.procedures)
50            } else {
51                None
52            },
53            &env.diagnostic,
54        );
55
56        // if we do not use encode-only to get the dumped viper program (e.g., just to see if encoding fails), we don't need to start a JVM
57        if !config::verify_mode().is_encode_only() || config::dump_viper_program() {
58            let program = request.program;
59
60            prusti_server::verify_programs(&env.diagnostic, vec![program]);
61        }
62    }
63}
