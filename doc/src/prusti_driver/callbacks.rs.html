31// Running `get_body_with_borrowck_facts` can be very slow, therefore we avoid it when not
32// necessary; for crates which won't be verified or spec_fns it suffices to load just the fn body
33
34#[allow(clippy::needless_lifetimes)]
35#[tracing::instrument(level = "debug", skip(tcx))]
36fn mir_borrowck<'tcx>(tcx: TyCtxt<'tcx>, def_id: LocalDefId) -> MirBorrowck<'tcx> {
37    let def_kind = tcx.def_kind(def_id.to_def_id());
38    let is_anon_const = matches!(def_kind, DefKind::AnonConst);
39    // Anon Const bodies have already been stolen and so will result in a crash
40    // when calling `get_body_with_borrowck_facts`. TODO: figure out if we need
41    // (anon) const bodies at all, and if so, how to get them?
42    if !is_anon_const {
43        let consumer_opts =
44            if is_spec_fn(tcx, def_id.to_def_id()) || config::verify_mode().is_compile_only() {
45                consumers::ConsumerOptions::RegionInferenceContext
46            } else {
47                consumers::ConsumerOptions::PoloniusInputFacts
48            };
49        let body_with_facts = consumers::get_bodies_with_borrowck_facts(tcx, def_id, consumer_opts);
50        for (def_id, body) in body_with_facts {
51            // SAFETY: This is safe because we are feeding in the same `tcx` that is
52            // going to be used as a witness when pulling out the data.
53            unsafe {
54                mir_storage::store_mir_body(tcx, def_id, body);
55            }
56        }
57    }
58    let mut providers = Providers::default();
59    prusti_rustc_interface::borrowck::provide(&mut providers);
60    let original_mir_borrowck = providers.mir_borrowck;
61    original_mir_borrowck(tcx, def_id)
62}

64#[allow(clippy::needless_lifetimes)]
65#[tracing::instrument(level = "debug", skip(tcx))]
66fn mir_promoted<'tcx>(
67    tcx: TyCtxt<'tcx>,
68    def_id: LocalDefId,
69) -> (
70    &'tcx Steal<mir::Body<'tcx>>,
71    &'tcx Steal<IndexVec<mir::Promoted, mir::Body<'tcx>>>,
72) {
73    let original_mir_promoted = prusti_rustc_interface::interface::DEFAULT_QUERY_PROVIDERS
74        .queries
75        .mir_promoted;
76    let result = original_mir_promoted(tcx, def_id);
77    // SAFETY: This is safe because we are feeding in the same `tcx` that is
78    // going to be used as a witness when pulling out the data.
79    unsafe {
80        mir_storage::store_promoted_mir_body(tcx, def_id, result.0.borrow().clone());
81    }
82    result
83}

90    fn config(&mut self, config: &mut Config) {
91        assert!(config.override_queries.is_none());
92        config.override_queries = Some(|_session: &Session, providers| {
93            providers.queries.mir_borrowck = mir_borrowck;
94            providers.queries.mir_promoted = mir_promoted;

98    fn after_crate_root_parsing(&mut self, compiler: &Compiler, krate: &mut Crate) -> Compilation {
99        // `prusti-rustc` will pass a `--extern` option indicating where to find
100        // the `prusti-contracts` crate. This crate must be loaded in order to
101        // include the default extern specs (for e.g build-in `assert!`
102        // statements). However, if the user does not explicitly import the
103        // crate in the code or use an `extern crate` statement, then the
104        // compiler will not load the crate.
105        //
106        // To work around this, we add a dummy `extern crate` statement to the
107        // crate if one does not already exist.
108        //
109        // Ideally we could load the prusti-contracts crate directly in the
110        // TyCtxt rather than modifying the crate AST.  However, it does not
111        // seem possible to load the crate directly in the TyCtxt in the
112        // `after_expansion` or `after_analysis` callback as the `cstore` object
113        // appears to be frozen at that point.
114
115        if compiler.sess.opts.externs.get("prusti_contracts").is_some() {
116            let prusti_contracts_symbol = Symbol::intern("prusti_contracts");
117
118            if !krate.items.iter().any(|item| match item.kind {
119                ItemKind::ExternCrate(Some(original_ident), _) => {
120                    original_ident == prusti_contracts_symbol
121                }
122                ItemKind::ExternCrate(None, ident) => ident.name == prusti_contracts_symbol,
123                _ => false,
124            }) {
125                krate.items.push(Box::new(Item {
126                    attrs: AttrVec::new(),
127                    id: DUMMY_NODE_ID,
128                    span: DUMMY_SP,
129                    vis: Visibility {
130                        kind: VisibilityKind::Public,
131                        span: DUMMY_SP,
132                    },
133                    kind: ItemKind::ExternCrate(
134                        None,
135                        Ident::new(prusti_contracts_symbol, DUMMY_SP),
136                    ),
137                    tokens: None,
138                }));
139            }
140        }
141        Compilation::Continue
142    }

144    #[tracing::instrument(level = "debug", skip_all)]
145    fn after_expansion<'tcx>(&mut self, compiler: &Compiler, tcx: TyCtxt<'tcx>) -> Compilation {
146        if compiler.sess.is_rust_2015() {
147            compiler
148                .sess
149                .dcx()
150                .struct_warn(
151                    "Prusti specifications are supported only from 2018 edition. Please \
152                    specify the edition with adding a command line argument `--edition=2018` or \
153                    `--edition=2021`.",
154                )
155                .emit();
156        }
157        compiler.sess.dcx().abort_if_errors();
158        if config::print_desugared_specs() {
159            // based on the implementation of rustc_driver::pretty::print_after_parsing
160            let sess = &compiler.sess;
161            let krate = &tcx.resolver_for_lowering().1.borrow();
162            let src_name = sess.io.input.file_name(sess);
163            let src = sess
164                .source_map()
165                .get_source_file(&src_name)
166                .expect("get_source_file")
167                .src
168                .as_ref()
169                .expect("src")
170                .to_string();
171            print!(
172                "{}",
173                prusti_rustc_interface::ast_pretty::pprust::print_crate(
174                    sess.source_map(),
175                    krate,
176                    src_name,
177                    src,
178                    &NoAnn,
179                    false,
180                    sess.edition(),
181                    &sess.psess.attr_id_generator,
182                )
183            );
184        }
185        Compilation::Continue
186    }

188    #[tracing::instrument(level = "debug", skip_all)]
189    fn after_analysis<'tcx>(&mut self, compiler: &Compiler, tcx: TyCtxt<'tcx>) -> Compilation {
190        compiler.sess.dcx().abort_if_errors();
191        let mut env = Environment::new(tcx, env!("CARGO_PKG_VERSION"));
192        let spec_checker = specs::checker::SpecChecker::new();
193        spec_checker.check(&env);
194        compiler.sess.dcx().abort_if_errors();
195
196        let mut spec_collector = specs::SpecCollector::new(&mut env);
197        spec_collector.collect_specs();
198
199        let mut def_spec = spec_collector.build_def_specs();
200        // Do print_typeckd_specs prior to importing cross crate
201        if config::print_typeckd_specs() {
202            for value in def_spec.all_values_debug(config::hide_uuids()) {
203                println!("{value}");
204            }
205        }
206        CrossCrateSpecs::import_export_cross_crate(&mut env, &mut def_spec);
207
208        // TODO: can we replace `get_annotated_procedures` with information
209        // that is already in `def_spec`?
210        let (annotated_procedures, types) = env.get_annotated_procedures_and_types();
211
212        if config::show_ide_info() && !config::verify_mode().is_compile_only() {
213            let compiler_info = IdeInfo::collect(&env, &annotated_procedures, &def_spec);
214            let out = serde_json::to_string(&compiler_info).unwrap();
215            PrustiError::message(format!("compilerInfo{out}"), DUMMY_SP.into())
216                .emit(&env.diagnostic);
217        }
218        // as long as we have to throw a fake error we need to check this
219        // note that is_single_file will still be false if this is run through x.py.
220        // it will find a manifest in prusti-launch but CARGO_PRIMARY_PACKAGE will still
221        // not be ok. meaning that selective verification can't currently be tested through
222        // x.py if the target is a single-file program.
223        let is_single_file = std::env::var("CARGO_MANIFEST_DIR").is_err();
224        let is_primary_package = is_single_file || std::env::var("CARGO_PRIMARY_PACKAGE").is_ok();
225
226        // collect and output Information used by IDE:
227        if !config::verify_mode().is_compile_only() && !config::skip_verification() {
228            let target_def_paths = config::verify_only_defpaths();
229            debug!(
230                "Received def paths: {target_def_paths:?}. Package is primary: {is_primary_package}, Package is single-file: {is_single_file}"
231            );
232            if !target_def_paths.is_empty() {
233                // if we do selective verification, then definitely only
234                // for methods of the primary package. Check needed because
235                // of the fake_error, otherwise verification stops early
236                // with local dependencies
237                if is_primary_package {
238                    let procedures = annotated_procedures
239                        .into_iter()
240                        .filter(|x| target_def_paths.contains(&env.name.get_unique_item_name(*x)))
241                        .collect::<Vec<_>>();
242                    if !procedures.is_empty() {
243                        let selective_task = VerificationTask {
244                            procedures,
245                            types,
246                            selective: true,
247                        };
248                        verify(env, def_spec, selective_task);
249                    } else {
250                        info!("Passed selective defpaths were empty or had no matching procedures - skipping verification.");
251                    }
252                    // fake_error(&env_diagnostic);
253                }
254            } else {
255                let verification_task = VerificationTask {
256                    procedures: annotated_procedures,
257                    types,
258                    selective: false,
259                };
260                verify(env, def_spec, verification_task);
261            }
262        } else if config::skip_verification()
263            && !config::verify_mode().is_compile_only()
264            && is_primary_package
265        {
266            // add a fake error, reason explained in issue #1261
267            fake_error(&env.diagnostic);
268        }
269
270        compiler.sess.dcx().abort_if_errors();
271        if config::full_compilation() {
272            Compilation::Continue
273        } else {
274            Compilation::Stop
275        }
276    }
