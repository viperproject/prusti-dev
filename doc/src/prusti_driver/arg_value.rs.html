8/// Source: https://github.com/rust-lang/rust-clippy/blob/master/src/driver.rs
9use std::ops::Deref;
10
11/// If a command-line option matches `find_arg`, then apply the predicate `pred` on its value. If
12/// true, then return it. The parameter is assumed to be either `--arg=value` or `--arg value`.
13pub fn arg_value<'a, T: Deref<Target = str>>(
14    args: &'a [T],
15    find_arg: &str,
16    pred: impl Fn(&str) -> bool,
17) -> Option<&'a str> {
18    let mut args = args.iter().map(Deref::deref);
19    while let Some(arg) = args.next() {
20        let mut arg = arg.splitn(2, '=');
21        if arg.next() != Some(find_arg) {
22            continue;
23        }
24
25        match arg.next().or_else(|| args.next()) {
26            Some(v) if pred(v) => return Some(v),
27            _ => {}
28        }
29    }
30    None
31}
32
33#[test]
34fn test_arg_value() {
35    let args = &["--bar=bar", "--foobar", "123", "--foo"];
36
37    assert_eq!(arg_value(&[] as &[&str], "--foobar", |_| true), None);
38    assert_eq!(arg_value(args, "--bar", |_| false), None);
39    assert_eq!(arg_value(args, "--bar", |_| true), Some("bar"));
40    assert_eq!(arg_value(args, "--bar", |p| p == "bar"), Some("bar"));
41    assert_eq!(arg_value(args, "--bar", |p| p == "foo"), None);
42    assert_eq!(arg_value(args, "--foobar", |p| p == "foo"), None);
43    assert_eq!(arg_value(args, "--foobar", |p| p == "123"), Some("123"));
44    assert_eq!(arg_value(args, "--foo", |_| true), None);
45}
