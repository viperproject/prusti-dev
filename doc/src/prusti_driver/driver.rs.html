29/// Link to report Prusti bugs
30const BUG_REPORT_URL: &str = "https://github.com/prusti/prusti/issues/new";

32fn get_prusti_version_info() -> String {
33    format!(
34        "{}, commit {} {}, built on {}",
35        env!("CARGO_PKG_VERSION"),
36        option_env!("COMMIT_HASH").unwrap_or("<unknown>"),
37        option_env!("COMMIT_TIME").unwrap_or("<unknown>"),
38        option_env!("BUILD_TIME").unwrap_or("<unknown>"),
39    )
40}

42/// Initialize Prusti and the Rust compiler loggers.
43fn init_loggers() -> Option<FlushGuard> {
44    // TODO: The `config::log() != ""` here is very bad; it makes us ignore the `log_tracing` flag
45    // It's enabled so that we only create a `trace.json` file if the user has explicitly requested logging
46    let guard = if config::log_tracing() && !config::log().is_empty() {
47        let log_dir = config::log_dir();
48        std::fs::create_dir_all(&log_dir).expect("failed to create log directory");
49        let filter = EnvFilter::new(config::log());
50        let (chrome_layer, guard) = ChromeLayerBuilder::new()
51            .file(log_dir.join("trace.json"))
52            .include_args(true)
53            .build();
54        tracing_subscriber::registry()
55            .with(filter)
56            .with(chrome_layer)
57            .init();
58        Some(guard)
59    } else {
60        env_logger::init_from_env(
61            env_logger::Env::new()
62                .filter_or("PRUSTI_LOG", config::log())
63                .write_style_or("PRUSTI_LOG_STYLE", config::log_style()),
64        );
65        None
66    };
67
68    let error_handler = EarlyDiagCtxt::new(session::config::ErrorOutputType::default());
69    prusti_rustc_interface::driver::init_rustc_env_logger(&error_handler);
70    guard
71}

73fn main() {
74    driver::install_ice_hook(BUG_REPORT_URL, |handler| {
75        let version_info = get_prusti_version_info();
76        handler
77            .handle()
78            .note(format!("Prusti version: {version_info}"));
79    });
80
81    // To measure how long Prusti takes to run
82    let stopwatch = Stopwatch::start("prusti", "main");
83
84    // Filter out Prusti's `-P<arg>=<val>` arguments
85    let original_rustc_args = config::get_filtered_args();
86
87    // If the environment asks us to actually be rustc, then run `rustc` instead of Prusti,
88    // or if we're building a build script where we can't retrieve MIR bodies.
89    let build_script_build = arg_value(&original_rustc_args, "--crate-name", |val| {
90        val == "build_script_build"
91    })
92    .is_some();
93
94    // This environment variable will not be set when building dependencies.
95    let is_primary_package = env::var("CARGO_PRIMARY_PACKAGE").is_ok();
96
97    if config::be_rustc()
98        || build_script_build
99        || (!is_primary_package && config::ignore_deps_contracts())
100    {
101        driver::main().exit_process();
102    }
103
104    // Initialize Prusti and the Rust compiler loggers.
105    // This must be done after the build script check, otherwise Tokio's global tracing will fail.
106    let _log_flush_guard = init_loggers();
107
108    // Is this crate a dependency when user doesn't want to verify dependencies
109    let is_no_verify_dep_crate = !is_primary_package && config::no_verify_deps();
110
111    // Would `cargo check` not report errors for this crate? That is, are lints disabled
112    // (i.e. is this a non-local crate)
113    let are_lints_disabled =
114        arg_value(&original_rustc_args, "--cap-lints", |val| val == "allow").is_some();
115
116    // Remote dependencies (e.g. from git/crates.io), or any dependencies if `no_verify_deps`,
117    // are not verified. However, we still run Prusti on them to export potential specs.
118    if is_no_verify_dep_crate || are_lints_disabled {
119        config::set_verify_mode(VerifyMode::CompileOnly);
120    }
121
122    // Disable incremental compilation because it causes mir_borrowck not to be called.
123    let mut rustc_args = Vec::new();
124    let mut is_codegen = false;
125    for arg in original_rustc_args {
126        if arg == "--codegen" || arg == "-C" {
127            is_codegen = true;
128        } else if is_codegen && arg.starts_with("incremental=") {
129            // Just drop the argument.
130            is_codegen = false;
131        } else {
132            if is_codegen {
133                rustc_args.push("-C".to_owned());
134                is_codegen = false;
135            }
136            rustc_args.push(arg);
137        }
138    }
139
140    // TODO: is this necessary, after compiler update there will never be an error?
141    let exit_code = driver::catch_with_exit_code(move || {
142        user::message(format!(
143            "{}\n{}\n{}\n",
144            r"  __          __        __  ___             ",
145            r" |__)  _\/_  |__) |  | /__`  |   ____\/_  | ",
146            r" |      /\   |  \ \__/ .__/  |       /\   | ",
147        ));
148        user::message(format!("Prusti version: {}", get_prusti_version_info()));
149        info!("Prusti version: {}", get_prusti_version_info());
150
151        if rustc_args.get(1).map(|s| s.as_ref()) == Some("-vV") {
152            // When cargo queries the verbose rustc version,
153            // also print the Prusti version to stdout.
154            // This ensures that the cargo build cache is
155            // invalidated when the Prusti version changes.
156            println!("Prusti version: {}", get_prusti_version_info());
157        }
158
159        rustc_args.push("-Zalways-encode-mir".to_owned());
160        rustc_args.push("-Zcrate-attr=feature(stmt_expr_attributes)".to_owned());
161        rustc_args.push("-Zcrate-attr=feature(register_tool)".to_owned());
162        rustc_args.push("-Zcrate-attr=register_tool(prusti)".to_owned());
163
164        if config::check_overflows() {
165            // Some crates might have a `overflow-checks = false` in their `Cargo.toml` to
166            // disable integer overflow checks, but we want to override that.
167            rustc_args.push("-Coverflow-checks=on".to_owned());
168        }
169
170        if config::dump_debug_info() {
171            rustc_args.push(format!(
172                "-Zdump-mir-dir={}",
173                config::log_dir()
174                    .join("mir")
175                    .to_str()
176                    .expect("failed to configure dump-mir-dir")
177            ));
178            rustc_args.push("-Zdump-mir=all".to_owned());
179            rustc_args.push("-Zdump-mir-graphviz".to_owned());
180            if !config::ignore_regions() {
181                rustc_args.push("-Zidentify-regions=yes".to_owned());
182            }
183        }
184        if config::dump_nll_facts() {
185            rustc_args.push("-Znll-facts=yes".to_string());
186            rustc_args.push(format!(
187                "-Znll-facts-dir={}",
188                config::log_dir()
189                    .join("nll-facts")
190                    .to_str()
191                    .expect("failed to configure nll-facts-dir")
192            ));
193        }
194
195        let mut callbacks = PrustiCompilerCalls;
196
197        driver::run_compiler(&rustc_args, &mut callbacks)
198    });
199
200    // Check if verifying a program in our test suite is taking too long
201    let duration = stopwatch.finish();
202    if let Some(deadline) = config::verification_deadline() {
203        // Check that we met the deadline.
204        assert!(
205            duration < std::time::Duration::from_secs(deadline),
206            "Prusti failed to finish within {deadline} seconds. It finished in {duration:?}.",
207        );
208    }
209
210    exit_code.exit_process();
211}
