11pub fn generate_constructor(
12    env: &JNIEnv,
13    class: &ClassName,
14    target_signature: Option<String>,
15    suffix: Option<String>,
16) -> Result<String> {
17    let clazz = env.find_class(class.path())?;
18
19    let constructors = env
20        .call_method(
21            clazz,
22            "getConstructors",
23            "()[Ljava/lang/reflect/Constructor;",
24            &[],
25        )?
26        .l()?;
27    let num_constructors = env.get_array_length(*constructors)?;
28
29    let mut indexed_constructors = BTreeMap::new();
30
31    for constructor_index in 0..num_constructors {
32        let constructor = env.get_object_array_element(*constructors, constructor_index)?;
33
34        let constructor_signature = java_str_to_string(
35            &env.get_string(
36                env.call_static_method(
37                    "org/objectweb/asm/Type",
38                    "getConstructorDescriptor",
39                    "(Ljava/lang/reflect/Constructor;)Ljava/lang/String;",
40                    &[JValue::Object(constructor)],
41                )?
42                .l()?
43                .into(),
44            )?,
45        )?;
46
47        indexed_constructors.insert(constructor_signature, constructor);
48    }
49
50    let (constructor_signature, constructor) = match target_signature {
51        None => {
52            if indexed_constructors.is_empty() {
53                return Err(ErrorKind::NoConstructors(class.full_name()).into());
54            }
55            if indexed_constructors.len() > 1 {
56                return Err(ErrorKind::AmbiguousConstructor(
57                    class.full_name(),
58                    indexed_constructors.keys().map(|k| k.to_string()).collect(),
59                )
60                .into());
61            }
62            indexed_constructors.pop_first().unwrap()
63        }
64        Some(sign) => match indexed_constructors.get(&sign) {
65            Some(constr) => (sign, *constr),
66            None => return Err(ErrorKind::NoMatchingConstructor(class.full_name(), sign).into()),
67        },
68    };
69
70    let parameters = env
71        .call_method(
72            constructor,
73            "getParameters",
74            "()[Ljava/lang/reflect/Parameter;",
75            &[],
76        )?
77        .l()?;
78    let num_parameters = env.get_array_length(*parameters)?;
79
80    let mut parameter_names: Vec<String> = Vec::with_capacity(num_parameters as usize);
81    let mut parameter_signatures: Vec<String> = Vec::with_capacity(num_parameters as usize);
82
83    for parameter_index in 0..num_parameters {
84        let parameter = env.get_object_array_element(*parameters, parameter_index)?;
85        let parameter_name = env.get_string(
86            env.call_method(parameter, "getName", "()Ljava/lang/String;", &[])?
87                .l()?
88                .into(),
89        )?;
90        let parameter_type = env
91            .call_method(parameter, "getType", "()Ljava/lang/Class;", &[])?
92            .l()?;
93        let parameter_signature = env.get_string(
94            env.call_static_method(
95                "org/objectweb/asm/Type",
96                "getDescriptor",
97                "(Ljava/lang/Class;)Ljava/lang/String;",
98                &[JValue::Object(parameter_type)],
99            )?
100            .l()?
101            .into(),
102        )?;
103
104        parameter_names.push(java_str_to_valid_rust_argument_name(&parameter_name)?);
105        parameter_signatures.push(java_str_to_string(&parameter_signature)?);
106    }
107
108    let constructor_name = match suffix {
109        None => "new".to_string(),
110        Some(s) => format!("new_{s}"),
111    };
112
113    Ok(generate(
114        class,
115        &constructor_name,
116        &constructor_signature,
117        &parameter_names,
118        &parameter_signatures,
119    ))
120}

122fn generate(
123    class: &ClassName,
124    constructor_name: &str,
125    constructor_signature: &str,
126    parameter_names: &[String],
127    parameter_signatures: &[String],
128) -> String {
129    assert_eq!(parameter_names.len(), parameter_signatures.len());
130
131    let mut code: Vec<String> = vec![];
132    code.push(format!(
133        "/// Calls a constructor of Java class `{}`.",
134        class.full_name()
135    ));
136    code.push("///".to_string());
137    code.push("/// Type and Java signature of parameters:".to_string());
138
139    for i in 0..parameter_names.len() {
140        let par_name = &parameter_names[i];
141        let par_sign = &parameter_signatures[i];
142        let par_type = generate_jni_type(par_sign);
143        code.push(format!("/// - `{par_name}`: `{par_type}` (`{par_sign}`)"));
144    }
145
146    code.push("///".to_string());
147    code.push(format!(
148        "/// Return type and Java signature: `JObject` (`L{};`)",
149        class.path()
150    ));
151
152    code.push(format!("pub fn {constructor_name}("));
153    code.push("    &self,".to_string());
154
155    for i in 0..parameter_names.len() {
156        let par_name = &parameter_names[i];
157        let par_sign = &parameter_signatures[i];
158        let par_type = generate_jni_type(par_sign);
159        code.push(format!("    {par_name}: {par_type},"));
160    }
161
162    code.push(") -> JNIResult<JObject<'a>> {".to_string());
163
164    code.push(format!("    let class_name = \"{}\";", class.path()));
165    code.push("    let method_name = \"<init>\";".to_string());
166    code.push(format!(
167        "    let method_signature = \"{constructor_signature}\";"
168    ));
169
170    for i in 0..parameter_names.len() {
171        let par_name = &parameter_names[i];
172        let par_sign = &parameter_signatures[i];
173        if is_signature_of_class_type(par_sign) {
174            let par_class = &par_sign[1..(par_sign.len() - 1)];
175            code.push(format!("    let {par_name}_class_name = \"{par_class}\";"));
176        }
177    }
178    code.push("".to_string());
179
180    // Generate dynamic type check for the arguments
181    for i in 0..parameter_names.len() {
182        let par_name = &parameter_names[i];
183        let par_sign = &parameter_signatures[i];
184        if is_signature_of_class_type(par_sign) {
185            code.push(generate_variable_type_check(
186                par_name,
187                &format!("{par_name}_class_name"),
188            ));
189        }
190    }
191
192    code.push(
193        r#"    static CLASS: OnceCell<GlobalRef> = OnceCell::new();
194    static METHOD_ID: OnceCell<JMethodID> = OnceCell::new();
195    let class = CLASS.get_or_try_init(|| {
196        let class = self.env.find_class(class_name)?;
197        self.env.new_global_ref(class)
198    })?;
199    let method_id = *METHOD_ID.get_or_try_init(|| {
200        self.env.get_method_id(
201            class_name,
202            method_name,
203            method_signature
204        )
205    })?;
206"#
207        .to_string(),
208    );
209
210    code.push("    let result = self.env.new_object_unchecked(".to_string());
211    code.push("        JClass::from(class.as_obj()),".to_string());
212    code.push("        method_id,".to_string());
213    code.push("        &[".to_string());
214
215    for i in 0..parameter_names.len() {
216        let par_name = &parameter_names[i];
217        let par_sign = &parameter_signatures[i];
218        let par_jvalue = generate_jvalue_wrapper(par_name, par_sign);
219        code.push(format!("            {par_jvalue},"));
220    }
221
222    code.push("        ]".to_string());
223    code.push("    );".to_string());
224    code.push("".to_string());
225
226    // Generate dynamic type check for the result
227    code.push(generate_result_type_check("class_name"));
228    code.push("    result".to_string());
229    code.push("}".to_string());
230
231    code.join("\n") + "\n"
232}
