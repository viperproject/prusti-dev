7use crate::{
8    class_name::*,
9    errors::*,

16use jni::JNIEnv;
17use log::warn;
18
19pub struct ClassGenerator<'a> {
20    env: &'a JNIEnv<'a>,
21    class: ClassName,
22    items: Vec<ItemWrapperSpec>,
23}
24
25impl<'a> ClassGenerator<'a> {
26    pub fn new(env: &'a JNIEnv<'a>, class: ClassName, items: Vec<ItemWrapperSpec>) -> Self {
27        ClassGenerator { env, class, items }
28    }
29
30    fn check(&self) -> Result<()> {
31        self.env
32            .find_class(self.class.path())
33            .map(|_| ())
34            .chain_err(|| ErrorKind::NoClass(self.class.full_name()))
35    }
36
37    pub fn generate(&self) -> Result<String> {
38        self.check()?;
39
40        let mut code: Vec<String> = vec![];
41        code.push(format!(
42            "//! Automatically generated code for '{}'\n",
43            self.class.full_name()
44        ));
45        code.push("#![allow(dead_code)]\n".to_string());
46        code.push("#![allow(non_snake_case)]\n".to_string());
47        code.push("#![allow(unused_imports)]\n".to_string());
48        code.push("#![allow(clippy::new_ret_no_self)]\n".to_string());
49        code.push("#![allow(clippy::wrong_self_convention)]\n".to_string());
50        code.push("#![allow(clippy::too_many_arguments)]\n".to_string());
51        code.push(self.generate_imports());
52        code.push(self.generate_struct());
53        code.push(self.generate_begin_impl());
54        code.push(self.generate_items()?);
55        code.push(self.generate_end_impl());
56
57        Ok(code.join("\n") + "\n")
58    }
59
60    fn generate_imports(&self) -> String {
61        vec![
62            "use jni::JNIEnv;",
63            "use jni::objects::GlobalRef;",
64            "use jni::objects::JObject;",
65            "use jni::objects::JMethodID;",
66            "use jni::objects::JStaticFieldID;",
67            "use jni::objects::JStaticMethodID;",
68            "use jni::objects::JValue;",
69            "use jni::objects::JClass;",
70            "use jni::errors::Result as JNIResult;",
71            "use jni::sys::*;",
72            "use jni::signature::*;",
73            "use once_cell::sync::OnceCell;",
74            "use std::str::FromStr;",
75            "use super::builtins::*;",
76        ]
77        .join("\n")
78            + "\n"
79    }
80
81    fn generate_struct(&self) -> String {
82        let mut code: Vec<String> = vec![];
83        code.push("#[allow(non_camel_case_types)]".to_string());
84        code.push(format!("pub struct {}<'a> {{", self.class.rust_name()));
85        code.push("    env: &'a JNIEnv<'a>,".to_string());
86        code.push("}".to_string());
87
88        code.join("\n")
89    }
90
91    fn generate_begin_impl(&self) -> String {
92        let mut code: Vec<String> = vec![];
93        code.push(format!("impl<'a> {}<'a> {{", self.class.rust_name()));
94        code.push("".to_string());
95        code.push("pub fn with(env: &'a JNIEnv<'a>) -> Self {".to_string());
96        code.push(format!("    {}{{ env }}", self.class.rust_name()));
97        code.push("}".to_string());
98        code.push("".to_string());
99
100        code.join("\n")
101    }
102
103    fn generate_end_impl(&self) -> String {
104        format!("}} // end of impl {}\n", self.class.rust_name())
105    }
106
107    fn generate_items(&self) -> Result<String> {
108        let mut gen_items: Vec<String> = vec![];
109
110        if self.items.is_empty() {
111            warn!("Class {} does not wrap any item", self.class.full_name())
112        }
113
114        for item in &self.items {
115            let gen = match *item {
116                ItemWrapperSpec::ScalaObjectGetter() => {
117                    generate_scala_object_getter(self.env, &self.class)?
118                }
119                ItemWrapperSpec::Constructor {
120                    ref signature,
121                    ref suffix,
122                } => {
123                    generate_constructor(self.env, &self.class, signature.clone(), suffix.clone())?
124                }
125                ItemWrapperSpec::Method {
126                    ref name,
127                    ref signature,
128                    ref suffix,
129                } => generate_method(
130                    self.env,
131                    &self.class,
132                    name,
133                    signature.clone(),
134                    suffix.clone(),
135                )?,
136                ItemWrapperSpec::FieldGetterSetter { ref field_name } => {
137                    generate_field_getter_setter_for_class(self.env, &self.class, field_name)?
138                }
139                ItemWrapperSpec::TraitFieldGetterSetter { ref field_name } => {
140                    generate_field_getter_setter_for_trait(self.env, &self.class, field_name)?
141                }
142            };
143            gen_items.push(gen)
144        }
145        Ok(gen_items.join("\n") + "\n")
146    }
147}
