12pub fn is_interface(env: &JNIEnv, class: &ClassName) -> Result<bool> {
13    let class = env.find_class(class.path())?;
14    Ok(env.call_method(class, "isInterface", "()Z", &[])?.z()?)
15}

17/// Generates Rust code to both retrieve and modify a value of
18/// the specified field of a given object of the specified interface
19///
20/// Such fields are translated from Scala to Java as methods
21pub fn generate_field_getter_setter_for_trait(
22    env: &JNIEnv,
23    class: &ClassName,
24    field_name: &str,
25) -> Result<String> {
26    if !is_interface(env, class)? {
27        return Err(ErrorKind::TraitField(class.full_name(), field_name.into(), false).into());
28    }
29    let setter_method_name = format!("{field_name}_$eq");
30    let getter_method_name = field_name.to_string();
31    let method::MethodInfo {
32        method_signature: setter_method_signature,
33        method: setter_method,
34        is_static: setter_is_static,
35    } = method::find_method(env, class, &setter_method_name, Option::None)?;
36    let method::MethodInfo {
37        method_signature: getter_method_signature,
38        method: getter_method,
39        is_static: getter_is_static,
40    } = method::find_method(env, class, &getter_method_name, Option::None)?;
41
42    // Check neither is static
43    if setter_is_static || getter_is_static {
44        return Err(ErrorKind::NoField(class.full_name(), field_name.into()).into());
45    }
46
47    let setter_parameters = env
48        .call_method(
49            setter_method,
50            "getParameters",
51            "()[Ljava/lang/reflect/Parameter;",
52            &[],
53        )?
54        .l()?;
55    let setter_num_parameters = env.get_array_length(*setter_parameters)?;
56
57    // Check setter has 1 parameter, it has the name "x$1" and it returns a void
58    if setter_num_parameters != 1 {
59        return Err(ErrorKind::NoField(class.full_name(), field_name.into()).into());
60    }
61
62    let setter_parameter = env.get_object_array_element(*setter_parameters, 0)?;
63    let setter_parameter_name = java_str_to_string(
64        &env.get_string(
65            env.call_method(setter_parameter, "getName", "()Ljava/lang/String;", &[])?
66                .l()?
67                .into(),
68        )?,
69    )?;
70
71    if setter_parameter_name != "x$1" {
72        return Err(ErrorKind::NoField(class.full_name(), field_name.into()).into());
73    }
74
75    // Check getter has 0 parameters
76    let getter_parameters = env
77        .call_method(
78            getter_method,
79            "getParameters",
80            "()[Ljava/lang/reflect/Parameter;",
81            &[],
82        )?
83        .l()?;
84    let getter_num_parameters = env.get_array_length(*getter_parameters)?;
85
86    if getter_num_parameters != 0 {
87        return Err(ErrorKind::NoField(class.full_name(), field_name.into()).into());
88    }
89
90    // Check the setter parameter and getter return type match
91    let setter_parameter_type = env
92        .call_method(setter_parameter, "getType", "()Ljava/lang/Class;", &[])?
93        .l()?;
94    let setter_parameter_signature = java_str_to_string(
95        &env.get_string(
96            env.call_static_method(
97                "org/objectweb/asm/Type",
98                "getDescriptor",
99                "(Ljava/lang/Class;)Ljava/lang/String;",
100                &[JValue::Object(setter_parameter_type)],
101            )?
102            .l()?
103            .into(),
104        )?,
105    )?;
106
107    let getter_return_signature = get_return_signature(&getter_method_signature);
108
109    if setter_parameter_signature != getter_return_signature {
110        return Err(ErrorKind::NoField(class.full_name(), field_name.into()).into());
111    }
112
113    let getter_code = method::generate(
114        class,
115        &getter_method_name,
116        &format!("get_{}", java_identifier_to_rust(field_name)),
117        &getter_method_signature,
118        &[],
119        &[],
120    );
121
122    let setter_code = method::generate(
123        class,
124        &setter_method_name,
125        &format!("set_{}", java_identifier_to_rust(field_name)),
126        &setter_method_signature,
127        &[format!("new_{field_name}")],
128        &[setter_parameter_signature],
129    );
130
131    Ok(format!("{setter_code}\n{getter_code}"))
132}
