15fn hierarchy_field_lookup<'a>(
16    env: &JNIEnv<'a>,
17    class: &ClassName,
18    lookup_name: &str,
19) -> Result<Option<JObject<'a>>> {
20    let mut class = env.find_class(class.path())?;
21    while !class.is_null() {
22        if let Some(field) = class_field_lookup(env, class, lookup_name)? {
23            return Ok(Some(field));
24        }
25
26        class = env.get_superclass(class)?;
27    }
28
29    Ok(None)
30}

32fn class_field_lookup<'a>(
33    env: &JNIEnv<'a>,
34    class: JClass<'a>,
35    lookup_name: &str,
36) -> Result<Option<JObject<'a>>> {
37    let fields = env
38        .call_method(
39            class,
40            "getDeclaredFields",
41            "()[Ljava/lang/reflect/Field;",
42            &[],
43        )?
44        .l()?;
45
46    let num_fields = env.get_array_length(*fields)?;
47
48    for field_index in 0..num_fields {
49        let field = env.get_object_array_element(*fields, field_index)?;
50
51        let field_name = java_str_to_string(
52            &env.get_string(
53                env.call_method(field, "getName", "()Ljava/lang/String;", &[])?
54                    .l()?
55                    .into(),
56            )?,
57        )?;
58
59        if field_name == lookup_name {
60            return Ok(Some(field));
61        }
62    }
63
64    Ok(None)
65}

67fn generate_field_getter(class: &ClassName, field_name: &str, type_signature: &str) -> String {
68    let rust_getter_name = format!("get_{}", java_identifier_to_rust(field_name));
69    let return_type = generate_return_jni_type(type_signature);
70
71    let mut code: Vec<String> = vec![];
72    code.push(format!(
73        "/// Returns the field `{}` of the scala class `{}`.",
74        field_name,
75        class.full_name()
76    ));
77    code.push("///".to_string());
78    code.push(format!(
79        "/// Return type and Java signature: `{return_type}` (`{type_signature}`)"
80    ));
81    code.push(format!("pub fn {rust_getter_name}("));
82    code.push("    &self,".to_string());
83    code.push("    receiver: JObject<'a>,".to_string());
84    code.push(format!(") -> JNIResult<{return_type}> {{"));
85    code.push(format!("    let class_name = \"{}\";", class.path()));
86    code.push(format!("    let field_name = \"{field_name}\";"));
87    code.push(format!("    let return_signature = \"{type_signature}\";"));
88    if is_signature_of_class_type(type_signature) {
89        let type_class = &type_signature[1..(type_signature.len() - 1)];
90        code.push(format!("    let return_class_name = \"{type_class}\";"))
91    }
92    code.push("".to_string());
93    code.push(generate_variable_type_check("receiver", "class_name"));
94    code.push("    let result = self.env.get_field(".to_string());
95    code.push("        receiver,".to_string());
96    code.push("        field_name,".to_string());
97    code.push("        return_signature,".to_string());
98    code.push(format!(
99        "    ).and_then(|x| x.{}());",
100        generate_jni_type_char(type_signature)
101    ));
102    code.push("".to_string());
103    if is_signature_of_class_type(type_signature) {
104        code.push(generate_result_type_check("return_class_name"));
105    }
106
107    code.push("    result".to_string());
108    code.push("}".to_string());
109
110    code.join("\n") + "\n"
111}

113fn generate_field_setter(class: &ClassName, field_name: &str, type_signature: &str) -> String {
114    let rust_setter_name = format!("set_{}", java_identifier_to_rust(field_name));
115    let parameter_name = format!("new_{field_name}");
116    let parameter_type = generate_jni_type(type_signature);
117
118    let mut code: Vec<String> = vec![];
119    code.push(format!(
120        "/// Sets the field `{}` of the scala class `{}`.",
121        field_name,
122        class.full_name()
123    ));
124    code.push("///".to_string());
125    code.push("/// Type and Java signature of parameters:".to_string());
126    code.push("///".to_string());
127    code.push(format!(
128        "/// - `{parameter_name}`: `{parameter_type}` (`{type_signature}`)"
129    ));
130    code.push("/// ".to_string());
131    code.push("/// Return type and Java signature: `()` (`V`)".to_string());
132    code.push(format!("pub fn {rust_setter_name}("));
133    code.push("    &self,".to_string());
134    code.push("    receiver: JObject<'a>,".to_string());
135    code.push(format!("    {parameter_name}: {parameter_type},"));
136    code.push(") -> JNIResult<()> {".to_string());
137    code.push(format!("    let class_name = \"{}\";", class.path()));
138    code.push(format!("    let field_name = \"{field_name}\";"));
139    code.push(format!(
140        "    let {parameter_name}_signature = \"{type_signature}\";"
141    ));
142    if is_signature_of_class_type(type_signature) {
143        let parameter_class = &type_signature[1..(type_signature.len() - 1)];
144        code.push(format!(
145            "    let {parameter_name}_class_name = \"{parameter_class}\";"
146        ));
147    }
148    code.push("".to_string());
149    code.push(generate_variable_type_check("receiver", "class_name"));
150    if is_signature_of_class_type(type_signature) {
151        code.push(generate_variable_type_check(
152            &parameter_name,
153            &format!("{parameter_name}_class_name"),
154        ));
155    }
156    code.push("    self.env.set_field(".to_string());
157    code.push("        receiver,".to_string());
158    code.push("        field_name,".to_string());
159    code.push(format!("        {parameter_name}_signature,"));
160    code.push(format!("        JValue::from({parameter_name})"));
161    code.push("    )".to_string());
162    code.push("}".to_string());
163
164    code.join("\n") + "\n"
165}

167/// Generates Rust code to both retrieve and modify a value of
168/// the specified field of a given object of the specified class
169///
170/// It works also for private fields and for inherited fields
171///
172/// It determines the type of the field by iterating over the
173/// inheritance hierarchy and checking for a field with the matching name
174///
175/// It also generates runtime checks verifying
176/// that the given object is of the specified class (or its descendant)
177pub fn generate_field_getter_setter_for_class(
178    env: &JNIEnv,
179    class: &ClassName,
180    field_name: &str,
181) -> Result<String> {
182    if trait_field_getter_setter::is_interface(env, class)? {
183        return Err(ErrorKind::TraitField(class.full_name(), field_name.into(), true).into());
184    }
185    let field_signature = match hierarchy_field_lookup(env, class, field_name)? {
186        Some(field) => {
187            let field_type = env
188                .call_method(field, "getType", "()Ljava/lang/Class;", &[])?
189                .l()?;
190
191            java_str_to_string(
192                &env.get_string(
193                    env.call_static_method(
194                        "org/objectweb/asm/Type",
195                        "getDescriptor",
196                        "(Ljava/lang/Class;)Ljava/lang/String;",
197                        &[JValue::Object(field_type)],
198                    )?
199                    .l()?
200                    .into(),
201                )?,
202            )?
203        }
204        _ => return Err(ErrorKind::NoField(class.full_name(), field_name.into()).into()),
205    };
206
207    let setter_code = generate_field_getter(class, field_name, &field_signature);
208    let getter_code = generate_field_setter(class, field_name, &field_signature);
209
210    Ok(format!("{setter_code}\n{getter_code}"))
211}
