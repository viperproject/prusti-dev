7use crate::{class_name::*, module_tree::*};
8use std::collections::BTreeMap;
9
10pub fn generate_module(class_names: Vec<&ClassName>) -> String {
11    let mut modules = ModuleTree::default();
12
13    for class_name in class_names {
14        let components: Vec<String> = class_name
15            .full_rust_components()
16            .iter()
17            .map(|x| x.to_string())
18            .collect();
19        modules = modules.insert(components);
20    }
21
22    let modules_tree = modules
23        .visit(|modules| {
24            let modules: BTreeMap<String, Option<String>> = modules;
25            let mut res: Vec<String> = vec![];
26            for (name, opt_rec_result) in modules {
27                match opt_rec_result {
28                    None => {
29                        res.push(format!("mod {name}_wrapper;\n"));
30                        res.push(format!("pub use self::{name}_wrapper::*;\n"));
31                    }
32                    Some(rec_result) => {
33                        res.push(format!("pub mod {name} {{\n"));
34                        res.push("use super::builtins;\n".to_string());
35                        res.push(rec_result.to_string());
36                        res.push(format!("}} // end of mod {name}\n"));
37                    }
38                }
39            }
40            res.join("")
41        })
42        .unwrap_or_else(|| "// No modules".to_string());
43
44    [
45        "//! Automatically generated code\n".to_string(),
46        "#![allow(non_snake_case)]\n".to_string(),
47        "pub mod builtins;\n".to_string(),
48        modules_tree,
49    ]
50    .join("\n")
51        + "\n"
52}
