15use std::{fs::create_dir_all, io::prelude::*, path::Path};

17#[derive(Default)]
18pub struct WrapperGenerator {
19    jars: Vec<String>,
20    classes: Vec<ClassWrapperSpec>,
21}
22
23fn save_file_atomic(content: &str, temp_dir: &Path, file_path: &Path) -> LocalResult<()> {
24    // Here we use `NamedTempFile` and `TempPath::persist` to make the write atomic.
25    let mut mod_file = tempfile::NamedTempFile::new_in(temp_dir)?;
26    mod_file.write_all(content.as_bytes())?;
27    mod_file.into_temp_path().persist(file_path)?;
28    Ok(())
29}
30
31impl WrapperGenerator {
32    pub fn new() -> Self {
33        Default::default()
34    }
35
36    pub fn use_jar(&mut self, jar_path: &str) -> &mut Self {
37        self.jars.push(jar_path.into());
38        self
39    }
40
41    pub fn use_jars(&mut self, jar_paths: &mut Vec<String>) -> &mut Self {
42        self.jars.append(jar_paths);
43        self
44    }
45
46    pub fn wrap(&mut self, class: ClassWrapperSpec) -> &mut Self {
47        self.classes.push(class);
48        self
49    }
50
51    pub fn wrap_all(&mut self, classes: Vec<ClassWrapperSpec>) -> &mut Self {
52        self.classes.extend(classes);
53        self
54    }
55
56    #[tracing::instrument(level = "debug", skip(self))]
57    pub fn generate(&mut self, out_dir: &Path) -> LocalResult<()> {
58        let classpath_separator = if cfg!(windows) { ";" } else { ":" };
59
60        let jvm_args = InitArgsBuilder::new()
61            .version(JNIVersion::V8)
62            .option(&format!(
63                "-Djava.class.path={}",
64                self.jars.join(classpath_separator)
65            ))
66            .build()?;
67
68        let jvm = JavaVM::new(jvm_args)?;
69        let env = jvm.attach_current_thread()?;
70
71        //remove_dir_all(out_dir)?;
72        create_dir_all(out_dir)?;
73        create_dir_all(out_dir.join("builtins"))?;
74
75        let utils_content = include_str!("../builtins/mod.rs");
76        save_file_atomic(utils_content, out_dir, &out_dir.join("builtins/mod.rs"))?;
77
78        for class in &self.classes {
79            let class_name = class.get_name();
80            debug!("Generate wrapper for '{}'", class_name.full_name());
81            let class_rel_mod_path = {
82                let mut class_components = class_name.full_components();
83                let last_index = class_components.len() - 1;
84                class_components.remove(last_index);
85                class_components.join("/")
86            };
87            let class_path = Path::new(out_dir)
88                .join(class_rel_mod_path)
89                .join(format!("{}_wrapper.rs", class_name.rust_name()));
90            debug!(
91                "Write wrapper for '{}' to '{}'",
92                class_name.full_name(),
93                class_path.display()
94            );
95            if let Some(parent_path) = class_path.parent() {
96                create_dir_all(parent_path)?;
97            }
98
99            let class_generator =
100                ClassGenerator::new(&env, class_name.to_owned(), class.get_items().to_vec());
101            let class_code = class_generator.generate()?;
102            save_file_atomic(&class_code, out_dir, &class_path)?;
103        }
104
105        {
106            let class_names: Vec<&ClassName> = self.classes.iter().map(|x| x.get_name()).collect();
107            let mod_code = generate_module(class_names);
108            save_file_atomic(&mod_code, out_dir, &out_dir.join("mod.rs"))?;
109        }
110
111        Ok(())
112    }
113}
