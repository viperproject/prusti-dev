7/// A queued trigger continuation. Stored as `'static` for the same reason as
8/// the encoder caches (see `encoder_cache!`): the captured data lives in the
9/// VIR arena, which outlives the drain.
10pub(crate) type Continuation = Box<dyn FnOnce()>;
11
12/// Continuations waiting for a task of encoder `E` to be requested.
13type Watchers<'vir, E> = FxIndexMap<<E as TaskEncoder>::TaskKey<'vir>, Vec<Continuation>>;
14pub type WatchersRef<'vir, E> = RefCell<Watchers<'vir, E>>;
15
16type WatchersStatic<E> = FxIndexMap<<E as TaskEncoder>::TaskKey<'static>, Vec<Continuation>>;
17pub type WatchersStaticRef<E> = RefCell<WatchersStatic<E>>;
18
19thread_local! {
20    static PENDING: RefCell<VecDeque<Continuation>> = RefCell::new(VecDeque::new());
21}
22
23/// Queues a continuation to run at the next [`drain_triggers`].
24pub(crate) fn queue(f: Continuation) {
25    PENDING.with(|pending| pending.borrow_mut().push_back(f));
26}
27
28pub(crate) fn mk_continuation<'vir>(f: impl FnOnce() + 'vir) -> Continuation {
29    // SAFETY: `'vir` outlives the drain, as for the encoder caches (see the [`Continuation`] docs and `encoder_cache!`).
30    unsafe { std::mem::transmute::<Box<dyn FnOnce() + 'vir>, Continuation>(Box::new(f)) }
31}

33/// Called when `key` first enters `E`'s cache: moves its waiting
34/// continuations to the pending queue.
35pub(crate) fn fire_watchers<'vir, E: TaskEncoder + 'vir + ?Sized>(key: &E::TaskKey<'vir>) {
36    let fired = E::with_watchers(|watchers| watchers.borrow_mut().swap_remove(key));
37    if let Some(fired) = fired {
38        PENDING.with(|pending| pending.borrow_mut().extend(fired));
39    }
40}
41
42/// Runs queued trigger continuations until quiescence. A continuation may
43/// request new tasks, which can fire further triggers (or register new
44/// ones), so this is a worklist fixpoint. Call once after the main encoding
45/// phase, before outputs are emitted.
46pub fn drain_triggers() {
47    while let Some(f) = PENDING.with(|pending| pending.borrow_mut().pop_front()) {
48        f();
49    }
50    debug_assert!(PENDING.with(|pending| pending.borrow().is_empty()));
51}
