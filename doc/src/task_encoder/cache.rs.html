3pub enum TaskEncoderCacheState<'vir, E: TaskEncoder + 'vir + ?Sized> {
4    // None, // indicated by absence in the cache
5    /// Task's encoding is running but has not yet emitted an output
6    /// reference. Requesting the task re-runs the encoding (see `Started`).
7    Encoding,
8
9    /// Task is being re-encoded while its first encoding (which has not yet
10    /// emitted an output reference, see `Encoding`) is suspended. Requesting
11    /// the task in this state is a cyclic dependency error: a third run
12    /// would only repeat the re-run's requests.
13    ReEncoding,
14
15    /// Task is currently being encoded. The output reference is available.
16    /// Full encoding is not available yet: querying for it re-runs the
17    /// encoding, which succeeds when the re-run makes progress (e.g. because
18    /// its dependencies now find this task's output reference or the outputs
19    /// cached by the first run), in which case the earlier run adopts its
20    /// result (`EncodeFullError::AlreadyEncoded`).
21    Started {
22        output_ref: <E as TaskEncoder>::OutputRef<'vir>,
23    },
24
25    /// Task is being re-encoded while its first encoding is suspended (see
26    /// `Started`). Requesting the task in this state is a cyclic dependency
27    /// error: a third run would only repeat the re-run's requests.
28    Restarted {
29        output_ref: <E as TaskEncoder>::OutputRef<'vir>,
30    },
31
32    /// Task was successfully encoded.
33    /// TODO: can still collect errors?
34    Encoded {
35        output_ref: <E as TaskEncoder>::OutputRef<'vir>,
36        deps: TaskEncoderDependencies<'vir, E>,
37        output_local: <E as TaskEncoder>::OutputFullLocal<'vir>,
38        output_dep: <E as TaskEncoder>::OutputFullDependency<'vir>,
39    },
40
41    /// An error occurred when enqueing the task.
42    ErrorEnqueue {
43        error: TaskEncoderError<E>,
44        spans: Vec<Span>,
45    },
46
47    /// An error occurred when encoding the task. The full "local" encoding is
48    /// not available. However, tasks which depend on this task may still
49    /// succeed, so the encoding for dependents may be present.
50    ///
51    /// As an example, encoding a method may fail, but it may still be possible
52    /// to encode its signature, to be included in dependents' programs.
53    ErrorEncode {
54        output_ref: <E as TaskEncoder>::OutputRef<'vir>,
55        deps: TaskEncoderDependencies<'vir, E>,
56        error: TaskEncoderError<E>,
57        output_dep: Option<<E as TaskEncoder>::OutputFullDependency<'vir>>,
58        spans: Vec<Span>,
59    },
60}

62/// Cache for a task encoder. See `TaskEncoderCacheState` for a description of
63/// the possible values in the encoding process.
64pub type Cache<'vir, E> =
65    FxIndexMap<<E as TaskEncoder>::TaskKey<'vir>, TaskEncoderCacheState<'vir, E>>;
66pub type CacheRef<'vir, E> = RefCell<Cache<'vir, E>>;
67
68pub type CacheStatic<E> =
69    FxIndexMap<<E as TaskEncoder>::TaskKey<'static>, TaskEncoderCacheState<'static, E>>;
70pub type CacheStaticRef<E> = RefCell<CacheStatic<E>>;

72/// Create the cache storage (a static `RefCell`) and a `with_cache`
73/// implementation within a `TaskEncoder` `impl` block. This should always be
74/// placed at the beginning of the `impl` block for consistency.
75///
76/// (Implementation notes: the implementation is always the same. However, it
77/// cannot be a method provided by the trait, because such an implementation
78/// would only create a single static; each cache storage must syntactically
79/// differ. A supertrait of `TaskEncoder` which only contains the cache and
80/// has a derive macro *might* work, but the `CacheRef` etc types make this a
81/// bit difficult without introducing a cyclic dependency in the two traits.)
82#[macro_export]
83macro_rules! encoder_cache {
84    ($encoder: ty) => {
85        fn with_cache<'vir, F, R>(f: F) -> R
86            where F: FnOnce(&'vir $crate::CacheRef<'vir, $encoder>) -> R,
87        {
88            ::std::thread_local! {
89                static CACHE: $crate::CacheStaticRef<$encoder> = ::std::cell::RefCell::new(Default::default());
90            }
91            CACHE.with(|cache| {
92                // SAFETY: the 'vir and 'tcx given to this function will always be
93                //   the same (or shorter) than the lifetimes of the VIR arena and
94                //   the rustc type context, respectively
95                let cache = unsafe { ::std::mem::transmute(cache) };
96                f(cache)
97            })
98        }
99
100        fn with_watchers<'vir, F, R>(f: F) -> R
101            where F: FnOnce(&'vir $crate::WatchersRef<'vir, $encoder>) -> R,
102        {
103            ::std::thread_local! {
104                static WATCHERS: $crate::WatchersStaticRef<$encoder> = ::std::cell::RefCell::new(Default::default());
105            }
106            WATCHERS.with(|watchers| {
107                // SAFETY: as for the cache above
108                let watchers = unsafe { ::std::mem::transmute(watchers) };
109                f(watchers)
110            })
111        }
112    };
113}
