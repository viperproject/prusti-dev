24/// An unsuccessful result occurring in `do_encode_full`.
25pub enum EncodeFullError<'vir, E: TaskEncoder + 'vir + ?Sized> {
26    /// Indicates that the current task has already been encoded. This can
27    /// occur when there are cyclic dependencies between multiple encoders.
28    /// This error is specifically returned when one encoder depends on
29    /// another encoder (using e.g. `TaskEncoderDependencies::require_ref`),
30    /// that latter encoder then depending on the former again, causing the
31    /// former encoder to complete its full encoding in the inner invocation.
32    /// The outer invocation remains on the stack, but will be aborted early
33    /// as soon as the control flow returns to it.
34    AlreadyEncoded,
35
36    /// An actual error occurred during encoding.
37    EncodingError(
38        <E as TaskEncoder>::EncodingError,
39        Option<Box<E::OutputFullDependency<'vir>>>,
40    ),
41
42    DependencyError(Vec<(&'static str, String, Vec<Span>)>),
43}

45// Manual implementation, since neither `E` nor `E::OutputFullDependency` are
46// required to be `Debug`.
47impl<'vir, E: TaskEncoder + 'vir + ?Sized> std::fmt::Debug for EncodeFullError<'vir, E> {
48    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
49        match self {
50            Self::AlreadyEncoded => write!(f, "AlreadyEncoded"),
51            Self::EncodingError(err, _output_dep) => f
52                .debug_tuple("EncodingError")
53                .field(err) /*.field(output_dep)*/
54                .finish(),
55            Self::DependencyError(contents) => write!(f, "DependencyError({contents:?})"),
56        }
57    }
58}

60pub enum TaskEncoderError<E: TaskEncoder + ?Sized> {
61    EnqueueingError(<E as TaskEncoder>::EnqueueingError),
62    EncodingError(<E as TaskEncoder>::EncodingError),
63    DependencyError(Vec<(&'static str, String, Vec<Span>)>),
64    CyclicError,
65    PanicError(String),
66}

68impl<E: TaskEncoder + ?Sized> std::fmt::Debug for TaskEncoderError<E>
69where
70    <E as TaskEncoder>::EncodingError: std::fmt::Debug,
71{
72    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
73        let mut helper = f.debug_struct("TaskEncoderError");
74        match self {
75            Self::EncodingError(err) => helper.field("EncodingError", err),
76            Self::EnqueueingError(err) => helper.field("EnqueueingError", err),
77            Self::DependencyError(stack) => helper.field("DependencyError", stack),
78            Self::CyclicError => helper.field("CyclicError", &""),
79            Self::PanicError(msg) => helper.field("PanicError", msg),
80        };
81        helper.finish()
82    }
83}

88        match self {
89            Self::EncodingError(err) => Self::EncodingError(err.clone()),
90            Self::EnqueueingError(err) => Self::EnqueueingError(err.clone()),
91            Self::DependencyError(stack) => Self::DependencyError(stack.clone()),
92            Self::CyclicError => Self::CyclicError,
93            Self::PanicError(msg) => Self::PanicError(msg.clone()),
94        }
95    }
