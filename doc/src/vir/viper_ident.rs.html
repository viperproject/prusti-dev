17impl<'vir> ViperIdent<'vir> {
18    pub fn new(ident: &'vir str) -> ViperIdent<'vir> {
19        assert!(is_valid_identifier(ident));
20        ViperIdent(ident)
21    }
22
23    pub fn sanitize(vcx: &'vir VirCtxt<'_>, ident: &str) -> ViperIdent<'vir> {
24        let ident = sanitize_str(ident);
25        // Just a sanity check, if this fails there is a problem in `sanitize`
26        assert!(is_valid_identifier(ident.as_str()));
27        ViperIdent(vcx.alloc_str(&ident))
28    }
29
30    /// The identifier fragment naming the Rust item `def_id`. Callers combine
31    /// this with a role-specific prefix, e.g.
32    /// `vir_format_identifier!(vcx, "m_{}", ViperIdent::from_def_id(vcx, def_id))`.
33    pub fn from_def_id(vcx: &'vir VirCtxt<'_>, def_id: DefId) -> ViperIdent<'vir> {
34        let name = match vcx.ident_style {
35            IdentStyle::DefPath => def_path_name(vcx, def_id),
36            IdentStyle::ItemName => short_name(vcx, def_id),
37        };
38        Self::sanitize(vcx, &name)
39    }
40
41    pub fn to_str(&self) -> &'vir str {
42        self.0
43    }
44}

46/// The def path of `def_id`, unique per item. `def_path_str` alone is not:
47/// it omits the disambiguators that tell apart same-named items, e.g. types
48/// defined in two `const _: () = { .. };` blocks of one module (as
49/// `bitflags!` generates), which would then share one Viper identity. Those
50/// disambiguators are appended (closures and impls already print theirs, or
51/// are told apart by what they print).
52fn def_path_name(vcx: &VirCtxt<'_>, def_id: DefId) -> String {
53    let mut name = vcx.tcx().def_path_str(def_id);
54    for (idx, component) in vcx.tcx().def_path(def_id).data.iter().enumerate() {
55        if component.disambiguator != 0
56            && matches!(
57                component.data,
58                DefPathData::TypeNs(_) | DefPathData::ValueNs(_) | DefPathData::MacroNs(_)
59            )
60        {
61            name.push_str(&format!("#{idx}_{}", component.disambiguator));
62        }
63    }
64    name
65}

67/// Asking for the `item_name` of a closure triggers an ICE in the compiler, so
68/// a closure is named after its nearest non-closure ancestor (closures can
69/// nest, e.g. a quantifier's closure inside an assertion's closure).
70fn short_name(vcx: &VirCtxt<'_>, mut def_id: DefId) -> String {
71    let mut key = vcx.tcx().def_key(def_id);
72    let mut suffix = String::new();
73    while let DefPathData::Closure = key.disambiguated_data.data {
74        suffix = format!("_Closure_{}{suffix}", key.disambiguated_data.disambiguator);
75        def_id.index = key.parent.unwrap();
76        key = vcx.tcx().def_key(def_id);
77    }
78    format!("{}{suffix}", vcx.tcx().item_name(def_id).to_ident_string())
79}

81fn sanitize_char(c: char) -> Option<String> {
82    match c {
83        '<' => Some("$lt$".to_string()),
84        '>' => Some("$gt$".to_string()),
85        ' ' => Some("$sp$".to_string()),
86        ',' => Some("$com$".to_string()),
87        ':' => Some("$col$".to_string()),
88        '\'' => Some("$sq$".to_string()),
89        '&' => Some("$amp$".to_string()),
90        '-' => Some("$hyp$".to_string()),
91        '(' => Some("$lp$".to_string()),
92        ')' => Some("$rp$".to_string()),
93        '[' => Some("$lb$".to_string()),
94        ']' => Some("$rb$".to_string()),
95        '{' => Some("$lc$".to_string()),
96        '}' => Some("$rc$".to_string()),
97        '?' => Some("$qm$".to_string()),
98        ';' => Some("$sc$".to_string()),
99        '#' => Some("$oc$".to_string()),
100        '/' => Some("$fs$".to_string()),
101        '*' => Some("$as$".to_string()),
102        '=' => Some("$eq$".to_string()),
103        '+' => Some("$pl$".to_string()),
104        '!' => Some("$ex$".to_string()),
105        _ => None,
106    }
107}
108
109fn sanitize_str(s: &str) -> String {
110    s.chars()
111        .map(|c| sanitize_char(c).unwrap_or_else(|| c.to_string()))
112        .collect()
113}
114
115fn is_valid_identifier(s: &str) -> bool {
116    s.chars().all(|c| sanitize_char(c).is_none())
117}

123    /// Call sites compose sanitized fragments into larger identifiers with
124    /// `vir_format_identifier!`, which sanitizes again; that must be a no-op.
125    #[test]
126    fn sanitize_is_idempotent() {
127        let path = "<core::ops::Range<T> as Foo>::bar::{closure#0}";
128        let once = sanitize_str(path);
129        assert!(is_valid_identifier(&once));
130        assert_eq!(sanitize_str(&once), once);
