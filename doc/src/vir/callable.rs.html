9/// An empty type that is only used to specify the arity of callable things.
10/// Without the `Many` wrapper, exactly one argument of the type is expected,
11/// with the wrapper we expect a slice of statically-unknown length.
12#[derive(
13    Debug, Clone, Copy, PartialEq, Eq, PartialOrd, Ord, Hash, serde::Serialize, serde::Deserialize,
14)]
15pub struct Many<T: CompType>(core::marker::PhantomData<T>);
16macro_rules! typed_wrapper {
17    ($wrapper:ident; $($type:ident => $name:ident$(<$($gen:tt),+>)?),+) => {
18        $(
19            pub type $name$(<$($gen),+>)? = $wrapper<$($($gen,)*)?$crate::$type>;
20        )+
21    }
22}
23typed_wrapper!(Many; Bool => ManyBool, Int => ManyInt, Ref => ManyRef, Perm => ManyPerm);
24typed_wrapper!(Many; CSnap => ManyCSnap, PSnap => ManyPSnap, TyVal => ManyTyVal);

39impl<'vir, R: CompType> DomainIdn<'vir, R> {
40    pub fn new(idn: ViperIdent<'vir>, params: usize) -> Self {
41        let ps = (0..params)
42            .map(|_| crate::TypeData::new(crate::TypeKind::Err))
43            .collect::<Vec<_>>();
44        let ps = ps.iter().collect::<Vec<_>>();
45        R::check(&crate::TypeData::<crate::Dyn>::new(
46            crate::TypeKind::Domain(idn.to_str(), &ps),
47        ));
48        Self {
49            idn,
50            params,
51            _p: core::marker::PhantomData,
52        }
53    }
54
55    pub fn name(&self) -> ViperIdent<'vir> {
56        self.idn
57    }
58
59    pub fn cast_ty<R1: CompType>(self) -> DomainIdn<'vir, R1> {
60        DomainIdn::new(self.idn, self.params)
61    }
62}

418    + Ord

434        locals: Self::Local<'a, 'vir>,
435    ) -> impl Borrow<[LocalDecl<'vir, Self::T>]>;
436    fn params<'a, 'vir>(param: &'a Self::Ty<'vir>) -> &'a [Type<'vir, Self::T>];
437}

483/// Something that indicates the arity and type of arguments. Implemented for
484/// tuples of size 0 to 12, where each element is an `Arg`. An `Arg` is either
485/// the exact type of the argument (e.g. `TypePerm<'vir>` = `T<'vir, Perm>`) or
486/// a slice of an unknown number of arguments of the same type (e.g.
487/// `&[TypeSnap<'vir>]` or `&[TypeDyn<'vir>]`).
488///
489/// This means that a `FunctionIdn<'vir, (TypeSnap<'vir>, &[TypeTyVal<'vir>],
490/// TypePerm<'vir>), TypeInt<'vir>>` must be called with `(ExprSnap<'vir>,
491/// &[ExprTyVal<'vir>], ExprPerm<'vir>)` and produces an `ExprInt<'vir>`.
492#[sealed]
493pub trait Arity:
494    'static
495    + Clone
496    + Copy
497    + core::fmt::Debug
498    + PartialEq
499    + Eq
500    + PartialOrd
501    + Ord
502    + Hash
503    + Serialize
504    + for<'de> Deserialize<'de>
505{
506    /// The arguments that this will be called with, must be a tuple. For
507    /// example `(ExprSnap, &'a [ExprTyVal], ExprPerm)`.
508    type Exprs<'a, 'vir: 'a, Curr: 'vir, Next: 'vir>: core::marker::Tuple;
509    type Locals<'a, 'vir: 'a>: core::marker::Tuple;
510    type Tys<'vir>: Copy + Debug;
511
512    /// Check that the arguments are of the correct type.
513    fn types_match<'vir, T: CompType, E: HasType<'vir, T>>(
514        params: Self::Tys<'vir>,
515        args: &[&'vir E],
516        debug_info: DebugInfo<'vir>,
517    ) {
518        let params = Self::params(params);
519        if params.len() != args.len() {
520            crate::typecheck_error!(
521                "Expected {} arguments, got {}, Debug info: {debug_info}",
522                params.len(),
523                args.len()
524            );
525        }
526        for (i, (param, arg)) in params.iter().zip(args).enumerate() {
527            if param.kind() != arg.ty().kind() {
528                crate::typecheck_error!(
529                    "Argument {i} has type {:?}, expected {:?}, Debug info: {debug_info}",
530                    arg.ty().ty(),
531                    param,
532                );
533            }
534        }
535    }
536
537    /// Convert the arguments to usable slice.
538    fn args<'a, 'vir: 'a, Curr: 'vir, Next: 'vir>(
539        vcx: &'vir VirCtxt,
540        args: Self::Exprs<'a, 'vir, Curr, Next>,
541    ) -> &'vir [ExprGenDyn<'vir, Curr, Next>];
542
543    fn locals<'a, 'vir: 'a>(
544        vcx: &'vir VirCtxt,
545        locals: Self::Locals<'a, 'vir>,
546    ) -> &'vir [LocalDeclDyn<'vir>];
547
548    fn params<'vir>(params: Self::Tys<'vir>) -> Vec<TypeDyn<'vir>>;
549}

552//     /// The number of arguments that this will be called with.

557macro_rules! tuple_arity {
558    ($($g:ident),*) => {
559        #[sealed]
560        impl<$($g: Arg),*> Arity for ($($g),*) {
561            type Exprs<'a, 'vir: 'a, Curr: 'vir, Next: 'vir> = ($($g::Expr<'a, 'vir, Curr, Next>),*);
562            type Locals<'a, 'vir: 'a> = ($($g::Local<'a, 'vir>),*);
563            type Tys<'vir> = ($($g::Ty<'vir>),*);
564            #[allow(unused_variables)]
565            fn args<'a, 'vir: 'a, Curr: 'vir, Next: 'vir>(vcx: &'vir VirCtxt, args: Self::Exprs<'a, 'vir, Curr, Next>) -> &'vir [ExprGenDyn<'vir, Curr, Next>] {
566                let a = core::iter::empty();
567                $(
568                    let t = $g::args(args.${index()});
569                    let a = a.chain(t.borrow().as_dyn().iter().copied());
570                )*
571                vcx.alloc_slice(&a.collect::<Vec<_>>())
572            }
573            #[allow(unused_variables)]
574            fn locals<'a, 'vir: 'a>(vcx: &'vir VirCtxt, locals: Self::Locals<'a, 'vir>) -> &'vir [LocalDeclDyn<'vir>] {
575                let a = core::iter::empty();
576                $(
577                    let t = $g::locals(locals.${index()});
578                    let a = a.chain(t.borrow().as_dyn().iter().copied());
579                )*
580                vcx.alloc_slice(&a.collect::<Vec<_>>())
581            }
582            fn params<'vir>(params: Self::Tys<'vir>) -> Vec<TypeDyn<'vir>> {
583                [$($g::params(&params.${index()}).as_dyn()),*]
584                    .into_iter()
585                    .flatten()
586                    .copied()
587                    .collect()
588            }
589        }
590    };
591}
592macro_rules! tuple_arity_many {
593    ($f:ident, $g:ident) => {
594        tuple_arity!($f, $g);
595    };
596    ($f:ident $(, $g:ident)*) => {
597        tuple_arity!($f $(, $g)*);
598        tuple_arity_many!($($g),*);
599    }
600}
