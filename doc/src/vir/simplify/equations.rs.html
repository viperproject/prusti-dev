1//! The equations, one function per kind of node they rewrite. Each gets the
2//! node's simplified operands and returns an equal expression, or `None` if
3//! no equation applies. Results are put together with the constructors at
4//! the end, which apply the equations to what they build in turn, so a
5//! result is simplified as a whole. Every equation makes the expression
6//! smaller or moves an operation towards the leaves, so this terminates.
7
8use super::walk::{occurrences, Binding, Simplifier};
9use crate::{data::*, gendata::*, refs::*, CastType};
10
11impl<'vir> Simplifier<'_, 'vir, '_> {
12    /// Applies the equations of `e`'s kind to `e`, whose subexpressions are
13    /// simplified.
14    pub(super) fn fold(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
15        match e.kind {
16            ExprKindGenData::UnOp(u) if u.kind == UnOpKind::Not => self.fold_not(u.expr.as_dyn()),
17            ExprKindGenData::BinOp(b) => self.fold_bin_op(b.kind, b.lhs, b.rhs),
18            ExprKindGenData::Ternary(t) => self.fold_ternary(t.cond.as_dyn(), t.then, t.else_),
19            ExprKindGenData::AdtDestructor(recv, destr) => self.fold_field(recv, destr),
20            ExprKindGenData::FuncApp(app) => self.fold_app(app),
21            ExprKindGenData::Forall(q) => self.fold_quantifier(q.body.as_dyn()),
22            ExprKindGenData::Exists(q) => self.fold_quantifier(q.body.as_dyn()),
23            ExprKindGenData::Let(l) => self.fold_let(l),
24            _ => None,
25        }
26    }

28    /// `let x = v in b`:
29    ///  - `let x = v in x` is `v`,
30    ///  - `let x = v in b` is `b` when `b` does not mention `x`. The
31    ///    traversal has replaced the uses of `x` it could by `v`.
32    fn fold_let(&mut self, l: &LetGenData<'vir, (), !>) -> Option<ExprDyn<'vir>> {
33        match l.expr.kind {
34            ExprKindGenData::Local(x) if x.name == l.name => Some(l.val),
35            _ => (occurrences(l.name, l.expr) == 0).then_some(l.expr),
36        }
37    }

39    /// `!e`:
40    ///  - `!b` for a literal `b` is the negated literal,
41    ///  - `!!a` is `a`,
42    ///  - `!(a < b)` is `a >= b`, and likewise for the other comparisons.
43    fn fold_not(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
44        if let Some(b) = as_bool(e) {
45            return Some(self.bool(!b));
46        }
47        match e.kind {
48            ExprKindGenData::UnOp(u) if u.kind == UnOpKind::Not => Some(u.expr.as_dyn()),
49            ExprKindGenData::BinOp(b) => {
50                let negated = match b.kind {
51                    BinOpKind::CmpEq => BinOpKind::CmpNe,
52                    BinOpKind::CmpNe => BinOpKind::CmpEq,
53                    BinOpKind::CmpGt => BinOpKind::CmpLe,
54                    BinOpKind::CmpLe => BinOpKind::CmpGt,
55                    BinOpKind::CmpGe => BinOpKind::CmpLt,
56                    BinOpKind::CmpLt => BinOpKind::CmpGe,
57                    _ => return None,
58                };
59                Some(self.bin_op(negated, b.lhs, b.rhs))
60            }
61            _ => None,
62        }
63    }

65    /// `l && r`, `l || r`, `l ==> r`, `l == r` and `l != r`:
66    ///  - `true && r` is `r` and `false && r` is `false`, `false || r` is `r`
67    ///    and `true || r` is `true`, and likewise with the literal on the
68    ///    right,
69    ///  - `true ==> r` is `r`, `false ==> r` and `l ==> true` are `true`,
70    ///  - `l == true` and `l != false` are `l`, `l == false` and `l != true`
71    ///    are `!l`, and likewise with the literal on the left,
72    ///  - `C(a_0, ..) == C(b_0, ..)` is `a_0 == b_0 && ..` since adt
73    ///    constructors are injective, and `!=` is its negation; the
74    ///    applications may be locals bound to them.
75    fn fold_bin_op(
76        &mut self,
77        kind: BinOpKind,
78        l: ExprDyn<'vir>,
79        r: ExprDyn<'vir>,
80    ) -> Option<ExprDyn<'vir>> {
81        use BinOpKind::*;
82        match (kind, as_bool(l), as_bool(r)) {
83            (And, Some(true), _) | (Or, Some(false), _) | (Implies, Some(true), _) => Some(r),
84            (And, _, Some(true)) | (Or, _, Some(false)) => Some(l),
85            (And, Some(false), _) | (And, _, Some(false)) => Some(self.bool(false)),
86            (Or, Some(true), _) | (Or, _, Some(true)) => Some(self.bool(true)),
87            (Implies, Some(false), _) | (Implies, _, Some(true)) => Some(self.bool(true)),
88            (CmpEq | CmpNe, Some(a), Some(b)) => Some(self.bool((a == b) == (kind == CmpEq))),
89            (CmpEq | CmpNe, Some(lit), None) => Some(self.eq_literal(kind, lit, r)),
90            (CmpEq | CmpNe, None, Some(lit)) => Some(self.eq_literal(kind, lit, l)),
91            (CmpEq | CmpNe, None, None) => self.constructor_eq(kind, l, r),
92            _ => None,
93        }
94    }

96    /// `e == lit` (or `e != lit`, by `kind`) for a boolean literal `lit`.
97    fn eq_literal(&mut self, kind: BinOpKind, lit: bool, e: ExprDyn<'vir>) -> ExprDyn<'vir> {
98        if lit == (kind == BinOpKind::CmpEq) {
99            e
100        } else {
101            self.not(e)
102        }
103    }

105    fn constructor_eq(
106        &mut self,
107        kind: BinOpKind,
108        l: ExprDyn<'vir>,
109        r: ExprDyn<'vir>,
110    ) -> Option<ExprDyn<'vir>> {
111        let (ExprKindGenData::FuncApp(x), ExprKindGenData::FuncApp(y)) =
112            (self.resolve(l).kind, self.resolve(r).kind)
113        else {
114            return None;
115        };
116        if !self.ctx.constructors.contains_key(x.target) || !same_arg_types(x, y) {
117            return None;
118        }
119        let mut conj = self.bool(true);
120        for (a, b) in x.args.iter().zip(y.args) {
121            let eq = self.bin_op(BinOpKind::CmpEq, a, b);
122            conj = self.bin_op(BinOpKind::And, conj, eq);
123        }
124        Some(if kind == BinOpKind::CmpEq {
125            conj
126        } else {
127            self.not(conj)
128        })
129    }

131    /// `c ? a : b`:
132    ///  - `c ? a : a` is `a`,
133    ///  - for booleans, a literal branch turns it into a connective:
134    ///    `c ? true : b` is `c || b` (for a pure `b`: Viper rejects e.g.
135    ///    `c || acc(x.f)`), `c ? false : b` is `!c && b`, `c ? a : true` is
136    ///    `c ==> a` and `c ? a : false` is `c && a`,
137    ///  - `c ? f(a_0, ..) : f(b_0, ..)` is `f(c ? a_0 : b_0, ..)` when `f` is
138    ///    an adt constructor or a function without a precondition. A function
139    ///    with a precondition stays inside: Silicon cannot find a permission
140    ///    whose receiver is a ternary unless the condition is decided.
141    fn fold_ternary(
142        &mut self,
143        c: ExprDyn<'vir>,
144        a: ExprDyn<'vir>,
145        b: ExprDyn<'vir>,
146    ) -> Option<ExprDyn<'vir>> {
147        if syntactic_eq(a, b) {
148            return Some(a);
149        }
150        match (as_bool(a), as_bool(b)) {
151            (Some(true), _) if !self.is_impure(b) => {
152                return Some(self.bin_op(BinOpKind::Or, c, b));
153            }
154            (Some(false), _) => {
155                let not_c = self.not(c);
156                return Some(self.bin_op(BinOpKind::And, not_c, b));
157            }
158            (_, Some(true)) => return Some(self.bin_op(BinOpKind::Implies, c, a)),
159            (_, Some(false)) => return Some(self.bin_op(BinOpKind::And, c, a)),
160            _ => {}
161        }
162        let (ExprKindGenData::FuncApp(x), ExprKindGenData::FuncApp(y)) = (a.kind, b.kind) else {
163            return None;
164        };
165        let hoistable = same_arg_types(x, y)
166            && (self.ctx.constructors.contains_key(x.target)
167                || self.ctx.total_functions.contains(x.target));
168        if !hoistable {
169            return None;
170        }
171        let args = x
172            .args
173            .iter()
174            .zip(y.args)
175            .map(|(p, q)| self.ternary(c, p, q))
176            .collect::<Vec<_>>();
177        Some(self.app(x, &args))
178    }

180    /// `r.f` for an adt field `f`:
181    ///  - `C(a_0, .., a_n).f_i` is `a_i` for the constructor `C` with fields
182    ///    `f_0, .., f_n`, also when `r` is a local bound to such an
183    ///    application,
184    ///  - `(c ? a : b).f` is `c ? a.f : b.f` when the read folds in a branch,
185    ///    so that it is never just duplicated,
186    ///  - `(let x = v in b).f` is `let x = v in b.f` when the read folds in
187    ///    `b`.
188    fn fold_field(
189        &mut self,
190        r: ExprDyn<'vir>,
191        f: AdtDestructor<'vir, crate::Dyn, crate::Dyn>,
192    ) -> Option<ExprDyn<'vir>> {
193        let resolved = self.resolve(r);
194        match resolved.kind {
195            ExprKindGenData::FuncApp(app) => {
196                let &(cons, idx) = self.ctx.destructors.get(f.name)?;
197                let arg = *app.args.get(idx)?;
198                (app.target == cons && resolved.ty() == f.input && arg.ty() == f.ty).then_some(arg)
199            }
200            ExprKindGenData::Ternary(t) => {
201                let a = self.fold_field(t.then, f);
202                let b = self.fold_field(t.else_, f);
203                if a.is_none() && b.is_none() {
204                    return None;
205                }
206                let a = a.unwrap_or_else(|| self.node(ExprKindGenData::AdtDestructor(t.then, f)));
207                let b = b.unwrap_or_else(|| self.node(ExprKindGenData::AdtDestructor(t.else_, f)));
208                Some(self.ternary(t.cond.as_dyn(), a, b))
209            }
210            ExprKindGenData::Let(l) => {
211                let binding = Binding::new(self.vcx, l.val, l.name, l.expr);
212                let expr = self.with_binding(l.name, binding, |s| s.fold_field(l.expr, f))?;
213                let (name, val) = (l.name, l.val);
214                // The traversal inlines a local the read leaves with one use.
215                let moved = self.node(ExprKindGenData::Let(self.vcx.alloc(LetGenData {
216                    name,
217                    val,
218                    expr,
219                })));
220                Some(self.simplify(moved).unwrap_or(moved))
221            }
222            _ => None,
223        }
224    }

226    /// `f(a_0, ..)`:
227    ///  - `C(p.f_0, .., p.f_n)` is `p` for the constructor `C` of a
228    ///    single-constructor adt, whose exclusivity axiom (from silver) states
229    ///    exactly this. For the variants of a multi-constructor adt such as
230    ///    `s_Param` it would need the variant of `p`.
231    ///  - `outer(inner(k))` is `k` for an integer literal `k` in one of the
232    ///    ranges of a [`LiteralInverse`](super::LiteralInverse) the encoder
233    ///    declares, e.g. `value(cons(5))` of a primitive snapshot domain.
234    fn fold_app(&mut self, app: &'vir FuncAppGenData<'vir, (), !>) -> Option<ExprDyn<'vir>> {
235        if self.ctx.eta.contains(app.target) {
236            let fields = *self.ctx.constructors.get(app.target)?;
237            if fields.len() != app.args.len() {
238                return None;
239            }
240            let mut recv = None;
241            for (arg, field) in app.args.iter().zip(fields) {
242                let ExprKindGenData::AdtDestructor(p, destr) = arg.kind else {
243                    return None;
244                };
245                if destr.name != field.name || !recv.is_none_or(|r| syntactic_eq(r, p)) {
246                    return None;
247                }
248                recv = Some(p);
249            }
250            return recv.filter(|p| p.ty() == app.result_ty);
251        }
252        let inverse = self.ctx.literal_inverses.get(app.target)?;
253        let [arg] = app.args else {
254            return None;
255        };
256        let ExprKindGenData::FuncApp(inner) = self.resolve(arg).kind else {
257            return None;
258        };
259        let [k] = inner.args else {
260            return None;
261        };
262        let v = as_int(k)?;
263        let in_range = |&(min, max): &(ExprInt<'vir>, ExprInt<'vir>)| {
264            // A bound beyond `i128` (the maximum of `u128`) bounds no literal.
265            let below_max = match max.kind {
266                ExprKindGenData::Const(ConstData::Int(max)) => {
267                    i128::try_from(*max).ok().is_none_or(|max| v <= max)
268                }
269                _ => false,
270            };
271            as_int(min.as_dyn()).is_some_and(|min| min <= v) && below_max
272        };
273        (inner.target == inverse.inner.to_str() && inverse.ranges.iter().any(in_range))
274            .then(|| self.node_of_kind(self.orig.unwrap(), k.kind))
275    }

277    /// `forall xs :: b` and `exists xs :: b` for a literal `b` are `b`:
278    /// Viper's types are inhabited.
279    fn fold_quantifier(&mut self, body: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
280        as_bool(body).map(|b| self.bool(b))
281    }

283    /// Follows locals to their bound values, to match patterns through them.
284    fn resolve(&self, e: ExprDyn<'vir>) -> ExprDyn<'vir> {
285        let mut cur = e;
286        while let ExprKindGenData::Local(l) = cur.kind {
287            match self.env.get(l.name) {
288                Some(b) => cur = b.val,
289                None => break,
290            }
291        }
292        cur
293    }

295    // The constructors for results. A node built here takes the span and
296    // debug information of the node being rewritten.
297
298    fn node(&self, kind: ExprKindGenData<'vir, (), !>) -> ExprDyn<'vir> {
299        self.node_at(self.orig.expect("a node is being rewritten"), kind)
300    }
301
302    fn build(&mut self, kind: ExprKindGenData<'vir, (), !>) -> ExprDyn<'vir> {
303        let e = self.node(kind);
304        self.fold(e).unwrap_or(e)
305    }

307    fn bool(&self, b: bool) -> ExprDyn<'vir> {
308        self.node(ExprKindGenData::Const(self.vcx.alloc(ConstData::Bool(b))))
309    }

311    fn not(&mut self, e: ExprDyn<'vir>) -> ExprDyn<'vir> {
312        self.build(ExprKindGenData::UnOp(self.vcx.alloc(UnOpGenData {
313            kind: UnOpKind::Not,
314            expr: e.inner_cast_ty(),
315        })))
316    }

318    fn bin_op(&mut self, kind: BinOpKind, lhs: ExprDyn<'vir>, rhs: ExprDyn<'vir>) -> ExprDyn<'vir> {
319        self.build(ExprKindGenData::BinOp(self.vcx.alloc(BinOpGenData {
320            kind,
321            lhs,
322            rhs,
323        })))
324    }

326    fn ternary(&mut self, c: ExprDyn<'vir>, a: ExprDyn<'vir>, b: ExprDyn<'vir>) -> ExprDyn<'vir> {
327        self.build(ExprKindGenData::Ternary(self.vcx.alloc(TernaryGenData {
328            cond: c.inner_cast_ty(),
329            then: a,
330            else_: b,
331        })))
332    }

334    /// The application `app` to other arguments.
335    fn app(
336        &mut self,
337        app: &'vir FuncAppGenData<'vir, (), !>,
338        args: &[ExprDyn<'vir>],
339    ) -> ExprDyn<'vir> {
340        self.build(ExprKindGenData::FuncApp(self.vcx.alloc(FuncAppGenData {
341            args: self.vcx.alloc_slice(args),
342            ..*app
343        })))
344    }

347fn as_bool(e: ExprDyn<'_>) -> Option<bool> {
348    match e.kind {
349        ExprKindGenData::Const(ConstData::Bool(b)) => Some(*b),
350        _ => None,
351    }

354/// An integer literal, a negative one written as the negation of its
355/// absolute value; `None` beyond `i128`.
356pub(super) fn as_int(e: ExprDyn<'_>) -> Option<i128> {
357    match e.kind {
358        ExprKindGenData::Const(ConstData::Int(v)) => i128::try_from(*v).ok(),
359        ExprKindGenData::UnOp(u) if u.kind == UnOpKind::Neg => match u.expr.kind {
360            ExprKindGenData::Const(ConstData::Int(v)) => 0i128.checked_sub_unsigned(*v),
361            _ => None,
362        },
363        _ => None,
364    }
365}

367/// Whether two applications have the same target, type arguments and number
368/// and types of arguments.
369fn same_arg_types(x: &FuncAppGenData<'_, (), !>, y: &FuncAppGenData<'_, (), !>) -> bool {
370    x.target == y.target
371        && x.typ_var_map == y.typ_var_map
372        && x.args.len() == y.args.len()
373        && x.args.iter().zip(y.args).all(|(a, b)| a.ty() == b.ty())
374}

376/// Syntactic equality of the expressions the equations compare (locals,
377/// constants, applications and field reads), `false` for anything else.
378fn syntactic_eq<'vir>(a: ExprDyn<'vir>, b: ExprDyn<'vir>) -> bool {
379    if std::ptr::eq(a.kind, b.kind) {
380        return true;
381    }
382    match (a.kind, b.kind) {
383        (ExprKindGenData::Local(x), ExprKindGenData::Local(y)) => x.name == y.name,
384        (ExprKindGenData::Const(x), ExprKindGenData::Const(y)) => x == y,
385        (ExprKindGenData::FuncApp(x), ExprKindGenData::FuncApp(y)) => {
386            same_arg_types(x, y)
387                && x.result_ty == y.result_ty
388                && x.args.iter().zip(y.args).all(|(a, b)| syntactic_eq(a, b))
389        }
390        (ExprKindGenData::AdtDestructor(r1, d1), ExprKindGenData::AdtDestructor(r2, d2)) => {
391            d1.name == d2.name && syntactic_eq(r1, r2)
392        }
393        _ => false,
394    }
