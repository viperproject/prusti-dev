14/// A `let`-bound local in scope.
15#[derive(Clone, Copy)]
16pub(super) struct Binding<'vir> {
17    pub(super) val: ExprDyn<'vir>,
18    /// The locals `val` mentions.
19    locals: &'vir [&'vir str],
20    /// Whether `val` is a local or constant: it means the same in any state
21    /// and costs nothing to copy.
22    trivial: bool,
23    /// How often the `let` body mentions the local.
24    uses: usize,
25}

27pub(super) struct Simplifier<'a, 'vir, 'tcx> {
28    pub(super) vcx: &'vir VirCtxt<'tcx>,
29    pub(super) ctx: &'a SimplifyCtx<'vir>,
30    /// The bindings whose values mean the same at the current node as at
31    /// their `let`: a binder hides the bindings it rebinds or whose values
32    /// mention a local it binds, and `old(..)` and `unfolding` hide those
33    /// with a non-trivial value.
34    pub(super) env: FxHashMap<&'vir str, Binding<'vir>>,
35    /// The node being rewritten, whose span and debug information the nodes
36    /// built for it take.
37    pub(super) orig: Option<ExprDyn<'vir>>,
38    /// Identifies the bindings in scope: a fresh number for every scope
39    /// entered.
40    scope: usize,
41    scopes: usize,
42    /// Memoized [`Self::simplify`], by node and scope: an expression shared
43    /// in a tree is simplified once.
44    simplified: FxHashMap<(Ptr<'vir>, usize), Option<ExprDyn<'vir>>>,
45    /// Memoized [`Self::is_impure`].
46    impure: FxHashMap<Ptr<'vir>, bool>,
47}

49/// Hands the roots of the expression trees of an item (contract clauses,

54impl<'vir> Folder<'vir, (), !> for Roots<'_, '_, 'vir, '_> {

56        self.0.root(e)
57    }

59    fn fold_wand(&mut self, _: Wand<'vir>) -> Option<Wand<'vir>> {

64/// Magic wands are left untouched, wherever they occur: Viper matches
65/// packaged wand instances syntactically, and evaluates the heap-dependent
66/// subterms of a wand only when applying it, in the order of its conjuncts,
67/// so moving them around inside a wand changes which permissions are there.
68impl<'vir> Folder<'vir, (), !> for Simplifier<'_, 'vir, '_> {
69    fn fold_expr(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
70        self.simplify(e)
71    }
72
73    fn fold_wand(&mut self, _: Wand<'vir>) -> Option<Wand<'vir>> {
74        None
75    }
76}

78impl<'a, 'vir, 'tcx> Simplifier<'a, 'vir, 'tcx> {
79    pub(super) fn new(vcx: &'vir VirCtxt<'tcx>, ctx: &'a SimplifyCtx<'vir>) -> Self {
80        Self {
81            vcx,
82            ctx,
83            env: FxHashMap::default(),
84            orig: None,
85            scope: 0,
86            scopes: 0,
87            simplified: FxHashMap::default(),
88            impure: FxHashMap::default(),
89        }
90    }
91
92    /// Simplifies the root of an expression tree, keeping the root's span.
93    /// Viper blames e.g. a failing postcondition on the clause's root node,
94    /// and the error handler is registered on the span of that node only (see
95    /// `realloc_span`); replacing the root by a subexpression would make the
96    /// error impossible to backtranslate.
97    pub(super) fn root(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
98        let simplified = self.simplify(e)?;
99        // A root without a span has no handler; keep the result's own.
100        Some(match e.span {
101            None => simplified,
102            Some(_) => self.node_of_kind(e, simplified.kind),
103        })
104    }
105
106    /// The simplified `e`, `None` if nothing about it changes.
107    pub(super) fn simplify(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
108        let key = (std::ptr::from_ref(e), self.scope);
109        if let Some(&simplified) = self.simplified.get(&key) {
110            return simplified;
111        }
112        let simplified = self.simplify_node(e);
113        self.simplified.insert(key, simplified);
114        simplified
115    }
116
117    /// The simplified `e`. A local bound to a trivial value, or used only
118    /// once, is replaced by its value; the `let` is then dropped by the
119    /// equations, unless a use the traversal does not substitute (in a
120    /// trigger, a wand, or where the binding is hidden) remains.
121    fn simplify_node(&mut self, e: ExprDyn<'vir>) -> Option<ExprDyn<'vir>> {
122        let children = match e.kind {
123            ExprKindGenData::Local(l) => {
124                let b = self.env.get(l.name)?;
125                return (b.trivial || b.uses == 1).then_some(b.val);
126            }
127            ExprKindGenData::Let(l) => {
128                let new_val = self.simplify(l.val);
129                let val = new_val.unwrap_or(l.val);
130                let binding = Binding::new(self.vcx, val, l.name, l.expr);
131                let mut body = self.with_binding(l.name, binding, |s| s.simplify(l.expr));
132                // Folding may leave a single use of a local used more than
133                // once before; a second pass substitutes it.
134                if let Some(b) = body.filter(|b| binding.uses > 1 && occurrences(l.name, b) == 1) {
135                    let binding = Binding::new(self.vcx, val, l.name, b);
136                    body = self
137                        .with_binding(l.name, binding, |s| s.simplify(b))
138                        .or(body);
139                }
140                (new_val.is_some() || body.is_some()).then(|| {
141                    let expr = body.unwrap_or(l.expr);
142                    let l = LetGenData {
143                        name: l.name,
144                        val,
145                        expr,
146                    };
147                    self.node_at(e, ExprKindGenData::Let(self.vcx.alloc(l)))
148                })
149            }
150            // `old(..)` evaluates in another heap and the body of `unfolding`
151            // with other permissions, so only trivial values move into them.
152            ExprKindGenData::Old(_) => self.scoped(|_, b| b.trivial, |s| e.super_fold_with(s)),
153            ExprKindGenData::Unfolding(u) => {
154                let target = u.target.fold_with(self);
155                let body = self.scoped(|_, b| b.trivial, |s| s.simplify(u.expr));
156                (target.is_some() || body.is_some()).then(|| {
157                    let u = UnfoldingGenData {
158                        target: target.unwrap_or(u.target),
159                        expr: body.unwrap_or(u.expr),
160                    };
161                    self.node_at(e, ExprKindGenData::Unfolding(self.vcx.alloc(u)))
162                })
163            }
164            // The triggers are left as encoded.
165            ExprKindGenData::Forall(ForallGenData {
166                qvars,
167                triggers,
168                body,
169            })
170            | ExprKindGenData::Exists(ExistsGenData {
171                qvars,
172                triggers,
173                body,
174            }) => self
175                .in_quantifier(qvars, |s| s.simplify(body.as_dyn()))
176                .map(|body| {
177                    let (qvars, triggers, body) = (*qvars, *triggers, body.inner_cast_ty());
178                    let kind = match e.kind {
179                        ExprKindGenData::Forall(_) => {
180                            let q = ForallGenData {
181                                qvars,
182                                triggers,
183                                body,
184                            };
185                            ExprKindGenData::Forall(self.vcx.alloc(q))
186                        }
187                        _ => {
188                            let q = ExistsGenData {
189                                qvars,
190                                triggers,
191                                body,
192                            };
193                            ExprKindGenData::Exists(self.vcx.alloc(q))
194                        }
195                    };
196                    self.node_at(e, kind)
197                }),
198            _ => e.super_fold_with(self),
199        };
200        let node = children.unwrap_or(e);
201        let outer = self.orig.replace(node);
202        let folded = self.fold(node);
203        self.orig = outer;
204        folded.or(children)
205    }

207    /// Runs `f` in the body of a `let` binding `local`. A value that mentions
208    /// the local it is bound to means the outer local, which the body cannot
209    /// see, so such a binding is not entered.
210    pub(super) fn with_binding<R>(
211        &mut self,
212        local: &'vir str,
213        binding: Binding<'vir>,
214        f: impl FnOnce(&mut Self) -> R,
215    ) -> R {
216        self.scoped(
217            |name, b| name != local && !b.mentions(local),
218            |s| {
219                if !binding.mentions(local) {
220                    s.env.insert(local, binding);
221                }
222                f(s)
223            },
224        )
225    }
226
227    /// Runs `f` in the body of a quantifier over `qvars`.
228    fn in_quantifier<R>(
229        &mut self,
230        qvars: &[LocalDeclDyn<'vir>],
231        f: impl FnOnce(&mut Self) -> R,
232    ) -> R {
233        self.scoped(
234            |name, b| qvars.iter().all(|q| q.name != name && !b.mentions(q.name)),
235            f,
236        )
237    }
238
239    /// Runs `f` with only the bindings that `visible` keeps in scope.
240    fn scoped<R>(
241        &mut self,
242        visible: impl Fn(&str, &Binding<'vir>) -> bool,
243        f: impl FnOnce(&mut Self) -> R,
244    ) -> R {
245        let saved = self.env.clone();
246        self.env.retain(|name, b| visible(name, b));
247        self.scopes += 1;
248        let outer = std::mem::replace(&mut self.scope, self.scopes);
249        let result = f(self);
250        self.scope = outer;
251        self.env = saved;
252        result
253    }

255    /// A node of kind `kind` with the span and debug information of `e`.
256    pub(super) fn node_at(
257        &self,
258        e: ExprDyn<'vir>,
259        kind: ExprKindGenData<'vir, (), !>,
260    ) -> ExprDyn<'vir> {
261        self.node_of_kind(e, self.vcx.alloc(kind))
262    }
263
264    /// A node of the allocated kind `kind` with the span and debug
265    /// information of `e`.
266    pub(super) fn node_of_kind(
267        &self,
268        e: ExprDyn<'vir>,
269        kind: &'vir ExprKindGenData<'vir, (), !>,
270    ) -> ExprDyn<'vir> {
271        self.vcx.alloc(ExprGenData::new_inner(
272            kind,
273            e.debug_info,
274            e.span,
275            kind.ty(),
276        ))
277    }

279    /// Whether `e` contains an assertion (an access predicate, magic wand or
280    /// inhale-exhale expression).
281    pub(super) fn is_impure(&mut self, e: ExprDyn<'vir>) -> bool {
282        if let Some(&impure) = self.impure.get(&std::ptr::from_ref(e)) {
283            return impure;
284        }
285        let impure = match e.kind {
286            ExprKindGenData::AccField(_)
287            | ExprKindGenData::PredicateApp(_)
288            | ExprKindGenData::Wand(_)
289            | ExprKindGenData::InhaleExhale(_) => true,
290            _ => {
291                let mut children = Children(Vec::new());
292                e.super_visit_with(&mut children);
293                children.0.into_iter().any(|child| self.is_impure(child))
294            }
295        };
296        self.impure.insert(std::ptr::from_ref(e), impure);
297        impure
298    }
299}

301impl<'vir> Binding<'vir> {
302    /// The binding of `name` to `val` in `body`.
303    pub(super) fn new(
304        vcx: &'vir VirCtxt<'_>,
305        val: ExprDyn<'vir>,
306        name: &str,
307        body: ExprDyn<'vir>,
308    ) -> Self {
309        let mut locals = FxHashSet::default();
310        collect_locals(val, &mut locals);
311        Self {
312            val,
313            locals: vcx.alloc_slice(&locals.into_iter().collect::<Vec<_>>()),
314            trivial: matches!(
315                val.kind,
316                ExprKindGenData::Local(_) | ExprKindGenData::Const(_)
317            ),
318            uses: occurrences(name, body),
319        }
320    }
321
322    fn mentions(&self, name: &str) -> bool {
323        self.locals.contains(&name)
324    }
325}
326
327/// How often `e` mentions the local `name`, also where it is rebound.
328pub(super) fn occurrences(name: &str, e: ExprDyn<'_>) -> usize {
329    struct Count<'n>(&'n str, usize);
330    impl<'vir> Visitor<'vir, (), !> for Count<'_> {
331        fn visit_expr(&mut self, e: ExprDyn<'vir>) {
332            if matches!(e.kind, ExprKindGenData::Local(l) if l.name == self.0) {
333                self.1 += 1;
334            }
335            e.super_visit_with(self);
336        }
337    }
338    let mut count = Count(name, 0);
339    e.visit_with(&mut count);
340    count.1
341}

343/// Collects the direct subexpressions of an expression.
344struct Children<'vir>(Vec<ExprDyn<'vir>>);
345
346impl<'vir> Visitor<'vir, (), !> for Children<'vir> {
347    fn visit_expr(&mut self, e: ExprDyn<'vir>) {
348        self.0.push(e);
349    }
