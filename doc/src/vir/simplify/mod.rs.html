1//! Peephole simplification of hole-free expressions, run over the final
2//! program (domain axioms, predicates, functions and methods).
3//!
4//! Composing independently encoded snapshot operations produces wrap/unwrap
5//! round trips: reference snapshots built and immediately dereferenced,
6//! generic casts cancelled out, snapshot constructors compared field by
7//! field. The pass undoes them with [equations](equations), listed by the
8//! kind of node they rewrite, each of which can be checked on its own. The
9//! [traversal](walk) applies them bottom-up and takes care of everything
10//! that depends on the context of a node:
11//!  - `let`-bound locals are replaced by their values where that means the
12//!    same: inside an `old(..)` or `unfolding` only locals and constants
13//!    (other values may depend on the heap or the permissions held), and
14//!    not values whose locals a binder (a quantifier or `let`) rebinds,
15//!  - the span of the root of an expression tree is kept, since Prusti
16//!    backtranslates Viper errors through it,
17//!  - quantifier triggers and magic wands are left as encoded.
18//!
19//! Equations may drop subexpressions together with their well-definedness
20//! checks: Prusti checks all side conditions explicitly in impure code.
21

31/// A declaration that `outer(inner(k))` is `k` for every integer literal `k`
32/// in one of the inclusive `ranges` (pairs of literal bounds), such as for the
33/// value and constructor of a primitive snapshot domain, whose axiom only
34/// holds for the values of the type.
35#[derive(Clone, Copy, Debug)]
36pub struct LiteralInverse<'vir> {
37    pub outer: ViperIdent<'vir>,
38    pub inner: ViperIdent<'vir>,
39    pub ranges: &'vir [(ExprInt<'vir>, ExprInt<'vir>)],
40}

42/// What the simplification knows about the program: its adt constructors and
43/// destructors, its total functions and the literal inverses its encoder
44/// declares, used to recognize the foldable applications.
45pub struct SimplifyCtx<'vir> {
46    /// Constructor name to its ordered field (destructor) declarations.
47    constructors: FxHashMap<&'vir str, &'vir [crate::LocalDeclDyn<'vir>]>,
48    /// Destructor name to its constructor's name and field index.
49    destructors: FxHashMap<&'vir str, (&'vir str, usize)>,
50    /// The constructors of single-constructor adts.
51    eta: FxHashSet<&'vir str>,
52    /// Names of the total functions: domain functions and functions without
53    /// preconditions (which therefore cannot read the heap either).
54    total_functions: FxHashSet<&'vir str>,
55    /// The declared literal inverses, by the name of `outer`.
56    literal_inverses: FxHashMap<&'vir str, LiteralInverse<'vir>>,
57}
58
59impl<'vir> SimplifyCtx<'vir> {
60    pub fn new(
61        adts: &[Adt<'vir>],
62        domains: &[Domain<'vir>],
63        functions: &[Function<'vir>],
64        literal_inverses: &[LiteralInverse<'vir>],
65    ) -> Self {
66        let total_functions = domains
67            .iter()
68            .flat_map(|d| d.functions.iter().map(|f| f.name.to_str()))
69            .chain(
70                functions
71                    .iter()
72                    .filter(|f| f.pres.is_empty())
73                    .map(|f| f.name),
74            )
75            .collect();
76        let mut constructors = FxHashMap::default();
77        let mut destructors = FxHashMap::default();
78        let mut eta = FxHashSet::default();
79        for adt in adts {
80            let eta_adt = adt.constructors.len() == 1;
81            for cons in adt.constructors {
82                constructors.insert(cons.name, cons.args);
83                for (idx, field) in cons.args.iter().enumerate() {
84                    destructors.insert(field.name, (cons.name, idx));
85                }
86                if eta_adt {
87                    eta.insert(cons.name);
88                }
89            }
90        }
91        Self {
92            constructors,
93            destructors,
94            eta,
95            total_functions,
96            literal_inverses: literal_inverses
97                .iter()
98                .map(|inverse| (inverse.outer.to_str(), *inverse))
99                .collect(),
100        }
101    }
102}

104/// Simplifies the expressions of a program item (a domain, predicate,
105/// function or method).
106pub fn simplify<'vir, 'tcx, T: Foldable<'vir, (), !> + Copy>(
107    vcx: &'vir VirCtxt<'tcx>,
108    ctx: &SimplifyCtx<'vir>,
109    item: T,
110) -> T {
111    let mut s = walk::Simplifier::new(vcx, ctx);
112    item.fold_with(&mut walk::Roots(&mut s)).unwrap_or(item)
113}
