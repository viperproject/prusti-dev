5/// A walk over the expressions in VIR data, in the style of rustc's
6/// `TypeVisitor`: [`Visitable::visit_with`] hands each expression to
7/// [`Self::visit_expr`], whose default continues into its subexpressions
8/// with [`ExprGenData::super_visit_with`]. An override calls that itself to
9/// continue.
10pub trait Visitor<'vir, Curr, Next> {
11    fn visit_expr(&mut self, e: ExprGenDyn<'vir, Curr, Next>) {
12        e.super_visit_with(self);
13    }
14}
15
16/// VIR data containing expressions, in the style of rustc's
17/// `TypeVisitable`: an expression is handed to the visitor, other data
18/// visits each expression it is made of. Derived with `VirVisitable` for the
19/// generic VIR data (visiting the fields `VirReify` descends into).
20pub trait Visitable<'vir, Curr, Next> {
21    fn visit_with<V: Visitor<'vir, Curr, Next> + ?Sized>(&self, v: &mut V);
22}
23
24impl<'vir, Curr, Next, T: CompType> ExprGenData<'vir, Curr, Next, T> {
25    /// Visits the subexpressions of this expression.
26    pub fn super_visit_with<V: Visitor<'vir, Curr, Next> + ?Sized>(&self, v: &mut V) {
27        self.kind.visit_with(v);
28    }
29}

31impl<'vir, Curr, Next, T: CompType> Visitable<'vir, Curr, Next>
32    for &'vir ExprGenData<'vir, Curr, Next, T>
33{
34    fn visit_with<V: Visitor<'vir, Curr, Next> + ?Sized>(&self, v: &mut V) {
35        v.visit_expr((*self).as_dyn());
36    }
37}

39impl<'vir, Curr, Next, X: Visitable<'vir, Curr, Next>> Visitable<'vir, Curr, Next> for [X] {
40    fn visit_with<V: Visitor<'vir, Curr, Next> + ?Sized>(&self, v: &mut V) {
41        for elem in self {
42            elem.visit_with(v);
43        }
44    }
45}

47impl<'vir, Curr, Next, X: Visitable<'vir, Curr, Next>> Visitable<'vir, Curr, Next> for Option<X> {
48    fn visit_with<V: Visitor<'vir, Curr, Next> + ?Sized>(&self, v: &mut V) {
49        if let Some(elem) = self {
50            elem.visit_with(v);
51        }
52    }

55/// Collects every local name occurring in `e` (a superset of its free
56/// locals: bound variables of quantifiers and lets are included).
57pub fn collect_locals<'vir, Curr, Next>(
58    e: ExprGenDyn<'vir, Curr, Next>,
59    out: &mut FxHashSet<&'vir str>,
60) {
61    struct Locals<'a, 'vir>(&'a mut FxHashSet<&'vir str>);
62    impl<'vir, Curr, Next> Visitor<'vir, Curr, Next> for Locals<'_, 'vir> {
63        fn visit_expr(&mut self, e: ExprGenDyn<'vir, Curr, Next>) {
64            if let ExprKindGenData::Local(local) = e.kind {
65                self.0.insert(local.name);
66            }
67            e.super_visit_with(self);
68        }
69    }
70    e.visit_with(&mut Locals(out));
71}

77    /// `forall x :: {old(t)} let y == z in (t == x ? y : x) == x`
78    fn sample<'vir>(vcx: &'vir crate::VirCtxt<'_>) -> ExprGenDyn<'vir, (), !> {
79        let decl = |name| vcx.mk_local_decl(name, crate::TYPE_INT);
80        let (x, y, z, t) = (decl("x"), decl("y"), decl("z"), decl("t"));
81        let ex = |decl| vcx.mk_local_ex::<(), !, _>(decl);
82        let trigger = vcx.mk_trigger(&[vcx.mk_old_expr(ex(t))]);
83        let cond = vcx.mk_eq_expr(ex(t), ex(x));
84        let let_ = vcx.mk_let_expr(y, ex(z), vcx.mk_ternary_expr(cond, ex(y), ex(x)));
85        let body = vcx.mk_eq_expr(let_, ex(x));
86        vcx.mk_forall_expr(vcx.alloc_slice(&[x]), vcx.alloc_slice(&[trigger]), body)
87            .as_dyn()
88    }

90    #[test]
91    fn collects_locals_in_binders_and_triggers() {
92        crate::init_vcx(crate::VirCtxt::new_without_tcx());
93        crate::with_vcx(|vcx| {
94            let mut locals = FxHashSet::default();
95            collect_locals(sample(vcx), &mut locals);
96            assert_eq!(locals, FxHashSet::from_iter(["x", "y", "z", "t"]));
97        });
98    }

100    #[test]
101    fn visits_every_subexpression() {
102        struct Count(usize);
103        impl<'vir> Visitor<'vir, (), !> for Count {
104            fn visit_expr(&mut self, e: ExprGenDyn<'vir, (), !>) {
105                self.0 += 1;
106                e.super_visit_with(self);
107            }
108        }
109        crate::init_vcx(crate::VirCtxt::new_without_tcx());
110        crate::with_vcx(|vcx| {
111            let mut count = Count(0);
112            sample(vcx).visit_with(&mut count);
113            // forall; old, t; ==, let, z, ternary, ==, t, x, y, x, x
114            assert_eq!(count.0, 13);
115        });
116    }
