26/// # Safety
27///
28/// Types must be safe to transmute between each other.
29pub unsafe trait TransmuteFrom<T: CompType>: Sized {}

31pub trait CastType<'a, 'vir: 'a, T: CompType>: private::UnsafeCastType<'a, 'vir, T> {
32    /// Only for `expr!` macro use. Use `downcast_ty`/`upcast_ty` instead.
33    fn inner_cast_ty<U: CompType>(&self) -> &Self::Output<U> {
34        self.cast::<U>()
35    }
36
37    /// Will panic if casting to an incorrect type.
38    fn downcast_ty<U: CompType>(&self) -> &Self::Output<U>
39    where
40        T: TransmuteFrom<U>,
41    {
42        self.cast::<U>()
43    }
44
45    /// Cannot panic.
46    fn upcast_ty<U: CompType + TransmuteFrom<T>>(&self) -> &Self::Output<U> {
47        unsafe { self.cast_unchecked::<U>() }
48    }
49
50    /// Cannot panic.
51    fn as_dyn(&self) -> &Self::Output<crate::Dyn> {
52        unsafe { self.cast_unchecked::<crate::Dyn>() }
53    }
54}

60macro_rules! impl_exp_type {
61    ($inner:ident[$name:ident = $($const:tt)+]$( => $($up:tt)|+)?, $doc:literal) => {
62        impl_exp_type!($inner[$name = $crate::TypeKind::$($const)*]$( => $($up)|+)?, $crate::TypeKind::$($const)*, $doc);
63    };
64    ($inner:ident$([$name:ident = $const:expr])?$( => $($up:tt)|+)?, $expected:pat$( if $cond:expr)?$( => $neg:literal)?, $doc:literal) => {
65        #[doc = $doc]
66        #[repr(transparent)]
67        #[derive(Clone, Copy, Debug, PartialEq, Eq, PartialOrd, Ord, Hash, Serialize, Deserialize)]
68        pub struct $inner;
69        #[sealed]
70        impl CompType for $inner {
71            fn check(ty: Type<impl CompType>) {
72                if !(matches!(**ty, $expected$( if $cond)?)$( == $neg)?) {
73                    $crate::typecheck_error!(
74                        "Expected type `{}` but got `{ty:?}`", stringify!($expected$( if $cond)?$( == $neg)?)
75                    )
76                }
77            }
78        }
79
80        $(pub const $name: $crate::Type<'static, $inner> = unsafe { &$crate::TypeData::new_unchecked($const) };)?
81        unsafe impl TransmuteFrom<$inner> for $inner {}
82        $(
83            $(
84                unsafe impl TransmuteFrom<$inner> for $up {}
85            )+
86        )?
87    };
88}

90// `Bool`, `Int`, `Perm` and the collection types are subtypes of `CSnap`
91// since Rust's `bool` and the `prusti_contracts` builtins (`Int`, `Real`,
92// `Set`, `Multiset`, `Seq`, `Map`) translate to these. The meaning
93// of `Snap` is the mathematical representation of any Rust type (which these
94// types are). Note that the `Prim` type is unrelated and other `Prim` types
95// are not `Snap`.
96// TODO: we could also have an `Impure` supertype of `Bool` which can contain `acc`s.
97impl_exp_type!(Bool[TYPE_BOOL = Bool] => Prim | CSnap | Snap | Dyn, "The Viper `Bool` type");
98impl_exp_type!(Int[TYPE_INT = Int] => Prim | CSnap | Snap | Dyn, "The Viper `Int` type");
99impl_exp_type!(Perm[TYPE_PERM = Perm] => Prim | CSnap | Snap | Dyn, "The Viper `Perm` type (reals)");
100impl_exp_type!(Ref[TYPE_REF = Ref] => Prim | Dyn, "The Viper `Ref` type");
101impl_exp_type!(Set => Prim | CSnap | Snap | Dyn, TypeKind::Set(_), "The Viper `Set` type");
102impl_exp_type!(Multiset => Prim | CSnap | Snap | Dyn, TypeKind::Multiset(_), "The Viper `Multiset` type");
103impl_exp_type!(Seq => Prim | CSnap | Snap | Dyn, TypeKind::Seq(_), "The Viper `Seq` type");
104impl_exp_type!(Map => Prim | CSnap | Snap | Dyn, TypeKind::Map(..), "The Viper `Map` type");
105
106impl_exp_type!(Err[TYPE_ERR = Err] => Prim | Dyn, "Type for encoding errors");
107
108impl_exp_type!(CSnap => Snap | Dyn, ref kind if matches!(kind, TypeKind::Bool | TypeKind::Int | TypeKind::Perm | TypeKind::Set(_) | TypeKind::Multiset(_) | TypeKind::Seq(_) | TypeKind::Map(..)) || matches!(kind, TypeKind::Domain(name, ..) if name.starts_with("s_") && *name != "s_Param"), "A concrete Prusti snapshot type");
109impl_exp_type!(PSnap[TYPE_PSNAP = Domain("s_Param", &[])] => Snap | Dyn, "The generic snapshot domain (`s_Param`)");
110impl_exp_type!(TyVal[TYPE_TYVAL = Domain("Type", &[])] => Dyn, "The type domain (`ExpType`) which gives values to types");
111
112impl_exp_type!(Pair => Dyn, TypeKind::Domain(name, args) if name == format!("Pair{}", args.len()), "A custom generic tuple type");
113
114impl_exp_type!(Prim => Dyn, TypeKind::Bool | TypeKind::Int | TypeKind::Perm | TypeKind::Ref, "Represents any primitive Viper type");
115impl_exp_type!(Snap => Dyn, ref kind if matches!(kind, TypeKind::Bool | TypeKind::Int | TypeKind::Perm | TypeKind::Set(_) | TypeKind::Multiset(_) | TypeKind::Seq(_) | TypeKind::Map(..)) || matches!(kind, TypeKind::Domain(name, ..) if name.starts_with("s_")), "A Prusti snapshot type, either concrete or generic");
116impl_exp_type!(Dyn, TypeKind::Unsupported(..) => false, "Represents a dynamically typed value");

136    pub trait UnsafeCastType<'a, 'vir: 'a, T: CompType>: 'a {
137        type Output<U: CompType>: 'a + ?Sized;
138        fn check<U: CompType>(&self);
139        unsafe fn cast_unchecked<U: CompType>(&self) -> &Self::Output<U>;
140
141        fn cast<U: CompType>(&self) -> &Self::Output<U> {
142            Self::check::<U>(self);
143            unsafe { self.cast_unchecked::<U>() }
144        }
145    }
146
147    impl<'a, 'vir: 'a, T: CompType, A: UnsafeCastType<'a, 'vir, T>> UnsafeCastType<'a, 'vir, T>
148        for [&'a A]
149    {
150        type Output<U: CompType> = [&'a A::Output<U>];
151        fn check<U: CompType>(&self) {
152            for x in self {
153                x.check::<U>();
154            }
155        }
156        unsafe fn cast_unchecked<U: CompType>(&self) -> &Self::Output<U> {
157            let other = self as *const Self as *const Self::Output<U>;
158            unsafe { &*other }
159        }
160    }

162    macro_rules! impl_unsafe_cast {
163        ($($name:ident$(<$($g:ident$(: $bound:ident)?),+>)?);+) => {
164            $(impl<'a, 'vir: 'a$($(, $g$(: $bound)?)*)?, T: CompType> UnsafeCastType<'a, 'vir, T> for $crate::$name<'vir$($(, $g)*)?, T> {
165                type Output<U: CompType> = crate::$name<'vir$($(, $g)*)?, U>;
166                fn check<U: CompType>(&self) {
167                    U::check(self.ty());
168                }
169                /// The most general type cast. Always use `upcast_ty` or `downcast_ty` if
170                /// possible. The only reason to use this if casting e.g. a generic
171                /// `ExprGen<T>` type to a `ExprDyn` type.
172                unsafe fn cast_unchecked<U: CompType>(&self) -> &Self::Output<U> {
173                    let other = self as *const Self as *const Self::Output<U>;
174                    unsafe { &*other }
175                }
176            })*
177        };
178    }
179    impl_unsafe_cast!(LocalData; LocalDeclData; FieldData; AdtDestructorData<I: CompType>; ExprGenData<Curr, Next>; TypeData);
