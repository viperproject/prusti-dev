22    span: Span,

40        D: serde::de::Deserializer<'de>,

51/// The span manager. Maintains a vector of all allocated spans, as well as
52/// the stack, used when allocating AST nodes.
53#[derive(Default)]
54pub struct VirSpanManager<'vir> {
55    /// Vector of all allocated spans. The `id` field of a `VirSpan` is an
56    /// index into this vector. The same `id` can thus be used as the position
57    /// ID given to Viper over JNI, and when backtranslating error positions
58    /// can be used to index into this vector again, to find any error
59    /// transformers.
60    all: Vec<&'vir crate::spans::VirSpan<'vir>>,
61
62    /// Stack of "current" spans. This is maintained such that an encoder can
63    /// walk down the MIR primitives recursively, adding their stacks onto the
64    /// stack as it works. At the same time, these spans will be linked with
65    /// their parent, i.e. the preceding span in the stack. This parent link
66    /// can be used during error backtranslation to find the closest ancestor
67    /// to an offending node with an error transformer.
68    // TODO: it might be good to insert sentinel nodes into the stack whenever
69    //   crossing an encoder context (e.g. when a different encoder is used as
70    //   part of a `deps.require_*` call) to avoid linking unrelated spans
71    //   together
72    stack: Vec<&'vir crate::spans::VirSpan<'vir>>,
73
74    handlers: FxHashMap<usize, VirSpanHandler<'vir>>,
75
76    early_errors: Vec<PrustiError>,
77    /// Vector of objects holding for each method call the spans of any
78    /// associated contracts. Intended for consuption by Prusti-Assistant.
79    call_contract_spans: Vec<SpanOfCallContracts>,
80
81    /// Map of identifiers composed of a method `DefId` and a label to the
82    /// span of the respective block.
83    /// Only populated if `GENERATE_BLOCK_MESSAGES` is set.
84    block_spans: FxHashMap<(DefId, String), Span>,
85
86    /// Map of specifically local, non-trusted, selected,
87    /// impure method identifier strings to their `DefId`s.
88    /// (See `prusti_encoder::ImpureFunctionEnc::encode`)
89    /// Used for backtranslation of viper names where the identifier is all to
90    /// go off of.
91    // TODO: If useful, extend to include other identifiers too. But take care
92    // to also change the usage as well - these currently server as a filter
93    // for EntitySuccess/FailureMessages in prusti-server's viper backend
94    viper_identifiers: FxHashMap<String, DefId>,
95}

97impl<'tcx> VirCtxt<'tcx> {
98    /// Execute the given function with the given span (temporarily) added to
99    /// the span stack.
100    pub fn with_span<T>(&'tcx self, span: Span, f: impl FnOnce(&'tcx Self) -> T) -> T {
101        let mut manager = self.spans.borrow_mut();
102        let span = self.alloc(VirSpan {
103            id: manager.all.len(),
104            span,
105            parent: manager.stack.last().copied(),
106        });
107        manager.all.push(span);
108        manager.stack.push(span);
109        let len_before = manager.stack.len();
110        drop(manager);
111        let res = f(self);
112        let mut manager = self.spans.borrow_mut();
113        debug_assert_eq!(manager.stack.len(), len_before);
114        manager.stack.pop().unwrap();
115        res
116    }
117
118    /// Execute the given function with an empty span stack, restoring the
119    /// current stack afterwards. Used when crossing an encoder-context
120    /// boundary (a task encoder starting to encode a new task): encodings are
121    /// demand-driven and cached, so without this the encoded item's spans
122    /// would be linked to whatever span happens to be ambient at the *first*
123    /// demand site, and error positions inside the shared encoding would
124    /// forever backtranslate to that first caller (e.g. every failing
125    /// `assert!` in a crate reporting its "failing precondition" note at the
126    /// first `assert!` encoded).
127    pub fn with_span_stack_isolated<T>(&'tcx self, f: impl FnOnce() -> T) -> T {
128        let saved = std::mem::take(&mut self.spans.borrow_mut().stack);
129        let res = f();
130        // Overwrite rather than assert emptiness: a panicking encoder (the
131        // panic is caught by the task encoder) leaves its pushed spans behind.
132        self.spans.borrow_mut().stack = saved;
133        res
134    }

136    /// Add an error handler to the span currently on top of the stack.
137    /// `error_kind` is the machine-readable identifier of an error, as
138    /// defined by Viper. The handler function should construct one or more
139    /// `PrustiError`s to report the error with the correct span etc.
140    pub fn handle_error(
141        &self,
142        error_kind: &'static str,
143        handler: impl Fn(Option<Span>) -> Option<Vec<PrustiError>> + 'tcx,
144    ) {
145        let top_span_id = self
146            .top_span()
147            .expect("an error handler must be registered under a span")
148            .id;
149        let mut manager = self.spans.borrow_mut();
150        let previous = manager.handlers.remove(&top_span_id);
151        manager.handlers.insert(
152            top_span_id,
153            VirSpanHandler {
154                error_kind,
155                handler: Box::new(handler),
156                next: previous.map(Box::new),
157            },
158        );
159    }

161    pub fn emit_early_error(&self, error: PrustiError) {
162        let mut manager = self.spans.borrow_mut();
163        manager.early_errors.push(error);
164    }

176    /// Attempt to backtranslate the given error at the given position.
177    pub fn backtranslate(
178        &self,
179        error_kind: &str,
180        offending_pos_id: usize,
181        reason_pos_id: Option<usize>,
182    ) -> Vec<PrustiError> {
183        let manager = self.spans.borrow();
184        let reason_span_opt = reason_pos_id
185            .and_then(|id| manager.all.get(id))
186            .map(|vir_span| vir_span.span);
187        // TODO: the `reason_pos_id` fallback works around Viper blaming the
188        //   *enclosing* node (e.g. method call statement) for a failed
189        //   well-definedness check of a native partial operation expression
190        //   (e.g. a sequence index), which leaves the expression that actually
191        //   failed in the `reason` rather than in the `offending` position.
192        //   See https://github.com/viperproject/silver/issues/928
193        for pos_id in [Some(offending_pos_id), reason_pos_id] {
194            let mut span_opt = pos_id.and_then(|id| manager.all.get(id));
195            while let Some(span) = span_opt {
196                let mut handler_opt = manager.handlers.get(&span.id);
197                while let Some(handler) = handler_opt {
198                    if handler.error_kind == error_kind {
199                        if let Some(errors) = (handler.handler)(reason_span_opt) {
200                            return errors;
201                        }
202                    }
203                    handler_opt = handler.next.as_deref();
204                }
205                span_opt = span.parent.as_ref();
206            }
207        }
208        // No handler found for the error, this must be a bug. We'll try to provide a span at least.
209        let span = manager
210            .all
211            .get(offending_pos_id)
212            .map(|vir_span| vir_span.span)
213            .unwrap_or(DUMMY_SP);
214        vec![PrustiError::internal(format!("A verification error occurred, but it could not be backtranslated: no handler found for error kind: {error_kind}."), span.into())]
215    }

287    /// Emit contract spans as diagnostic. (For Prusti-Assistant)
288    pub fn emit_contract_spans(&self, env_diagnostic: &EnvDiagnostic<'_>) {
289        let mut call_contract_spans = self.spans.borrow().call_contract_spans.clone();
290        // sort, so the order is deterministic
291        call_contract_spans.sort_by(|a, b| a.defpath.cmp(&b.defpath));
292
293        let encoding_info = EncodingInfo {
294            call_contract_spans,
295        };
296        PrustiError::message(
297            format!("encodingInfo{}", encoding_info.to_json_string()),
298            DUMMY_SP.into(),
299        )
300        .emit(env_diagnostic);
301    }
