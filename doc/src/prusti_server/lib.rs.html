43/// Verify a list of programs.
44pub fn verify_programs(
45    env_diagnostic: &EnvDiagnostic<'_>,
46    programs: Vec<vir::ProgramRef>,
47) -> VerificationResult {
48    let verification_requests = programs
49        .into_iter()
50        .map(|program| {
51            let procedures = vir::with_vcx(|vcx| vcx.get_viper_identifiers());
52            let backend = config::viper_backend().parse().unwrap();
53            VerificationRequest {
54                program,
55                procedures,
56                backend_config: ViperBackendConfig::new(backend, config::verify_mode()),
57            }
58        })
59        .collect();
60
61    let stopwatch = Stopwatch::start("prusti-server", "verifying Viper program");
62    // runtime used either for client connecting to server sequentially
63    // or to sequentially verify the requests -> single thread is sufficient
64    // why single thread if everything is asynchronous? VerificationRequestProcessing in
65    // prusti-server/src/process_verification.rs only has a single thread and the underlying
66    // viper instance already uses as many cores as possible
67    let rt = Builder::new_current_thread()
68        .thread_name("prusti-viper")
69        .enable_all()
70        .build()
71        .expect("failed to construct Tokio runtime");
72
73    let overall_result = rt.block_on(async {
74        if let Some(server_address) = config::server_address() {
75            let verification_messages =
76                verify_requests_server(verification_requests, server_address);
77            handle_stream(env_diagnostic, verification_messages).await
78        } else {
79            let cache = Lazy::new(move || {
80                Arc::new(sync::Mutex::new(PersistentCache::load_cache(
81                    config::cache_path(),
82                )))
83            });
84            let vrp = Lazy::new(VerificationRequestProcessing::new);
85            let verification_messages = verify_requests_local(verification_requests, &cache, &vrp);
86            handle_stream(env_diagnostic, verification_messages).await
87        }
88    });
89    stopwatch.finish();
90    overall_result
91}

93async fn handle_stream(
94    env_diagnostic: &EnvDiagnostic<'_>,
95    verification_messages: impl Stream<Item = ServerMessage>,
96) -> VerificationResult {
97    let mut overall_result = VerificationResult::Success;
98    // let encoding_errors_count = self.encoder.count_encoding_errors();
99
100    // we want quantifier_pos_ID + program_name + q_name as identifier because there are
101    // different q_names for the same ID and each program reports independent results
102    // key: (pos_id, program_name), key to result: q_name result: num_instantiations
103    let mut quantifier_instantiations: FxHashMap<(usize, String), FxHashMap<String, u64>> =
104        FxHashMap::default();
105
106    let mut prusti_errors: Vec<_> = vec![];
107
108    pin_mut!(verification_messages);
109
110    while let Some(server_msg) = verification_messages.next().await {
111        match server_msg {
112            ServerMessage::Termination(result) => handle_termination_message(
113                env_diagnostic,
114                result,
115                &mut prusti_errors,
116                &mut overall_result,
117            ),
118            ServerMessage::MethodTermination {
119                viper_method_name,
120                result,
121                verification_time,
122            } => handle_method_termination_message(
123                env_diagnostic,
124                viper_method_name,
125                result,
126                verification_time,
127                &mut prusti_errors,
128                &mut overall_result,
129            ),
130            ServerMessage::QuantifierInstantiation {
131                q_name,
132                insts,
133                pos_id,
134            } => handle_quantifier_instantiation_message(
135                env_diagnostic,
136                q_name,
137                insts,
138                pos_id,
139                &mut quantifier_instantiations,
140            ),
141            ServerMessage::QuantifierChosenTriggers {
142                viper_quant,
143                triggers,
144                pos_id,
145            } => handle_quantifier_chosen_triggers_message(
146                env_diagnostic,
147                viper_quant,
148                triggers,
149                pos_id,
150            ),
151            ServerMessage::BlockReached {
152                viper_method,
153                vir_label,
154                path_id,
155            } => handle_block_processing_message(
156                env_diagnostic,
157                viper_method,
158                Some(vir_label),
159                path_id,
160                None,
161            ),
162            ServerMessage::BlockFailure {
163                viper_method,
164                vir_label,
165                path_id,
166            } => handle_block_processing_message(
167                env_diagnostic,
168                viper_method,
169                Some(vir_label),
170                path_id,
171                // for now, we don't report any details about failures here
172                Some("Failure".to_owned()),
173            ),
174            ServerMessage::PathProcessed {
175                viper_method,
176                path_id,
177                result: _,
178            } => handle_block_processing_message(env_diagnostic, viper_method, None, path_id, None),
179        }
180    }
181
182    // if we are in an ide, we already emit the errors asynchronously, otherwise we wait for
183    // all of them because we want the errors to be reported sortedly
184    if !config::show_ide_info() {
185        prusti_errors.sort();
186
187        for prusti_error in prusti_errors {
188            debug!("Prusti error: {prusti_error:?}");
189            prusti_error.emit(env_diagnostic);
190        }
191    }
192
193    overall_result
194}

196fn handle_result(
197    env_diagnostic: &EnvDiagnostic<'_>,
198    result: viper::VerificationResult,
199    prusti_errors: &mut Vec<PrustiError>,
200    overall_result: &mut VerificationResult,
201) {
202    match result.kind {
203        // nothing to do
204        viper::VerificationResultKind::Success => (),
205        viper::VerificationResultKind::ConsistencyErrors(errors) => {
206            for error in errors {
207                PrustiError::internal(format!("consistency error: {error:?}"), DUMMY_SP.into())
208                    .emit(env_diagnostic);
209            }
210            *overall_result = VerificationResult::Failure;
211        }
212        viper::VerificationResultKind::Failure(errors) => {
213            errors
214                .into_iter()
215                .flat_map(|error| {
216                    let Some(offending_pos_id) = error
217                        .offending_pos_id
218                        .as_ref()
219                        .and_then(|id| id.parse::<usize>().ok())
220                    else {
221                        // An error without a position (e.g. inside a
222                        // specification of a generated method) cannot be
223                        // backtranslated; report it as-is.
224                        return vec![PrustiError::internal(
225                            format!("verification error without a position: {error:?}"),
226                            DUMMY_SP.into(),
227                        )]
228                        .into_iter();
229                    };
230                    prusti_encoder::backtranslate_error(
231                        &error.full_id,
232                        offending_pos_id,
233                        error.reason_pos_id.and_then(|id| id.parse::<usize>().ok()),
234                    )
235                    .into_iter()
236                })
237                .for_each(|prusti_error| {
238                    debug!("Prusti error: {prusti_error:?}");
239                    if prusti_error.is_disabled() {
240                        prusti_error.cancel();
241                    } else if config::show_ide_info() {
242                        prusti_error.emit(env_diagnostic);
243                    } else {
244                        prusti_errors.push(prusti_error);
245                    }
246                });
247
248            *overall_result = VerificationResult::Failure;
249        }
250        viper::VerificationResultKind::JavaException(exception) => {
251            error!("Java exception: {}", exception.get_stack_trace());
252            PrustiError::internal(format!("in: {exception}"), DUMMY_SP.into()).emit(env_diagnostic);
253            *overall_result = VerificationResult::Failure;
254        }
255        viper::VerificationResultKind::Timeout(seconds) => {
256            PrustiError::verification(
257                format!("verification timed out after {seconds} seconds"),
258                DUMMY_SP.into(),
259            )
260            .set_help("the time limit is set by the `--timeout` option in `extra_verifier_args`")
261            .emit(env_diagnostic);
262            *overall_result = VerificationResult::Failure;
263        }
264    }
265}

267fn handle_method_termination_message(
268    env_diagnostic: &EnvDiagnostic<'_>,
269    viper_method_name: String,
270    result_kind: viper::VerificationResultKind,
271    verification_time: u128,
272    prusti_errors: &mut Vec<PrustiError>,
273    overall_result: &mut VerificationResult,
274) {
275    if let Some(rust_method) = vir::with_vcx(|vcx| vcx.viper_to_rust_identifier(&viper_method_name))
276    {
277        let result = viper::VerificationResult {
278            item_name: rust_method,
279            kind: result_kind,
280            cached: false,
281            time_ms: verification_time,
282        };
283
284        handle_termination_message(env_diagnostic, result, prusti_errors, overall_result);
285    } else {
286        debug!(
287            "Could not map method identifier to def id in termination message: {viper_method_name}"
288        );
289    }
290}

292fn handle_termination_message(
293    env_diagnostic: &EnvDiagnostic<'_>,
294    result: viper::VerificationResult,
295    prusti_errors: &mut Vec<PrustiError>,
296    overall_result: &mut VerificationResult,
297) {
298    debug!(
299        "Received termination message for {} with result {result:?} during verification",
300        result.item_name
301    );
302    if config::show_ide_info() {
303        PrustiError::message(
304            format!(
305                "ideVerificationResult{}",
306                serde_json::to_string(&IdeVerificationResult {
307                    item_name: result.item_name.clone(),
308                    success: result.is_success(),
309                    cached: result.cached,
310                    time_ms: result.time_ms,
311                })
312                .unwrap()
313            ),
314            DUMMY_SP.into(),
315        )
316        .emit(env_diagnostic);
317    }
318    handle_result(env_diagnostic, result, prusti_errors, overall_result);
319}

321fn handle_quantifier_instantiation_message(
322    env_diagnostic: &EnvDiagnostic<'_>,
323    q_name: String,
324    insts: u64,
325    pos_id: usize,
326    quantifier_instantiations: &mut FxHashMap<(usize, String), FxHashMap<String, u64>>,
327) {
328    if config::report_viper_messages() {
329        debug!("Received #{insts} quantifier instantiations of {q_name} for position id {pos_id} durign verification");
330        vir::with_vcx(|vcx| {
331            match vcx.get_span_from_id(pos_id) {
332                Some(span) => {
333                    let program_name = "program".to_owned();
334                    let key = (pos_id, program_name.clone());
335                    if !quantifier_instantiations.contains_key(&key) {
336                        quantifier_instantiations.insert(key.clone(), FxHashMap::default());
337                    }
338                    let map = quantifier_instantiations.get_mut(&key).unwrap();
339                    // for some reason, the aux quantifiers by the same z3 instance (same uniqueId
340                    // in silicon) can have different amount of instantiations.
341                    // e.g. we receive a message with 100 instantiations for a certain quantifier
342                    // and afterwards a message with 20 instantiations for the same one.
343                    // All verifying the same viper program and by the same z3 instance.
344                    // Since I don see a better way to take this into account than taking the
345                    // maximum, that is exactly what we do here.
346                    let old_inst = map.get(&q_name).unwrap_or(&0);
347                    map.insert(q_name, std::cmp::max(insts, *old_inst));
348                    let mut n: u64 = 0;
349                    for (q_name, insts) in map.iter() {
350                        debug!("Key: {q_name}, Value: {insts}");
351                        n += *insts;
352                    }
353                    PrustiError::message(
354                        format!("quantifierInstantiationsMessage{}",
355                            // TODO: in assistant, this quantifier message becomes an inlay hint at the given
356                            // span. on hover of this inlay, a list of method names and their instantiation count
357                            // for this range are displayed. currently, methods aren't resolved here, meaning the
358                            // hover text will only display the isntantiations of the last message for the range.
359                            // which is probably ok since all methods are in the same program. but the hover is useless.
360                            // it should count which quantifier was instantiated how many times _by_ which method.
361                            // resolving the method name here would merely tell what method the quantifier belongs
362                            // to, not which one instantiated it. this is an artifact of verifying the entire program
363                            // at once rather than having different programs for each method (as in the original PR).
364                            // if it stays that way, the logic can be simplified.
365                            json!({"instantiations": n, "method": "program"}),
366                        ), span.into()
367                    ).emit(env_diagnostic);
368                },
369                None => error!(
370                  "#{insts} quantifier instantiations of {q_name} for unknown position id {pos_id} during verification"
371                ),
372            }
373        });
374    }
375}

377fn handle_quantifier_chosen_triggers_message(
378    env_diagnostic: &EnvDiagnostic<'_>,
379    viper_quant: String,
380    triggers: String,
381    pos_id: usize,
382) {
383    if config::report_viper_messages() && pos_id != 0 {
384        debug!("Received quantifier triggers {triggers} for quantifier {viper_quant} for position id {pos_id} during verification");
385        vir::with_vcx(|vcx| {
386            match vcx.get_span_from_id(pos_id) {
387                Some(span) => {
388                    PrustiError::message(
389                        format!("quantifierChosenTriggersMessage{}",
390                            json!({"viper_quant": viper_quant, "triggers": triggers}),
391                        ), span.into()
392                    ).emit(env_diagnostic);
393                },
394                None => error!("Invalid position id {pos_id} for viper quantifier {viper_quant} during verification"),
395            }
396        });
397    }
398}

400fn handle_block_processing_message(
401    env_diagnostic: &EnvDiagnostic<'_>,
402    viper_method: String,
403    vir_label: Option<String>,
404    path_id: i32,
405    result: Option<String>,
406) {
407    if config::report_viper_messages() && config::report_block_messages() {
408        let token = if result.is_none() && vir_label.is_none() {
409            "pathProcessedMessage"
410        } else if result.is_none() {
411            "blockReachedMessage"
412        } else {
413            "blockFailureMessage"
414        };
415        debug!("Received {token}: {{ method: {viper_method} message, vir_label: {vir_label:?}, path_id: {path_id}, result: {result:?} }}");
416
417        vir::with_vcx(|vcx| {
418            if let Some(def_id) = vcx.get_viper_identifier(&viper_method) {
419                let rust_method = vcx.get_unique_item_name(&def_id);
420                if let Some(vir_label) = vir_label {
421                    if vir_label == "start" {
422                        return;
423                    }
424                    let key = (def_id, vir_label);
425                    if let Some(span) = vcx.get_block_span(&key) {
426                        PrustiError::message(
427                            format!("{}{}",
428                                token,
429                                if token == "blockFailureMessage" {
430                                    json!({"method": rust_method, "path_id": path_id, "result": result.unwrap()})
431                                } else {
432                                    json!({"method": rust_method, "path_id": path_id})
433                                },
434                            ), span.into()
435                        ).emit(env_diagnostic);
436                    } else {
437                        debug!(
438                            "Could not map vir label {} to a position in {rust_method}",
439                            key.1
440                        )
441                    }
442                } else {
443                    // no label means this is a pathProcessedMessage. This also covers the case of label == "end"
444                    PrustiError::message(
445                        format!(
446                            "{}{}",
447                            "pathProcessedMessage",
448                            json!({"method": rust_method, "path_id": path_id})
449                        ),
450                        DUMMY_SP.into(),
451                    )
452                    .emit(env_diagnostic);
453                }
454            } else {
455                debug!("Could not map method identifier to def id: {viper_method}")
456            }
457        })
458    }
459}

461fn verify_requests_server(
462    verification_requests: Vec<VerificationRequest>,
463    server_address: String,
464) -> impl Stream<Item = ServerMessage> {
465    let server_address = if server_address == "MOCK" {
466        spawn_server_thread().to_string()
467    } else {
468        server_address
469    };
470    info!("Connecting to Prusti server at {server_address}");
471    let verification_stream = stream! {
472        for request in verification_requests {
473            yield PrustiClient::verify(server_address.clone(), request).await;
474        }
475    };
476    verification_stream.flatten()
477}

479fn verify_requests_local<'a>(
480    verification_requests: Vec<VerificationRequest>,
481    cache: &'a Lazy<
482        Arc<sync::Mutex<PersistentCache>>,
483        impl FnOnce() -> Arc<sync::Mutex<PersistentCache>>,
484    >,
485    vrp: &'a Lazy<
486        VerificationRequestProcessing,
487        impl FnMut() -> VerificationRequestProcessing + 'a,
488    >,
489) -> impl Stream<Item = ServerMessage> + 'a {
490    let verification_stream = stream! {
491        for request in verification_requests {
492            let request_hash = request.get_hash();
493            if config::enable_cache() {
494                if let Some(mut result) = Lazy::force(cache).get(request_hash) {
495                    info!(
496                        "Using cached result {:?}",
497                        result,
498                    );
499                    result.cached = true;
500                    yield futures::stream::once(async move {
501                        ServerMessage::Termination(result)
502                    }).left_stream();
503                    continue;
504                }
505            }
506            yield vrp.verify(request).map(move |msg| {
507                if let ServerMessage::Termination(result) = &msg {
508                    if config::enable_cache() && !matches!(result.kind, viper::VerificationResultKind::JavaException(_) | viper::VerificationResultKind::Timeout(_)) {
509                        info!(
510                            "Storing new cached result {:?}",
511                            result,
512                        );
513                        cache.insert(request_hash, result.clone());
514                    }
515                }
516                msg
517            }).right_stream();
518        }
519    };
520    verification_stream.flatten()
521}
