7use crate::{ServerMessage, ServerRequest, VerificationRequest};
8use futures::{lock, stream::Stream};
9use log::{debug, info};
10use std::{
11    sync::{self, mpsc},
12    thread,
13};
14
15struct ThreadJoin {
16    handle: Option<thread::JoinHandle<()>>,
17}
18
19// we join the thread after dropping the sender for the ServerRequests, so
20// that the verification thread actually terminates
21impl Drop for ThreadJoin {
22    fn drop(&mut self) {
23        self.handle.take().unwrap().join().unwrap();
24    }
25}
26
27pub struct VerificationRequestProcessing {
28    mtx_rx_servermsg: lock::Mutex<mpsc::Receiver<ServerMessage>>,
29    mtx_tx_verreq: sync::Mutex<mpsc::Sender<ServerRequest>>,
30    // mtx_tx_verreq has to be dropped before thread_join
31    #[allow(dead_code)]
32    thread_join: ThreadJoin,
33}
34
35impl Default for VerificationRequestProcessing {
36    fn default() -> Self {
37        Self::new()
38    }
39}
40
41/// A structure that lives for all the requests and has a single thread working on all the
42/// requests sequentially.
43/// On reception of a verification request, we send it through a channel to the already running
44/// thread.
45impl VerificationRequestProcessing {
46    pub fn new() -> Self {
47        let (tx_servermsg, rx_servermsg) = mpsc::channel();
48        let (tx_verreq, rx_verreq) = mpsc::channel();
49        let mtx_rx_servermsg = lock::Mutex::new(rx_servermsg);
50        let mtx_tx_verreq = sync::Mutex::new(tx_verreq);
51
52        let handle = thread::spawn(move || verification_thread(rx_verreq, tx_servermsg));
53        Self {
54            mtx_rx_servermsg,
55            mtx_tx_verreq,
56            thread_join: ThreadJoin {
57                handle: Some(handle),
58            },
59        }
60    }
61
62    pub fn verify(&self, request: VerificationRequest) -> impl Stream<Item = ServerMessage> + '_ {
63        let hash = request.get_hash();
64        info!(
65            "Verification request hash: {} - for program {}",
66            hash,
67            request.program.get_name(),
68        );
69
70        request.send(&self.mtx_tx_verreq);
71
72        futures::stream::unfold(false, move |done: bool| async move {
73            if request.backend_config.verify_mode.is_encode_only() || done {
74                return None;
75            }
76            let msg = self.mtx_rx_servermsg.lock().await.recv().unwrap();
77            let mut done = false;
78            if let ServerMessage::Termination(_) = msg {
79                done = true;
80            }
81            Some((msg, done))
82        })
83    }
84}
85
86fn verification_thread(
87    rx_verreq: mpsc::Receiver<ServerRequest>,
88    tx_servermsg: mpsc::Sender<ServerMessage>,
89) {
90    debug!("Verification thread started.");
91
92    while let Ok(request) = rx_verreq.recv() {
93        match request {
94            ServerRequest::Verification(verification_request) => {
95                verification_request.process(&tx_servermsg)
96            }
97        }
98    }
99    debug!("Verification thread finished.");
100}
