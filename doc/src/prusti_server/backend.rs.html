10pub enum Backend<'a> {
11    Viper(
12        viper::Verifier<'a>,
13        &'a VerificationContext<'a>,
14        jni::objects::GlobalRef,
15    ),
16}
17
18impl<'a> Backend<'a> {
19    pub fn verify(
20        &mut self,
21        procedures: prusti_rustc_interface::data_structures::fx::FxHashSet<String>,
22        sender: mpsc::Sender<ServerMessage>,
23    ) -> VerificationResultKind {
24        // Convert FxHashSet to std HashSet for viper compatibility
25        #[allow(clippy::disallowed_types)]
26        let procedures_hashset: std::collections::HashSet<String> =
27            procedures.iter().cloned().collect();
28
29        match self {
30            Backend::Viper(ref mut verifier, context, viper_program_ref) => {
31                let ast_utils = context.new_ast_utils();
32
33                ast_utils.with_local_frame(16, || {
34                    let viper_program = viper::Program::new(viper_program_ref.as_obj());
35
36                    // If report_viper_messages is activated, we need to check if we only run consistency checks here.
37                    // Otherwise, we will keep waiting for the messages to come in
38                    if prusti_utils::config::report_viper_messages() && verifier.is_verify() {
39                        verify_and_poll_msgs(
40                            verifier,
41                            context,
42                            viper_program,
43                            procedures_hashset,
44                            sender,
45                        )
46                    } else {
47                        verifier.verify(viper_program, None)
48                    }
49                })
50            }
51        }
52    }
53}

55#[allow(clippy::disallowed_types)]
56fn verify_and_poll_msgs(
57    verifier: &mut viper::Verifier<'_>,
58    verification_context: &viper::VerificationContext,
59    viper_program: viper::Program,
60    procedures: std::collections::HashSet<String>,
61    sender: mpsc::Sender<ServerMessage>,
62) -> VerificationResultKind {
63    let mut kind = VerificationResultKind::Success;
64
65    // get the reporter global reference outside of the thread scope because it needs to
66    // be dropped by thread attached to the jvm. This is also why we pass it as reference
67    // and not per value
68    let env = &verification_context.env();
69    let jni = JniUtils::new(env);
70    let verifier_wrapper = silver::verifier::Verifier::with(env);
71    let reporter = jni.unwrap_result(verifier_wrapper.call_reporter(*verifier.verifier_instance()));
72    let rep_glob_ref = env.new_global_ref(reporter).unwrap();
73
74    debug!("Starting viper message polling thread");
75
76    // start thread for polling messages
77    let (stop, stopped) = mpsc::channel::<()>();
78    thread::scope(|scope| {
79        let polling_thread =
80            scope.spawn(|| polling_function(&rep_glob_ref, procedures, sender, stopped));
81        kind = verifier.verify(viper_program, Some(polling_thread));
82        drop(stop);
83    });
84    debug!("Viper message polling thread terminated");
85    kind
86}

88#[allow(clippy::disallowed_types)]
89fn polling_function(
90    rep_glob_ref: &jni::objects::GlobalRef,
91    procedures: std::collections::HashSet<String>,
92    sender: mpsc::Sender<ServerMessage>,
93    stopped: mpsc::Receiver<()>,
94) -> std::collections::HashSet<u64> {
95    debug!("attach polling thread to JVM.");
96    let verification_context = VIPER
97        .get()
98        .expect("Viper was not instantiated before polling")
99        .attach_current_thread();
100    let env = verification_context.env();
101    let jni = JniUtils::new(env);
102    let reporter_instance = rep_glob_ref.as_obj();
103    let reporter_wrapper = silver::reporter::PollingReporter::with(env);
104
105    let mut error_hashes: std::collections::HashSet<u64> = std::collections::HashSet::new();
106    loop {
107        while reporter_wrapper
108            .call_hasNewMessage(reporter_instance)
109            .unwrap()
110        {
111            let msg = reporter_wrapper
112                .call_getNewMessage(reporter_instance)
113                .unwrap();
114            debug!("Polling thread received {}", jni.class_name(msg).as_str());
115            match jni.class_name(msg).as_str() {
116                "viper.silver.reporter.QuantifierInstantiationsMessage" => {
117                    let msg_wrapper = silver::reporter::QuantifierInstantiationsMessage::with(env);
118                    let q_name =
119                        jni.get_string(jni.unwrap_result(msg_wrapper.call_quantifier(msg)));
120                    let q_inst = jni.unwrap_result(msg_wrapper.call_instantiations(msg));
121                    // debug!("QuantifierInstantiationsMessage: {} {}", q_name, q_inst);
122                    // also matches the "-aux" and "_precondition" quantifiers generated
123                    // we encoded the position id in the line and column number since this is not used by
124                    // prusti either way
125                    if q_name.starts_with("prog.l") {
126                        let no_pref = q_name.strip_prefix("prog.l").unwrap();
127                        let stripped = no_pref
128                            .strip_suffix("-aux")
129                            .or(no_pref.strip_suffix("_precondition"))
130                            .unwrap_or(no_pref);
131                        let parsed = stripped.parse::<usize>();
132                        match parsed {
133                            Ok(pos_id) => {
134                                sender
135                                    .send(ServerMessage::QuantifierInstantiation {
136                                        q_name,
137                                        insts: u64::try_from(q_inst).unwrap(),
138                                        pos_id,
139                                    })
140                                    .unwrap();
141                            }
142                            _ => info!("Unexpected quantifier name {q_name}"),
143                        }
144                    }
145                }
146                "viper.silver.reporter.QuantifierChosenTriggersMessage" => {
147                    let obj_wrapper = java::lang::Object::with(env);
148                    let positioned_wrapper = silver::ast::Positioned::with(env);
149                    let msg_wrapper = silver::reporter::QuantifierChosenTriggersMessage::with(env);
150
151                    let viper_quant = jni.unwrap_result(msg_wrapper.call_quantifier(msg));
152                    let viper_quant_str =
153                        jni.get_string(jni.unwrap_result(obj_wrapper.call_toString(viper_quant)));
154                    // we encoded the position id in the line and column number since this is not used by
155                    // prusti either way
156                    let pos = jni.unwrap_result(positioned_wrapper.call_pos(viper_quant));
157                    let pos_string =
158                        jni.get_string(jni.unwrap_result(obj_wrapper.call_toString(pos)));
159                    if let Some(pos_id_index) = pos_string.rfind('.') {
160                        let pos_id = pos_string[pos_id_index + 1..].parse::<usize>().unwrap();
161
162                        let viper_triggers = jni
163                            .get_string(jni.unwrap_result(msg_wrapper.call_triggers__string(msg)));
164                        debug!(
165                            "QuantifierChosenTriggersMessage: {viper_quant_str} {viper_triggers} {pos_id}"
166                        );
167                        sender
168                            .send(ServerMessage::QuantifierChosenTriggers {
169                                viper_quant: viper_quant_str,
170                                triggers: viper_triggers,
171                                pos_id,
172                            })
173                            .unwrap();
174                    } else {
175                        debug!("quantifier chosen trigger for {viper_quant_str} had no position.");
176                    }
177                }
178                "viper.silver.reporter.VerificationTerminationMessage" => return error_hashes,
179                "viper.silver.reporter.EntitySuccessMessage" => {
180                    let msg_wrapper = silver::reporter::EntitySuccessMessage::with(env);
181                    let concerning = jni.unwrap_result(msg_wrapper.call_concerning(msg));
182                    if jni.is_instance_of(concerning, "viper/silver/ast/Method") {
183                        let method_wrapper = silver::ast::Method::with(env);
184                        let method_name =
185                            jni.get_string(jni.unwrap_result(method_wrapper.call_name(concerning)));
186                        debug!("Entity success for method: {method_name}");
187                        // this should only match local methods
188                        if procedures.contains(&method_name) {
189                            let verification_time =
190                                jni.unwrap_result(msg_wrapper.call_verificationTime(msg));
191                            // verification_time is a long -> i64. but we are using u128
192                            if verification_time >= 0 {
193                                let verification_time_u128 = verification_time as u64 as u128;
194                                sender
195                                    .send(ServerMessage::MethodTermination {
196                                        viper_method_name: method_name.to_string(),
197                                        result: VerificationResultKind::Success,
198                                        verification_time: verification_time_u128,
199                                    })
200                                    .unwrap();
201                            } else {
202                                debug!(
203                                      "EntitySuccessMessage for {method_name} had negative verification time {verification_time}"
204                                  );
205                            }
206                        }
207                    } else {
208                        debug!("Entity is not a method");
209                    }
210                }
211                "viper.silver.reporter.EntityFailureMessage" => {
212                    let msg_wrapper = silver::reporter::EntityFailureMessage::with(env);
213                    let concerning = jni.unwrap_result(msg_wrapper.call_concerning(msg));
214                    if jni.is_instance_of(concerning, "viper/silver/ast/Method") {
215                        let method_wrapper = silver::ast::Method::with(env);
216                        let method_name =
217                            jni.get_string(jni.unwrap_result(method_wrapper.call_name(concerning)));
218                        debug!("Entity failure for method: {method_name}");
219                        // this should only match local methods
220                        if procedures.contains(&method_name) {
221                            let verification_time =
222                                jni.unwrap_result(msg_wrapper.call_verificationTime(msg));
223                            // verification_time is a long -> i64. but we are using u128
224                            if verification_time >= 0 {
225                                let viper_result = jni.unwrap_result(msg_wrapper.call_result(msg));
226                                let result = viper::extract_errors(
227                                    &jni,
228                                    env,
229                                    viper_result,
230                                    Some(&mut error_hashes),
231                                );
232                                let verification_time_u128 = verification_time as u64 as u128;
233                                sender
234                                    .send(ServerMessage::MethodTermination {
235                                        viper_method_name: method_name.to_string(),
236                                        result,
237                                        verification_time: verification_time_u128,
238                                    })
239                                    .unwrap();
240                            } else {
241                                debug!(
242                                      "EntityFailureMessage for {method_name} had negative verification time {verification_time}"
243                                  );
244                            }
245                        }
246                    } else {
247                        debug!("Received entity was not a method");
248                    }
249                }
250                _ => (),
251            }
252        }
253        if !matches!(
254            stopped.recv_timeout(time::Duration::from_millis(10)),
255            Err(RecvTimeoutError::Timeout)
256        ) {
257            return error_hashes;
258        }
259    }
260}
