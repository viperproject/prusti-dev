7use std::{
8    fmt::Display,
9    marker::PhantomData,
10    time::{Duration, Instant},
11};
12
13pub trait LogLevel {
14    fn log_start(prefix: &str, name: &str);
15    fn log_finish(prefix: &str, name: &str, duration: Duration);
16}
17
18pub struct Stopwatch<Level: LogLevel> {
19    start_time: Instant,
20    prefix: String,
21    section_name: String,
22    is_finished: bool,
23    level: PhantomData<Level>,
24}
25
26impl Stopwatch<log_level::Info> {
27    /// Starts a stopwatch logging at info level, within the given domain, timing a section with the given name.
28    pub fn start<D: Display, S: ToString>(domain: D, section_name: S) -> Self {
29        Self::start_info(domain, section_name)
30    }
31}
32
33impl<Level: LogLevel> Stopwatch<Level> {
34    fn _start(prefix: String, section_name: String) -> Self {
35        Level::log_start(&prefix, &section_name);
36
37        Self {
38            start_time: Instant::now(),
39            prefix,
40            section_name,
41            is_finished: false,
42            level: PhantomData,
43        }
44    }
45
46    /// Finishes up the current section, logging the time taken, and starts timing the next one.
47    pub fn start_next<S: ToString>(&mut self, section_name: S) {
48        let now = Instant::now();
49        Level::log_finish(
50            &self.prefix,
51            &self.section_name,
52            now.duration_since(self.start_time),
53        );
54        self.section_name = section_name.to_string();
55        Level::log_start(&self.prefix, &self.section_name);
56        self.start_time = now;
57    }
58
59    /// Finishes up the current section, logging the time taken.
60    ///
61    /// - Note: Simply dropping the stopwatch has the same effect.
62    pub fn finish(mut self) -> Duration {
63        self._finish()
64    }
65
66    fn _finish(&mut self) -> Duration {
67        let duration = self.start_time.elapsed();
68        Level::log_finish(&self.prefix, &self.section_name, duration);
69        self.is_finished = true;
70        duration
71    }
72}
73
74impl<Level: LogLevel> Drop for Stopwatch<Level> {
75    fn drop(&mut self) {
76        if !self.is_finished {
77            self._finish();
78        }
79    }
80}

82pub mod log_level {
83    use log::{debug, error, info, trace, warn};
84
85    use super::*;
86
87    macro_rules! log_level {
88        ($name:ident, $level:ident, $start:ident) => {
89            pub struct $name;
90
91            impl LogLevel for $name {
92                fn log_start(prefix: &str, name: &str) {
93                    $level!("{}Starting: {}", prefix, name);
94                }
95
96                fn log_finish(prefix: &str, name: &str, duration: Duration) {
97                    $level!(
98                        "{}Completed: {} ({}.{:03} seconds)",
99                        prefix,
100                        name,
101                        duration.as_secs(),
102                        duration.subsec_millis()
103                    )
104                }
105            }
106
107            impl Stopwatch<$name> {
108                /// Starts a stopwatch logging at this level, within the given domain, timing a section with the given name.
109                pub fn $start<D: Display, S: ToString>(domain: D, section_name: S) -> Self {
110                    Self::_start(format!("[{}] ", domain), section_name.to_string())
111                }
112            }
113        };
114    }
115
116    log_level!(Error, error, start_error);
117    log_level!(Warn, warn, start_warn);
118    log_level!(Info, info, start_info);
119    log_level!(Debug, debug, start_debug);
120    log_level!(Trace, trace, start_trace);
121}
