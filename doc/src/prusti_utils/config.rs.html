241fn get_keys(settings: &Config) -> HashSet<String> {
242    settings
243        .cache
244        .clone()
245        .into_table()
246        .unwrap()
247        .into_keys()
248        .collect()
249}
250
251fn check_keys(settings: &Config, allowed_keys: &HashSet<String>, source: &str) {
252    for key in settings.cache.clone().into_table().unwrap().keys() {
253        assert!(
254            allowed_keys.contains(key),
255            "Found an unknown configuration flag `{key}` in {source}",
256        );
257    }
258}

267/// Generate a dump of the settings
268pub fn dump() -> String {
269    let settings = SETTINGS.read().unwrap();
270    let map = config::Source::collect(&*settings).unwrap();
271    let mut pairs: Vec<_> = map
272        .iter()
273        .map(|(key, value)| format!("{key}={value:#?}"))
274        .collect();
275    pairs.sort();
276    pairs.join("\n\n")
277}

279fn read_optional_setting<T>(name: &'static str) -> Option<T>
280where
281    T: Deserialize<'static>,
282{
283    SETTINGS.read().unwrap().get(name).ok()
284}
285
286fn read_setting<T>(name: &'static str) -> T
287where
288    T: Deserialize<'static>,
289{
290    SETTINGS
291        .read()
292        .unwrap()
293        .get(name)
294        .unwrap_or_else(|e| panic!("Failed to read setting {name} due to {e}"))
295}
296
297fn write_setting<T: Into<config::Value>>(key: &'static str, value: T) {
298    SETTINGS
299        .write()
300        .unwrap()
301        .set(key, value)
302        .unwrap_or_else(|e| panic!("Failed to write setting {key} due to {e}"));
303}

354/// The path to the Viper JARs. `prusti-rustc` is expected to set this
355/// configuration flag to the correct path.
356pub fn viper_home() -> String {
357    if let Some(path) = read_setting::<Option<String>>("viper_home") {
358        path
359    } else {
360        // If we are running tests, the VIPER_HOME should be in a directory
361        // relative to us.
362        let current_executable_dir = get_current_executable_dir();
363        if let Some(path) = find_viper_home(&current_executable_dir) {
364            path.to_str().unwrap().to_owned()
365        } else {
366            panic!("Failed to detect Vipe home, please set viper_home configuration flag")
367        }
368    }
369}

722/// Comma-separated list of optimizations to enable, or `"all"` to enable all.
723/// Possible values in the list are:
724///
725/// - `"inline_constant_functions"`
726/// - `"delete_unused_predicates"`
727/// - `"optimize_folding"`
728/// - `"remove_empty_if"`
729/// - `"purify_vars"`
730/// - `"fix_quantifiers"`
731/// - `"fix_unfoldings"`
732/// - `"remove_unused_vars"`
733/// - `"remove_trivial_assertions"`
734/// - `"clean_cfg"`
735pub fn optimizations() -> Optimizations {
736    let optimizations_string = read_setting::<String>("optimizations");
737
738    let mut opt = Optimizations::all_disabled();
739
740    for s in optimizations_string.split(',') {
741        let trimmed = s.trim();
742        match trimmed {
743            "all" => opt = Optimizations::all_enabled(),
744            "inline_constant_functions" => opt.inline_constant_functions = true,
745            "delete_unused_predicates" => opt.delete_unused_predicates = true,
746            "optimize_folding" => opt.optimize_folding = true,
747            "remove_empty_if" => opt.remove_empty_if = true,
748            "purify_vars" => opt.purify_vars = true,
749            "fix_quantifiers" => opt.fix_quantifiers = true,
750            "fix_unfoldings" => opt.fix_unfoldings = true,
751            "remove_unused_vars" => opt.remove_unused_vars = true,
752            "remove_trivial_assertions" => opt.remove_trivial_assertions = true,
753            "clean_cfg" => opt.clean_cfg = true,
754            _ => warn!("Ignoring Unknown optimization '{trimmed}'"),
755        }
756    }
757
758    opt
759}

1011#[derive(Debug, Hash, PartialEq, Eq, serde::Deserialize, serde::Serialize, Clone, Copy)]
1012pub enum VerifyMode {
1013    CompileOnly,
1014    EncodeOnly,
1015    ConsistencyCheck,
1016    Verify,
1017}
1018
1019impl VerifyMode {
1020    pub fn is_compile_only(&self) -> bool {
1021        matches!(*self, VerifyMode::CompileOnly)
1022    }
1023
1024    pub fn is_encode_only(&self) -> bool {
1025        matches!(*self, VerifyMode::EncodeOnly)
1026    }
1027
1028    pub fn is_consistency_check(&self) -> bool {
1029        matches!(*self, VerifyMode::ConsistencyCheck)
1030    }
1031
1032    pub fn is_verify(&self) -> bool {
1033        matches!(*self, VerifyMode::Verify)
1034    }
1035}
1036
1037impl FromStr for VerifyMode {
1038    type Err = String;
1039
1040    fn from_str(s: &str) -> Result<Self, Self::Err> {
1041        match s {
1042            "compile-only" => Ok(Self::CompileOnly),
1043            "encode-only" => Ok(Self::EncodeOnly),
1044            "consistency-check" => Ok(Self::ConsistencyCheck),
1045            "verify" => Ok(Self::Verify),
1046            _ => Err(format!("Found invalid value {s} for option VERIFY_MODE (possible options are: \"compile-only\", \"encode-only\", \"consistency-check\", or \"verify\")")),
1047        }
1048    }
1049}
