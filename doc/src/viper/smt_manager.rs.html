8#[derive(Default)]
9pub struct SmtManager {
10    connection: Option<Connection>,
11    preserve_trace_files: bool,
12    write_smt_statistics: bool,
13    quantifier_instantiations_ignore_builtin: bool,
14    quantifier_instantiations_bound_global_kind: Option<u64>,
15    quantifier_instantiations_bound_trace: Option<u64>,
16    quantifier_instantiations_bound_trace_kind: Option<u64>,
17    unique_triggers_bound: Option<u64>,
18    unique_triggers_bound_total: Option<u64>,
19}
20
21struct Connection {
22    smt_listener: JoinHandle<Vec<String>>,
23    termination_shot: oneshot::Sender<()>,
24    port: u16,
25}
26
27impl SmtManager {
28    #[allow(clippy::too_many_arguments)]
29    pub fn new(
30        log_path: PathBuf,
31        preserve_trace_files: bool,
32        write_smt_statistics: bool,
33        quantifier_instantiations_ignore_builtin: bool,
34        quantifier_instantiations_bound_global_kind: Option<u64>,
35        quantifier_instantiations_bound_trace: Option<u64>,
36        quantifier_instantiations_bound_trace_kind: Option<u64>,
37        unique_triggers_bound: Option<u64>,
38        unique_triggers_bound_total: Option<u64>,
39    ) -> Self {
40        let log_path = std::fs::canonicalize(log_path).unwrap();
41        let (termination_sender, termination_receiver) = oneshot::channel();
42        let (port_sender, port_receiver) = std::sync::mpsc::channel();
43        let smt_listener = thread::spawn(move || {
44            let runtime = Builder::new_current_thread()
45                .thread_name("prusti-smt-manager")
46                .enable_all()
47                .build()
48                .expect("failed to construct Tokio runtime");
49            runtime.block_on(async {
50                let listener = TcpListener::bind("127.0.0.1:0").await.unwrap();
51                let address = listener.local_addr().unwrap();
52                port_sender.send(address.port()).unwrap();
53
54                let mut termination_receiver = termination_receiver.fuse();
55
56                let mut counter = 0;
57                let mut trace_files = Vec::new();
58
59                loop {
60                    let accept = listener.accept().fuse();
61                    futures::pin_mut!(accept);
62                    futures::select! {
63                        value = termination_receiver => {
64                            value.unwrap();
65                            break;
66                        }
67                        socket = accept => {
68                            let (mut socket, _) = socket.unwrap();
69                            let log_file = log_path.join(format!("wrapper_{counter}.log"));
70                            let trace_file = log_path.join(format!("trace{counter}.log"));
71                            let log_file = log_file.into_os_string().into_string().unwrap();
72                            let trace_file = trace_file.into_os_string().into_string().unwrap();
73                            socket.write_all(log_file.as_bytes()).await.unwrap();
74                            socket.write_all(b"\n").await.unwrap();
75                            socket.write_all(trace_file.as_bytes()).await.unwrap();
76                            socket.write_all(b"\n").await.unwrap();
77                            trace_files.push(trace_file);
78                            socket.flush().await.unwrap();
79                            socket.shutdown().await.unwrap();
80                            counter += 1;
81                        }
82                    }
83                }
84                trace_files
85            })
86        });
87        let connection = Connection {
88            smt_listener,
89            termination_shot: termination_sender,
90            port: port_receiver.recv().unwrap(),
91        };
92        Self {
93            connection: Some(connection),
94            preserve_trace_files,
95            write_smt_statistics,
96            quantifier_instantiations_ignore_builtin,
97            quantifier_instantiations_bound_global_kind,
98            quantifier_instantiations_bound_trace,
99            quantifier_instantiations_bound_trace_kind,
100            unique_triggers_bound,
101            unique_triggers_bound_total,
102        }
103    }
104
105    pub fn port(&self) -> u16 {
106        self.connection.as_ref().unwrap().port
107    }
108
109    pub fn stop_and_check(&mut self) {
110        if let Some(connection) = self.connection.take() {
111            connection.termination_shot.send(()).unwrap();
112            let trace_files = connection.smt_listener.join().unwrap();
113            for trace_file in trace_files {
114                let trace_file = PathBuf::from(trace_file);
115                // Since Silicon kills the SMT solver, there is no guarantee
116                // that the scopes will be fully popped.
117                let expected_scopes_count = None;
118                let settings = smt_log_analyzer::Settings {
119                    write_statistics: self.write_smt_statistics,
120                    quantifier_instantiations_ignore_builtin: self
121                        .quantifier_instantiations_ignore_builtin,
122                    quantifier_instantiations_bound_global_kind: self
123                        .quantifier_instantiations_bound_global_kind,
124                    quantifier_instantiations_bound_trace: self
125                        .quantifier_instantiations_bound_trace,
126                    quantifier_instantiations_bound_trace_kind: self
127                        .quantifier_instantiations_bound_trace_kind,
128                    unique_triggers_bound: self.unique_triggers_bound,
129                    unique_triggers_bound_total: self.unique_triggers_bound_total,
130                    check_active_scopes_count: expected_scopes_count,
131                    pop_scopes_by_one: false,
132                    trace_quantifier_triggers: None,
133                };
134                smt_log_analyzer::analyze(&trace_file, settings).unwrap();
135                if !self.preserve_trace_files {
136                    std::fs::remove_file(trace_file).unwrap();
137                }
138            }
139        }
140    }
141}
