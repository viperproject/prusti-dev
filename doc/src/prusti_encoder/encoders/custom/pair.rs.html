20    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {

24    fn do_encode_full<'vir>(

27    ) -> task_encoder::EncodeFullResult<'vir, Self> {
28        deps.emit_output_ref(task_key.clone(), ())?;
29        let tuple = deps.require_dep::<PairEnc>(task_key.len())?;
30
31        vir::with_vcx(|vcx| {
32            let params = vcx.alloc_slice(task_key);
33            let ty = (tuple.self_ty)(params);
34
35            let constructor =
36                vir::FunctionIdn::new(vir::ViperIdent::new(tuple.constructor.name), params, ty);
37
38            let destructors = tuple
39                .constructor
40                .args
41                .iter()
42                .zip(params)
43                .map(|(local, field)| vcx.mk_adt_destructor(local.name, ty, field))
44                .collect::<Vec<_>>();
45            Ok((
46                (),
47                PairUse {
48                    ty,
49                    constructor,
50                    destructors,
51                },
52            ))
53        })

58struct Pair<'vir> {
59    self_ty: vir::DomainIdn<'vir, vir::Pair>,
60    constructor: vir::AdtConstructor<'vir>,
61}
62
63struct PairEnc;
64
65impl TaskEncoder for PairEnc {
66    task_encoder::encoder_cache!(PairEnc);
67    const ENCODER_NAME: &'static str = "pair encoder";
68
69    type TaskDescription<'vir> = usize;
70    type OutputFullDependency<'vir> = Pair<'vir>;
71    type OutputFullLocal<'vir> = vir::Adt<'vir>;
72
73    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
74        *task
75    }
76
77    fn do_encode_full<'vir>(
78        task_key: &Self::TaskKey<'vir>,
79        deps: &mut task_encoder::TaskEncoderDependencies<'vir, Self>,
80    ) -> task_encoder::EncodeFullResult<'vir, Self> {
81        deps.emit_output_ref(*task_key, ())?;
82        vir::with_vcx(|vcx| {
83            let idn = vir::ViperIdent::new(vir::vir_format!(vcx, "Pair{}", *task_key));
84            let self_ty = vir::DomainIdn::new(idn, *task_key);
85            let typarams = (0..*task_key)
86                .map(|i| {
87                    vcx.alloc(vir::DomainParamData {
88                        name: vir::vir_format!(vcx, "Ty{}", i),
89                        index: i,
90                    })
91                })
92                .collect::<Vec<_>>();
93            let typarams = vcx.alloc_slice(&typarams);
94            let locals = typarams
95                .iter()
96                .enumerate()
97                .map(|(i, dpd)| {
98                    let dt = vir::TypeKind::DomainTypeParam(**dpd);
99                    let ty = vcx.alloc(vir::TypeData::<vir::Dyn>::new(dt));
100                    vcx.mk_local_decl(vir::vir_format!(vcx, "_{task_key}_{i}"), ty)
101                })
102                .collect::<Vec<_>>();
103            let locals = vcx.alloc_slice(&locals);
104            let constructor =
105                vcx.mk_adt_constructor(vir::vir_format!(vcx, "T{}", *task_key), locals);
106            let constructors = vcx.alloc_array(&[constructor]);
107            let adt = vcx.mk_adt(idn, typarams, constructors);
108
109            Ok((
110                adt,
111                Pair {
112                    self_ty,
113                    constructor,
114                },
115            ))
116        })
117    }
118
119    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
120        let outputs = Self::all_outputs_local_no_errors(program);
121        for output in outputs {
122            program.add_adt(output);
123        }
124    }
125}
