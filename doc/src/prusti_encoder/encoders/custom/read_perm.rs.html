1use task_encoder::{EncodeFullResult, TaskEncoder, TaskEncoderDependencies};
2use vir::{CallableIdn, CastType, FunctionIdn};
3
4/// Encodes `read()`, an unspecified permission amount strictly between `none`
5/// and `write`, for predicates that hold part of a permission and leave the
6/// splitting of it to the verifier.
7///
8/// It is a domain function rather than a regular one: a function whose body
9/// unfolds a predicate using `read()` only sees the limited version of a
10/// regular function, which carries no postcondition, so the fraction could
11/// not be shown to be non-negative there.
12pub struct ReadPermEnc;
13
14#[derive(Debug, Clone, Copy)]
15pub struct ReadPerm<'vir> {
16    pub read: FunctionIdn<'vir, (), vir::Perm>,
17}
18
19impl TaskEncoder for ReadPermEnc {
20    task_encoder::encoder_cache!(ReadPermEnc);
21    const ENCODER_NAME: &'static str = "read permission encoder";
22
23    type TaskDescription<'vir> = ();
24    type OutputFullLocal<'vir> = vir::Domain<'vir>;
25    type OutputFullDependency<'vir> = ReadPerm<'vir>;
26
27    fn task_to_key<'vir>(_task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {}
28
29    fn do_encode_full<'vir>(
30        task_key: &Self::TaskKey<'vir>,
31        deps: &mut TaskEncoderDependencies<'vir, Self>,
32    ) -> EncodeFullResult<'vir, Self> {
33        deps.emit_output_ref(*task_key, ())?;
34        let read = FunctionIdn::new(vir::ViperIdent::new("read"), (), vir::TYPE_PERM);
35        let domain = vir::with_vcx(|vcx| {
36            let bounds = vcx.mk_domain_axiom(
37                vir::vir_format_identifier!(vcx, "{}_bounds", read.name()),
38                vcx.mk_conj(&[
39                    vcx.mk_bin_op_expr(vir::BinOpKind::CmpLt, vcx.mk_no_perm(), read())
40                        .downcast_ty(),
41                    vcx.mk_bin_op_expr(vir::BinOpKind::CmpLt, read(), vcx.mk_full_perm())
42                        .downcast_ty(),
43                ]),
44            );
45            vcx.mk_domain(
46                vir::ViperIdent::new("ReadPerm"),
47                &[],
48                vcx.alloc_slice(&[bounds]),
49                vcx.alloc_slice(&[vcx.mk_domain_function(read, false, None)]),
50                None,
51            )
52        });
53        Ok((domain, ReadPerm { read }))
54    }
55
56    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
57        for domain in Self::all_outputs_local_no_errors(program) {
58            program.add_domain(domain);
59        }
60    }
61}
