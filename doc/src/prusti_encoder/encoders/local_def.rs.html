27#[derive(Clone, Copy)]
28pub struct MirLocalDefEncOutput<'vir> {
29    /// The definitions of the locals; `None` for the locals only serving
30    /// specification-only arms, which are invisible in the encoding.
31    locals: &'vir IndexVec<mir::Local, Option<LocalDef<'vir>>>,
32    pub arg_count: usize,
33}
34
35impl<'vir> MirLocalDefEncOutput<'vir> {
36    /// Returns the definition of the given local, or `None` if the local
37    /// only serves specification-only arms.
38    pub fn get(&self, local: mir::Local) -> Option<LocalDef<'vir>> {
39        self.locals[local]
40    }
41
42    /// Returns the definitions for the function return value.
43    pub fn ret(&self) -> LocalDef<'vir> {
44        self[mir::RETURN_PLACE]
45    }
46
47    /// The snapshot type of the fn return value. Used to construct e.g. the
48    /// `FunctionIdn`.
49    pub fn snap_ty_return(&self) -> vir::TypeSnap<'vir> {
50        self.ret().local_snap.ty()
51    }
52
53    fn arg_locals(&self) -> impl Iterator<Item = mir::Local> + '_ {
54        (1..=self.arg_count).map(mir::Local::from)
55    }
56
57    /// Creates an iterator of all fn arguments. Can be used to construct e.g.
58    /// type assertions.
59    pub fn args(&self) -> impl Iterator<Item = LocalDef<'vir>> + '_ {
60        self.arg_locals().map(|local| self[local])
61    }
62
63    /// Creates an iterator of the snapshot type of all fn arguments. Used to
64    /// construct e.g. the `FunctionIdn`.
65    pub fn snap_ty_args(&self) -> impl Iterator<Item = vir::TypeSnap<'vir>> + '_ {
66        self.args().map(|arg| arg.local_snap.ty())
67    }
68
69    /// Creates an iterator of the snapshot type of all fn arguments. Used to
70    /// construct e.g. the `FunctionIdn`.
71    pub fn local_decl_args(&self) -> impl Iterator<Item = vir::LocalDeclSnap<'vir>> + '_ {
72        self.args().map(|arg| arg.local_snap)
73    }
74
75    pub fn local_decl_ret(&self) -> vir::LocalDeclSnap<'vir> {
76        self.ret().local_snap
77    }
78}

82#[derive(Clone, Copy)]
83pub struct LocalDef<'vir> {
84    pub local: vir::LocalDeclRef<'vir>,
85    pub local_snap: vir::LocalDeclSnap<'vir>,
86    pub local_ex: vir::ExprRef<'vir>,
87    pub impure_snap: vir::ExprSnap<'vir>,
88    pub impure_pred: vir::ExprBool<'vir>,
89}

91#[derive(Copy, Clone, Debug, PartialEq, Eq, Hash)]
92pub enum MirLocalDefEncTask<'vir> {
93    // TODO: is this case needed?
94    #[allow(dead_code)]
95    ExternSpec(DefId),
96    Local {
97        def_id: DefId,
98        all_locals: bool,
99    },
100    LocalSubsts {
101        def_id: DefId,
102        context_def_id: DefId,
103        substs: ty::GenericArgsRef<'vir>,
104        all_locals: bool,
105    },
106}
107
108impl<'vir> MirLocalDefEncTask<'vir> {
109    fn all_locals(self) -> bool {
110        match self {
111            MirLocalDefEncTask::ExternSpec(_) => true,
112            MirLocalDefEncTask::Local { all_locals, .. } => all_locals,
113            MirLocalDefEncTask::LocalSubsts { all_locals, .. } => all_locals,
114        }
115    }
116
117    fn substs(self, vcx: &vir::VirCtxt<'vir>) -> ty::GenericArgsRef<'vir> {
118        match self {
119            MirLocalDefEncTask::ExternSpec(def_id) | MirLocalDefEncTask::Local { def_id, .. } => {
120                ty::GenericArgs::identity_for_item(vcx.tcx(), def_id)
121            }
122            MirLocalDefEncTask::LocalSubsts { substs, .. } => substs,
123        }
124    }
125
126    /// The body whose locals are encoded, or `None` if there is none to
127    /// encode, in which case the locals are derived from the signature
128    /// instead.
129    ///
130    /// The body is never instantiated: the encoding is generic, and the
131    /// callers that do pass substs only need them for the signature.
132    fn body(self) -> Option<MirBody<'vir>> {
133        match self {
134            MirLocalDefEncTask::ExternSpec(def_id) => Some(crate::encoders::spec_body(def_id)),
135            MirLocalDefEncTask::Local { def_id, .. }
136            | MirLocalDefEncTask::LocalSubsts { def_id, .. } => {
137                crate::encoders::impure_body(def_id)
138            }
139        }
140    }
141
142    fn def_id(self) -> DefId {
143        match self {
144            MirLocalDefEncTask::ExternSpec(def_id) => def_id,
145            MirLocalDefEncTask::Local { def_id, .. } => def_id,
146            MirLocalDefEncTask::LocalSubsts { def_id, .. } => def_id,
147        }
148    }
149
150    fn context_def_id(self) -> DefId {
151        match self {
152            MirLocalDefEncTask::ExternSpec(def_id) => def_id,
153            MirLocalDefEncTask::Local { def_id, .. } => def_id,
154            MirLocalDefEncTask::LocalSubsts { context_def_id, .. } => context_def_id,
155        }
156    }
157}

159impl TaskEncoder for MirLocalDefEnc {
160    task_encoder::encoder_cache!(MirLocalDefEnc);
161    const ENCODER_NAME: &'static str = "MIR local def encoder";
162
163    type TaskDescription<'vir> = MirLocalDefEncTask<'vir>;
164
165    type OutputRef<'vir> = MirLocalDefEncOutputRef;
166    type OutputFullDependency<'vir> = MirLocalDefEncOutput<'vir>;
167
168    type EncodingError = MirLocalDefEncError;
169
170    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
171        *task
172    }
173
174    fn do_encode_full<'vir>(
175        task_key: &Self::TaskKey<'vir>,
176        deps: &mut TaskEncoderDependencies<'vir, Self>,
177    ) -> EncodeFullResult<'vir, Self> {
178        fn mk_local_def<'vir>(
179            vcx: &'vir vir::VirCtxt<'vir>,
180            local: mir::Local,
181            ty: TyUseImpure<'vir>,
182        ) -> LocalDef<'vir> {
183            let ref_local = vir::vir_format!(vcx, "_{}p", local.index());
184            let snap_local = vir::vir_format!(vcx, "_{}s", local.index());
185            let local = vcx.mk_local_decl(ref_local, vir::TYPE_REF);
186            let local_snap = vcx.mk_local_decl(snap_local, ty.snapshot());
187            let local_ex = vcx.mk_local_ex(local);
188            let impure_snap = ty.ref_to_snap(local_ex);
189            let impure_pred = ty.ref_to_pred(vcx, local_ex, None);
190            LocalDef {
191                local,
192                local_snap,
193                local_ex,
194                impure_snap,
195                impure_pred,
196            }
197        }
198
199        vir::with_vcx(|vcx| {
200            // TODO: refactor this a bit: split into one encoder for arguments (only)
201            //   and one for locals (only)
202            let data = if let Some(body) = task_key.body() {
203                deps.emit_output_ref(
204                    *task_key,
205                    MirLocalDefEncOutputRef {
206                        arg_count: body.arg_count,
207                    },
208                )?;
209                // Locals only serving specification-only arms are never
210                // used in the encoding; give them no definition so that
211                // their types are not encoded. Arguments and the return
212                // place never serve only a spec arm, so args-only tasks skip
213                // the analysis.
214                let spec_only_locals = if task_key.all_locals() {
215                    deps.require_dep::<crate::encoders::mir_fn::SpecBlocksEnc>(task_key.def_id())?
216                        .spec_arms
217                        .spec_only_locals
218                } else {
219                    Default::default()
220                };
221                let locals = IndexVec::from_fn_n(
222                    |local: mir::Local| {
223                        if spec_only_locals.contains(&local) {
224                            return None;
225                        }
226                        let rust_ty = body.local_decls[local].ty;
227                        let rust_ty_task = RustTyDecomposition::from_ty(rust_ty, task_key.def_id());
228                        let ty = deps.require_dep::<TyUseImpureEnc>(rust_ty_task).unwrap();
229                        Some(mk_local_def(vcx, local, ty))
230                    },
231                    if task_key.all_locals() {
232                        body.local_decls.len()
233                    } else {
234                        // return + arguments
235                        1 + body.arg_count
236                    },
237                );
238                MirLocalDefEncOutput {
239                    locals: vcx.alloc(locals),
240                    arg_count: body.arg_count,
241                }
242            } else {
243                let typing_env = ty::TypingEnv::post_analysis(vcx.tcx(), task_key.context_def_id());
244                // `identity_fn_sig` rather than the `fn_sig` query, which
245                // closures do not have: their body signature (the
246                // environment followed by the untupled parameters) is read
247                // off their type instead.
248                let sig = FunctionData::new(task_key.def_id()).identity_fn_sig(vcx.tcx());
249                let sig = vcx.tcx().instantiate_and_normalize_erasing_regions(
250                    task_key.substs(vcx),
251                    typing_env,
252                    ty::EarlyBinder::bind(vcx.tcx(), sig),
253                );
254                deps.emit_output_ref(
255                    *task_key,
256                    MirLocalDefEncOutputRef {
257                        arg_count: sig.inputs().len(),
258                    },
259                )?;
260
261                let locals = (0..sig.inputs_and_output.len())
262                    .map(mir::Local::from)
263                    .map(|local: mir::Local| {
264                        let rust_ty = if local == mir::RETURN_PLACE {
265                            sig.output()
266                        } else {
267                            sig.inputs()[local.index() - 1]
268                        };
269                        let rust_ty_task =
270                            RustTyDecomposition::from_ty(rust_ty, task_key.context_def_id());
271                        let ty = deps.require_dep::<TyUseImpureEnc>(rust_ty_task)?;
272                        Ok(Some(mk_local_def(vcx, local, ty)))
273                    })
274                    .collect::<Result<IndexVec<_, _>, _>>()?;
275
276                MirLocalDefEncOutput {
277                    locals: vcx.alloc(locals),
278                    arg_count: sig.inputs().len(),
279                }
280            };
281            Ok(((), data))
282        })
283    }
284}

286impl<'vir> Index<mir::Local> for MirLocalDefEncOutput<'vir> {
287    type Output = LocalDef<'vir>;
288    fn index(&self, index: mir::Local) -> &Self::Output {
289        self.locals[index]

291            .unwrap_or_else(|| panic!("spec-only local {index:?} has no definition"))
