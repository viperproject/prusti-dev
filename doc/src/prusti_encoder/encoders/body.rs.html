10/// Whether the body of `def_id` is encoded, rather than only its contract.
11/// It is not if the function is trusted (we take its contract on faith), or
12/// if it has no body of its own to encode (an external function, or a trait
13/// method without a default implementation).
14pub fn encodes_body(def_id: DefId) -> bool {
15    if crate::encoders::is_function_trusted(def_id) {
16        tracing::info!("function {def_id:?} is trusted, not encoding its body");
17        return false;
18    }
19    vir::with_vcx(|vcx| {
20        is_function_with_body(vcx.tcx(), def_id) && EnvQuery::new(vcx.tcx()).has_body(def_id)
21    })
22}

24/// The MIR body of `def_id` to encode, or `None` if there is none (see
25/// [encodes_body]). The body is the one the compiler produced: as the
26/// encoding is generic it is not instantiated, which also keeps the regions
27/// that the PCG needs.
28pub fn impure_body<'vir>(def_id: DefId) -> Option<MirBody<'vir>> {
29    let def_id = local_to_encode(def_id)?;
30    vir::with_vcx(|vcx| Some(vcx.body_mut().get_impure_fn_body_identity(def_id)))
31}
32
33/// As [impure_body], together with the borrowck facts needed by the PCG.
34pub fn impure_body_with_facts<'vir>(def_id: DefId) -> Option<BodyWithBorrowckFacts<'vir>> {
35    let def_id = local_to_encode(def_id)?;
36    vir::with_vcx(|vcx| Some(vcx.body_mut().get_impure_fn_body_with_facts(def_id)))
37}

39/// The MIR body of the pure function `def_id`. Only the bodies of functions
40/// whose specification marks them pure are preloaded (they must be exported
41/// across crates); a local function is pure by inheritance from the trait
42/// method it implements, so its body is taken like any other local body.
43pub fn pure_body<'vir>(def_id: DefId) -> MirBody<'vir> {
44    impure_body(def_id).unwrap_or_else(|| {
45        vir::with_vcx(|vcx| {
46            let substs = ty::GenericArgs::identity_for_item(vcx.tcx(), def_id);
47            vcx.body_mut().get_pure_fn_body(def_id, substs, None)
48        })
49    })
50}

52/// The MIR body of the specification `def_id`, taken generically (i.e. with
53/// its own parameters, see [instantiated_spec_body] for the exception).
54pub fn spec_body<'vir>(def_id: DefId) -> MirBody<'vir> {
55    vir::with_vcx(|vcx| {
56        let substs = ty::GenericArgs::identity_for_item(vcx.tcx(), def_id);
57        vcx.body_mut().get_spec_body(def_id, substs, None)
58    })
59}
60
61fn local_to_encode(def_id: DefId) -> Option<LocalDefId> {
62    encodes_body(def_id).then(|| def_id.as_local()).flatten()
63}
