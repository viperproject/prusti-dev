19/// Takes as input the most generic version (c.f. [`MostGenericTy`]) of a Rust
20/// type, and generates functions to convert the generic Viper representation of
21/// a Rust expression with that type to its concrete representation, and
22/// vice-versa. If the provided type is generic, it does nothing, returning
23/// [`CastFunctions::AlreadyGeneric`].
24///
25/// In the pure case, each pair becomes one variant of the `s_Param` adt (see
26/// [`CastersEnc::<Pure>::emit_outputs`]): `make_generic` is the variant's
27/// constructor (the concrete snapshot and the type/const arguments are its
28/// fields) and `make_concrete` is the value field's destructor.
29pub(super) struct CastersEnc<T>(PhantomData<T>);
30
31pub trait PurityCasters: Purity {
32    type MakeGeneric<'vir>: Debug + Clone + Copy;
33    type MakeConcrete<'vir>: Debug + Clone + Copy;
34}
35
36impl PurityCasters for Pure {
37    type MakeGeneric<'vir> =
38        FunctionIdn<'vir, (vir::CSnap, vir::ManyTyVal, vir::ManyCSnap), vir::PSnap>;
39    type MakeConcrete<'vir> = vir::AdtDestructor<'vir, vir::PSnap, vir::CSnap>;
40}

42impl PurityCasters for Impure {
43    type MakeGeneric<'vir> = MethodIdn<'vir, (vir::Ref, vir::ManyTyVal, vir::ManyCSnap)>;
44    type MakeConcrete<'vir> = MethodIdn<'vir, (vir::Ref, vir::ManyTyVal, vir::ManyCSnap)>;
45}

49/// A pair's contribution to the program: its variant of the `s_Param` adt,
50/// which [`CastersEnc::<Pure>::emit_outputs`] gathers into the one adt
51/// declaration, and the axioms relating that variant to its type.
52#[derive(Clone)]
53pub(super) struct PureCaster<'vir> {
54    constructor: vir::AdtConstructor<'vir>,
55    axioms: vir::Domain<'vir>,
56}

58impl TaskEncoder for CastersEnc<Pure> {
59    task_encoder::encoder_cache!(CastersEnc<Pure>);
60    const ENCODER_NAME: &'static str = "pure casters encoder";
61
62    type TaskDescription<'vir> = (RustTy<'vir>, RustTy<'vir>);
63    type OutputRef<'vir> = GArgCasters<'vir, Pure>;
64    type OutputFullLocal<'vir> = PureCaster<'vir>;
65
66    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
67        *task
68    }
69
70    fn do_encode_full<'vir>(
71        task_key: &Self::TaskKey<'vir>,
72        deps: &mut TaskEncoderDependencies<'vir, Self>,
73    ) -> EncodeFullResult<'vir, Self> {
74        let (param, concrete) = task_key;
75        assert!(param.specifics.is_param() && !concrete.specifics.is_param());
76        vir::with_vcx(|vcx| {
77            use vir::CastType;
78            let domain_ref = deps.require_ref::<TyPureEnc>(concrete)?;
79            let generic_snap = vir::TYPE_PSNAP;
80            let self_ty = domain_ref.snapshot.downcast_ty();
81            let base_name = concrete.name();
82            let ty_constructor = deps.require_ref::<TyConstructorEnc>(concrete)?;
83            let generics = deps.require_dep::<GenericParamsEnc>(concrete.params)?;
84
85            let make_generic_ident = FunctionIdn::new(
86                vir::vir_format_identifier!(vcx, "make_generic_s_{base_name}"),
87                (self_ty, generics.ty_args(), generics.const_args()),
88                generic_snap,
89            );
90
91            let make_concrete_destr = vcx.mk_adt_destructor(
92                vir::vir_format!(vcx, "make_concrete_s_{base_name}"),
93                generic_snap,
94                self_ty,
95            );
96
97            deps.emit_output_ref(
98                *task_key,
99                GArgCasters {
100                    make_generic: make_generic_ident,
101                    make_concrete: make_concrete_destr,
102                },
103            )?;
104
105            // The variant's fields: the concrete value (whose destructor is
106            // `make_concrete`), then the type and const arguments. The latter
107            // make `s_Param_typ` definable on instantiated generics.
108            let fields = std::iter::once(
109                vcx.mk_local_decl(make_concrete_destr.name, self_ty)
110                    .as_dyn(),
111            )
112            .chain(generics.ty_decls().iter().map(|d| {
113                vcx.mk_local_decl(
114                    vir::vir_format!(vcx, "make_generic_s_{base_name}_typaram_{}", d.name),
115                    vir::TYPE_TYVAL,
116                )
117                .as_dyn()
118            }))
119            .chain(generics.const_decls().iter().map(|d| {
120                vcx.mk_local_decl(
121                    vir::vir_format!(vcx, "make_generic_s_{base_name}_constparam_{}", d.name),
122                    d.ty,
123                )
124                .as_dyn()
125            }))
126            .collect::<Vec<vir::LocalDeclDyn<'vir>>>();
127            let constructor = vcx
128                .mk_adt_constructor(make_generic_ident.name().to_str(), vcx.alloc_slice(&fields));
129
130            // A zero-field struct has a single value, so `make_concrete`
131            // returns it; this merges otherwise-distinct unit terms (e.g. the
132            // thin pointer metadata read out of a reference's snapshot with
133            // the freshly constructed one).
134            let unit_value = if concrete
135                .get_structlike()
136                .is_some_and(|data| data.fields.is_empty())
137            {
138                let cons = deps
139                    .require_dep::<TyPureEnc>(concrete)?
140                    .expect_structlike()
141                    .field_snaps_to_snap;
142                Some(cons.call()(&[]))
143            } else {
144                None
145            };
146
147            // Two axioms relating the variant to its type: the definition
148            // `s_Param_typ(make_generic_T(x, ts..)) == T_type(ts..)` and the
149            // variant bridge, which makes the reconstruction
150            // `make_generic_T(make_concrete_T(p), ts..) == p` hold wherever
151            // `p`'s type is known to be `T_type(ts..)`.
152            let param_typ = deps.require_dep::<ParamTypEnc>(())?;
153            let typ_idn = param_typ.typ;
154            let mut axioms = Vec::new();
155            let x_decl = vcx.mk_local_decl("x", self_ty);
156            let tys = generics
157                .ty_decls()
158                .iter()
159                .map(|d| vcx.mk_local_ex(*d))
160                .collect::<Vec<_>>();
161            let consts = generics
162                .const_decls()
163                .iter()
164                .map(|d| vcx.mk_local_ex(*d))
165                .collect::<Vec<_>>();
166            let mg_app = make_generic_ident(vcx.mk_local_ex(x_decl), &tys, &consts);
167            let qvars = std::iter::once(x_decl.as_dyn())
168                .chain(generics.ty_decls().iter().map(|d| d.as_dyn()))
169                .chain(generics.const_decls().iter().map(|d| d.as_dyn()))
170                .collect::<Vec<vir::LocalDeclDyn<'vir>>>();
171            let def = vcx.mk_forall_expr(
172                vcx.alloc_slice(&qvars),
173                vcx.alloc_slice(&[vcx.mk_trigger(&[mg_app])]),
174                vcx.mk_eq_expr(
175                    typ_idn(mg_app),
176                    (ty_constructor.ty_constructor)(&tys, &consts),
177                ),
178            );
179            axioms.push(vcx.mk_domain_axiom(
180                vir::vir_format_identifier!(vcx, "{}_def_{}", typ_idn.name(), constructor.name),
181                def,
182            ));
183
184            // No discriminators: their tag terms trigger the N-way
185            // exhaustiveness axioms of the `Type` and `s_Param` adts.
186            let p_decl = vcx.mk_local_decl("p", vir::TYPE_PSNAP);
187            let p = vcx.mk_local_ex(p_decl);
188            let mc_p = make_concrete_destr.call()(p);
189            let typ_p = typ_idn(p);
190            let bridge = match unit_value {
191                // A param of a zero-field type is never read; the trigger
192                // names the type so that a param fires only its own bridge.
193                Some(unit_value) => {
194                    let ty = (ty_constructor.ty_constructor)(&tys, &consts);
195                    let bridge_qvars = std::iter::once(p_decl.as_dyn())
196                        .chain(generics.ty_decls().iter().map(|d| d.as_dyn()))
197                        .chain(generics.const_decls().iter().map(|d| d.as_dyn()))
198                        .collect::<Vec<vir::LocalDeclDyn<'vir>>>();
199                    vcx.mk_forall_expr(
200                        vcx.alloc_slice(&bridge_qvars),
201                        vcx.alloc_slice(&[vcx.mk_trigger(&[(param_typ.has_typ)(p, ty)])]),
202                        vcx.mk_bin_op_expr(
203                            vir::BinOpKind::Implies,
204                            vcx.mk_eq_expr(typ_p, ty),
205                            vcx.mk_eq_expr(make_generic_ident(unit_value, &tys, &consts), p),
206                        )
207                        .downcast_ty(),
208                    )
209                }
210                // Type arguments are read off `p`'s type, so `p` need not be
211                // a constructor term.
212                None => {
213                    let ty_args = ty_constructor
214                        .ty_param_accessors
215                        .iter()
216                        .map(|a| a.call()(typ_p))
217                        .collect::<Vec<_>>();
218                    let const_args = ty_constructor
219                        .const_param_accessors
220                        .iter()
221                        .map(|a| a.call()(typ_p))
222                        .collect::<Vec<_>>();
223                    vcx.mk_forall_expr(
224                        vcx.alloc_slice(&[p_decl]),
225                        vcx.alloc_slice(&[vcx.mk_trigger(&[mc_p.as_dyn(), typ_p.as_dyn()])]),
226                        vcx.mk_bin_op_expr(
227                            vir::BinOpKind::Implies,
228                            vcx.mk_eq_expr(
229                                typ_p,
230                                (ty_constructor.ty_constructor)(&ty_args, &const_args),
231                            ),
232                            vcx.mk_eq_expr(make_generic_ident(mc_p, &ty_args, &const_args), p),
233                        )
234                        .downcast_ty(),
235                    )
236                }
237            };
238            axioms.push(vcx.mk_domain_axiom(
239                vir::vir_format_identifier!(vcx, "{}_variant_{}", typ_idn.name(), constructor.name),
240                bridge,
241            ));
242
243            if let Some(unit_value) = unit_value {
244                let unit = vcx.mk_forall_expr(
245                    vcx.alloc_slice(&[p_decl]),
246                    vcx.alloc_slice(&[vcx.mk_trigger(&[mc_p])]),
247                    vcx.mk_eq_expr(mc_p, unit_value),
248                );
249                axioms.push(vcx.mk_domain_axiom(
250                    vir::vir_format_identifier!(vcx, "{}_unit", constructor.name),
251                    unit,
252                ));
253            }
254
255            let axioms = vcx.mk_domain(
256                vir::ViperIdent::new(vir::vir_format!(vcx, "ParamTyp_{}", constructor.name)),
257                &[],
258                vcx.alloc_slice(&axioms),
259                &[],
260                None,
261            );
262            Ok((
263                PureCaster {
264                    constructor,
265                    axioms,
266                },
267                (),
268            ))
269        })
270    }
271
272    /// Emits the `s_Param` adt: one variant per generic cast pair, plus the
273    /// fallback variant for values of types without a cast pair in this
274    /// program (mirroring `Unknown_type` in the `Type` adt). An adt is a
275    /// single declaration, so this is the one part that cannot be built per
276    /// pair; each pair's axioms are emitted as built.
277    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
278        let outputs = Self::all_outputs_local_no_errors(program);
279        vir::with_vcx(|vcx| {
280            let mut constructors = Vec::new();
281            for pc in outputs {
282                constructors.push(pc.constructor);
283                program.add_domain(pc.axioms);
284            }
285            let unknown_args =
286                vcx.alloc_array(&[vcx.mk_local_decl(Self::UNKNOWN_PARAM_ID, vir::TYPE_INT)]);
287            constructors.push(vcx.mk_adt_constructor(Self::UNKNOWN_PARAM_NAME, unknown_args));
288            let vir::TypeKind::Domain(param_adt_name, _) = **vir::TYPE_PSNAP else {
289                unreachable!()
290            };
291            program.add_adt(vcx.mk_adt(
292                vir::ViperIdent::new(param_adt_name),
293                &[],
294                vcx.alloc_slice(&constructors),
295            ));
296        })
297    }
298}

306impl TaskEncoder for CastersEnc<Impure> {
307    task_encoder::encoder_cache!(CastersEnc<Impure>);
308    const ENCODER_NAME: &'static str = "impure casters encoder";
309
310    type TaskDescription<'vir> = (RustTy<'vir>, RustTy<'vir>);
311    type OutputRef<'vir> = GArgCasters<'vir, Impure>;
312    type OutputFullLocal<'vir> = Vec<vir::Method<'vir>>;
313    type EncodingError = ();
314
315    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
316        *task
317    }
318
319    fn do_encode_full<'vir>(
320        task_key: &Self::TaskKey<'vir>,
321        deps: &mut TaskEncoderDependencies<'vir, Self>,
322    ) -> EncodeFullResult<'vir, Self> {
323        let (param, concrete) = task_key;
324        assert!(param.specifics.is_param() && !concrete.specifics.is_param());
325        vir::with_vcx(|vcx| {
326            use vir::CastType;
327            let base_name = concrete.name();
328            let ty_constructor = deps.require_ref::<TyConstructorEnc>(concrete)?;
329            let generics = deps.require_dep::<GenericParamsEnc>(concrete.params)?;
330
331            let make_generic_ident = MethodIdn::new(
332                vir::vir_format_identifier!(vcx, "make_generic_{base_name}"),
333                (vir::TYPE_REF, generics.ty_args(), generics.const_args()),
334            );
335
336            let make_concrete_ident = MethodIdn::new(
337                vir::vir_format_identifier!(vcx, "make_concrete_{base_name}"),
338                (vir::TYPE_REF, generics.ty_args(), generics.const_args()),
339            );
340
341            deps.emit_output_ref(
342                *task_key,
343                GArgCasters {
344                    make_generic: make_generic_ident,
345                    make_concrete: make_concrete_ident,
346                },
347            )?;
348            let make_generic_pure = deps
349                .require_ref::<CastersEnc<Pure>>(*task_key)?
350                .make_generic;
351            let self_decl = vcx.mk_local_decl("self", vir::TYPE_REF);
352            let self_expr = vcx.mk_local_ex(self_decl);
353            let decls = (self_decl, generics.ty_decls(), generics.const_decls());
354
355            let predicate_ref = deps.require_ref::<TyImpureEnc>(concrete)?;
356            let generic_ref = deps.require_ref::<TyImpureEnc>(param)?;
357
358            let concrete_predicate = (predicate_ref.ref_to_pred)(
359                self_expr,
360                generics.ty_exprs(),
361                generics.const_exprs(),
362            )(None);
363
364            let concrete_snap =
365                (predicate_ref.ref_to_snap)(self_expr, generics.ty_exprs(), generics.const_exprs())
366                    .downcast_ty();
367
368            let concrete_predicate = vcx.mk_predicate_app_expr(concrete_predicate);
369
370            let lifted_ty_expr =
371                (ty_constructor.ty_constructor)(generics.ty_exprs(), generics.const_exprs());
372
373            let generic_predicate =
374                (generic_ref.ref_to_pred)(self_expr, &[lifted_ty_expr], &[])(None);
375
376            let generic_snap = (generic_ref.ref_to_snap)(self_expr, &[lifted_ty_expr], &[])
377                .downcast_ty::<vir::PSnap>();
378
379            let generic_predicate = vcx.mk_predicate_app_expr(generic_predicate);
380
381            let make_generic_same_snap = vcx.mk_eq_expr(
382                vcx.mk_old_expr(make_generic_pure(
383                    concrete_snap,
384                    generics.ty_exprs(),
385                    generics.const_exprs(),
386                )),
387                generic_snap,
388            );
389
390            let make_concrete_same_snap = vcx.mk_eq_expr(
391                vcx.mk_old_expr(generic_snap),
392                make_generic_pure(concrete_snap, generics.ty_exprs(), generics.const_exprs()),
393            );
394
395            let make_generic = vcx.mk_method(
396                make_generic_ident,
397                decls,
398                &[],
399                vcx.alloc_slice(&[concrete_predicate]),
400                vcx.alloc_slice(&[generic_predicate, make_generic_same_snap]),
401                None,
402            );
403
404            let make_concrete = vcx.mk_method(
405                make_concrete_ident,
406                decls,
407                &[],
408                vcx.alloc_slice(&[generic_predicate]),
409                vcx.alloc_slice(&[concrete_predicate, make_concrete_same_snap]),
410                None,
411            );
412            Ok((vec![make_generic, make_concrete], ()))
413        })
414    }
415
416    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
417        for output in Self::all_outputs_local_no_errors(program) {
418            for method in output {
419                program.add_method(method);
420            }
421        }
422    }
