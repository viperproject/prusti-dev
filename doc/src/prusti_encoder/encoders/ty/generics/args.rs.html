5/// The instantiation of generic arguments, typically found in `TyKind::Adt` and
6/// `TyKind::FnDef`.
7#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
8pub struct GArgs<'tcx> {
9    pub(super) context: GParams<'tcx>,
10    pub(super) args: &'tcx [ty::GenericArg<'tcx>],
11}
12
13pub enum GParamVariant<'tcx> {
14    Param(ty::ParamTy),
15    Alias(ty::AliasTy<'tcx>),
16}
17
18impl<'tcx> GArgs<'tcx> {
19    pub fn new(context: impl Into<GParams<'tcx>>, args: &'tcx [ty::GenericArg<'tcx>]) -> Self {
20        let context: GParams<'tcx> = context.into();
21        // Sanity check that all generic values in args are bound (i.e. defined
22        // in context).
23        for arg in args.iter().flat_map(|arg| arg.walk()) {
24            let valid = context.check_arg(arg);
25            assert!(valid, "context: {context:#?}, args: {args:#?}");
26        }
27        GArgs { context, args }
28    }
29
30    pub fn context(self) -> GParams<'tcx> {
31        self.context
32    }
33
34    pub fn args(self) -> &'tcx [ty::GenericArg<'tcx>] {
35        self.args
36    }
37
38    /// Drops the generic context, keeping the (ground) `args`. Use when the
39    /// context is irrelevant to the encoding (e.g. a builtin's concrete result
40    /// type) so that clients in different contexts (different where-clauses)
41    /// share one task key. Panics (via `new`) if the `args` are not ground.
42    pub fn with_empty_context(self) -> Self {
43        Self::new(GParams::empty(), self.args)
44    }
45
46    /// Substitutes type arguments and try to normalize associated types
47    pub fn normalize(self, ty: ty::Ty<'tcx>) -> ty::Ty<'tcx> {
48        // Substitute type parameters
49        let ty = vir::with_vcx(|vcx| {
50            ty::EarlyBinder::bind(vcx.tcx(), ty).instantiate(vcx.tcx(), self.args)
51        })
52        .skip_normalization();
53        // Normalize associated types
54        self.context.normalize(ty)
55    }
56
57    pub fn expect_param(self) -> GParamVariant<'tcx> {
58        assert_eq!(self.args.len(), 1);
59        match self.args[0].expect_ty().kind() {
60            ty::TyKind::Param(p) => GParamVariant::Param(*p),
61            ty::TyKind::Alias(_k, t) => GParamVariant::Alias(*t),
62            other => panic!("expected type parameter, {other:?}"),
63        }
64    }
65
66    /// Given the definitions:
67    /// ```
68    /// struct S0<'a, T0, U0>
69    /// struct S1<'a, T1, U1> {
70    ///     field: S0<'a, U1, T1>,
71    /// }
72    /// # { field: &'a mut T }
73    /// fn foo<'x>(x: S1<'x, u32, bool>)
74    /// # {}
75    /// ```
76    /// We will decompose `S1<'x, u32, bool>` into a `GArgs` of `['x, u32, bool]`.
77    /// When traversing into the definition of `S1`, the field `S0<'a, U1, T1>` is
78    /// decomposed into a `GArgs` of `['a, U1, T1]`. This function will substitute
79    /// the first into the second resulting in the `GArgs` of `['x, bool, u32]`.
80    /// This is useful when encoding the field of `S1` in the context of `foo`
81    /// (without any e.g. predicate that needs to be general for any use of `S1`).
82    pub fn substitute(self, to_sub_in: GArgs<'tcx>) -> GArgs<'tcx> {
83        assert_eq!(self.context.rust_params().len(), to_sub_in.args.len());
84        let args = vir::with_vcx(|vcx| {
85            let args = self.args.iter().map(|arg| {
86                ty::EarlyBinder::bind(vcx.tcx(), *arg)
87                    .instantiate(vcx.tcx(), to_sub_in.args)
88                    .skip_normalization()
89            });
90            vcx.tcx().mk_args_from_iter(args)
91        });
92        GArgs {
93            context: to_sub_in.context,
94            args,
95        }
96    }
97}
