39impl TaskEncoder for TraitFnEnc {
40    task_encoder::encoder_cache!(TraitFnEnc);
41    const ENCODER_NAME: &'static str = "trait fn encoder";
42
43    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
44        *task
45    }
46
47    type TaskDescription<'vir> = DefId;
48
49    type OutputRef<'vir> = TraitFnEncOutputRef<'vir>;
50    type OutputFullLocal<'vir> = (
51        vir::Domain<'vir>,
52        Vec<vir::Function<'vir>>,
53        Vec<vir::Method<'vir>>,
54    );
55
56    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
57        for (dom, funcs, methods) in Self::all_outputs_local_no_errors(program) {
58            program.add_domain(dom);
59            for func in funcs {
60                program.add_function(func);
61            }
62            for method in methods {
63                program.add_method(method);
64            }
65        }
66    }
67
68    fn do_encode_full<'vir>(
69        task_key: &Self::TaskKey<'vir>,
70        deps: &mut TaskEncoderDependencies<'vir, Self>,
71    ) -> EncodeFullResult<'vir, Self> {
72        vir::with_vcx(|vcx| {
73            let tcx = vcx.tcx();
74
75            let assoc_item = tcx
76                .opt_associated_item(*task_key)
77                .expect("task key should be the associated item of a trait");
78            let def_id = assoc_item.def_id;
79            let span = vcx.tcx().def_span(def_id);
80            assert!(matches!(assoc_item.kind, ty::AssocKind::Fn { .. }));
81            assert_eq!(def_id, *task_key);
82
83            // Prusti specifications on trait methods emit additional spec-
84            // only fn items (with default implementations). These should never
85            // be passed here, even though they are part of the trait as far as
86            // Rust typing is concerned.
87            assert!(!is_spec_fn(tcx, def_id));
88
89            let trait_def_id = assoc_item
90                .trait_container(tcx)
91                .expect("task key should be the associated item of a trait");
92
93            let trait_name = ViperIdent::from_def_id(vcx, trait_def_id);
94
95            let mut axioms = Vec::new();
96            let mut funcs = Vec::new();
97            let mut dom_funcs = Vec::new();
98            let mut methods = Vec::new();
99
100            // item_generics also includes parameters of trait itself
101            let item_params = GParams::from(def_id);
102            let item_generics = deps.require_dep::<GenericParamsEnc>(item_params)?;
103            let item_name = ViperIdent::from_def_id(vcx, def_id);
104
105            let local_defs = deps.require_dep::<MirLocalDefEnc>(MirLocalDefEncTask::Local {
106                def_id,
107                all_locals: false,
108            })?;
109            let arg_count = local_defs.arg_count + 1;
110            let arg_types = vcx.alloc_slice(&local_defs.snap_ty_args().collect::<Vec<_>>());
111            let return_type = local_defs.snap_ty_return();
112            let ref_args = vcx.alloc_slice(&vec![vir::TYPE_REF; arg_count]);
113
114            let is_pure = crate::encoders::is_function_pure(
115                def_id,
116                GArgs::new(item_params, item_params.rust_params()),
117            );
118
119            let pre_func = FunctionIdn::new(
120                vir_format_identifier!(vcx, "{trait_name}_fn_pre_{item_name}"),
121                (
122                    arg_types,
123                    item_generics.ty_args(),
124                    item_generics.const_args(),
125                ),
126                vir::TYPE_BOOL,
127            );
128            let post_func = FunctionIdn::new(
129                vir_format_identifier!(vcx, "{trait_name}_fn_post_{item_name}"),
130                // TODO: old(arg) types (if applicable)
131                (
132                    return_type,
133                    arg_types,
134                    item_generics.ty_args(),
135                    item_generics.const_args(),
136                ),
137                vir::TYPE_BOOL,
138            );
139            // TODO: spec functions for each pledge
140
141            let call_stub_impure = (!is_pure).then(|| {
142                MethodIdn::new(
143                    vir_format_identifier!(vcx, "{trait_name}_fn_stub_{item_name}"),
144                    (
145                        ref_args,
146                        item_generics.ty_args(),
147                        item_generics.const_args(),
148                    ),
149                )
150            });
151            let call_stub_pure_caller = is_pure.then(|| {
152                FunctionIdn::new(
153                    vir_format_identifier!(vcx, "{trait_name}_cfn_stub_{item_name}"),
154                    (
155                        arg_types,
156                        item_generics.ty_args(),
157                        item_generics.const_args(),
158                    ),
159                    return_type,
160                )
161            });
162            let call_stub_pure_function = is_pure.then(|| {
163                FunctionIdn::new(
164                    vir_format_identifier!(vcx, "{trait_name}_fn_stub_{item_name}"),
165                    (
166                        arg_types,
167                        item_generics.ty_args(),
168                        item_generics.const_args(),
169                    ),
170                    return_type,
171                )
172            });
173            deps.emit_output_ref(
174                *task_key,
175                TraitFnEncOutputRef {
176                    pre_func,
177                    post_func,
178                    call_stub_impure,
179                    call_stub_pure_caller,
180                    call_stub_pure_function,
181                },
182            )?;
183            // The stubs' semantics are given by the impls' axioms; require the
184            // trait so that its impls' condition/axiom triggers are registered
185            // (foreign traits are not encoded by `encode_all_in_crate`).
186            deps.require_ref::<TraitEnc>(trait_def_id)?;
187            dom_funcs.push(vcx.mk_domain_function(pre_func, false, None));
188            dom_funcs.push(vcx.mk_domain_function(post_func, false, None));
189
190            // The stub emitted below is only useful together with the axioms
191            // bridging the abstract pre/post functions to concrete impl
192            // specs. Unlock, per impl of the trait, the axioms of just the
193            // item implementing this function once the impl's constructor
194            // keys are requested - the same gating as the impl's condition
195            // (see `TraitEnc`), but calling a trait function must not pull in
196            // the trait's whole machinery. An impl that inherits a pure
197            // default body instead unlocks the axiom stating that body at its
198            // trait refs. Only final definitions are assumed (see
199            // `final_leaf_def`).
200            let has_body = is_function_with_body(vcx.tcx(), def_id);
201            for impl_did in trait_impls::implementing_impls(tcx, trait_def_id) {
202                let Some(leaf) = trait_impls::final_leaf_def(tcx, impl_did, def_id) else {
203                    continue;
204                };
205                let keys = trait_impls::impl_unlock_keys(impl_did);
206                let impl_span = tcx.def_span(impl_did);
207                if !leaf.defining_node.is_from_trait() {
208                    let item_did = leaf.item.def_id;
209                    TyConstructorEnc::on_all_requested(keys, move || {
210                        let _ = trait_impls::TraitImplItemEnc::encode(
211                            (impl_did, item_did),
212                            false,
213                            impl_span,
214                        );
215                    });
216                } else if has_body && is_pure {
217                    TyConstructorEnc::on_all_requested(keys, move || {
218                        let _ = trait_impls::TraitImplDefaultFnEnc::encode(
219                            (impl_did, def_id),
220                            false,
221                            impl_span,
222                        );
223                    });
224                }
225            }
226
227            let func_args = local_defs.local_decl_args().collect::<Vec<_>>();
228            let func_arg_exprs = vcx.alloc_slice(
229                &func_args
230                    .iter()
231                    .map(|arg| vcx.mk_local_ex(arg))
232                    .collect::<Vec<_>>(),
233            );
234            let func_ret = local_defs.local_decl_ret();
235
236            let spec = deps.require_dep_spanned::<MirSpecEnc>(
237                (def_id, def_id, MirSpecEncMode::PureWithoutResult),
238                span,
239            )?;
240            let pres = vcx.mk_conj(&spec.pre_exprs().collect::<Vec<_>>());
241            let pre_func_call = pre_func.call()(
242                func_arg_exprs,
243                item_generics.ty_exprs(),
244                item_generics.const_exprs(),
245            );
246            axioms.push(vcx.mk_domain_axiom(
247                vir_format_identifier!(
248                    vcx,
249                    "{trait_name}_fn_pre_{item_name}_base",
250                ),
251                vir::expr! {
252                    forall ..[func_args], ..[item_generics.ty_decls()], ..[item_generics.const_decls()] :: {[pre_func_call]}
253                        (pres) ==> (pre_func_call)
254                },
255            ));
256            // A default body is not part of the trait's contract: impls may
257            // override it with a different result.
258            let posts = vcx.mk_conj(&spec.post_exprs().collect::<Vec<_>>());
259            let post_func_call = post_func.call()(
260                vcx.mk_local_ex(func_ret),
261                func_arg_exprs,
262                item_generics.ty_exprs(),
263                item_generics.const_exprs(),
264            );
265            // Impls may weaken the preconditions, so a call can be valid
266            // outside them; the trait's postconditions are only promised
267            // where they held. `post_func` receives the arguments' pre-state.
268            axioms.push(vcx.mk_domain_axiom(
269                vir_format_identifier!(
270                    vcx,
271                    "{trait_name}_fn_post_{item_name}_base",
272                ),
273                vir::expr! {
274                    forall [func_ret], ..[func_args], ..[item_generics.ty_decls()], ..[item_generics.const_decls()] :: {[post_func_call]}
275                        (post_func_call) ==> ((pres) ==> (posts))
276                },
277            ));
278
279            if is_pure {
280                let mut stub_pres = Vec::new();
281                let mut stub_posts = Vec::new();
282                stub_pres.push(pre_func.call()(
283                    vcx.alloc_slice(
284                        &local_defs
285                            .args()
286                            .map(|arg| vcx.mk_local_ex(arg.local_snap))
287                            .collect::<Vec<_>>(),
288                    ),
289                    item_generics.ty_exprs(),
290                    item_generics.const_exprs(),
291                ));
292                stub_posts.push(post_func.call()(
293                    vcx.mk_result(local_defs.snap_ty_return()),
294                    vcx.alloc_slice(
295                        &local_defs
296                            .args()
297                            .map(|arg| vcx.mk_local_ex(arg.local_snap))
298                            .collect::<Vec<_>>(),
299                    ),
300                    item_generics.ty_exprs(),
301                    item_generics.const_exprs(),
302                ));
303
304                // If the call succeeds, then its return type is definitely inhabited
305                // We need this postcondition to generate impure wrapper fns
306                // See tests/verify/pass/extern-spec/module-arg.rs
307                let ret_ty = tcx
308                    .instantiate_and_normalize_erasing_regions(
309                        ty::GenericArgs::identity_for_item(tcx, def_id),
310                        ty::TypingEnv::post_analysis(tcx, def_id),
311                        tcx.fn_sig(def_id),
312                    )
313                    .skip_binder()
314                    .output();
315                let ret_ty = RustTyDecomposition::from_ty(ret_ty, def_id);
316                stub_posts.push(deps.require_ref::<TyUseInhabitedEnc>(ret_ty)?.inhabited());
317                // stub_posts.push(local_defs.ret().inhabited);
318
319                let wrapped_call = call_stub_pure_function.unwrap().call()(
320                    func_arg_exprs,
321                    item_generics.ty_exprs(),
322                    item_generics.const_exprs(),
323                );
324                funcs.push(vcx.mk_function(
325                    call_stub_pure_caller.unwrap(),
326                    (
327                        &func_args,
328                        item_generics.ty_decls(),
329                        item_generics.const_decls(),
330                    ),
331                    vcx.alloc_slice(&stub_pres),
332                    vcx.alloc_slice(&stub_posts),
333                    Some(&vir::DecreasesGenData::Star),
334                    Some(wrapped_call),
335                ));
336                funcs.push(vcx.mk_function(
337                    call_stub_pure_function.unwrap(),
338                    (
339                        &func_args,
340                        item_generics.ty_decls(),
341                        item_generics.const_decls(),
342                    ),
343                    &[],
344                    vcx.alloc_slice(&stub_posts),
345                    None,
346                    None,
347                ));
348            } else {
349                let mut stub_pres = Vec::new();
350                let mut stub_posts = Vec::new();
351                let mut args = Vec::with_capacity(arg_count + item_params.count());
352                for arg_idx in (0..arg_count).map(mir::Local::from) {
353                    let name_p = local_defs[arg_idx].local.name;
354                    args.push(vir::vir_local_decl! { vcx; [name_p] : Ref });
355                    if arg_idx != mir::RETURN_PLACE {
356                        stub_pres.push(local_defs[arg_idx].impure_pred);
357                    }
358                }
359                stub_posts.push(local_defs[mir::RETURN_PLACE].impure_pred);
360                // TODO: wands
361
362                stub_pres.push(pre_func.call()(
363                    vcx.alloc_slice(
364                        &local_defs
365                            .args()
366                            .map(|arg| arg.impure_snap)
367                            .collect::<Vec<_>>(),
368                    ),
369                    item_generics.ty_exprs(),
370                    item_generics.const_exprs(),
371                ));
372                // TODO: mutable arguments should also have a post-state
373                stub_posts.push(post_func.call()(
374                    local_defs.ret().impure_snap,
375                    vcx.alloc_slice(
376                        &local_defs
377                            .args()
378                            .map(|arg| vcx.mk_old_expr(arg.impure_snap))
379                            .collect::<Vec<_>>(),
380                    ),
381                    item_generics.ty_exprs(),
382                    item_generics.const_exprs(),
383                ));
384
385                methods.push(vcx.mk_method(
386                    call_stub_impure.unwrap(),
387                    (
388                        args.as_slice(),
389                        item_generics.ty_decls(),
390                        item_generics.const_decls(),
391                    ),
392                    &[],
393                    vcx.alloc_slice(&stub_pres),
394                    vcx.alloc_slice(&stub_posts),
395                    None,
396                ));
397            }
398
399            let trait_domain = vcx.mk_domain(
400                vir_format_identifier!(vcx, "trait_fns_{trait_name}_{item_name}"),
401                &[],
402                vcx.alloc_slice(&axioms),
403                vcx.alloc_slice(&dom_funcs),
404                None,
405            );
406
407            Ok(((trait_domain, funcs, methods), ()))
408        })
409    }
