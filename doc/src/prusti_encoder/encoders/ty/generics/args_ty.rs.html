8/// Encodes type arguments when calling a function in this context
9pub struct GArgsTyEnc;
10
11#[derive(Debug, Clone, Copy)]
12pub struct GArgsTy<'vir> {
13    ty_args: &'vir [vir::ExprTyVal<'vir>],
14    const_args: &'vir [vir::ExprCSnap<'vir>],
15}
16
17impl<'vir> GArgsTy<'vir> {
18    /// If possible, use `GArgsTyEnc` instead.
19    ///
20    /// Builds the type/const arguments directly (e.g. to instantiate a builtin
21    /// method with synthetic type parameters that don't have a corresponding
22    /// Rust `GArgs`).
23    pub fn new(
24        ty_args: &'vir [vir::ExprTyVal<'vir>],
25        const_args: &'vir [vir::ExprCSnap<'vir>],
26    ) -> Self {
27        GArgsTy {
28            ty_args,
29            const_args,
30        }
31    }
32
33    pub fn get_ty<Curr, Next>(&self) -> &'vir [vir::ExprGenTyVal<'vir, Curr, Next>] {
34        let args = self.ty_args as *const [vir::ExprTyVal<'vir>]
35            as *const [vir::ExprGenTyVal<'vir, Curr, Next>];
36        unsafe { &*args }
37    }
38
39    pub fn get_const<Curr, Next>(&self) -> &'vir [vir::ExprGenCSnap<'vir, Curr, Next>] {
40        let args = self.const_args as *const [vir::ExprCSnap<'vir>]
41            as *const [vir::ExprGenCSnap<'vir, Curr, Next>];
42        unsafe { &*args }
43    }
44}
45
46impl TaskEncoder for GArgsTyEnc {
47    task_encoder::encoder_cache!(GArgsTyEnc);
48    const ENCODER_NAME: &'static str = "generic args type encoder";
49    type TaskDescription<'tcx> = GArgs<'tcx>;
50    type OutputFullDependency<'vir> = GArgsTy<'vir>;
51
52    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
53        *task
54    }
55
56    fn do_encode_full<'vir>(
57        task_key: &Self::TaskKey<'vir>,
58        deps: &mut TaskEncoderDependencies<'vir, Self>,
59    ) -> EncodeFullResult<'vir, Self> {
60        deps.emit_output_ref(*task_key, ())?;
61        let ty_args = task_key
62            .args
63            .iter()
64            .copied()
65            .filter_map(ty::GenericArg::as_type)
66            .map(|arg| {
67                let decomp = RustTyDecomposition::from_ty(arg, task_key.context);
68                deps.require_dep::<TyExprEnc>(decomp)
69            })
70            .collect::<Result<Vec<_>, _>>()?;
71        let const_args = task_key
72            .args
73            .iter()
74            .copied()
75            .filter_map(ty::GenericArg::as_const)
76            .map(|const_| {
77                // If the constant is a value, we already know its type.
78                // Otherwise, we will look it up in the param environment.
79                // TODO: what about the other ConstKind variants?
80                let ty = match const_.kind() {
81                    ty::ConstKind::Value(v) => v.ty,
82                    ty::ConstKind::Param(p) => task_key.context.expect_const(p.index as usize).1,
83                    other => unreachable!("unexpected ConstKind: {other:?}"),
84                };
85                deps.require_dep::<ConstEnc>(ConstEncTask::Ty {
86                    const_,
87                    ty,
88                    context: task_key.context,
89                })
90            })
91            .collect::<Result<Vec<_>, _>>()?;
92        let args = vir::with_vcx(|vcx| GArgsTy {
93            ty_args: vcx.alloc_slice(&ty_args),
94            const_args: vcx.alloc_slice(&const_args),
95        });
96        Ok(((), args))
97    }
98}
