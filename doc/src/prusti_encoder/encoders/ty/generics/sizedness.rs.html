11/// Encodes a type constructor's facts about the sizedness traits `Sized` and
12/// `MetaSized` (`PointeeSized` holds of every type, and bounds on it are
13/// skipped, see `TraitImplEnc::context_bounds`). These are implemented by
14/// the compiler rather than by impls, so where an ordinary trait's
15/// `impl_fun` is established by its impls' conditions, these are
16/// established by the constructors, each with an axiom triggered on the
17/// application at the constructor:
18///
19/// ```text
20/// axiom { Sized_impl(s_Char_type()) }
21/// axiom { forall T :: {Sized_impl(s_W_type(T))} Sized_impl(T) ==> Sized_impl(s_W_type(T)) }
22/// ```
23///
24/// A constructor that does not implement a trait (e.g. `str` for `Sized`)
25/// has no axiom for it, which leaves it open rather than false.
26pub struct SizednessEnc;
27
28impl TaskEncoder for SizednessEnc {
29    task_encoder::encoder_cache!(SizednessEnc);
30    const ENCODER_NAME: &'static str = "sizedness encoder";
31
32    type TaskDescription<'vir> = RustTy<'vir>;
33    type OutputFullLocal<'vir> = vir::Domain<'vir>;
34
35    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
36        *task
37    }
38
39    fn do_encode_full<'vir>(
40        task_key: &Self::TaskKey<'vir>,
41        deps: &mut TaskEncoderDependencies<'vir, Self>,
42    ) -> EncodeFullResult<'vir, Self> {
43        deps.emit_output_ref(*task_key, ())?;
44        vir::with_vcx(|vcx| {
45            let lang_items = vcx.tcx().lang_items();
46            // For each trait, `None` if the constructor does not implement
47            // it, otherwise the type it is conditional on (if any).
48            let (sized, meta_sized) = match task_key.sizedness {
49                RustTySizedness::None => unreachable!(),
50                RustTySizedness::Sized {
51                    sized_if,
52                    meta_sized_if,
53                } => (Some(sized_if), Some(meta_sized_if)),
54                RustTySizedness::MetaSized => (None, Some(None)),
55                RustTySizedness::PointeeSized => (None, None),
56            };
57            let facts = [
58                ("Sized", lang_items.sized_trait(), sized),
59                ("MetaSized", lang_items.meta_sized_trait(), meta_sized),
60            ];
61
62            let params = deps.require_dep::<GenericParamsEnc>(task_key.params)?;
63            let constructor = deps.require_ref::<TyConstructorEnc>(*task_key)?;
64            let ty = constructor.ty_constructor.call()(params.ty_exprs(), params.const_exprs());
65            let base_name = task_key.name();
66
67            let mut axioms = Vec::new();
68            for (trait_name, trait_did, fact) in facts {
69                let (Some(trait_did), Some(condition)) = (trait_did, fact) else {
70                    continue;
71                };
72                let impl_fun = deps.require_ref::<TraitEnc>(trait_did)?.impl_fun;
73                let implemented = impl_fun(&[ty], &[]);
74                let body = match condition {
75                    None => implemented,
76                    Some(tail) => {
77                        let tail = RustTyDecomposition::from_ty(tail, task_key.params);
78                        let tail_implemented = impl_fun(&[params.ty_expr(deps, tail)?], &[]);
79                        vir::expr! { (tail_implemented) ==> (implemented) }
80                    }
81                };
82                let decls = params.ty_decls().iter().map(|decl| decl.upcast_ty());
83                let decls = decls.chain(params.const_decls().iter().map(|decl| decl.upcast_ty()));
84                axioms.push(vcx.mk_domain_axiom(
85                    vir_format_identifier!(vcx, "s_{base_name}_{trait_name}"),
86                    vcx.mk_forall_expr(
87                        vcx.alloc_slice(&decls.collect::<Vec<vir::LocalDeclDyn>>()),
88                        vcx.alloc_slice(&[vcx.mk_trigger(&[implemented])]),
89                        body,
90                    ),
91                ));
92            }
93
94            let domain = vcx.mk_domain(
95                vir_format_identifier!(vcx, "sizedness_{base_name}"),
96                &[],
97                vcx.alloc_slice(&axioms),
98                &[],
99                None,
100            );
101            Ok((domain, ()))
102        })
103    }
104
105    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
106        for domain in Self::all_outputs_local_no_errors(program) {
107            program.add_domain(domain);
108        }
109    }
110}
111
112impl SizednessEnc {
113    /// Encodes the facts of a type constructor. Called for every constructor
114    /// `TyConstructorEnc` encodes.
115    pub fn require(constructor: RustTy<'_>) {
116        if !matches!(constructor.sizedness, RustTySizedness::None) {
117            let _ = Self::encode(constructor, false, DUMMY_SP);
118        }
119    }
120}
