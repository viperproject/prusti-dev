46/// Use-side data for a `prusti_contracts` builtin. The collection builtins
47/// store their elements as generic snapshots (`s_Param`), so their operations
48/// cast the elements with the casters here; `Int`/`Real` have no elements.
49#[derive(Debug, Clone, Copy)]
50pub struct TyUsePureBuiltinData<'vir> {
51    pub pure: <PureTyDatas as TyDatas<'vir>>::BuiltinData,
52    /// `[elem, NoCast]` for `Set`/`Multiset`/`Seq`, `[key, val]` for `Map`,
53    /// and `[NoCast; 2]` for `Int`/`Real`.
54    casters: [FieldCaster<'vir>; 2],
55}
56
57impl<'vir> TyUsePureBuiltinData<'vir> {
58    /// The element caster of a `Set`/`Multiset`/`Seq`.
59    pub fn elem_caster(&self) -> &FieldCaster<'vir> {
60        assert!(matches!(
61            self.pure,
62            super::pure::TyPureBuiltinData::Set
63                | super::pure::TyPureBuiltinData::Multiset
64                | super::pure::TyPureBuiltinData::Seq
65        ));
66        &self.casters[0]
67    }
68
69    pub fn map_key_caster(&self) -> &FieldCaster<'vir> {
70        assert!(matches!(self.pure, super::pure::TyPureBuiltinData::Map));
71        &self.casters[0]
72    }
73
74    pub fn map_val_caster(&self) -> &FieldCaster<'vir> {
75        assert!(matches!(self.pure, super::pure::TyPureBuiltinData::Map));
76        &self.casters[1]
77    }
78}

148impl TaskEncoder for TyUsePureEnc {
149    task_encoder::encoder_cache!(TyUsePureEnc);

151    type TaskDescription<'vir> = super::RustTyDecomposition<'vir>;
152
153    type OutputRef<'vir> = TyUsePureRef<'vir>;
154    type OutputFullDependency<'vir> = TyUsePure<'vir>;
155
156    type TaskKey<'tcx> = Self::TaskDescription<'tcx>;
157
158    type EncodingError = ();
159
160    const ENCODER_NAME: &'static str = "pure type encoder";
161
162    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
163        *task
164    }
165
166    fn do_encode_full<'vir>(
167        task_key: &Self::TaskKey<'vir>,
168        deps: &mut task_encoder::TaskEncoderDependencies<'vir, Self>,
169    ) -> EncodeFullResult<'vir, Self> {
170        let ty_pure_ref = deps.require_ref::<TyPureEnc>(task_key.ty)?;
171        let args = deps.require_dep::<GArgsTyEnc>(task_key.args)?;
172        let snapshot = ty_pure_ref.snapshot;
173        let inner = TyUsePureRef {
174            args,
175            snapshot,
176            ty_pure_ref,
177        };
178        deps.emit_output_ref(*task_key, inner)?;
179
180        let ty_pure = deps.require_dep::<TyPureEnc>(task_key.ty)?;
181        let ty = task_key.ty.zip(ty_pure);
182        let mut walker = TyUsePureWalker::new(deps, task_key.args)?;
183        let specifics = walker.encode_ty(ty)?;
184        let ty_use_pure = TyData::new(inner, specifics);
185        Ok(((), ty_use_pure.alloc()))
186    }
187
188    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
189        TyPureEnc::emit_outputs(program)
190    }

199impl<'a, 'vir> TyUsePureWalker<'a, 'vir> {
200    fn new(
201        deps: &'a mut task_encoder::TaskEncoderDependencies<'vir, TyUsePureEnc>,
202        args: GArgs<'vir>,
203    ) -> EncResult<'vir, Self> {
204        let args_t = deps.require_dep::<GArgsTyEnc>(args)?;
205        Ok(TyUsePureWalker { deps, args_t, args })
206    }
207
208    fn encode_ty(
209        &mut self,
210        ty: TyData<'vir, (RustTyDatas, PureTyDatas)>,
211    ) -> EncResult<'vir, TySpecifics<'vir, UsePureTyDatas>> {
212        let specifics = match &ty.specifics {
213            TySpecifics::Param(data) => {
214                let _: () = *data.1;
215                TySpecifics::mk_param(())
216            }
217            TySpecifics::Opaque(data) => TySpecifics::mk_opaque(*data.1),
218            TySpecifics::Raw((data, raw_domain)) => {
219                let metadata_caster = self.encode_normalized(data.metadata, ty.0.params)?;
220                TySpecifics::mk_raw(TyUsePureRaw {
221                    metadata_caster,
222                    pure: **raw_domain,
223                })
224            }
225            TySpecifics::Primitive(data) => TySpecifics::mk_primitive(*data.1),
226            TySpecifics::ImmRef((data, ref_domain)) => {
227                let referent_caster = self.encode_normalized(data.referent, ty.0.params)?;
228                let metadata_caster = self.encode_normalized(data.metadata, ty.0.params)?;
229                TySpecifics::mk_immref(TyUsePureImmRef {
230                    referent_caster,
231                    metadata_caster,
232                    pure: **ref_domain,
233                })
234            }
235            TySpecifics::MutRef((data, ref_domain)) => {
236                let referent_caster = self.encode_normalized(data.referent, ty.0.params)?;
237                let metadata_caster = self.encode_normalized(data.metadata, ty.0.params)?;
238                TySpecifics::mk_mutref(TyUsePureMutRef {
239                    referent_caster,
240                    metadata_caster,
241                    pure: **ref_domain,
242                })
243            }
244            TySpecifics::ArrayLike(data) => {
245                TySpecifics::ArrayLike(self.encode_array(data, ty.0.params)?)
246            }
247            TySpecifics::StructLike(data) => {
248                let box_data = match ty.0.special {
249                    RustTySpecial::Box => Some(TyUsePureBoxData {
250                        metadata_caster: self
251                            .encode_normalized(ty.box_metadata_ty(), ty.0.params)?,
252                    }),
253                    RustTySpecial::None => None,
254                };
255                TySpecifics::StructLike(self.encode_structlike(data, ty.0.params, box_data)?)
256            }
257            TySpecifics::EnumLike(data) => {
258                TySpecifics::EnumLike(self.encode_enumlike(data, ty.0.params)?)
259            }
260            TySpecifics::Builtin(data) => {
261                use super::rust_ty::RustBuiltinData;
262                let no_cast = GArgCaster::NoCast;
263                let casters = match data.0 {
264                    RustBuiltinData::Int | RustBuiltinData::Real => [no_cast, no_cast],
265                    RustBuiltinData::Set(elem)
266                    | RustBuiltinData::Multiset(elem)
267                    | RustBuiltinData::Seq(elem) => {
268                        [self.encode_normalized(*elem, ty.0.params)?, no_cast]
269                    }
270                    RustBuiltinData::Map(key, val) => [
271                        self.encode_normalized(*key, ty.0.params)?,
272                        self.encode_normalized(*val, ty.0.params)?,
273                    ],
274                };
275                TySpecifics::mk_builtin(TyUsePureBuiltinData {
276                    pure: *data.1,
277                    casters,
278                })
279            }
280        };
281        Ok(specifics)
282    }
283
284    fn encode_normalized(
285        &mut self,
286        inner: LazyRustTy<'vir>,
287        params: GParams<'vir>,
288    ) -> EncResult<'vir, FieldCaster<'vir>> {
289        let normalized = inner.decompose_compare_normalize(params, self.args);
290        self.deps.require_dep::<GArgsCastEnc<Pure>>(normalized)
291    }
292
293    fn encode_array(
294        &mut self,
295        data: &ArrayData<'vir, (RustTyDatas, PureTyDatas)>,
296        params: GParams<'vir>,
297    ) -> EncResult<'vir, ArrayData<'vir, UsePureTyDatas>> {
298        let caster = self.encode_normalized(*data.0, params)?;
299        let slice = data.slice;
300        let data = TyUsePureArrayData {
301            caster,
302            args: self.args_t,
303            pure: *data.data.1,
304        };
305        Ok(ArrayData::new(data, slice))
306    }
307
308    fn encode_structlike(
309        &mut self,
310        data: &StructData<'vir, (RustTyDatas, PureTyDatas)>,
311        params: GParams<'vir>,
312        box_data: Option<TyUsePureBoxData<'vir>>,
313    ) -> EncResult<'vir, StructData<'vir, UsePureTyDatas>> {
314        let fields = data
315            .fields
316            .iter()
317            .map(|field| {
318                let caster = self.encode_normalized(field.0.ty(), params)?;
319                Ok(TyUsePureField {
320                    caster,
321                    args: self.args_t,
322                    pure: *field.1,
323                })
324            })
325            .collect::<EncResult<'vir, Vec<_>>>()?;
326        let data = TyUsePureStructData {
327            args: self.args_t,
328            pure: *data.1,
329            box_data,
330        };
331        Ok(StructData::new(data, fields))
332    }
333
334    fn encode_enumlike(
335        &mut self,
336        data: &EnumData<'vir, (RustTyDatas, PureTyDatas)>,
337        params: GParams<'vir>,
338    ) -> EncResult<'vir, EnumData<'vir, UsePureTyDatas>> {
339        let variants = data
340            .variants
341            .iter()
342            .map(|variant| {
343                let structlike = self.encode_structlike(&variant.inner, params, None)?;
344                Ok(VariantData::new(*variant.1, structlike))
345            })
346            .collect::<EncResult<'vir, Vec<_>>>()?;
347        Ok(EnumData::new(*data.1, variants))
348    }
349}

357impl<'vir> TyData<'vir, UsePureTyDatas> {
358    /// The snapshot of a zero-sized type such as `()`: all values are equal, so
359    /// it is built with the regular (zero-field) constructor. Returns `None`
360    /// when the type is not a fieldless struct-like, leaving error handling to
361    /// the caller.
362    pub fn zst_to_snap<Curr, Next>(&self) -> Option<vir::ExprGenCSnap<'vir, Curr, Next>> {
363        match &self.specifics {
364            TySpecifics::StructLike(data) if data.fields.is_empty() => {
365                Some(data.field_snaps_to_snap(Vec::new()))
366            }
367            _ => None,
368        }
369    }
370
371    pub fn metadata_access<Curr, Next>(
372        &self,
373        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
374    ) -> vir::ExprGenSnap<'vir, Curr, Next> {
375        match &self.specifics {
376            TySpecifics::Raw(data) => data.metadata_access(snap),
377            TySpecifics::ImmRef(data) => data.metadata_access(snap),
378            TySpecifics::MutRef(data) => data.metadata_access(snap),
379            // Cannot be reached as per `UnOp::PtrMetadata` description
380            _ => unreachable!("metadata_access called on non-ref type"),
381        }
382    }
383
384    /// Replaces the address of a shared reference snapshot with a dummy value,
385    /// leaving any other snapshot unchanged. Applied to the arguments of a
386    /// pure function call so that the result cannot depend on the identity of
387    /// a shared reference, only on the value it points to.
388    pub fn dummy_ref_address<Curr, Next>(
389        &self,
390        snap: vir::ExprGenSnap<'vir, Curr, Next>,
391    ) -> vir::ExprGenSnap<'vir, Curr, Next> {
392        match &self.specifics {
393            TySpecifics::ImmRef(data) => data.value_snap_of(snap.downcast_ty()).upcast_ty(),
394            _ => snap,
395        }
396    }
397}

399impl<'vir> TyUsePureImmRef<'vir> {
400    pub fn prim_to_snap<Curr, Next>(
401        &self,
402        ref_: vir::ExprGenRef<'vir, Curr, Next>,
403        metadata: vir::ExprGenSnap<'vir, Curr, Next>,
404        inner: vir::ExprGenSnap<'vir, Curr, Next>,
405    ) -> vir::ExprGenCSnap<'vir, Curr, Next> {
406        let metadata = self.metadata_caster.cast_to_callee_ctx(metadata);
407        let inner = self.referent_caster.cast_to_callee_ctx(inner);
408        self.pure.prim_to_snap.call()(ref_, metadata.downcast_ty(), inner.downcast_ty())
409    }
410
411    pub fn addr_access<Curr, Next>(
412        &self,
413        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
414    ) -> vir::ExprGenRef<'vir, Curr, Next> {
415        self.pure.deref_access.call()(snap)
416    }
417
418    pub fn metadata_access<Curr, Next>(
419        &self,
420        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
421    ) -> vir::ExprGenSnap<'vir, Curr, Next> {
422        let metadata = self.pure.metadata_access.call()(snap);
423        self.metadata_caster
424            .cast_to_caller_ctx(metadata.upcast_ty())
425    }
426
427    /// The snapshot with the address replaced by `null`.
428    pub fn value_snap_of<Curr, Next>(
429        &self,
430        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
431    ) -> vir::ExprGenCSnap<'vir, Curr, Next> {
432        vir::with_vcx(|vcx| {
433            self.pure.prim_to_snap.call()(
434                vcx.mk_null().lazy(),
435                self.pure.metadata_access.call()(snap),
436                self.pure.value_access.call()(snap),
437            )
438        })
439    }
440
441    pub fn value_access<Curr, Next>(
442        &self,
443        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
444    ) -> vir::ExprGenSnap<'vir, Curr, Next> {
445        let value = self.pure.value_access.call()(snap);
446        self.referent_caster.cast_to_caller_ctx(value.upcast_ty())
447    }
448}

568impl<'vir> TyUsePureStruct<'vir> {
569    /// The `Unique` field's snapshot of a `Box`, where its pointer lives.
570    #[track_caller]
571    fn box_unique_snap<Curr, Next>(
572        &self,
573        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
574    ) -> vir::ExprGenCSnap<'vir, Curr, Next> {
575        assert!(self.data.box_data.is_some(), "expected box");
576        self[abi::FieldIdx::ZERO].read(snap).downcast_ty()
577    }
578
579    /// Function to access the address of the boxed `T` (read out of the raw
580    /// pointer's snapshot).
581    #[track_caller]
582    pub fn box_address_access<Curr, Next>(
583        &self,
584        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
585    ) -> vir::ExprGenRef<'vir, Curr, Next> {
586        assert!(self.data.box_data.is_some(), "expected box");
587        let TyPureFieldRef::Dynamic(address_access) =
588            self.fields.last().unwrap().pure.ref_to_field_ref
589        else {
590            unreachable!()
591        };
592        address_access.call()(self.box_unique_snap(snap))
593    }
594
595    /// Function to access the pointer metadata of a `Box`.
596    #[track_caller]
597    pub fn box_metadata_access<Curr, Next>(
598        &self,
599        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
600    ) -> vir::ExprGenSnap<'vir, Curr, Next> {
601        let box_data = self.data.pure.box_data.expect("expected box");
602        let metadata = box_data.metadata_access.call()(self.box_unique_snap(snap));
603        self.data
604            .box_data
605            .expect("expected box")
606            .metadata_caster
607            .cast_to_caller_ctx(metadata.upcast_ty())
608    }
609
610    /// Function to access the value of the boxed `T` (the box's last field).
611    #[track_caller]
612    pub fn box_value_access<Curr, Next>(
613        &self,
614        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
615    ) -> vir::ExprGenSnap<'vir, Curr, Next> {
616        assert!(self.data.box_data.is_some(), "expected box");
617        self.fields.last().unwrap().read(snap)
618    }
619
620    /// The snapshot of a `Box` (given as `snap`) with the same address and
621    /// other fields, but new pointer metadata and value; used for unsize
622    /// coercions. `metadata` and `value` are given in this box's context.
623    #[track_caller]
624    pub fn box_unsize_snap<Curr, Next>(
625        &self,
626        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
627        metadata: vir::ExprGenSnap<'vir, Curr, Next>,
628        value: vir::ExprGenSnap<'vir, Curr, Next>,
629    ) -> vir::ExprGenCSnap<'vir, Curr, Next> {
630        let box_data = self.data.box_data.expect("expected box");
631        // The `Unique` around the same address with the new metadata.
632        let metadata = box_data.metadata_caster.cast_to_callee_ctx(metadata);
633        let unique = self.data.pure.box_data.unwrap().mk_unique.call()(
634            self.box_unique_snap(snap),
635            self.box_address_access(snap),
636            metadata.downcast_ty(),
637        );
638        // The box itself: the new `Unique`, the other fields, the new value.
639        let mut snaps = vec![unique.upcast_ty()];
640        snaps.extend(
641            self.fields[1..self.fields.len() - 1]
642                .iter()
643                .map(|f| f.read(snap)),
644        );
645        snaps.push(value);
646        self.field_snaps_to_snap(snaps)
647    }
648
649    pub fn field_snaps_to_snap<Curr, Next>(
650        &self,
651        mut snaps: Vec<vir::ExprGenSnap<'vir, Curr, Next>>,
652    ) -> vir::ExprGenCSnap<'vir, Curr, Next> {
653        assert_eq!(snaps.len(), self.fields.len());
654        for (snap, field) in snaps.iter_mut().zip(&self.fields) {
655            *snap = field.caster.cast_to_callee_ctx(*snap);
656        }
657        self.pure.field_snaps_to_snap.call()(&snaps)
658    }
659}

661impl<'vir> TyUsePureField<'vir> {
662    pub fn read<Curr, Next>(
663        &self,
664        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
665    ) -> vir::ExprGenSnap<'vir, Curr, Next> {
666        let res = self.pure.read.call()(snap);
667        self.caster.cast_to_caller_ctx(res)
668    }
669
670    /// Get the (Ref) address of a field. Identical to the function one would
671    /// call in `use_impure`.
672    #[track_caller]
673    pub fn field_ref<Curr, Next>(
674        &self,
675        self_ref: vir::ExprGenRef<'vir, Curr, Next>,
676    ) -> vir::ExprGenRef<'vir, Curr, Next> {
677        match self.pure.ref_to_field_ref {
678            TyPureFieldRef::Constant(ref_to_field_ref) => {
679                ref_to_field_ref.call()(self_ref, self.args.get_ty(), self.args.get_const())
680            }
681            TyPureFieldRef::Dynamic(_) => {
682                panic!("field without a pure address (the value of a box)")
683            }
684        }
685    }
686}
