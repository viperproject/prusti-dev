112impl<'vir> TyPurePrimData<'vir> {
113    /// The Viper primitive representation of this type (the argument of
114    /// [`Self::prim_to_snap`]; for floats the raw bits).
115    pub fn prim_type(&self) -> vir::TypePrim<'vir> {
116        match &self.kind {
117            TyPurePrimDataKind::Bool => vir::TYPE_BOOL.upcast_ty(),
118            TyPurePrimDataKind::Int(_) | TyPurePrimDataKind::Float(_) => vir::TYPE_INT.upcast_ty(),
119        }
120    }
121
122    /// Constructs the snapshot from the Viper primitive value.
123    pub fn prim_to_snap<Curr: 'vir, Next: 'vir>(
124        &self,
125        prim: vir::ExprGenPrim<'vir, Curr, Next>,
126    ) -> vir::ExprGenCSnap<'vir, Curr, Next> {
127        match &self.kind {
128            TyPurePrimDataKind::Bool => prim.downcast_ty::<vir::Bool>().upcast_ty(),
129            TyPurePrimDataKind::Int(int) => int.prim_to_snap.call()(prim),
130            TyPurePrimDataKind::Float(float) => float.prim_to_snap.call()(prim),
131        }
132    }
133
134    /// Extracts the Viper primitive value from the snapshot.
135    pub fn snap_to_prim<Curr: 'vir, Next: 'vir>(
136        &self,
137        snap: vir::ExprGenCSnap<'vir, Curr, Next>,
138    ) -> vir::ExprGenPrim<'vir, Curr, Next> {
139        match &self.kind {
140            TyPurePrimDataKind::Bool => snap.downcast_ty::<vir::Bool>().upcast_ty(),
141            TyPurePrimDataKind::Int(int) => int.snap_to_prim.call()(snap),
142            TyPurePrimDataKind::Float(_) => panic!("float snapshots have no primitive value"),
143        }
144    }
145}

268impl TaskEncoder for TyPureEnc {
269    task_encoder::encoder_cache!(TyPureEnc);
270    const ENCODER_NAME: &'static str = "type pure encoder";
271    type TaskDescription<'vir> = RustTy<'vir>;
272
273    type OutputRef<'vir> = TyPureRef<'vir>;
274    type OutputFullDependency<'vir> = TyPure<'vir>;
275
276    /// Nothing is encoded here for generic Param types: the `s_Param` adt is
277    /// emitted by `CastersEnc<Pure>`, which knows the variants (one per
278    /// generic cast pair encoded in the program).
279    type OutputFullLocal<'vir> = Option<TyPureEncLocal<'vir>>;
280
281    type EncodingError = TyPureEncError;
282
283    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
284        *task
285    }
286
287    fn do_encode_full<'vir>(
288        task_key: &Self::TaskKey<'vir>,
289        deps: &mut TaskEncoderDependencies<'vir, Self>,
290    ) -> EncodeFullResult<'vir, Self> {
291        vir::with_vcx(|vcx| {
292            let mut builder = TyPureBuilder::new(deps, vcx, task_key);
293            let output_ref = builder.output_ref();
294            deps.emit_output_ref(*task_key, output_ref)?;
295
296            let specifics = match &task_key.specifics {
297                TySpecifics::Param(param) => {
298                    // No domain for generic params: the `s_Param` adt is
299                    // emitted by `CastersEnc<Pure>` (one variant per generic
300                    // cast pair).
301                    TySpecifics::Param(super::kinds::param::ty_pure(param, deps, &mut builder)?)
302                }
303                TySpecifics::Opaque(opaque) => {
304                    let builder = builder.set_domain_builder();
305                    TySpecifics::Opaque(super::kinds::opaque::ty_pure(opaque, deps, builder)?)
306                }
307                TySpecifics::ArrayLike(array) => {
308                    let builder = builder.set_domain_builder();
309                    TySpecifics::ArrayLike(super::kinds::arraylike::ty_pure(array, deps, builder)?)
310                }
311                TySpecifics::Primitive(prim) => TySpecifics::Primitive(
312                    super::kinds::primitive::ty_pure(vcx, prim, deps, &mut builder)?,
313                ),
314                TySpecifics::ImmRef(immref) => {
315                    let builder = builder.set_adt_builder();
316                    TySpecifics::ImmRef(super::kinds::immref::ty_pure(
317                        task_key, immref, deps, builder,
318                    )?)
319                }
320                TySpecifics::MutRef(mutref) => {
321                    let builder = builder.set_adt_builder();
322                    TySpecifics::MutRef(super::kinds::mutref::ty_pure(
323                        task_key, mutref, deps, builder,
324                    )?)
325                }
326                TySpecifics::Raw(raw) => {
327                    let builder = builder.set_adt_builder();
328                    TySpecifics::Raw(super::kinds::raw::ty_pure(task_key, raw, deps, builder)?)
329                }
330                TySpecifics::StructLike(structlike) => {
331                    let builder = builder.set_adt_builder();
332                    TySpecifics::StructLike(super::kinds::structlike::ty_pure(
333                        task_key, structlike, deps, builder,
334                    )?)
335                }
336                TySpecifics::EnumLike(enumlike) => {
337                    let builder = builder.set_adt_builder();
338                    TySpecifics::EnumLike(super::kinds::enumlike::ty_pure(
339                        task_key, enumlike, deps, builder,
340                    )?)
341                }
342                TySpecifics::Builtin(builtin) => {
343                    TySpecifics::Builtin(super::kinds::builtin::ty_pure(builtin, &mut builder)?)
344                }
345            };
346            let output = TyData::new(output_ref, specifics).alloc();
347            Ok((Some(builder.build()), output))
348        })
349    }
350
351    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
352        for output in Self::all_outputs_local_no_errors(program) {
353            let Some(output) = output else {
354                continue;
355            };
356            program.add_function(output.unreachable_to_snap);
357            for function in output.functions {
358                program.add_function(function);
359            }
360            match output.kind {
361                TyPureEncLocalKind::Domain {
362                    domain,
363                    literal_inverse,
364                } => {
365                    program.add_domain(domain);
366                    if let Some(literal_inverse) = literal_inverse {
367                        program.add_literal_inverse(literal_inverse);
368                    }
369                }
370                TyPureEncLocalKind::Adt { adt, discr_fn } => {
371                    program.add_adt(adt);
372                    if let Some(discr_fn) = discr_fn {
373                        program.add_function(discr_fn);
374                    }
375                }
376                TyPureEncLocalKind::None => {}
377            }
378        }
379    }
380}

461impl<'vir> TyPureBuilder<'vir> {
462    pub(crate) fn new<E: TaskEncoder>(
463        deps: &mut TaskEncoderDependencies<'vir, E>,
464        vcx: &'vir vir::VirCtxt<'vir>,
465        ty: RustTy<'vir>,
466    ) -> Self {
467        let params = deps.require_dep::<GenericParamsEnc>(ty.params).unwrap();
468        let name = vir::ViperIdent::new(vir::vir_format!(vcx, "s_{}", ty.name()));
469        // The `Int`/`Real` builtins and Rust's `bool` are represented directly
470        // by the native Viper `Int`/`Perm`/`Bool` types; nothing is emitted
471        // for them.
472        let self_type = match &ty.specifics {
473            TySpecifics::Builtin(RustBuiltinData::Int) => vir::TYPE_INT.upcast_ty(),
474            TySpecifics::Builtin(RustBuiltinData::Real) => vir::TYPE_PERM.upcast_ty(),
475            // The collection builtins are encoded at their most generic
476            // instantiation, so the elements are generic snapshots.
477            TySpecifics::Builtin(RustBuiltinData::Set(inner)) => {
478                assert!(inner.decompose(ty.params).ty.specifics.is_param());
479                vcx.mk_ty_set(vir::TYPE_PSNAP).upcast_ty()
480            }
481            TySpecifics::Builtin(RustBuiltinData::Multiset(inner)) => {
482                assert!(inner.decompose(ty.params).ty.specifics.is_param());
483                vcx.mk_ty_multiset(vir::TYPE_PSNAP).upcast_ty()
484            }
485            TySpecifics::Builtin(RustBuiltinData::Seq(inner)) => {
486                assert!(inner.decompose(ty.params).ty.specifics.is_param());
487                vcx.mk_ty_seq(vir::TYPE_PSNAP).upcast_ty()
488            }
489            TySpecifics::Builtin(RustBuiltinData::Map(inner1, inner2)) => {
490                assert!(inner1.decompose(ty.params).ty.specifics.is_param());
491                assert!(inner2.decompose(ty.params).ty.specifics.is_param());
492                vcx.mk_ty_map(vir::TYPE_PSNAP, vir::TYPE_PSNAP).upcast_ty()
493            }
494            TySpecifics::Primitive(prim) if prim.is_bool() => vir::TYPE_BOOL.upcast_ty(),
495            _ => DomainIdnSnap::new(name, 0)(),
496        };
497        let unreachable_to_snap = FunctionIdn::new(
498            vir::ViperIdent::new(vir::vir_format!(vcx, "{name}_unreachable")),
499            (params.ty_args(), params.const_args()),
500            self_type,
501        );
502        TyPureBuilder {
503            vcx,
504            name,
505            self_type,
506            unreachable_to_snap,
507            params,
508            data: BuilderData::None,
509        }
510    }

549    pub(crate) fn build(self) -> TyPureEncLocal<'vir> {
550        vir::with_vcx(|vcx| {
551            let unreachable_to_snap = vcx.mk_function(
552                self.unreachable_to_snap,
553                (self.params.ty_decls(), self.params.const_decls()),
554                &[],
555                vcx.alloc_array(&[vcx.mk_bool::<false>()]),
556                None,
557                None,
558            );
559            let functions = match &self.data {
560                BuilderData::Adt(data) => data.functions.as_slice(),
561                _ => &[],
562            };
563            let functions = vcx.alloc_slice(functions);
564            let kind = self.build_kind();
565            TyPureEncLocal {
566                unreachable_to_snap,
567                kind,
568                functions,
569            }
570        })
571    }
572
573    fn build_kind(self) -> TyPureEncLocalKind<'vir> {
574        match self.data {
575            BuilderData::Domain(data) => {
576                let domain = self.vcx.mk_domain(
577                    self.name,
578                    &[],
579                    self.vcx.alloc_slice(data.axioms.as_slice()),
580                    self.vcx.alloc_slice(data.functions.as_slice()),
581                    data.interpretation,
582                );
583                TyPureEncLocalKind::Domain {
584                    domain,
585                    literal_inverse: data.literal_inverse,
586                }
587            }
588            BuilderData::Adt(data) => {
589                let adt = self.vcx.mk_adt(
590                    self.name,
591                    &[],
592                    self.vcx.alloc_slice(data.constructors.as_slice()),
593                );
594                let discr_fn = data.discr_fn.map(|df| {
595                    let DiscrFnBuilder::Built(df) = df else {
596                        panic!("discriminant function not built");
597                    };
598                    df
599                });
600                TyPureEncLocalKind::Adt { adt, discr_fn }
601            }
602            // Natively-represented types (`Int`/`Real`/`bool`) emit nothing.
603            BuilderData::None => TyPureEncLocalKind::None,
604        }
605    }
606}

608impl<'vir> AdtBuilder<'vir> {
609    pub(crate) fn constructor<A: vir::Arity>(
610        &mut self,
611        prefix: &str,
612        fields: A::Tys<'vir>,
613        discr: Option<vir::ExprCSnap<'vir>>,
614    ) -> (
615        FunctionIdn<'vir, A, vir::CSnap>,
616        Vec<vir::AdtDestructor<'vir, vir::CSnap, vir::Dyn>>,
617    ) {
618        let name = format!("{prefix}cons");
619        let self_ty = self.self_type();
620        assert!(
621            self.data().discr_fn.is_none() || discr.is_some(),
622            "discr was passed previously, but now it wasn't"
623        );
624        let self_name = self.name;
625        let name = vir::vir_format!(self.vcx, "{self_name}_{name}",);
626        let locals = self.vcx.alloc_slice(
627            &A::params(fields)
628                .into_iter()
629                .enumerate()
630                .map(|(i, ty)| {
631                    self.vcx
632                        .mk_local_decl(vir::vir_format!(self.vcx, "{self_name}_{prefix}{i}",), ty)
633                })
634                .collect::<Vec<_>>(),
635        );
636        let constructor = self.vcx.mk_adt_constructor(name, locals);
637        self.data().constructors.push(constructor);
638        let ident = FunctionIdn::new(vir::ViperIdent::new(name), fields, self_ty);
639        if let Some(discr) = discr {
640            let df = self.data().discr_fn.take().map(|df| {
641                let DiscrFnBuilder::Building { param, recv, acc } = df else {
642                    panic!("discriminant function was already built");
643                };
644                let acc = self.vcx.mk_ternary_expr(
645                    self.vcx.mk_adt_discriminator_expr(recv, name),
646                    discr,
647                    acc,
648                );
649                DiscrFnBuilder::Building { param, recv, acc }
650            });
651            let df = df.unwrap_or_else(|| {
652                let param = self.vcx.mk_local_decl("self", self_ty);
653                DiscrFnBuilder::Building {
654                    param,
655                    recv: self.vcx.mk_local_ex(param),
656                    acc: discr,
657                }
658            });
659            self.data().discr_fn = Some(df)
660        }
661        (
662            ident,
663            locals
664                .iter()
665                .map(|arg| self.vcx.mk_adt_destructor(arg.name, self_ty, arg.ty))
666                .collect(),
667        )
668    }
669
670    pub(crate) fn build_discr_fn(
671        &mut self,
672        ty: vir::TypeCSnap<'vir>,
673    ) -> vir::FunctionIdn<'vir, vir::CSnap, vir::CSnap> {
674        let self_ty = self.self_type();
675        let param = self.vcx.mk_local_decl("self", self_ty);
676        let ident = FunctionIdn::new(
677            vir::ViperIdent::new(vir::vir_format!(self.vcx, "{}_discr", self.name)),
678            param.ty,
679            ty,
680        );
681        let (expr, posts) = if let Some(df) = self.data().discr_fn {
682            let DiscrFnBuilder::Building { acc, .. } = df else {
683                panic!("discriminant function already built");
684            };
685            (Some(acc), &[][..])
686        } else {
687            // We get here if we didn't add any constructors to the ADT (i.e.
688            // this is an empty enum = uninhabitable type). Viper forbids this
689            // (see silver#693, silver#696), so here we will just add a dummy
690            // constructor that won't actually be used by the encoders. Instead
691            // we encode the uninhabitability by adding an `ensures false` to
692            // the discriminant function.
693            // TODO: https://github.com/Aurel300/prusti-dev/pull/89#discussion_r2263306839
694            let self_name = self.name;
695            let name = vir::vir_format!(self.vcx, "{self_name}_DummyConstructor",);
696            let constructor = self.vcx.mk_adt_constructor::<(), !, vir::Dyn>(name, &[]);
697            self.data().constructors.push(constructor);
698            (None, self.vcx.alloc_slice(&[self.vcx.mk_bool::<false>()]))
699        };
700        let built_fn = self
701            .vcx
702            .mk_function(ident, (param,), &[], posts, None, expr);
703        self.data().discr_fn = Some(DiscrFnBuilder::Built(built_fn));
704        ident
705    }
706
707    #[allow(clippy::too_many_arguments)]
708    pub(crate) fn function<A: Arity, T: CompType>(
709        &mut self,
710        name: &str,
711        args: A::Tys<'vir>,
712        ret: vir::Type<'vir, T>,
713        params: A::Locals<'_, 'vir>,
714        pres: &[vir::ExprBool<'vir>],
715        posts: &[vir::ExprBool<'vir>],
716        expr: Option<vir::Expr<'vir, T>>,
717    ) -> FunctionIdn<'vir, A, T> {
718        let name = vir::vir_format!(self.vcx, "{}_{name}", self.name);
719        let ident = FunctionIdn::new(vir::ViperIdent::new(name), args, ret);
720        let function = self.vcx.mk_function(
721            ident,
722            params,
723            self.vcx.alloc_slice(pres),
724            self.vcx.alloc_slice(posts),
725            None,
726            expr,
727        );
728        self.data().functions.push(function);
729        ident
730    }
731}

796impl<'vir> TyPurePrimData<'vir> {
797    pub fn expr_from_bits(&self, ty: ty::Ty<'vir>, value: u128) -> vir::ExprPrim<'vir> {
798        match &self.kind {
799            TyPurePrimDataKind::Bool => {
800                vir::with_vcx(|vcx| vcx.mk_const_expr(vir::ConstData::Bool(value != 0)))
801            }
802            TyPurePrimDataKind::Int(_) | TyPurePrimDataKind::Float(_) => {
803                let (bit_width, signed) = match ty.kind() {
804                    TyKind::Int(IntTy::Isize) => ((std::mem::size_of::<isize>() * 8) as u64, true),
805                    TyKind::Int(ty) => (ty.bit_width().unwrap(), true),
806                    TyKind::Uint(UintTy::Usize) => {
807                        ((std::mem::size_of::<usize>() * 8) as u64, false)
808                    }
809                    TyKind::Uint(ty) => (ty.bit_width().unwrap(), false),
810                    TyKind::Char => (32, false),
811                    // The float prim_to_snap takes the raw bits as an unsigned integer.
812                    TyKind::Float(..) => (0, false),
813                    kind => unreachable!("{kind:?}"),
814                };
815                let size = abi::Size::from_bits(bit_width);
816                let negative_value = if signed {
817                    let value = size.sign_extend(value);
818                    Some(value).filter(|value| value.is_negative())
819                } else {
820                    None
821                };
822                match negative_value {
823                    Some(value) => vir::with_vcx(|vcx| {
824                        let value = vcx.mk_const_expr(vir::ConstData::Int(value.unsigned_abs()));
825                        vcx.mk_unary_op_expr(vir::UnOpKind::Neg, value)
826                    }),
827                    None => vir::with_vcx(|vcx| vcx.mk_const_expr(vir::ConstData::Int(value))),
828                }
829            }
830        }
831    }
