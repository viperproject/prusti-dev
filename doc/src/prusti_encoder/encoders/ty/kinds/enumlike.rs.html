17pub(crate) fn ty_pure<'vir>(
18    task_key: &TyData<'vir, RustTyDatas>,
19    data: &EnumData<'vir, RustTyDatas>,
20    deps: &mut TaskEncoderDependencies<'vir, TyPureEnc>,
21    builder: &mut AdtBuilder<'vir>,
22) -> Result<EnumData<'vir, PureTyDatas>, EncodeFullError<'vir, TyPureEnc>> {
23    let discr_ty =
24        deps.require_dep::<TyPureEnc>(RustTyDecomposition::from_prim_ty(data.discr).ty)?;
25    let discr_prim = discr_ty.expect_primitive();
26    let discr_ty = discr_ty.snapshot;
27
28    let variants = data
29        .variants
30        .iter()
31        .map(|variant| {
32            let var_idx_num = variant.vid.as_u32();
33            let discr =
34                discr_prim.prim_to_snap(discr_prim.expr_from_bits(data.discr, variant.discr_val));
35
36            let specifics = super::structlike::ty_pure_variant(
37                &format!("{var_idx_num}_"),
38                Some(discr),
39                task_key,
40                &variant.inner,
41                deps,
42                builder,
43            )?;
44
45            Ok(VariantData::new(TyPureVariantData { discr }, specifics))
46        })
47        .collect::<Result<Vec<_>, _>>()?;
48
49    // discriminant can only have the selected values
50    let snap_to_discr_snap = builder.build_discr_fn(discr_ty.downcast_ty());
51
52    Ok(EnumData::new(
53        TyPureEnumData {
54            discr_ty,
55            discr_prim: *discr_prim,
56            snap_to_discr_snap,
57        },
58        variants,
59    ))
60}

62pub(crate) fn ty_impure<'vir>(
63    task_key: &TyData<'vir, (RustTyDatas, PureTyDatas)>,
64    data: &EnumData<'vir, (RustTyDatas, PureTyDatas)>,
65    deps: &mut TaskEncoderDependencies<'vir, TyImpureEnc>,
66    builder: &mut PredicateBuilder<'vir>,
67) -> Result<EnumData<'vir, ImpureTyDatas>, EncodeFullError<'vir, TyImpureEnc>> {
68    let ref_self_decl = builder.ref_self_decl();
69    let ref_self = builder.vcx.mk_local_ex(ref_self_decl);
70
71    // first encode the discriminant's type
72    let task = RustTyDecomposition::from_prim_ty(data.0.discr);
73    let discr_ty_impure = deps.require_dep::<TyUseImpureEnc>(task)?;
74
75    // Ref-to-Ref function for the discriminant field
76    let fdisc_func = builder.function(
77        "field_discr",
78        ref_self_decl.ty,
79        vir::TYPE_REF,
80        (ref_self_decl,),
81        &[],
82        &[vir::expr! { ((ref_self) == (null)) == ((result: Ref) == (null)) }],
83        None,
84    );
85    let ref_disc = fdisc_func(ref_self);
86    let snap_disc = discr_ty_impure.ref_to_snap(ref_disc).downcast_ty();
87
88    let variants = data
89        .variants
90        .iter()
91        .map(|variant| {
92            let var_idx_num = variant.0.vid.as_u32();
93
94            let (inner, variant_pred, variant_snap_expr) =
95                super::structlike::ty_impure_variant(
96                &format!("{var_idx_num}_"),
97                task_key,
98                &variant.inner,
99                deps,
100                builder,
101            )?;
102            let variant_inhabited = builder.vcx.mk_conj(
103                &variant
104                    .inner
105                    .fields
106                    .iter()
107                    .map(|field| {
108                        let ty = field.0.ty().decompose(task_key.0.params);
109                        Ok(deps
110                            .require_ref::<TyUseInhabitedEnc>(ty)?
111                            .inhabited())
112                    })
113                    .collect::<Result<Vec<_>, EncodeFullError<'vir, TyImpureEnc>>>()?,
114            );
115
116            let variant_pred_expr = vir::expr! {
117                (([snap_disc])
118                    == ([variant.1.discr])) ==> (([variant_inhabited])
119                        && ([variant_pred](ref_self, [..[builder.params.ty_exprs()]], [..[builder.params.const_exprs()]])))
120            };
121            let variant_snap_expr = vir::expr! {
122                unfolding ([variant_pred](ref_self, [..[builder.params.ty_exprs()]], [..[builder.params.const_exprs()]])) in (variant_snap_expr)
123            };
124
125            Ok((
126                variant_snap_expr,
127                variant_pred_expr,
128                variant.1.discr,
129                VariantData::new(TyImpureVariantData {
130                    predicate: variant_pred,
131                }, inner)
132            ))
133        })
134        .collect::<Result<Vec<_>, _>>()?;
135
136    // main predicate
137    let variant_predicate = discr_ty_impure.ref_to_pred(builder.vcx, ref_disc, None);
138    let variant_values = builder.vcx.mk_disj(
139        &variants
140            .iter()
141            .map(|variant| vir::expr! { ([snap_disc]) == ([variant.2]) })
142            .collect::<Vec<_>>(),
143    );
144    let variant_predicates = builder
145        .vcx
146        .mk_conj(&variants.iter().map(|v| v.1).collect::<Vec<_>>());
147    builder.mk_predicate(
148        "",
149        Some(vir::expr! {
150            ([variant_predicate])
151            && (([variant_values])
152            && ([variant_predicates]))
153        }),
154    );
155
156    // Ref-to-snap
157    let base =
158        (task_key.1.unreachable_to_snap)(builder.params.ty_exprs(), builder.params.const_exprs())
159            .downcast_ty();
160    let inner = variants.iter().fold(base, |else_, variant| {
161        builder.vcx.mk_ternary_expr(
162            vir::expr! { ([snap_disc]) == ([variant.2]) },
163            variant.0,
164            else_,
165        )
166    });
167    builder.mk_snap_function(Some(inner), &[]);
168
169    Ok(EnumData::new(
170        TyImpureEnumData {
171            discr: fdisc_func,
172            discr_ty: discr_ty_impure,
173        },
174        variants.into_iter().map(|v| v.3).collect::<Vec<_>>(),
175    ))
176}
