1//! Encoding for raw pointer types (`*const T` / `*mut T`).
2//!
3//! Modelled like a reference (see [`super::immref`]) but conservatively: the
4//! snapshot carries the pointer address and the pointer metadata, and the
5//! pointee is left opaque (there is no `value_access` into it, and we do not
6//! require the pointee type to be encoded. This keeps raw pointers to
7//! otherwise-unencodable types, e.g. the panic/formatting machinery, working).
8
9use crate::encoders::{
10    TyUseImpureEnc, TyUsePureEnc,
11    ty::{
12        RustRaw, RustTyDatas,
13        data::TyData,
14        generics::{ParamTypEnc, TyExprEnc},
15        impure::{PredicateBuilder, TyImpureEnc, TyImpureRaw, TyImpureRawData},
16        pure::{AdtBuilder, PureTyDatas, TyPureEnc, TyPureRaw, TyPureRawData},
17    },
18};
19use task_encoder::{EncodeFullError, TaskEncoderDependencies};
20use vir::CastType;
21
22pub(crate) fn ty_pure<'vir>(
23    task_key: &TyData<'vir, RustTyDatas>,
24    data: &RustRaw<'vir>,
25    deps: &mut TaskEncoderDependencies<'vir, TyPureEnc>,
26    builder: &mut AdtBuilder<'vir>,
27) -> Result<TyPureRaw<'vir>, EncodeFullError<'vir, TyPureEnc>> {
28    let ty = data.metadata.decompose(task_key.params);
29    let metadata = deps.require_ref::<TyUsePureEnc>(ty)?.snapshot.downcast_ty();
30
31    // Unlike a reference, there is no value field for the (opaque) pointee.
32    let (field_snaps_to_snap, field_access) =
33        builder.constructor("", (vir::TYPE_REF, metadata), None);
34
35    Ok(TyPureRawData {
36        prim_to_snap: field_snaps_to_snap,
37        address_access: field_access[0].downcast_ty(),
38        metadata_access: field_access[1].downcast_ty(),
39    })
40}

42pub(crate) fn ty_impure<'vir>(
43    task_key: &TyData<'vir, (RustTyDatas, PureTyDatas)>,
44    data: &(&RustRaw<'vir>, &TyPureRaw<'vir>),
45    deps: &mut TaskEncoderDependencies<'vir, TyImpureEnc>,
46    builder: &mut PredicateBuilder<'vir>,
47) -> Result<TyImpureRaw<'vir>, EncodeFullError<'vir, TyImpureEnc>> {
48    let metadata_type = data.0.metadata.decompose(task_key.0.params);
49    deps.require_dep::<TyUseImpureEnc>(metadata_type)?;
50    // The pointee is opaque; we deliberately do not require its encoding.
51
52    let ref_self_decl = builder.ref_self_decl();
53    let ref_self = builder.vcx.mk_local_ex(ref_self_decl);
54    let vcx = builder.vcx;
55
56    let metadata_snap = deps
57        .require_ref::<TyUsePureEnc>(metadata_type)?
58        .snapshot
59        .downcast_ty();
60
61    // fields: the address and the pointer metadata, which is all a raw
62    // pointer's snapshot holds.
63    let addr_field = builder.field("addr", vir::TYPE_REF);
64    let metadata_field = builder.field("metadata", metadata_snap);
65
66    let metadata_ty = deps.require_dep::<TyExprEnc>(metadata_type)?;
67    let typ = deps.require_dep::<ParamTypEnc>(())?.typ;
68    let metadata_typ = |metadata| vcx.mk_eq_expr(typ(metadata), metadata_ty);
69
70    // main predicate
71    builder.mk_predicate(
72        "",
73        Some(vcx.mk_conj(&[
74            vir::expr! { acc((ref_self).[addr_field]) },
75            vir::expr! { acc((ref_self).[metadata_field]) },
76            metadata_typ(vir::expr! { [metadata_field](ref_self) }),
77        ])),
78    );
79
80    // Ref-to-snap
81    builder.mk_snap_function(
82        Some(data.1.prim_to_snap.call()(
83            vir::expr! { [addr_field](ref_self) },
84            vir::expr! { [metadata_field](ref_self) },
85        )),
86        &[metadata_typ(data.1.metadata_access.call()(
87            vcx.mk_result(builder.csnap_type()),
88        ))],
89    );
90
91    Ok(TyImpureRawData {})
