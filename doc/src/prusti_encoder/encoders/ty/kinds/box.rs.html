18/// The address/metadata of the boxed `T` (the value field of a `Box`), read
19/// out of the raw pointer's snapshot down the `Unique`/`NonNull` chain, and
20/// the reverse: the `Unique` snapshot rebuilt around a new pointer. All are
21/// functions of the `Unique` field's snapshot: that is where the pointer
22/// lives, and (unlike the whole box) it is what the impure side has a
23/// snapshot of when the address is needed. Returns the box data and the
24/// address (of the value field).
25pub(super) fn mk_pure_box_data<'vir>(
26    task_key: &TyData<'vir, RustTyDatas>,
27    data: &StructData<'vir, RustTyDatas>,
28    deps: &mut TaskEncoderDependencies<'vir, TyPureEnc>,
29    builder: &mut AdtBuilder<'vir>,
30) -> Result<
31    (
32        TyPureBoxData<'vir>,
33        vir::FunctionIdn<'vir, vir::CSnap, vir::Ref>,
34    ),
35    EncodeFullError<'vir, TyPureEnc>,
36> {
37    let first = |ty: &RustTyDecomposition<'vir>| {
38        ty.ty.expect_structlike().fields[abi::FieldIdx::ZERO.as_usize()]
39            .decompose_context(ty.ty.params, ty.args)
40    };
41    let unique_ty = data.fields[0].ty().decompose(task_key.params);
42    let nonnull_ty = first(&unique_ty);
43    let raw_ty = first(&nonnull_ty);
44    let unique_use = deps.require_dep::<TyUsePureEnc>(unique_ty)?;
45    let nonnull_use = deps.require_dep::<TyUsePureEnc>(nonnull_ty)?;
46    let raw_use = deps.require_dep::<TyUsePureEnc>(raw_ty)?;
47
48    let unique_decl = builder
49        .vcx
50        .mk_local_decl("ptr", unique_use.snapshot.downcast_ty::<vir::CSnap>());
51    let unique_snap = builder.vcx.mk_local_ex(unique_decl);
52    let unique = unique_use.expect_structlike();
53    let nonnull = nonnull_use.expect_structlike();
54    let raw = raw_use.expect_raw();
55    let nonnull_snap = unique[abi::FieldIdx::ZERO].read(unique_snap).downcast_ty();
56    let raw_snap = nonnull[abi::FieldIdx::ZERO]
57        .read(nonnull_snap)
58        .downcast_ty();
59
60    let address_access = builder.function(
61        "address",
62        unique_decl.ty(),
63        vir::TYPE_REF,
64        (unique_decl,),
65        &[],
66        &[],
67        Some(raw.address_access(raw_snap)),
68    );
69    let metadata_access = builder.function(
70        "metadata",
71        unique_decl.ty(),
72        vir::TYPE_PSNAP,
73        (unique_decl,),
74        &[],
75        &[],
76        Some(raw.metadata_access(raw_snap).downcast_ty()),
77    );
78
79    // The `Unique` with its raw pointer replaced (its other fields kept).
80    let address_decl = builder.vcx.mk_local_decl("address", vir::TYPE_REF);
81    let metadata_decl = builder.vcx.mk_local_decl("metadata", vir::TYPE_PSNAP);
82    let new_raw = raw.prim_to_snap(
83        builder.vcx.mk_local_ex(address_decl),
84        builder.vcx.mk_local_ex(metadata_decl).upcast_ty(),
85    );
86    let new_nonnull = nonnull.field_snaps_to_snap(vec![new_raw.upcast_ty()]);
87    let mut unique_snaps = vec![new_nonnull.upcast_ty()];
88    unique_snaps.extend(unique.fields[1..].iter().map(|f| f.read(unique_snap)));
89    let mk_unique = builder.function(
90        "mk_unique",
91        (unique_decl.ty(), address_decl.ty(), metadata_decl.ty()),
92        unique_decl.ty(),
93        (unique_decl, address_decl, metadata_decl),
94        &[],
95        &[],
96        Some(unique.field_snaps_to_snap(unique_snaps)),
97    );
98
99    Ok((
100        TyPureBoxData {
101            metadata_access,
102            mk_unique,
103        },
104        address_access,
105    ))
106}

108/// The heap-dependent `address`/`metadata` functions of a `Box`: the pure
109/// accessors applied to the `Unique` field's snapshot, read out of its
110/// (folded) predicate. Returns the box data and the `address` accessor.
111pub(super) fn mk_impure_box_data<'vir>(
112    data: &StructData<'vir, (RustTyDatas, PureTyDatas)>,
113    fields: &[TyUseImpure<'vir>],
114    builder: &mut PredicateBuilder<'vir>,
115) -> (
116    TyImpureBoxData<'vir>,
117    vir::FunctionIdn<'vir, (vir::Ref, vir::ManyTyVal, vir::ManyCSnap), vir::Ref>,
118) {
119    let ref_decl = builder.ref_self_decl();
120    let ref_self = builder.vcx.mk_local_ex(ref_decl);
121    let TyPureFieldRef::Constant(unique_field_ref) = data.fields[0].1.ref_to_field_ref else {
122        unreachable!()
123    };
124    let b_params = &builder.params;
125    let unique_ref = unique_field_ref(ref_self, b_params.ty_exprs(), b_params.const_exprs());
126    let unique_snap = fields[0].ref_to_snap(unique_ref).downcast_ty();
127    let unique_pred = fields[0].ref_to_pred(builder.vcx, unique_ref, None);
128    let pure = data.1.box_data.unwrap();
129    let TyPureFieldRef::Dynamic(address_access) = data.fields.last().unwrap().1.ref_to_field_ref
130    else {
131        unreachable!()
132    };
133    let args = (ref_decl.ty(), b_params.ty_args(), b_params.const_args());
134    let params = (ref_decl, b_params.ty_decls(), b_params.const_decls());
135    let address = builder.inner.function(
136        "address",
137        args,
138        vir::TYPE_REF,
139        params,
140        &[unique_pred],
141        &[],
142        Some(address_access.call()(unique_snap)),
143    );
144    let metadata = builder.inner.function(
145        "metadata",
146        args,
147        vir::TYPE_PSNAP,
148        params,
149        &[unique_pred],
150        &[],
151        Some(pure.metadata_access.call()(unique_snap)),
152    );
153    (TyImpureBoxData { metadata }, address)
154}
