14pub(crate) fn ty_pure<'vir>(
15    vcx: &'vir VirCtxt<'vir>,
16    data: &RustPrimitive<'vir>,
17    deps: &mut TaskEncoderDependencies<'vir, TyPureEnc>,
18    builder: &mut TyPureBuilder<'vir>,
19) -> Result<TyPurePrimitive<'vir>, EncodeFullError<'vir, TyPureEnc>> {
20    let ty = data;
21    let ty_kind = ty.kind();
22
23    if matches!(ty_kind, ty::TyKind::Bool) {
24        // Represented directly by the native Viper `Bool` type (see
25        // `TyPureBuilder::new`): the primitive and the snapshot coincide, so
26        // the conversions are casts and there is nothing to emit.
27        return Ok(TyPurePrimData {
28            kind: TyPurePrimDataKind::Bool,
29        });
30    }
31    let builder = builder.set_domain_builder();
32
33    let prim_type = vir::TYPE_INT.upcast_ty();
34    let cons_ident = builder.function("cons", prim_type, builder.self_type());
35
36    let kind = match ty_kind {
37        ty::TyKind::Float(float) => {
38            let data = ty_pure_float(vcx, deps, builder, *float, cons_ident)?;
39            TyPurePrimDataKind::Float(vcx.alloc(data))
40        }
41        ty::TyKind::Char | ty::TyKind::Int(_) | ty::TyKind::Uint(_) => {
42            let value_ident = builder.function("value", builder.self_type(), prim_type);
43
44            builder.axiom("cons", vir::expr! {
45                forall s: [builder.self_type()] :: {[value_ident](s)} ([cons_ident]([value_ident](s))) == (s)
46            });
47
48            // The values of the type, as inclusive ranges: those of a `char`
49            // are the Unicode scalar values, all code points but surrogates.
50            let ranges = match ty_kind {
51                ty::TyKind::Char => vec![
52                    (vcx.mk_uint::<0>(), vcx.mk_uint::<0xD7FF>()),
53                    (vcx.mk_uint::<0xE000>(), vcx.mk_uint::<0x10FFFF>()),
54                ],
55                _ => vec![(vcx.get_min_int(ty_kind), vcx.get_max_int(ty_kind))],
56            };
57            let in_range = |x: vir::ExprInt<'vir>| {
58                let ranges = ranges
59                    .iter()
60                    .map(|&(min, max)| vir::expr! { (([min]) <= ([x])) && (([x]) <= ([max])) })
61                    .collect::<Vec<_>>();
62                vcx.mk_disj(&ranges)
63            };
64            let s_decl = vcx.mk_local_decl("s", builder.self_type());
65            let s = vcx.mk_local_ex(s_decl);
66            let s_value = vir::expr! { ([value_ident](s)) as Int };
67            builder.axiom(
68                "bounds",
69                vir::expr! {
70                    forall [s_decl] :: {[value_ident](s)} [in_range(s_value)]
71                },
72            );
73            let value_decl = vcx.mk_local_decl("value", prim_type);
74            let value = vcx.mk_local_ex(value_decl);
75            builder.axiom(
76                "value",
77                vir::expr! {
78                    forall [value_decl] :: {[cons_ident](value)}
79                        ([in_range(vir::expr! { (value) as Int })])
80                            ==> (([value_ident]([cons_ident](value))) == (value))
81                },
82            );
83            builder.literal_inverse(value_ident, cons_ident, vcx.alloc_slice(&ranges));
84            TyPurePrimDataKind::Int(TyPurePrimDataInt {
85                prim_to_snap: cons_ident,
86                snap_to_prim: value_ident,
87            })
88        }
89        _ => unreachable!(),
90    };
91    Ok(TyPurePrimData { kind })
92}

94pub(crate) fn ty_impure<'vir>(
95    _data: &(&RustPrimitive<'vir>, &TyPurePrimitive<'vir>),
96    _deps: &mut TaskEncoderDependencies<'vir, TyImpureEnc>,
97    builder: &mut PredicateBuilder<'vir>,

103/// A predicate holding a single `val` field with the snapshot value.
104pub(super) fn set_primitive<'vir>(builder: &mut PredicateBuilder<'vir>) {
105    let snap_type = builder.csnap_type();
106
107    let ref_self_decl = builder.ref_self_decl();
108    let ref_self = builder.vcx.mk_local_ex(ref_self_decl);
109
110    // fields
111    let prim_field = builder.field("val", snap_type);
112
113    // main predicate
114    builder.mk_predicate("", Some(vir::expr! { acc((ref_self).[prim_field]) }));
115
116    // Ref-to-snap
117    builder.mk_snap_function(Some(vir::expr! { [prim_field](ref_self) }), &[]);
118}
