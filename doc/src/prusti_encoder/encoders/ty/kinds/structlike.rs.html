28pub(super) fn ty_pure_variant<'vir>(
29    prefix: &str,
30    discr: Option<vir::ExprCSnap<'vir>>,
31    task_key: &TyData<'vir, RustTyDatas>,
32    data: &StructData<'vir, RustTyDatas>,
33    deps: &mut TaskEncoderDependencies<'vir, TyPureEnc>,
34    builder: &mut AdtBuilder<'vir>,
35) -> Result<StructData<'vir, PureTyDatas>, EncodeFullError<'vir, TyPureEnc>> {
36    let field_tys = data
37        .fields
38        .iter()
39        .map(|f| {
40            let ty = f.decompose(task_key.params);
41            Ok(deps.require_ref::<TyUsePureEnc>(ty)?.snapshot)
42        })
43        .collect::<Result<Vec<_>, _>>()?;
44    let field_tys = builder.vcx.alloc_slice(&field_tys);
45    let (field_snaps_to_snap, des) = builder.constructor(prefix, field_tys, discr);
46
47    // For a `Box`, its extras and the address of its (dynamic) value field.
48    let box_data = if let RustTySpecial::Box = task_key.special {
49        Some(super::r#box::mk_pure_box_data(
50            task_key, data, deps, builder,
51        )?)
52    } else {
53        None
54    };
55
56    let ref_self_decl = builder.vcx.mk_local_decl("self", vir::TYPE_REF);
57    let ref_self = builder.vcx.mk_local_ex(ref_self_decl);
58    let params = builder.params.clone();
59
60    assert_eq!(des.len(), data.fields.len());
61    let des = des
62        .iter()
63        .zip(&data.fields)
64        .enumerate()
65        .map(|(idx, (read, field))| {
66            let ref_to_field_ref = match field.address {
67                RustFieldAddress::Constant => {
68                    let args = (ref_self_decl.ty(), params.ty_args(), params.const_args());
69                    let params = (ref_self_decl, params.ty_decls(), params.const_decls());
70                    TyPureFieldRef::Constant(builder.function(
71                        &format!("{prefix}field_{idx}"),
72                        args,
73                        vir::TYPE_REF,
74                        params,
75                        &[],
76                        &[vir::expr! { ((ref_self) == (null)) == ((result: Ref) == (null)) }],
77                        None,
78                    ))
79                }
80                RustFieldAddress::Dynamic => TyPureFieldRef::Dynamic(box_data.unwrap().1),
81            };
82            TyPureFieldData {
83                read: read.downcast_ty(),
84                ref_to_field_ref,
85            }
86        })
87        .collect::<Vec<_>>();
88    Ok(StructData::new(
89        TyPureStructData {
90            field_snaps_to_snap,
91            box_data: box_data.map(|(box_data, _)| box_data),
92        },
93        des,
94    ))
95}

97pub(crate) fn ty_impure<'vir>(
98    task_key: &TyData<'vir, (RustTyDatas, PureTyDatas)>,
99    data: &StructData<'vir, (RustTyDatas, PureTyDatas)>,
100    deps: &mut TaskEncoderDependencies<'vir, TyImpureEnc>,
101    builder: &mut PredicateBuilder<'vir>,
102) -> Result<StructData<'vir, ImpureTyDatas>, EncodeFullError<'vir, TyImpureEnc>> {
103    let (data, _, snap_expr) = ty_impure_variant("", task_key, data, deps, builder)?;
104
105    // Ref-to-snap
106    builder.mk_snap_function(Some(snap_expr), &[]);
107    Ok(data)
108}

116pub(crate) fn ty_impure_variant<'vir>(
117    prefix: &str,
118    task_key: &TyData<'vir, (RustTyDatas, PureTyDatas)>,
119    data: &StructData<'vir, (RustTyDatas, PureTyDatas)>,
120    deps: &mut TaskEncoderDependencies<'vir, TyImpureEnc>,
121    builder: &mut PredicateBuilder<'vir>,
122) -> Result<ImpureVariant<'vir>, EncodeFullError<'vir, TyImpureEnc>> {
123    let fields = data
124        .fields
125        .iter()
126        .map(|f| {
127            let ty = f.0.decompose(task_key.0.params);
128            deps.require_dep::<TyUseImpureEnc>(ty)
129        })
130        .collect::<Result<Vec<_>, _>>()?;
131
132    let ref_self_decl = builder.ref_self_decl();
133    let ref_self = builder.vcx.mk_local_ex(ref_self_decl);
134
135    // For a `Box`, its extras and the (heap-dependent) address of its dynamic
136    // value field. Since the value field comes after the `Unique` field, its
137    // predicate conjunct below is framed.
138    let box_data = if let RustTySpecial::Box = task_key.0.special {
139        Some(super::r#box::mk_impure_box_data(data, &fields, builder))
140    } else {
141        None
142    };
143
144    // Ref-to-Ref function for every field
145    let field_accessors = data
146        .fields
147        .iter()
148        .map(|field| TyImpureFieldData {
149            ref_to_field_ref: match field.1.ref_to_field_ref {
150                TyPureFieldRef::Constant(ref_to_field_ref) => ref_to_field_ref,
151                TyPureFieldRef::Dynamic(_) => box_data.unwrap().1,
152            },
153        })
154        .collect::<Vec<_>>();
155
156    // main variant predicate
157    let mut pred_name = String::new();
158    if !prefix.is_empty() {
159        pred_name = format!("{prefix}owned");
160    }
161    let pred_expr = builder.vcx.mk_conj(
162        &fields
163            .iter()
164            .zip(&field_accessors)
165            .map(|(field, TyImpureFieldData { ref_to_field_ref })| {
166                field.ref_to_pred(
167                    builder.vcx,
168                    ref_to_field_ref(
169                        ref_self,
170                        builder.params.ty_exprs(),
171                        builder.params.const_exprs(),
172                    ),
173                    None,
174                )
175            })
176            .collect::<Vec<_>>(),
177    );
178    let pred_owned = builder.mk_predicate(&pred_name, Some(pred_expr));
179
180    // Ref-to-snap
181    let snap_args: Vec<&'vir vir::ExprGenData<'vir, (), !, vir::Snap>> = fields
182        .iter()
183        .zip(&field_accessors)
184        .map(|(field, TyImpureFieldData { ref_to_field_ref })| {
185            field.ref_to_snap(ref_to_field_ref(
186                ref_self,
187                builder.params.ty_exprs(),
188                builder.params.const_exprs(),
189            ))
190        })
191        .collect::<Vec<_>>();
192    let variant_snap_expr = data.1.field_snaps_to_snap.call()(snap_args.as_slice());
193
194    Ok((
195        StructData::new(
196            TyImpureStructData {
197                box_data: box_data.map(|(box_data, _)| box_data),
198            },
199            field_accessors,
200        ),
201        pred_owned,
202        variant_snap_expr,
203    ))
