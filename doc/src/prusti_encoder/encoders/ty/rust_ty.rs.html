11#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
12pub struct RustTyDecomposition<'tcx> {
13    pub ty: RustTy<'tcx>,
14    pub args: GArgs<'tcx>,
15}
16
17impl<'tcx> RustTyDecomposition<'tcx> {
18    fn new(ty: RustTy<'tcx>, args: GArgs<'tcx>) -> Self {
19        ty.params.check(args.args());
20        RustTyDecomposition { ty, args }
21    }
22
23    /// Decomposes a rustc `ty::Ty` into the core type used to generate a Viper
24    /// domain/predicate and its type arguments (not used for the Viper
25    /// definition). For example, for the function:
26    /// ```no_run
27    /// struct MyStruct<I: Iterator> {
28    ///     field: I::Item
29    /// }
30    /// fn foo<T: Iterator<Item = i32>>(x: MyStruct<T>) { ... }
31    /// ```
32    /// when encoding the argument type, this should be called as
33    /// ```no_run
34    /// let decomp = from_ty(tcx, "MyStruct<T>", "<T: Iterator<Item = i32>>")
35    /// // which yields
36    /// RustTyDecomposition {
37    ///     ty: TyData { params: "<I: Iterator>", specifics: "MyStruct(I::Item)" }
38    ///     args: GArgs { args: "<T>", context: "<T: Iterator<Item = i32>>" }
39    /// }
40    /// ```
41    /// The `ty` field is agnostic of the client's generic arguments while the
42    /// `args` field captures everything from the client's side. Note that we
43    /// guarantee that `decomp.ty.params.len() == decomp.args.len()`.
44    ///
45    /// To recursively encode the struct itself, one should walk the
46    /// `decomp.ty.specifics` and call `RustFieldData::decompose` with
47    /// `decomp.ty.params`.
48    ///
49    /// To figure out which casts are required from the client side (e.g. when
50    /// unfolding), one should walk the `decomp.ty.specifics` and call
51    /// `RustFieldData::decompose_compare_normalize` with `decomp.ty.params`
52    /// and `decomp.args`.
53    pub fn from_ty(ty: ty::Ty<'tcx>, context: impl Into<GParams<'tcx>>) -> Self {
54        TyData::<'tcx, RustTyDatas>::from_ty(ty, context.into())
55    }
56
57    /// Same as `from_ty` to get a `RustTyDecomposition` for use in encoding,
58    /// but requires fewer arguments when the type is known to be primitive.
59    pub fn from_prim_ty(ty: ty::Ty<'tcx>) -> Self {
60        assert!(ty.is_primitive());
61        TyData::<'tcx, RustTyDatas>::from_prim_ty(ty)
62    }
63
64    /// When you only have a `RustTy<'tcx>` but you want to use one of the
65    /// type-use encoders.
66    pub fn identity(ty: RustTy<'tcx>) -> Self {
67        let args = GArgs::new(ty.params, ty.params.rust_params());
68        Self::new(ty, args)
69    }
70
71    pub fn param() -> RustTy<'tcx> {
72        let gty = TyData::<RustTyDatas>::args_from_tys([TySpecifics::new_param_ty(0)]);
73        let data = RustTyData {
74            name: symbol::Symbol::intern("Param"),
75            params: GParams::empty_env(gty),
76            special: RustTySpecial::None,
77            sizedness: RustTySizedness::None,
78        };
79        let specifics = TySpecifics::Param(RustParamData::Generic);
80        TyData::<RustTyDatas>::new(data, specifics).alloc()
81    }
82}

90/// A to-be decomposed Rust type. We need this since we cannot infinitely
91/// decompose recursive datatypes (instead fields are left as `LazyRustTy` and
92/// decomposed as needed when recursing).
93#[repr(transparent)]
94#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
95pub struct LazyRustTy<'tcx>(ty::Ty<'tcx>);
96
97impl<'tcx> LazyRustTy<'tcx> {
98    pub fn new(ty: ty::Ty<'tcx>) -> Self {
99        Self(ty)
100    }
101
102    pub fn new_slice(tys: &'tcx [ty::Ty<'tcx>]) -> &'tcx [Self] {
103        // SAFETY: `LazyRustTy` is `repr(transparent)` over `ty::Ty`
104        let ptr = tys as *const [ty::Ty<'tcx>] as *const [Self];
105        unsafe { &*ptr }
106    }
107
108    /// The (sized) type parameter at the given index. `decompose`d against a
109    /// `GParams` that declares it, this yields a `Param` decomposition whose
110    /// argument is that type variable. Used to make a builtin method generic
111    /// over one of the reference's type parameters.
112    ///
113    /// You very likely do NOT want to use this: it's a hacky thing to make
114    /// `MirBuiltinCastEnc` work.
115    pub fn new_param_ty(index: u32) -> Self {
116        Self::new(TySpecifics::new_param_ty(index))
117    }
118
119    /// The pointer metadata type of a pointer to this type.
120    fn pointee_metadata(self) -> Self {
121        vir::with_vcx(|vcx| {
122            let metadata_did = vcx
123                .tcx()
124                .require_lang_item(LangItem::Metadata, prusti_rustc_interface::span::DUMMY_SP);
125            Self::new(ty::Ty::new_projection(
126                vcx.tcx(),
127                ty::IsRigid::No,
128                metadata_did,
129                [self.0],
130            ))
131        })
132    }
133}

135impl<'tcx> LazyRustTy<'tcx> {
136    /// Decomposes the field's type into a `RustTyDecomposition` (to be used
137    /// when recursing over the fields of a containing `RustTy` to construct
138    /// e.g. a predicate - i.e. when the definition of the predicate is
139    /// independent of the context/generic args).
140    /// The passed `params` should be those of the containing `RustTy::params`.
141    ///
142    /// For example a `Foo<i32>` with definition `struct Foo<T>(T);`, then
143    /// decomposing the field of the struct would yield `TySpecifics::Param`
144    /// with arguments `<T>` (i.e. the `i32` from the context is lost).
145    pub fn decompose(&self, params: GParams<'tcx>) -> RustTyDecomposition<'tcx> {
146        RustTyDecomposition::from_ty(self.0, params)
147    }
148
149    /// Decomposes the field's type into a `RustTyDecomposition` (to be used
150    /// when recursing over the fields of a containing `RustTy`
151    /// non-transparently, e.g. when predicates of fields should be added
152    /// directly to a method itself).
153    /// The passed `args` should be those of the containing `RustTyDecomposition::args`.
154    pub fn decompose_context(
155        &self,
156        params: GParams<'tcx>,
157        args: GArgs<'tcx>,
158    ) -> RustTyDecomposition<'tcx> {
159        let mut decomp = self.decompose(params);
160        decomp.args = decomp.args.substitute(args);
161        decomp
162    }
163
164    /// Decomposes the field's type into a `RustTyDecomposition` (to be used
165    /// when recursing over the fields of a containing `RustTy`).
166    /// The passed `args` should be those of the containing `RustTyDecomposition::args`.
167    ///
168    /// This differs from `Self::decompose` in that it substitutes the `args`
169    /// removing definitional generics. For example a `Foo<i32>` with definition
170    /// `struct Foo<T>(T);` would yield `i32` instead of `T` when called on the
171    /// field of `Foo`.
172    pub fn decompose_normalize(&self, args: GArgs<'tcx>) -> RustTyDecomposition<'tcx> {
173        RustTyDecomposition::from_ty(args.normalize(self.0), args.context())
174    }
175
176    /// Similarly to `Self::decompose`, this decomposes the fields type.
177    /// However, it tries to normalize the type first and only returns a
178    /// decomposition if the type was a `TySpecifics::Param` and is now a
179    /// concrete type. For example, when called on the `field: I::Item` of the
180    /// following example:
181    /// ```no_run
182    /// struct MyStruct<I: Iterator> {
183    ///     field: I::Item
184    /// }
185    /// fn foo<T: Iterator<Item = i32>>(x: MyStruct<T>) { ... }
186    /// ```
187    /// For which the initial decomposition of the argument `MyStruct<T>` was
188    /// ```no_run
189    /// let decomp = RustTyDecomposition {
190    ///     ty: TyData { params: "<I: Iterator>", specifics: "MyStruct(I::Item)" }
191    ///     args: GArgs { args: "<T>", context: "<T: Iterator<Item = i32>>" }
192    /// };
193    /// // one would call
194    /// let field = decomp.ty.specifics.expect_struct().fields[0];
195    /// let decomp_field = field.decompose_compare_normalize(decomp.ty.params, decomp.args)
196    /// // where `decomp_field` would be
197    /// Some(RustTyDecomposition {
198    ///     ty: TyData { params: "", specifics: "i32" }
199    ///     args: GArgs { args: "", context: "<T: Iterator<Item = i32>>" }
200    /// });
201    /// ```
202    pub fn decompose_compare_normalize(
203        &self,
204        params: GParams<'tcx>,
205        args: GArgs<'tcx>,
206    ) -> Option<RustTyNormalized<'tcx>> {
207        let param = self.decompose(params).ty;
208        let TySpecifics::Param(..) = &param.specifics else {
209            return None;
210        };
211        let concrete = self.decompose_normalize(args);
212        if let TySpecifics::Param(..) = &concrete.ty.specifics {
213            None
214        } else {
215            Some(RustTyNormalized { param, concrete })
216        }
217    }
218}

271/// Which of the sizedness traits (`Sized`, `MetaSized`, `PointeeSized`) the
272/// constructor's instances implement. These traits have no impls, so their
273/// facts are encoded per constructor instead (see `SizednessEnc`). Mirrors
274/// `Ty::has_trivial_sizedness`.
275#[derive(Debug, Clone, Copy, PartialEq, Eq, Hash)]
276pub enum RustTySizedness<'tcx> {
277    /// Not a type constructor (a type parameter or projection).
278    None,
279    /// `Sized` and `MetaSized` exactly when the respective tail type, over
280    /// the constructor's parameters, is; unconditionally if there is none.
281    Sized {
282        sized_if: Option<ty::Ty<'tcx>>,
283        meta_sized_if: Option<ty::Ty<'tcx>>,
284    },
285    /// `MetaSized` but not `Sized`: `str`, slices and trait objects.
286    MetaSized,
287    /// Only `PointeeSized`: extern types.
288    PointeeSized,
289}
290
291impl<'tcx> RustTySizedness<'tcx> {
292    fn from_ty(ty: ty::Ty<'tcx>) -> Self {
293        let tail = |kind| match *ty.kind() {
294            ty::TyKind::Adt(adt, _) => vir::with_vcx(|vcx| {
295                adt.sizedness_constraint(vcx.tcx(), kind)
296                    .map(ty::EarlyBinder::instantiate_identity)
297                    .map(ty::Unnormalized::skip_normalization)
298            }),
299            ty::TyKind::Tuple(tys) => tys
300                .len()
301                .checked_sub(1)
302                .map(|last| TySpecifics::new_param_ty(last as u32)),
303            _ => None,
304        };
305        match ty.kind() {
306            ty::TyKind::Param(_) | ty::TyKind::Alias(..) => Self::None,
307            ty::TyKind::Str | ty::TyKind::Slice(_) | ty::TyKind::Dynamic(..) => Self::MetaSized,
308            ty::TyKind::Foreign(_) => Self::PointeeSized,
309            _ => Self::Sized {
310                sized_if: tail(ty::SizedTraitKind::Sized),
311                meta_sized_if: tail(ty::SizedTraitKind::MetaSized),
312            },
313        }
314    }
315}

404    /// The type of the value stored in a `Box` (its hardcoded last field, see
405    /// `TySpecifics::from_adt`).
406    pub fn box_value_ty(&self) -> LazyRustTy<'tcx> {

411    fn from_ty(ty: ty::Ty<'tcx>, context: GParams<'tcx>) -> RustTyDecomposition<'tcx> {
412        // We normalize since we may be translating a type such as the field of
413        // `struct MyStruct<T: Iterator<Item = i32>>(T::Item);` where `ty` is
414        // `T::Item` and `context` is `<T: Iterator<Item = i32>>`. In this case
415        // we want to encode the struct as if it had an `i32` field (without any
416        // Param generics).
417        let ty = context.normalize(ty);
418
419        let name = Self::ty_name(ty);
420        let (params, args) = Self::identity_for_ty(ty, context.is_trait_extern_spec());
421        let args = GArgs::new(context, args);
422        let specifics = TySpecifics::from_ty(ty);
423        let data = RustTyData {
424            name: symbol::Symbol::intern(&name),
425            params,
426            special: RustTySpecial::from_ty(ty),
427            sizedness: RustTySizedness::from_ty(ty),
428        };
429        RustTyDecomposition::new(Self::new(data, specifics).alloc(), args)
430    }
431
432    fn from_prim_ty(ty: ty::Ty<'tcx>) -> RustTyDecomposition<'tcx> {
433        let name = Self::prim_ty_name(ty);
434        let (params, args) = Self::identity_for_prim_ty(ty);
435        let args = GArgs::new(params, args);
436        let data = RustTyData {
437            name: symbol::Symbol::intern(&name),
438            params,
439            special: RustTySpecial::None,
440            sizedness: RustTySizedness::from_ty(ty),
441        };
442        let specifics = TySpecifics::from_prim_ty(ty);
443        RustTyDecomposition::new(Self::new(data, specifics).alloc(), args)
444    }
445
446    fn ty_name(ty: ty::Ty<'tcx>) -> String {
447        let def_id_name = |def_id| {
448            vir::with_vcx(|vcx| {
449                vir::ViperIdent::from_def_id(vcx, def_id)
450                    .to_str()
451                    .to_owned()
452            })
453        };
454        match ty.kind() {
455            _ if ty.is_primitive() => Self::prim_ty_name(ty),
456            ty::TyKind::Str => String::from("Str"),
457            ty::TyKind::Adt(adt, _) => def_id_name(adt.did()),
458            ty::TyKind::Tuple(params) => format!("{}_Tuple", params.len()),
459            ty::TyKind::Never => String::from("Never"),
460            ty::TyKind::Ref(_, _, ty::Mutability::Not) => String::from("Ref_immutable"),
461            ty::TyKind::Ref(_, _, ty::Mutability::Mut) => String::from("Ref_mutable"),
462            ty::TyKind::RawPtr(_, ty::Mutability::Not) => String::from("RawPtr_immutable"),
463            ty::TyKind::RawPtr(_, ty::Mutability::Mut) => String::from("RawPtr_mutable"),
464            ty::TyKind::Param(_) | ty::TyKind::Alias(..) => String::from("Param"),
465            ty::TyKind::Closure(def_id, _) => def_id_name(*def_id),
466            ty::TyKind::FnPtr(..) => String::from("FnPtr"),
467            ty::TyKind::Array(..) => String::from("Array"),
468            ty::TyKind::Slice(..) => String::from("Slice"),
469            ty::TyKind::Dynamic(..) => String::from("Dyn"),
470            ty::TyKind::Foreign(def_id) | ty::TyKind::FnDef(def_id, _) => def_id_name(*def_id),
471            ty::TyKind::Pat(ty, _) => Self::ty_name(*ty),
472            other => unimplemented!("ty_name for {:?}", other),
473        }
474    }
475
476    fn prim_ty_name(ty: ty::Ty<'tcx>) -> String {
477        assert!(ty.is_primitive());
478        match ty.kind() {
479            ty::TyKind::Bool => String::from("Bool"),
480            ty::TyKind::Char => String::from("Char"),
481            ty::TyKind::Int(kind) => format!("Int_{}", kind.name_str()),
482            ty::TyKind::Uint(kind) => format!("UInt_{}", kind.name_str()),
483            ty::TyKind::Float(kind) => format!("Float_{}", kind.name_str()),
484            _ => unreachable!(),
485        }
486    }
487
488    /// For the ty `MyStruct<i32>` (with defn
489    /// `struct MyStruct<T: Iterator<Item = i32>> { ... }`), returns
490    /// `([<T: Iterator<Item = i32>>], [i32])`.
491    pub(super) fn identity_for_ty(
492        ty: ty::Ty<'tcx>,
493        is_trait_extern_spec: bool,
494    ) -> (GParams<'tcx>, ty::GenericArgsRef<'tcx>) {
495        let (params, args) = match *ty.kind() {
496            _ if ty.is_primitive() => return Self::identity_for_prim_ty(ty),
497            ty::TyKind::Adt(adt, args) => (GParams::from(adt.did()), args),
498            ty::TyKind::Tuple(tys) => {
499                let gtys = (0..tys.len()).map(|idx| TySpecifics::new_param_ty(idx as u32));
500                (
501                    GParams::empty_env(Self::args_from_tys(gtys)),
502                    Self::args_from_tys(tys),
503                )
504            }
505            ty::TyKind::Array(ty, cst) => vir::with_vcx(|vcx| {
506                let gcst = TySpecifics::new_param_const(0).into();
507                let gty = TySpecifics::new_param_ty(1).into();
508                let gparams = Self::args_from_generics([gcst, gty]);
509                let predicate =
510                    vcx.tcx()
511                        .mk_predicate(ty::Binder::dummy(ty::PredicateKind::Clause(
512                            ty::ClauseKind::ConstArgHasType(
513                                gcst.expect_const(),
514                                vcx.tcx().types.usize,
515                            ),
516                        )));
517                let param_env =
518                    ty::ParamEnv::new(vcx.tcx().mk_clauses(&[predicate.expect_clause()]));
519                (
520                    GParams::new(gparams, param_env, false),
521                    Self::args_from_generics([cst.into(), ty.into()]),
522                )
523            }),
524            ty::TyKind::Slice(ty) => {
525                let gty = Self::args_from_tys([TySpecifics::new_param_ty(0)]);
526                (GParams::empty_env(gty), Self::args_from_tys([ty]))
527            }
528            ty::TyKind::RawPtr(pointee, _) => {
529                // Like a reference, a raw pointer's pointer-metadata type is
530                // derived from the pointee via `<pointee as Pointee>::Metadata`
531                // (see `RefData` below), so it has a single generic: the pointee.
532                let pointee_ty = TySpecifics::new_param_ty(0).into();
533                let gty = Self::args_from_generics([pointee_ty]);
534                (
535                    GParams::empty_env(gty),
536                    Self::args_from_generics([pointee.into()]),
537                )
538            }
539            ty::TyKind::Ref(region, inner, _) => {
540                // The pointer-metadata type is derived from the referent (see
541                // `RefData` below and `LazyRustTy::pointee_metadata`), so a
542                // reference has just two generics: the lifetime and the referent.
543                // TODO: what lifetime should we use here?
544                let param_region = vir::with_vcx(|vcx| vcx.tcx().lifetimes.re_erased.into());
545                let param_ty = TySpecifics::new_param_ty(1).into();
546                let gty = Self::args_from_generics([param_region, param_ty]);
547                (
548                    GParams::empty_env(gty),
549                    Self::args_from_generics([region.into(), inner.into()]),
550                )
551            }
552            ty::TyKind::Alias(..) | ty::TyKind::Param(_) => {
553                let gty = Self::args_from_tys([TySpecifics::new_param_ty(0)]);
554                (GParams::empty_env(gty), Self::args_from_tys([ty]))
555            }
556            ty::TyKind::Closure(did, args) => vir::with_vcx(|vcx| {
557                let identity = ty::List::identity_for_item(vcx.tcx(), did);
558                let gargs = vcx.tcx().mk_args(identity.as_closure().parent_args());
559                let args = vcx.tcx().mk_args(args.as_closure().parent_args());
560                (
561                    GParams::new(gargs, vcx.tcx().param_env(did), is_trait_extern_spec),
562                    args,
563                )
564            }),
565            ty::TyKind::FnDef(did, args) => vir::with_vcx(|vcx| {
566                let gargs = identity_params(vcx.tcx(), did);
567                (
568                    GParams::new(gargs, vcx.tcx().param_env(did), is_trait_extern_spec),
569                    args.skip_binder(),
570                )
571            }),
572            ty::TyKind::Pat(ty, _) => Self::identity_for_ty(ty, is_trait_extern_spec),
573            ty::TyKind::Never
574            | ty::TyKind::Str
575            | ty::TyKind::FnPtr(..)
576            | ty::TyKind::Dynamic(..)
577            | ty::TyKind::Foreign(_) => (GParams::empty(), ty::GenericArgs::empty()),
578            _ => todo!("instantiate_identity_for_type for {:?}", ty),
579        };
580        params.check(args);
581        (params, args)
582    }
583
584    fn identity_for_prim_ty(ty: ty::Ty<'tcx>) -> (GParams<'tcx>, ty::GenericArgsRef<'tcx>) {
585        assert!(ty.is_primitive());
586        (GParams::empty(), ty::GenericArgs::empty())
587    }
588
589    fn args_from_tys(tys: impl IntoIterator<Item = ty::Ty<'tcx>>) -> ty::GenericArgsRef<'tcx> {
590        Self::args_from_generics(tys.into_iter().map(ty::GenericArg::from))
591    }
592
593    fn args_from_generics(
594        tys: impl IntoIterator<Item = ty::GenericArg<'tcx>>,
595    ) -> ty::GenericArgsRef<'tcx> {
596        vir::with_vcx(|vcx| vcx.tcx().mk_args_from_iter(tys.into_iter()))
597    }
598}

600/// For a `RustTy` zipped with another (encoded) type data.
601impl<'tcx, D: TyDatas<'tcx>> TyData<'tcx, (RustTyDatas, D)> {
602    pub fn box_metadata_ty(&self) -> LazyRustTy<'tcx> {
603        assert_eq!(self.0.special, RustTySpecial::Box);
604        let box_value_ty = self.expect_structlike().fields.last().unwrap().0.ty();
605        box_value_ty.pointee_metadata()
606    }
607}

609impl RustTySpecial {
610    fn from_ty(ty: ty::Ty<'_>) -> Self {
611        match ty.kind() {
612            ty::TyKind::Adt(adt, _) if adt.is_box() => RustTySpecial::Box,
613            _ => RustTySpecial::None,
614        }
615    }
616}
617
618impl<'tcx> TySpecifics<'tcx, RustTyDatas> {
619    fn from_ty(ty: ty::Ty<'tcx>) -> Self {
620        if ty.is_primitive() {
621            return Self::from_prim_ty(ty);
622        }
623        if crate::encoders::is_type_trusted(ty) {
624            return TySpecifics::mk_opaque(());
625        }
626
627        match ty.kind() {
628            ty::TyKind::Adt(adt, _) => Self::from_adt(*adt),
629            ty::TyKind::Tuple(args) => {
630                let fields = args
631                    .iter()
632                    .enumerate()
633                    .map(|(i, _)| RustFieldData {
634                        name: symbol::Symbol::intern(&format!("_{i}")),
635                        fid: abi::FieldIdx::from_usize(i),
636                        ty: LazyRustTy(Self::new_param_ty(i as u32)),
637                        address: RustFieldAddress::Constant,
638                    })
639                    .collect::<Vec<_>>();
640                TySpecifics::mk_structlike((), fields)
641            }
642            ty::TyKind::Array(_, _) => TySpecifics::ArrayLike(ArrayData {
643                slice: false,
644                data: LazyRustTy(Self::new_param_ty(1)),
645            }),
646            ty::TyKind::Slice(_) => TySpecifics::ArrayLike(ArrayData {
647                slice: true,
648                data: LazyRustTy(Self::new_param_ty(0)),
649            }),
650            ty::TyKind::Ref(_, _, mutability) => {
651                // The referent is generic parameter 1; its pointer metadata is
652                // derived as `<referent as Pointee>::Metadata` rather than being
653                // a separate generic parameter.
654                let referent = LazyRustTy(Self::new_param_ty(1));
655                let data = RefData {
656                    metadata: referent.pointee_metadata(),
657                    referent,
658                };
659                match mutability {
660                    ty::Mutability::Mut => TySpecifics::mk_mutref(data),
661                    ty::Mutability::Not => TySpecifics::mk_immref(data),
662                }
663            }
664            // Raw pointers get their own conservative `Raw` kind: we don't
665            // reason about the pointee through them, but (like a reference) the
666            // snapshot carries the address and pointer metadata so `PtrMetadata`
667            // (e.g. a slice's length) can be read out. The pointee is generic
668            // parameter 0; the metadata is derived from it as
669            // `<pointee as Pointee>::Metadata`.
670            ty::TyKind::RawPtr(..) => {
671                let referent = LazyRustTy(Self::new_param_ty(0));
672                TySpecifics::mk_raw(RefData {
673                    metadata: referent.pointee_metadata(),
674                    referent,
675                })
676            }
677            ty::TyKind::Alias(..) | ty::TyKind::Param(_) => {
678                TySpecifics::mk_param(RustParamData::Generic)
679            }
680            ty::TyKind::Closure(_, args) => {
681                let captured = args.as_closure().upvar_tys();
682                let fields = vir::with_vcx(|vcx| {
683                    captured
684                        .iter()
685                        .enumerate()
686                        .map(|(i, ty)| RustFieldData {
687                            name: symbol::Symbol::intern(&format!("c{i}")),
688                            fid: abi::FieldIdx::from_usize(i),
689                            ty: LazyRustTy(vcx.tcx().erase_and_anonymize_regions(ty)),
690                            address: RustFieldAddress::Constant,
691                        })
692                        .collect::<Vec<_>>()
693                });
694                TySpecifics::mk_structlike((), fields)
695            }
696            ty::TyKind::Never => {
697                let data = vir::with_vcx(|vcx| RustEnumData {
698                    discr: vcx.tcx().types.isize,
699                });
700                TySpecifics::mk_enumlike(data, Vec::new())
701            }
702            // TODO: add str support
703            ty::TyKind::Str => TySpecifics::mk_opaque(()),
704            // TODO: give dyn Trait a type witness parameter (the concrete type behind the
705            // pointer), enabling virtual dispatch and distinguishing dyn TraitA from dyn TraitB.
706            ty::TyKind::Dynamic(..) => TySpecifics::mk_param(RustParamData::Dyn),
707            // TODO: encode the pattern as narrowed bounds on the base type, so that e.g.
708            // the `u32 is 1..` inside a `NonZeroU32` is known to be non-zero. Doing so
709            // also needs a `ty_name` that distinguishes the pattern from its base type,
710            // since the name and the specifics together are the interning key.
711            ty::TyKind::Pat(ty, _) => Self::from_ty(*ty),
712            _ => TySpecifics::mk_opaque(()),
713        }
714    }
715
716    fn from_prim_ty(ty: ty::Ty<'tcx>) -> Self {
717        assert!(ty.is_primitive());
718        TySpecifics::mk_primitive(ty)
719    }
720
721    fn from_adt(adt: ty::AdtDef<'tcx>) -> Self {
722        if vir::with_vcx(|vcx| vcx.tcx().lang_items().get(LangItem::DynMetadata) == Some(adt.did()))
723        {
724            // `DynMetadata<dyn Trait>` is the metadata of a `&dyn`/`*dyn`
725            // wide pointer. We never reason about vtable contents, so encode it
726            // as an opaque snapshot rather than recursing into its `Foreign`
727            // `VTable` field (which has no `ty_name`).
728            TySpecifics::mk_opaque(())
729        } else if vir::with_vcx(|vcx| {
730            EnvQuery::new(vcx.tcx()).is_adt_in_crate(adt, "prusti_contracts")
731        }) {
732            match adt.non_enum_variant().name.to_string().as_str() {
733                "Int" => Self::Builtin(RustBuiltinData::Int),
734                "Real" => Self::Builtin(RustBuiltinData::Real),
735                "Set" => Self::Builtin(RustBuiltinData::Set(LazyRustTy(Self::new_param_ty(0)))),
736                "Multiset" => {
737                    Self::Builtin(RustBuiltinData::Multiset(LazyRustTy(Self::new_param_ty(0))))
738                }
739                "Seq" => Self::Builtin(RustBuiltinData::Seq(LazyRustTy(Self::new_param_ty(0)))),
740                "Map" => Self::Builtin(RustBuiltinData::Map(
741                    LazyRustTy(Self::new_param_ty(0)),
742                    LazyRustTy(Self::new_param_ty(1)),
743                )),
744                // `Ghost<T>` is encoded as if it were `struct Ghost<T>(T)`:
745                // the snapshot wraps the value of `T`.
746                "Ghost" => {
747                    let fields = vec![RustFieldData {
748                        name: symbol::Symbol::intern("val"),
749                        fid: abi::FieldIdx::from_usize(0),
750                        ty: LazyRustTy(Self::new_param_ty(0)),
751                        address: RustFieldAddress::Constant,
752                    }];
753                    TySpecifics::mk_structlike((), fields)
754                }
755                // TODO: support other builtins (e.g. `Seq`, `Map`, `Set`, etc.)
756                s => todo!("Unimplemented builtin {s}"),
757            }
758        } else {
759            match adt.adt_kind() {
760                ty::AdtKind::Struct => {
761                    let mut data = Self::from_struct(adt.non_enum_variant());
762                    if adt.is_box() {
763                        // The hardcoded value slot for the boxed `T` is
764                        // appended as a regular field; only its (heap-dependent)
765                        // field accessor and the pointer metadata are
766                        // special-cased (see the structlike encoder).
767                        data.fields.push(RustFieldData {
768                            name: symbol::Symbol::intern("value"),
769                            fid: abi::FieldIdx::from_usize(data.fields.len()),
770                            ty: LazyRustTy(Self::new_param_ty(0)),
771                            address: RustFieldAddress::Dynamic,
772                        });
773                    }
774                    Self::StructLike(data)
775                }
776                ty::AdtKind::Enum => {
777                    let data = Self::from_enum(adt);
778                    Self::EnumLike(data)
779                }
780                ty::AdtKind::Union => {
781                    // TODO: add union support
782                    Self::mk_opaque(())
783                }
784            }
785        }
786    }
787
788    fn from_struct(variant: &ty::VariantDef) -> StructData<'tcx, RustTyDatas> {
789        let fields = Self::from_fields(&variant.fields);
790        StructData::new((), fields)
791    }
792
793    fn from_enum(adt: ty::AdtDef<'tcx>) -> EnumData<'tcx, RustTyDatas> {
794        vir::with_vcx(|vcx| {
795            use ty::util::IntTypeExt;
796            let discr = adt.repr().discr_type().to_ty(vcx.tcx());
797            let data = RustEnumData { discr };
798            let variants = adt
799                .discriminants(vcx.tcx())
800                .map(|(vid, discr)| {
801                    let variant = adt.variant(vid);
802                    let fields = Self::from_fields(&variant.fields);
803                    VariantData::new(
804                        RustVariantData {
805                            name: variant.name,
806                            vid,
807                            discr_val: discr.val,
808                        },
809                        StructData::new((), fields),
810                    )
811                })
812                .collect::<Vec<_>>();
813            EnumData::new(data, variants)
814        })
815    }
816
817    fn from_fields(
818        fields: &index::IndexVec<abi::FieldIdx, ty::FieldDef>,
819    ) -> Vec<RustFieldData<'tcx>> {
820        fields
821            .iter_enumerated()
822            .map(|(fid, field)| {
823                let ty = vir::with_vcx(|vcx| vcx.tcx().type_of(field.did).instantiate_identity());
824                RustFieldData {
825                    name: field.name,
826                    fid,
827                    ty: LazyRustTy(ty.skip_normalization()),
828                    address: RustFieldAddress::Constant,
829                }
830            })
831            .collect::<Vec<_>>()
832    }
833
834    fn new_param_ty(index: u32) -> ty::Ty<'tcx> {
835        let name = match index {
836            0 => symbol::Symbol::intern("T"),
837            1 => symbol::Symbol::intern("U"),
838            2 => symbol::Symbol::intern("V"),
839            other => symbol::Symbol::intern(&format!("T{other}")),
840        };
841        vir::with_vcx(|vcx| ty::Ty::new_param(vcx.tcx(), index, name))
842    }
843
844    fn new_param_const(index: u32) -> ty::Const<'tcx> {
845        let name = match index {
846            0 => symbol::Symbol::intern("M"),
847            1 => symbol::Symbol::intern("N"),
848            other => symbol::Symbol::intern(&format!("N{other}")),
849        };
850        let param = ty::ParamConst { index, name };
851        vir::with_vcx(|vcx| ty::Const::new_param(vcx.tcx(), param))
852    }
