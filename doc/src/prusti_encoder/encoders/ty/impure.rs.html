134impl TaskEncoder for TyImpureEnc {
135    task_encoder::encoder_cache!(TyImpureEnc);
136    const ENCODER_NAME: &'static str = "type impure encoder";
137    type TaskDescription<'vir> = RustTy<'vir>;
138
139    type OutputRef<'vir> = TyImpureRef<'vir>;
140    type OutputFullDependency<'vir> = TyImpure<'vir>;
141    type OutputFullLocal<'vir> = TyImpureEncLocal<'vir>;
142
143    type EncodingError = TyImpureEncError;
144
145    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
146        *task
147    }
148
149    fn do_encode_full<'vir>(
150        task_key: &Self::TaskKey<'vir>,
151        deps: &mut TaskEncoderDependencies<'vir, Self>,
152    ) -> EncodeFullResult<'vir, Self> {
153        let snap = deps.require_dep::<TyPureEnc>(*task_key)?;
154        let snapshot = snap.snapshot;
155
156        let ty = task_key.zip(snap);
157
158        vir::with_vcx(|vcx| {
159            let mut builder = PredicateBuilder::new(deps, vcx, task_key, snapshot);
160
161            let ref_self_decl = builder.ref_self_decl();
162            let ref_self = vcx.mk_local_ex(ref_self_decl);
163
164            // assign method
165            let value_decl = vcx.mk_local_decl("value", snapshot);
166            let value = vcx.mk_local_ex(value_decl);
167            let method_assign = builder.inner.method(
168                "assign",
169                (ref_self_decl.ty(), builder.params.ty_args(), builder.params.const_args(), snapshot),
170                &[],
171                (ref_self_decl, builder.params.ty_decls(), builder.params.const_decls(), value_decl),
172                &[],
173                &[
174                    vir::expr! { [builder.ref_to_pred](ref_self, [..[builder.params.ty_exprs()]], [..[builder.params.const_exprs()]]) },
175                    vir::expr! { ([builder.ref_to_snap](ref_self, [..[builder.params.ty_exprs()]], [..[builder.params.const_exprs()]])) == (value) },
176                ],
177            );
178
179            let data = TyImpureRef {
180                ref_to_pred: builder.ref_to_pred,
181                ref_to_snap: builder.ref_to_snap,
182                method_assign,
183            };
184            deps.emit_output_ref(*task_key, data)?;
185
186            let specifics = match &ty.specifics {
187                TySpecifics::Param(param) => {
188                    TySpecifics::Param(super::kinds::param::ty_impure(param, deps, &mut builder)?)
189                }
190                TySpecifics::Opaque(opaque) => TySpecifics::Opaque(
191                    super::kinds::opaque::ty_impure(opaque, deps, &mut builder)?,
192                ),
193                TySpecifics::ArrayLike(array) => TySpecifics::ArrayLike(
194                    super::kinds::arraylike::ty_impure(&ty, array, deps, &mut builder)?,
195                ),
196                TySpecifics::Primitive(prim) => TySpecifics::Primitive(
197                    super::kinds::primitive::ty_impure(prim, deps, &mut builder)?,
198                ),
199                TySpecifics::ImmRef(immref) => TySpecifics::ImmRef(
200                    super::kinds::immref::ty_impure(&ty, immref, deps, &mut builder)?,
201                ),
202                TySpecifics::MutRef(mutref) => TySpecifics::MutRef(
203                    super::kinds::mutref::ty_impure(&ty, mutref, deps, &mut builder)?,
204                ),
205                TySpecifics::Raw(raw) => {
206                    TySpecifics::Raw(super::kinds::raw::ty_impure(&ty, raw, deps, &mut builder)?)
207                }
208                TySpecifics::StructLike(structlike) => TySpecifics::StructLike(
209                    super::kinds::structlike::ty_impure(&ty, structlike, deps, &mut builder)?,
210                ),
211                TySpecifics::EnumLike(enumlike) => TySpecifics::EnumLike(
212                    super::kinds::enumlike::ty_impure(&ty, enumlike, deps, &mut builder)?,
213                ),
214                TySpecifics::Builtin(builtin) => TySpecifics::Builtin(
215                    super::kinds::builtin::ty_impure(builtin, deps, &mut builder)?,
216                ),
217            };
218            let output = TyData::new(data, specifics).alloc();
219
220            Ok((builder.inner.build(), output))
221        })
222    }
223
224    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
225        for output in Self::all_outputs_local_no_errors(program) {
226            for field in output.fields {
227                program.add_field(field);
228            }
229            for field_projection in output.functions {
230                program.add_function(field_projection);
231            }
232            program.add_function(output.function_snap);
233            for pred in output.predicates {
234                program.add_predicate(pred);
235            }
236            program.add_method(output.method_assign);
237            for method in output.methods {
238                program.add_method(method);
239            }
240        }
241    }
242}

268impl<'vir> PredicateBuilder<'vir> {
269    pub(crate) fn new<E: TaskEncoder>(
270        deps: &mut TaskEncoderDependencies<'vir, E>,
271        vcx: &'vir vir::VirCtxt<'vir>,
272        ty: RustTy<'vir>,
273        snapshot: vir::TypeSnap<'vir>,
274    ) -> Self {
275        let params = deps.require_dep::<GenericParamsEnc>(ty.params).unwrap();
276        let name = vir::vir_format!(vcx, "p_{}", ty.name());
277        let inner = PredicateBuilderInner {
278            vcx,
279            name,
280            fields: Vec::new(),
281            functions: Vec::new(),
282            methods: Vec::new(),
283            predicates: Vec::new(),
284            function_snap: None,
285        };
286
287        let ref_self_decl = inner.ref_self_decl();
288        let args = (ref_self_decl.ty(), params.ty_args(), params.const_args());
289        let ref_to_pred = inner.predicate_ident("", args);
290        let ref_to_snap = inner.function_ident("snap", args, snapshot);
291
292        PredicateBuilder {
293            params,
294            snapshot,
295            ref_to_pred,
296            ref_to_snap,
297            inner,
298        }
299    }
300
301    pub(crate) fn csnap_type(&self) -> vir::TypeCSnap<'vir> {
302        self.snapshot.downcast_ty()
303    }
304
305    pub(crate) fn mk_predicate(
306        &mut self,
307        name: &str,
308        expr: Option<vir::ExprBool<'vir>>,
309    ) -> PredicateIdn<'vir, (vir::Ref, vir::ManyTyVal, vir::ManyCSnap)> {
310        let ref_self_decl = self.ref_self_decl();
311        let args = (
312            ref_self_decl.ty(),
313            self.params.ty_args(),
314            self.params.const_args(),
315        );
316        let params = (
317            ref_self_decl,
318            self.params.ty_decls(),
319            self.params.const_decls(),
320        );
321        self.inner.predicate(name, args, params, expr)
322    }
323
324    /// Creates the `snap` function, sets the precondition to
325    /// `acc(ref_to_pred(self, ...))`, and the body as
326    /// `unfolding acc(ref_to_pred(self, ...)) in inner`. Note that the `inner`
327    /// will be wrapped in an unfolding and should not include it.
328    pub(crate) fn mk_snap_function(
329        &mut self,
330        inner: Option<vir::ExprCSnap<'vir>>,
331        posts: &[vir::ExprBool<'vir>],
332    ) {
333        let ref_self_decl = self.ref_self_decl();
334        let ref_self = self.vcx.mk_local_ex(ref_self_decl);
335        let params = (
336            ref_self_decl,
337            self.params.ty_decls(),
338            self.params.const_decls(),
339        );
340        let pred = vir::expr! {
341            acc([self.ref_to_pred](ref_self, [..[self.params.ty_exprs()]], [..[self.params.const_exprs()]]))
342        };
343        let expr = inner.map(|e| vir::expr! {
344            unfolding ([self.ref_to_pred](ref_self, [..[self.params.ty_exprs()]], [..[self.params.const_exprs()]])) in (e)
345        }.upcast_ty());
346        let function = self
347            .inner
348            .mk_function(self.ref_to_snap, params, &[pred], posts, expr);
349        self.inner.function_snap = Some(function);
350    }
351}

366impl<'vir> PredicateBuilderInner<'vir> {
367    pub(super) fn ref_self_decl(&self) -> vir::LocalDeclRef<'vir> {
368        self.vcx.mk_local_decl("self", vir::TYPE_REF)
369    }
370
371    fn ident_str(&self, name: &str) -> &'vir str {
372        let prefix = self.name;
373        if name.is_empty() {
374            prefix
375        } else {
376            vir::vir_format!(self.vcx, "{prefix}_{name}")
377        }
378    }
379
380    pub(crate) fn field<T: vir::CompType>(
381        &mut self,
382        name: &str,
383        typ: vir::Type<'vir, T>,
384    ) -> vir::Field<'vir, T> {
385        let name = self.ident_str(name);
386        let field = self.vcx.mk_field(name, typ);
387        self.fields.push(field.as_dyn());
388        field
389    }
390
391    pub(crate) fn predicate_ident<A: vir::Arity>(
392        &self,
393        name: &str,
394        args: A::Tys<'vir>,
395    ) -> vir::PredicateIdn<'vir, A> {
396        let name = self.ident_str(name);
397        vir::PredicateIdn::new(vir::ViperIdent::new(name), args)
398    }
399
400    pub(crate) fn predicate<A: vir::Arity>(
401        &mut self,
402        name: &str,
403        args: A::Tys<'vir>,
404        params: A::Locals<'_, 'vir>,
405        expr: Option<vir::ExprBool<'vir>>,
406    ) -> vir::PredicateIdn<'vir, A> {
407        let ident = self.predicate_ident(name, args);
408        self.predicates
409            .push(self.vcx.mk_predicate(ident, params, expr));
410        ident
411    }
412
413    pub(crate) fn function_ident<A: vir::Arity, T: vir::CompType>(
414        &self,
415        name: &str,
416        args: A::Tys<'vir>,
417        ret: vir::Type<'vir, T>,
418    ) -> vir::FunctionIdn<'vir, A, T> {
419        let name = self.ident_str(name);
420
421        vir::FunctionIdn::new(vir::ViperIdent::new(name), args, ret)
422    }
423
424    fn mk_function<A: vir::Arity, T: vir::CompType>(
425        &self,
426        ident: FunctionIdn<'vir, A, T>,
427        params: A::Locals<'_, 'vir>,
428        pres: &[vir::ExprBool<'vir>],
429        posts: &[vir::ExprBool<'vir>],
430        expr: Option<vir::Expr<'vir, T>>,
431    ) -> vir::Function<'vir> {
432        self.vcx.mk_function(
433            ident,
434            params,
435            self.vcx.alloc_slice(pres),
436            self.vcx.alloc_slice(posts),
437            None,
438            expr,
439        )
440    }
441
442    #[allow(clippy::too_many_arguments)]
443    pub(crate) fn function<A: vir::Arity, T: vir::CompType>(
444        &mut self,
445        name: &str,
446        args: A::Tys<'vir>,
447        ret: vir::Type<'vir, T>,
448        params: A::Locals<'_, 'vir>,
449        pres: &[vir::ExprBool<'vir>],
450        posts: &[vir::ExprBool<'vir>],
451        expr: Option<vir::Expr<'vir, T>>,
452    ) -> vir::FunctionIdn<'vir, A, T> {
453        let name = self.ident_str(name);
454        let ident = vir::FunctionIdn::new(vir::ViperIdent::new(name), args, ret);
455        self.functions
456            .push(self.mk_function(ident, params, pres, posts, expr));
457        ident
458    }
459
460    pub(crate) fn method<A: vir::Arity>(
461        &mut self,
462        name: &str,
463        args: A::Tys<'vir>,
464        rets: &[vir::LocalDeclDyn<'vir>],
465        params: A::Locals<'_, 'vir>,
466        pres: &[vir::ExprBool<'vir>],
467        posts: &[vir::ExprBool<'vir>],
468    ) -> vir::MethodIdn<'vir, A> {
469        let name = self.ident_str(name);
470        let ident = MethodIdn::new(
471            vir::ViperIdent::new(name),
472            args,
473            //ret,
474        );
475        self.methods.push(self.vcx.mk_method(
476            ident,
477            params,
478            self.vcx.alloc_slice(rets),
479            self.vcx.alloc_slice(pres),
480            self.vcx.alloc_slice(posts),
481            None,
482        ));
483        ident
484    }
485
486    pub(crate) fn build(mut self) -> TyImpureEncLocal<'vir> {
487        // TODO: don't rely on assignment being index 0, use separate field...
488        let method_assign = self.methods.remove(0);
489        TyImpureEncLocal {
490            fields: self.fields,
491            predicates: self.predicates,
492            function_snap: self.function_snap.unwrap(),
493            functions: self.functions,
494            method_assign,
495            methods: self.methods,
496        }
497    }
