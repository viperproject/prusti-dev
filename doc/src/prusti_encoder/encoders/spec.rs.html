38pub fn with_proc_spec<'tcx, F, R>(query: SpecQuery<'tcx>, f: F) -> Option<R>
39where
40    F: FnOnce(&ProcedureSpecification) -> R,
41{
42    vir::with_vcx(|vcx| {
43        let specs = vcx.specs.as_ref().unwrap();
44        specs
45            .borrow_mut()
46            .get_and_refine_proc_spec(vcx.tcx(), query)
47            .map(f)
48    })

51/// Whether a function's spec is trusted -- assumed rather than verified. This
52/// holds if the function itself is marked `#[trusted]` (never inherited; see
53/// `ProcedureSpecification::refine`), or if the spec derives from an
54/// `#[extern_spec]` and the function is foreign: such specs are always
55/// assumed (the mandatory `#[trusted]` inside the extern spec covers them).
56/// In particular a foreign impl inheriting the spec of an extern-spec'd trait
57/// method is assumed, necessarily: the annotation postdates the compilation
58/// of the defining crate (or that crate is not compiled by Prusti at all,
59/// e.g. `std`), so no body for it can ever have been exported. A *local* impl
60/// of such a trait has a body and is verified.
61pub fn spec_is_trusted(proc_spec: &ProcedureSpecification, def_id: DefId) -> bool {
62    let user_trusted = proc_spec
63        .trusted
64        .extract_with_selective_replacement()
65        .copied()
66        .unwrap_or_default();
67    let extern_trusted = proc_spec.extern_spec.is_some() && !def_id.is_local();
68    user_trusted || extern_trusted
69}

71pub fn is_function_trusted(def_id: DefId) -> bool {
72    let substs = ty::GenericArgs::identity_for_item(vir::with_vcx(|vcx| vcx.tcx()), def_id);
73    with_proc_spec(
74        SpecQuery::GetProcKind(def_id, substs),
75        |proc_spec: &ProcedureSpecification| spec_is_trusted(proc_spec, def_id),
76    )
77    .unwrap_or_default()
78}

80pub fn is_function_pure<'tcx>(def_id: DefId, args: GArgs<'tcx>) -> bool {
81    with_proc_spec(
82        SpecQuery::GetProcKind(def_id, args.args()),
83        |proc_spec: &ProcedureSpecification| kind_is_pure(&proc_spec.kind),
84    )
85    .unwrap_or_default()

88/// `kind.is_pure()`, treating an invalid trait-to-impl kind refinement as
89/// impure. This is a pure query; the refinement error itself is reported to
90/// the user separately by [`report_kind_refinement_error`].
91pub fn kind_is_pure(kind: &SpecificationItem<typed::ProcedureSpecificationKind>) -> bool {
92    kind.is_pure().unwrap_or(false)
93}

95/// Emit a user error for an invalid trait-to-impl kind refinement (e.g. an
96/// `impl` of a `#[pure]` trait method that is not itself `#[pure]`); a no-op if
97/// the refinement is valid. Kept separate from the purity query so it can be
98/// called once per function, at encoding time, rather than on every query.
99pub fn report_kind_refinement_error(
100    def_id: DefId,
101    kind: &SpecificationItem<typed::ProcedureSpecificationKind>,
102) {
103    use typed::ProcedureSpecificationKind::*;
104    let Err(typed::ProcedureSpecificationKindError::InvalidSpecKindRefinement(base, refined)) =
105        kind.is_pure()
106    else {
107        return;
108    };
109    vir::with_vcx(|vcx| {
110        let name = vcx.tcx().def_path_str(def_id);
111        let span = vcx.tcx().def_span(def_id).into();
112        let error = match (base, refined) {
113            (Pure, Impure) => {
114                let mut error = prusti_interface::PrustiError::incorrect(
115                    format!("`{name}` implements a `#[pure]` trait method and so must itself be `#[pure]`"),
116                    span,
117                )
118                .set_help("add `#[pure]` to the implementation");
119                // Point at the `#[pure]` in the trait definition (its
120                // `specs_version` marker is spanned at the annotation), when
121                // the trait method is available locally.
122                if let Some(trait_item) = vcx
123                    .tcx()
124                    .opt_associated_item(def_id)
125                    .and_then(|item| item.trait_item_def_id())
126                    && let Some(pure_span) =
127                        prusti_interface::utils::get_prusti_attr_of(vcx.tcx(), trait_item, "pure")
128                            .map(|item| item.span)
129                {
130                    error = error.add_note(
131                        "the trait method is declared `#[pure]` here",
132                        Some(pure_span),
133                    );
134                }
135                error
136            }
137            _ => prusti_interface::PrustiError::incorrect(
138                format!("the specification of `{name}` is incompatible with the trait declaration"),
139                span,
140            ),
141        };
142        vcx.emit_early_error(error);
143    });
144}

146pub fn is_type_trusted(ty: ty::Ty) -> bool {
147    match ty.kind() {
148        prusti_rustc_interface::middle::ty::TyKind::Adt(adt_def, _) => with_type_spec(|def_spec| {
149            def_spec
150                .get_type_spec(&adt_def.did())
151                .map(|type_spec| type_spec.trusted.extract_inherit().unwrap_or_default())
152                .unwrap_or_default()
153        }),
154        _ => false,
155    }
156}

185    fn do_encode_full<'vir>(
186        task_key: &Self::TaskKey<'vir>,
187        deps: &mut TaskEncoderDependencies<'vir, Self>,
188    ) -> EncodeFullResult<'vir, Self> {
189        deps.emit_output_ref(*task_key, ())?;
190        vir::with_vcx(|vcx| {
191            let (extern_spec, pres, posts, pledges) = with_proc_spec(
192                SpecQuery::GetProcKind(
193                    task_key.0,
194                    ty::List::identity_for_item(vcx.tcx(), task_key.0),
195                ),
196                |specs| {
197                    // TODO: handle specs other than `empty_or_inherent`
198                    let pres = specs.pres.map(|items| vcx.alloc_slice(items));
199                    let posts = specs.posts.map(|items| vcx.alloc_slice(items));
200                    let pledges = specs.pledges.map(|items| vcx.alloc_slice(items));
201                    (specs.extern_spec, pres, posts, pledges)
202                },
203            )
204            .unwrap_or((
205                None,
206                SpecificationItem::Empty,
207                SpecificationItem::Empty,
208                SpecificationItem::Empty,
209            ));
210            Ok((
211                (),
212                SpecEncOutput {
213                    extern_spec,
214                    pres,
215                    posts,
216                    pledges,
217                },
218            ))
219        })
220    }
221}

223/// The items to encode for one kind of specification, and whether they are
224/// expressed in the generics of the item they were inherited from rather than
225/// those of the item being encoded.
226pub fn spec_items<'vir, T>(spec: &SpecificationItem<&'vir [T]>) -> (&'vir [T], bool) {
227    match spec {
228        SpecificationItem::Empty => (&[], false),
229        SpecificationItem::Inherent(items) => (items, false),
230        SpecificationItem::Inherited(items) => (items, true),
231        SpecificationItem::Refined(_from, to) => {
232            // Here we ignore the original specs: to get to this branch, the
233            // task key given to `SpecEnc` was the `DefId` of an trait method
234            // implementation, which will happen when encoding the definition
235            // of that implementation.
236            //
237            // At callsites, `MethodCallEnc` will direct the call to the stub
238            // method, which uses the `DefId` of the trait item for emitting
239            // its specifications.
240            (to, false)
241        }
242    }
243}
