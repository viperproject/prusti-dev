33    pub(crate) fn get_loop_inv(
34        &mut self,
35        cfpcs: &PcgBasicBlock<'_, 'vir>,
36        loop_place_usages: &PlaceUsages<'vir>,
37        ctxt: impl HasCompilerCtxt<'a, 'vir>,
38    ) -> EncodeResult<
39        'vir,
40        (
41            Vec<vir::ExprBool<'vir>>,
42            FxHashMap<Place<'vir>, CapabilityKind>,
43        ),
44        E,
45    > {
46        let mut inv = Vec::new();
47        let start = &cfpcs.statements[0];
48        let state = &start.states[EvalStmtPhase::PreOperands];
49        let loop_invariant_place_capabilities =
50            cfpcs.loop_invariant_place_capabilities(loop_place_usages, ctxt);
51
52        for (place, capability) in loop_invariant_place_capabilities.iter() {
53            if capability.is_write() {
54                continue; // No permissions are encoded for places with write capabilities currently
55            }
56            let (place_res, _snap, _, _) = self.encode_place_with_snap(*place)?;
57            let ty = (*place).ty(self.pcg_ctxt());
58            let task = RustTyDecomposition::from_ty(ty.ty, self.def_id);
59            let ty_out = self.deps.require_dep::<TyUseImpureEnc>(task).unwrap();
60            let pred = ty_out.ref_to_pred(self.vcx, place_res.expr.expect_predicate(), None);
61            inv.push(pred);
62            // gets all the lifetime projections of the place
63            // for example, if we add an invariant for x and x is a reference, then we also want to add
64            // an invariant for *x
65            let projs = ctxt.extract_lifetime_projections(place.with_inherent_region(ctxt));
66            for proj in projs {
67                let indirect = self
68                    .deps
69                    .require_dep::<IndirectPredicatesEnc>(proj.with_base(task))
70                    .unwrap();
71                inv.extend(indirect.predicate_applications.iter().map(|p| {
72                    p.reify(
73                        self.vcx,
74                        ty_out.ref_to_snap(place_res.expr.expect_predicate()),
75                    )
76                }));
77            }
78        }
79
80        for (inputs, outputs) in self.get_abstraction_edges(state.borrow_pcg().graph()) {
81            let mut let_bind = WandOldOuter::LetBind(Vec::new());
82            let mut wand_rhs = Vec::new();
83            for i in inputs {
84                self.encode_pcg_node(&i, &mut wand_rhs, &mut let_bind)?;
85            }
86            let mut wand_lhs = Vec::new();
87            for i in outputs {
88                let i = match *i {
89                    PcgNode::LifetimeProjection(region_projection) => region_projection,
90                    PcgNode::Place(_) => unreachable!(),
91                };
92                let exprs = self.encode_lifetime_projection(i, &mut let_bind)?;
93                wand_lhs.extend(exprs);
94            }
95            let wand = self.vcx.mk_wand(
96                self.vcx.mk_conj(self.vcx.alloc_slice(&wand_lhs)),
97                self.vcx.mk_conj(self.vcx.alloc_slice(&wand_rhs)),
98            );
99            let mut wand = self.vcx.mk_wand_expr(wand);
100            let WandOldOuter::LetBind(let_bind) = let_bind else {
101                unreachable!()
102            };
103            for lb in let_bind {
104                wand = lb.map_or_else(
105                    |(d, e)| self.vcx.mk_let_expr(d, e, wand),
106                    |(d, e)| self.vcx.mk_let_expr(d, e, wand),
107                );
108            }
109            inv.push(wand);
110        }
111
112        Ok((inv, loop_invariant_place_capabilities.into_iter().collect()))
113    }

115    pub(super) fn encode_pcg_node<T: PcgLifetimeProjectionBaseLike<'vir>>(
116        &mut self,
117        node: &PcgNode<'vir, MaybeLabelledPlace<'vir>, T>,
118        wand_rhs: &mut Vec<vir::ExprBool<'vir>>,
119        old_outer: &mut WandOldOuter<'vir>,
120    ) -> EncodeResult<'vir, (), E> {
121        match node {
122            PcgNode::Place(place) => {
123                let p = place.place();
124                let ty = (*p).ty(self.local_decls, self.vcx.tcx());
125                let task = RustTyDecomposition::from_ty(ty.ty, self.def_id);
126                let ty_out = self.deps.require_dep::<TyUseImpureEnc>(task).unwrap();
127                let p = self.encode_place(p)?;
128                let p = self.configure_old((*place).into(), p.expr.expect_predicate(), old_outer);
129
130                let pred = ty_out.ref_to_pred(self.vcx, p, None);
131                wand_rhs.push(pred);
132            }
133            PcgNode::LifetimeProjection(r) => {
134                let exprs = self.encode_lifetime_projection(*r, old_outer)?;
135                wand_rhs.extend(exprs);
136            }
137        }
138        Ok(())
139    }

141    pub(super) fn encode_lifetime_projection<T: PcgLifetimeProjectionBaseLike<'vir>>(
142        &mut self,
143        r: LifetimeProjection<'vir, T>,
144        old_outer: &mut WandOldOuter<'vir>,
145    ) -> EncodeResult<'vir, Vec<vir::ExprBool<'vir>>, E> {
146        let place = r.base().to_pcg_lifetime_projection_base();
147        let (place_snap, ty, _) = match place {
148            PcgLifetimeProjectionBase::Place(p) => {
149                self.encode_maybe_remote_place_snap(p, old_outer)?
150            }
151            PcgLifetimeProjectionBase::Const(c) => todo!("{c:?}"),
152        };
153        let ty = RustTyDecomposition::from_ty(ty.ty, self.def_id);
154        let indirect = self
155            .deps
156            .require_dep::<IndirectPredicatesEnc>(r.with_base(ty))
157            .unwrap();
158        Ok(indirect
159            .predicate_applications
160            .into_iter()
161            .map(|expr| expr.reify(self.vcx, place_snap))
162            .collect::<Vec<_>>())
163    }

165    fn get_place(place: MaybeRemotePlace<'vir>) -> Place<'vir> {
166        match place {
167            MaybeRemotePlace::Local(MaybeLabelledPlace::Current(place)) => place,
168            MaybeRemotePlace::Local(MaybeLabelledPlace::Labelled(place)) => place.place(),
169            MaybeRemotePlace::Remote(r) => r.assigned_local().into(),
170        }
171    }
172
173    fn encode_maybe_remote_place_snap(
174        &mut self,
175        place: MaybeRemotePlace<'vir>,
176        old_outer: &mut WandOldOuter<'vir>,
177    ) -> EncodeResult<'vir, (vir::ExprSnap<'vir>, mir::PlaceTy<'vir>, TyUseImpure<'vir>), E> {
178        let p = Self::get_place(place);
179        let (_, place_snap, ty, ty_out) = self.encode_place_with_snap(p)?;
180        let place_snap = self.configure_old(place, place_snap, old_outer);
181        Ok((place_snap, ty, ty_out))
182    }
183
184    fn configure_old<T: SnapOrRef>(
185        &mut self,
186        place: MaybeRemotePlace,
187        expr: vir::Expr<'vir, T>,
188        old_outer: &mut WandOldOuter<'vir>,
189    ) -> vir::Expr<'vir, T> {
190        match place {
191            MaybeRemotePlace::Local(MaybeLabelledPlace::Current { .. }) => {
192                self.mk_wand_outer(expr, old_outer)
193            }
194            MaybeRemotePlace::Local(MaybeLabelledPlace::Labelled(place)) => {
195                let label = self.get_location_label(place.at());
196                self.vcx.mk_old(expr, label)
197            }
198            MaybeRemotePlace::Remote(_) => self.vcx.mk_old_expr(expr),
199        }
200    }
201
202    fn mk_wand_outer<T: SnapOrRef>(
203        &mut self,
204        expr: vir::Expr<'vir, T>,
205        old_outer: &mut WandOldOuter<'vir>,
206    ) -> vir::Expr<'vir, T> {
207        match old_outer {
208            WandOldOuter::LetBind(let_bind) => {
209                let ident = vir::vir_format!(self.vcx, "_lb{}", let_bind.len());
210                let decl = self.vcx.mk_local_decl(ident, expr.ty());
211                let_bind.push(T::as_result(decl, expr));
212                self.vcx.mk_local_ex(decl)
213            }
214            WandOldOuter::Label(label) => {
215                let label = *label.get_or_insert_with(|| self.new_label("outer_package"));
216                self.vcx.mk_local_labelled_old_expr(expr, label)
217            }
218        }
219    }
220}

222type LetBind<'vir> = Result<
223    (vir::LocalDeclSnap<'vir>, vir::ExprSnap<'vir>),
224    (vir::LocalDeclRef<'vir>, vir::ExprRef<'vir>),
225>;
226
227trait SnapOrRef: vir::CompType {
228    fn as_result<'vir>(d: vir::LocalDecl<'vir, Self>, e: vir::Expr<'vir, Self>) -> LetBind<'vir>;
229}
230
231impl SnapOrRef for vir::Snap {
232    fn as_result<'vir>(d: vir::LocalDecl<'vir, Self>, e: vir::Expr<'vir, Self>) -> LetBind<'vir> {
233        Ok((d, e))
234    }
235}
236
237impl SnapOrRef for vir::Ref {
238    fn as_result<'vir>(d: vir::LocalDecl<'vir, Self>, e: vir::Expr<'vir, Self>) -> LetBind<'vir> {
239        Err((d, e))
240    }
