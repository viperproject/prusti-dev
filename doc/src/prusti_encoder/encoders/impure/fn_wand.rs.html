32impl<'vir, E: TaskEncoder> ImpureEncVisitor<'vir, '_, E> {
33    pub fn package_wands(
34        &mut self,
35        final_borrow_state: &BorrowsState<'_, 'vir>,
36    ) -> EncodeResult<'vir, Vec<vir::Stmt<'vir>>, E> {
37        let mut wand_packages = Vec::new();
38        let label = self.new_label("package_post");
39        let result = self.local_defs[mir::RETURN_PLACE].impure_snap;
40        let result = self.vcx.mk_local_labelled_old_expr(result, label);
41        let args = self
42            .local_defs
43            .args()
44            .map(|a| self.vcx.mk_old_expr(a.impure_snap));
45        let args = PledgeExpr::pledge_args(result, args);
46
47        for wand_data in self.wands.viper_wands() {
48            let Some(wand) = self
49                .wands
50                .mk_wand(&wand_data, args, None, None, self.vcx, self.deps)
51            else {
52                continue;
53            };
54            let mut package_script = Vec::new();
55            for rhs in wand_data.rhs.iter() {
56                let ug = UnblockGraph::for_node(
57                    mir::Place::from(rhs.mir_local()),
58                    final_borrow_state,
59                    self.pcg_ctxt(),
60                );
61                let actions = ug.actions(self.pcg_ctxt()).unwrap();
62                let unblock = self.block(|visitor| {
63                    visitor.pcs_unblock_actions(final_borrow_state, &actions, Some(label))
64                })?;
65                package_script.extend(unblock);
66            }
67
68            if !wand_data.pledges.is_empty() {
69                // Statements in the package script only see resources already
70                // in the package state. A resource that is not obtained from
71                // the LHS (e.g. an argument the result does not borrow from)
72                // is only moved there when consumed, which for the RHS happens
73                // after the script. Asserting the RHS resources moves them in
74                // early, so that the pledge exhales below can read them.
75                let resources = wand_data
76                    .rhs
77                    .iter()
78                    .filter_map(|g| {
79                        self.wands.encode_predicates_for_function_shape_node(
80                            self.vcx,
81                            self.deps,
82                            *g,
83                            None,
84                            |i| args[i],
85                        )
86                    })
87                    .collect::<Vec<_>>();
88                package_script.push(self.vcx.mk_assert_stmt(self.vcx.mk_conj(&resources)));
89            }
90
91            for EncodedPledge {
92                expiry_postcondition,
93                ..
94            } in &wand_data.pledges
95            {
96                let span = expiry_postcondition.span();
97                self.vcx.with_span(span, |vcx| {
98                    vcx.handle_error("exhale.failed:assertion.false", move |_| {
99                        Some(vec![PrustiError::verification(
100                            "pledge postcondition might not hold",
101                            span.into(),
102                        )])
103                    });
104                    package_script.push(vcx.mk_exhale_stmt(expiry_postcondition.expr(args)));
105                });
106            }
107            wand_packages.push(
108                self.vcx
109                    .mk_package_stmt(wand, self.vcx.alloc_slice(&package_script)),
110            );
111        }
112        Ok(wand_packages)
113    }
114}

118/// Not tied to a caller or callee context. `indirect_pres`, `indirect_posts`,
119/// `wand_posts`, and `package_wands` are identity-substituted and intended for
120/// use in the callee's own contract; `apply_wands` is for caller use and
121/// re-substitutes via a [`WandCallContext`].
122#[derive(Clone)]
123pub struct WandEncOutput<'vir> {
124    /// Information about the corresponding function.
125    function_data: FunctionData<'vir>,
126
127    /// The lifetime projections of all arguments to the function.
128    inputs: Vec<FunctionShapeInput<Generalized>>,
129
130    /// The lifetime projections of all function outputs (according to the
131    /// corresponding [`FunctionShape`]). This *includes* lifetime projections
132    /// of nested lifetimes in the function arguments.
133    outputs: Vec<FunctionShapeOutput<Generalized>>,
134
135    /// Encoded VIR expressions for the magic wands.
136    wands: Vec<WandData<'vir>>,
137}

139/// Substitution context for instantiating a wand at a call site. When `None`,
140/// the wand is encoded using the callee's identity substitution (appropriate
141/// when emitting wands inside the function being defined). When `Some`, the
142/// wand is re-encoded with the call-site substitutions and the caller's
143/// generic parameters, so that placeholders like `Self` or other callee
144/// generics are replaced by concrete types from the caller's perspective.
145pub type WandCallContext<'vir> = Option<GArgs<'vir>>;

147impl<'vir> WandEncOutput<'vir> {
148    pub(crate) fn fn_sig(
149        &self,
150        vcx: &'vir vir::VirCtxt<'vir>,
151        call_ctx: WandCallContext<'vir>,
152    ) -> ty::FnSig<'vir> {
153        match call_ctx {
154            // TODO: change pcg's `fn_sig` to take `&[GenericArg]` instead of `GenericArgsRef`
155            Some(ctx) => self
156                .function_data
157                .fn_sig(vcx.tcx(), vcx.tcx().mk_args(ctx.args())),
158            None => self.function_data.identity_fn_sig(vcx.tcx()),
159        }
160    }

162    pub(crate) fn g_params(
163        &self,
164        vcx: &'vir vir::VirCtxt<'vir>,
165        call_ctx: WandCallContext<'vir>,
166    ) -> GParams<'vir> {
167        match call_ctx {
168            Some(ctx) => ctx.context(),
169            None => GParams::new(
170                self.function_data.identity_substs(vcx.tcx()),
171                self.function_data.param_env(vcx.tcx()),
172                false,
173            ),
174        }
175    }

177    /// The (unreified) predicates associated with the given node, or `None` if
178    /// there are no resources associated with it.
179    #[allow(clippy::type_complexity)]
180    fn predicates_for_function_shape_node(
181        &self,
182        vcx: &'vir vir::VirCtxt<'vir>,
183        deps: &mut TaskEncoderDependencies<'vir, impl TaskEncoder>,
184        g: FunctionShapeNode<Generalized>,
185        call_ctx: WandCallContext<'vir>,
186    ) -> Option<Vec<vir::ExprGenBool<'vir, vir::ExprSnap<'vir>, vir::ExprKind<'vir>>>> {
187        let arg_ty = g.ty(self.fn_sig(vcx, call_ctx));
188        let decomp = RustTyDecomposition::from_ty(arg_ty, self.g_params(vcx, call_ctx));
189        let region_proj =
190            projection_for_generalized_idx(arg_ty, g.region_idx(), decomp, vcx.tcx())?;
191        let predicates = deps
192            .require_dep::<IndirectPredicatesEnc>(region_proj)
193            .unwrap()
194            .predicate_applications;
195        (!predicates.is_empty()).then_some(predicates)
196    }
197
198    fn encode_predicates_for_function_shape_node(
199        &self,
200        vcx: &'vir vir::VirCtxt<'vir>,
201        deps: &mut TaskEncoderDependencies<'vir, impl TaskEncoder>,
202        g: impl Into<FunctionShapeNode<Generalized>>,
203        call_ctx: WandCallContext<'vir>,
204        mut snap: impl FnMut(mir::Local) -> vir::ExprSnap<'vir>,
205    ) -> Option<vir::ExprBool<'vir>> {
206        use vir::Reify;
207        let g = g.into();
208        let predicates = self.predicates_for_function_shape_node(vcx, deps, g, call_ctx)?;
209
210        let local = g.mir_local();
211        let local_snap = snap(local);
212        Some(
213            vcx.mk_conj(
214                &predicates
215                    .iter()
216                    .map(|p| p.reify(vcx, local_snap))
217                    .collect::<Vec<_>>(),
218            ),
219        )
220    }

222    pub fn indirect_pres<'a, E: TaskEncoder>(
223        &'a self,
224        vcx: &'vir vir::VirCtxt<'vir>,
225        local_defs: &'a MirLocalDefEncOutput<'vir>,
226        deps: &'a mut TaskEncoderDependencies<'vir, E>,
227    ) -> impl Iterator<Item = vir::ExprBool<'vir>> + 'a {
228        self.inputs().filter_map(|g| {
229            self.encode_predicates_for_function_shape_node(vcx, deps, g, None, |i| {
230                local_defs[i].impure_snap
231            })
232        })
233    }

235    pub fn indirect_posts<'a, E: TaskEncoder>(
236        &'a self,
237        vcx: &'vir vir::VirCtxt<'vir>,
238        local_defs: &'a MirLocalDefEncOutput<'vir>,
239        deps: &'a mut TaskEncoderDependencies<'vir, E>,
240    ) -> impl Iterator<Item = vir::ExprBool<'vir>> + 'a {
241        // The encoded predicates for the input lifetime projections that are
242        // not blocked by any of the result lifetime projections. These will be
243        // encoded as part of the postcondition of the function (in contrast,
244        // the predicates for the blocked inputs will appear on the right-hand
245        // side of a magic wand in the postcondition).
246        let unblocked_input_posts = self
247            .inputs()
248            .filter(|i| !self.blocked_inputs().contains(i))
249            .filter_map(|lp| {
250                self.encode_predicates_for_function_shape_node(vcx, deps, lp, None, |i| {
251                    vcx.mk_old_expr(local_defs[i].impure_snap)
252                })
253            })
254            .collect::<Vec<_>>()
255            .into_iter();
256
257        let output_posts = self.outputs().filter_map(|g| {
258            self.encode_predicates_for_function_shape_node(vcx, deps, g, None, |i| {
259                local_defs[i].impure_snap
260            })
261        });
262        unblocked_input_posts.chain(output_posts)
263    }

265    pub fn wand_posts<'a, E: TaskEncoder>(
266        &'a self,
267        vcx: &'vir vir::VirCtxt<'vir>,
268        local_defs: &'a MirLocalDefEncOutput<'vir>,
269        deps: &'a mut TaskEncoderDependencies<'vir, E>,
270    ) -> impl Iterator<Item = vir::ExprBool<'vir>> + 'a {
271        let wand_result =
272            vcx.mk_local_decl("wand_result", local_defs[mir::RETURN_PLACE].local_snap.ty());
273        let wand_result_expr = vcx.mk_local_ex(wand_result);
274        let args = local_defs
275            .args()
276            .map(|arg| vcx.mk_old_expr(arg.impure_snap));
277        let args = PledgeExpr::pledge_args(wand_result_expr, args);
278
279        // TODO: wands for late-bound regions
280        self.viper_wands().into_iter().filter_map(move |wand_data| {
281            let wand = self.mk_wand(&wand_data, args, None, None, vcx, deps)?;
282            Some(vcx.mk_let_expr(
283                wand_result,
284                local_defs[mir::RETURN_PLACE].impure_snap,
285                vcx.mk_wand_expr(wand),
286            ))
287        })
288    }

290    pub fn apply_wands<E: TaskEncoder>(
291        &self,
292        arguments: &[vir::ExprSnap<'vir>],
293        label_pre: &'vir str,
294        label_post: &'vir str,
295        call_ctx: GArgs<'vir>,
296        visitor: &mut ImpureEncVisitor<'vir, '_, E>,
297    ) {
298        let result = visitor
299            .vcx
300            .mk_local_labelled_old_expr(arguments[mir::RETURN_PLACE.as_usize()], label_post);
301        let args = (1..arguments.len()).map(|l| {
302            visitor
303                .vcx
304                .mk_local_labelled_old_expr(arguments[l], label_pre)
305        });
306        let args = PledgeExpr::pledge_args(result, args);
307        for wand_data in self.viper_wands() {
308            let Some(wand) = self.mk_wand(
309                &wand_data,
310                args,
311                Some(label_pre),
312                Some(call_ctx),
313                visitor.vcx,
314                visitor.deps,
315            ) else {
316                continue;
317            };
318            visitor.stmt(visitor.vcx.mk_apply_stmt(wand));
319        }
320    }

322    fn mk_wand<E: TaskEncoder>(
323        &self,
324        wand_data: &WandData<'vir>,
325        pledge_args: PledgeArgs<'vir>,
326        pledge_old_label: Option<&'vir str>,
327        call_ctx: WandCallContext<'vir>,
328        vcx: &'vir vir::VirCtxt<'vir>,
329        deps: &mut TaskEncoderDependencies<'vir, E>,
330    ) -> Option<vir::Wand<'vir>> {
331        debug_assert!(!wand_data.lhs.is_empty());
332        let generics = self.generics_subst(call_ctx, vcx, deps);
333        let pledge_expr = |pledge: &PledgeExpr<'vir>| match pledge_old_label {
334            Some(label) => {
335                vcx.with_local_subst(generics, || pledge.expr_at_label(pledge_args, label))
336            }
337            None => pledge.expr(pledge_args),
338        };
339        let rhs = wand_data.rhs.iter().filter_map(|g| {
340            self.encode_predicates_for_function_shape_node(vcx, deps, *g, call_ctx, |i| {
341                pledge_args[i]
342            })
343        });
344        let rhs = rhs
345            .chain(
346                wand_data
347                    .pledges
348                    .iter()
349                    .map(|pledge| pledge_expr(&pledge.expiry_postcondition)),
350            )
351            .collect::<Vec<_>>();
352        if rhs.is_empty() {
353            // We skip emitting the wand when there is nothing on the RHS, i.e.,
354            // nothing would be unblocked by applying this wand, nor are there
355            // any pledge postconditions.
356            return None;
357        }
358        let rhs = vcx.mk_conj(&rhs);
359        let lhs = wand_data.lhs.iter().filter_map(|g| {
360            self.encode_predicates_for_function_shape_node(vcx, deps, *g, call_ctx, |i| {
361                pledge_args[i]
362            })
363        });
364        let lhs = lhs
365            .chain(
366                wand_data
367                    .pledges
368                    .iter()
369                    .filter_map(|pledge| pledge.expiry_obligation.as_ref().map(pledge_expr)),
370            )
371            .collect::<Vec<_>>();
372        let lhs = vcx.mk_conj(&lhs);
373        Some(vcx.mk_wand(lhs, rhs))
374    }
375
376    /// The pledges are encoded once, at the callee's identity substitution,
377    /// so they refer to the callee's generic parameters. At a call site, these
378    /// are replaced by the call's generic arguments, as Viper does for the
379    /// callee's postcondition (from which the caller holds the wand).
380    fn generics_subst<E: TaskEncoder>(
381        &self,
382        call_ctx: WandCallContext<'vir>,
383        vcx: &'vir vir::VirCtxt<'vir>,
384        deps: &mut TaskEncoderDependencies<'vir, E>,
385    ) -> &'vir FxHashMap<&'vir str, vir::ExprDyn<'vir>> {
386        let Some(call_args) = call_ctx else {
387            return vcx.alloc(FxHashMap::default());
388        };
389        let params = deps
390            .require_dep::<GenericParamsEnc>(self.g_params(vcx, None))
391            .unwrap();
392        let args = deps.require_dep::<GArgsTyEnc>(call_args).unwrap();
393        let tys = params
394            .ty_decls()
395            .iter()
396            .zip(args.get_ty::<(), !>())
397            .map(|(decl, arg)| (decl.name, arg.as_dyn()));
398        let consts = params
399            .const_decls()
400            .iter()
401            .zip(args.get_const::<(), !>())
402            .map(|(decl, arg)| (decl.name, arg.as_dyn()));
403        vcx.alloc(tys.chain(consts).collect())
404    }
405}

461        task.clone()
462    }

464    fn describe_error(error: Self::EncodingError) -> String {
465        match error {

469
470    fn do_encode_full<'vir>(
471        task_key: &Self::TaskKey<'vir>,
472        deps: &mut TaskEncoderDependencies<'vir, Self>,
473    ) -> EncodeFullResult<'vir, Self> {
474        deps.emit_output_ref(task_key.clone(), ())?;
475        vir::with_vcx(|vcx| {
476            let def_id = task_key.def_id();
477
478            let shape = task_key.function_shape(vcx).map_err(|e| {
479                EncodeFullError::EncodingError(
480                    WandEncError::Unsupported(format!("function shape: {e:?}")),
481                    None,
482                )
483            })?;
484
485            let coupled_edges = shape.coupled_edges();
486            let edges: FxHashSet<_> = shape.edges().map(|e| (e.input(), e.output())).collect();
487
488            let (inputs, outputs) = shape.take_inputs_and_outputs();
489            let spec = deps.require_dep::<MirSpecEnc>((def_id, def_id, MirSpecEncMode::Impure))?;
490            if coupled_edges.is_empty() {
491                assert!(spec.pledges.is_empty());
492                return Ok((
493                    (),
494                    WandEncOutput {
495                        function_data: task_key.data,
496                        inputs,
497                        outputs,
498                        wands: vec![],
499                    },
500                ));
501            }
502            let pledges = spec.pledges;
503            let wands: Vec<WandData<'vir>> = coupled_edges
504                .into_iter()
505                .filter_map(|hyper_edge| {
506                    let (sources, mut targets) = hyper_edge.into_tuple();
507                    // We don't want to emit an identity wand, like P --* P. This can happen when
508                    // PCG returns self-edges, like for fn(x: &'a mut &'b i32) where 'b is in
509                    // invariant position and we therefore have an edge x|'b -> x|'b.
510                    // Currently, these edges also prevent us from emitting indirect postconditions.
511                    // TODO: we might want to emit these identity wands in the future to attach functional
512                    // specifications to them. We still need to emit the resources on the wand's LHS.
513                    let mut sources_as_nodes = sources
514                        .iter()
515                        .map(|&s| s.to_function_shape_node())
516                        .collect::<Vec<_>>();
517                    sources_as_nodes.sort();
518                    targets.sort();
519                    if sources_as_nodes == targets {
520                        return None;
521                    }
522                    Some(WandData::new(targets, sources, pledges.clone()))
523                })
524                .collect();
525            let mut output: WandEncOutput<'vir> = WandEncOutput {
526                function_data: task_key.data,
527                inputs,
528                outputs,
529                wands: Vec::new(),
530            };
531            output.wands = output
532                .select_wands(wands, !pledges.is_empty(), &edges, vcx, deps)
533                .map_err(|err| EncodeFullError::EncodingError(err, None))?;
534            Ok(((), output))
535        })
536    }
537}

539impl<'vir> WandEncOutput<'vir> {
540    /// Selects the wands to emit among those of the coupled edges, rejecting
541    /// the shapes that cannot be encoded precisely.
542    fn select_wands<E: TaskEncoder>(
543        &self,
544        wands: Vec<WandData<'vir>>,
545        has_pledges: bool,
546        edges: &FxHashSet<(WandRhsKey, WandLhsKey)>,
547        vcx: &'vir vir::VirCtxt<'vir>,
548        deps: &mut TaskEncoderDependencies<'vir, E>,
549    ) -> Result<Vec<WandData<'vir>>, WandEncError> {
550        let mut has_resources = |g: FunctionShapeNode<Generalized>| {
551            self.predicates_for_function_shape_node(vcx, deps, g, None)
552                .is_some()
553        };
554        // A wand is only needed if it gives back a resource, or to carry the
555        // pledges if none does (e.g. for shared references).
556        let (mut wands, resourceless): (Vec<_>, Vec<_>) = wands
557            .into_iter()
558            .partition(|wand_data| wand_data.rhs.iter().any(|g| has_resources((*g).into())));
559        // A wand gives back its sources only once all of its targets have
560        // expired. That matches the signature only if each source with
561        // resources flows into each target with resources.
562        for wand_data in &wands {
563            let sources = wand_data
564                .rhs
565                .iter()
566                .filter(|g| has_resources((**g).into()))
567                .collect::<Vec<_>>();
568            let targets = wand_data
569                .lhs
570                .iter()
571                .filter(|g| has_resources(**g))
572                .collect::<Vec<_>>();
573            let precise = sources.iter().all(|source| {
574                let mut targets = targets.iter();
575                targets.all(|target| edges.contains(&(**source, **target)))
576            });
577            if !precise {
578                return Err(WandEncError::Unsupported(
579                    "borrows in the result that expire separately but depend on a common \
580                     argument are not supported"
581                        .to_string(),
582                ));
583            }
584        }
585        if has_pledges {
586            if wands.is_empty() {
587                wands.extend(resourceless.into_iter().take(1));
588            } else if wands.len() > 1 {
589                // It is unclear which expiry the pledges refer to.
590                return Err(WandEncError::Unsupported(
591                    "pledges on a function whose result contains borrows that expire \
592                     separately are not supported"
593                        .to_string(),
594                ));
595            }
596        }
597        Ok(wands)
598    }

600    pub fn viper_wands(&self) -> Vec<WandData<'vir>> {
601        self.wands.clone()
602    }

604    /// All lifetime projections in the arguments that are blocked by any of the
605    /// lifetime projections in the function's result.
606    pub fn blocked_inputs(&self) -> FxHashSet<FunctionShapeInput<Generalized>> {
607        self.wands
608            .iter()
609            .flat_map(|wand| wand.rhs.iter().copied())
610            .collect()
611    }
