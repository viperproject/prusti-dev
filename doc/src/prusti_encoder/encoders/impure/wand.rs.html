15type Input<'a> = AbstractionInputTarget<'a>;
16type Output<'a> = AbstractionOutputTarget<'a>;
17type Inputs<'a> = Vec<Input<'a>>;
18type Outputs<'a> = Vec<Output<'a>>;
19
20impl<'vir, 'enc, E: TaskEncoder> ImpureEncVisitor<'vir, 'enc, E> {
21    pub(super) fn get_abstraction_edges<'a>(
22        &self,
23        g: &'a BorrowsGraph<'vir>,
24    ) -> Vec<(Inputs<'vir>, Outputs<'vir>)> {
25        g.coupled_edges()
26            .into_iter()
27            .map(|edge| (edge.value().inputs(self.pcg_ctxt()), edge.value().outputs()))
28            .collect()
29    }
30
31    pub(crate) fn pcs_handle_wand(
32        &mut self,
33        borrows_state: &BorrowsState<'_, 'vir>,
34        package: bool,
35        edge: &PcgCoupledEdgeKind<'vir>,
36        label: Option<&'vir str>,
37        edge_to_loop: bool,
38    ) -> EncodeResult<'vir, (), E> {
39        // TODO: there is something in the pcs which emits spurious? opaque edge creation, skip these
40        if package && !edge_to_loop {
41            return Ok(());
42        }
43        let inputs = edge.inputs(self.pcg_ctxt());
44        let outputs = edge.outputs();
45        let mut old_outer = WandOldOuter::Label(label);
46        let mut proof_block = Vec::new();
47        let mut wand_rhs = Vec::new();
48        for i in inputs {
49            self.encode_pcg_node(&*i, &mut wand_rhs, &mut old_outer)?;
50            if package {
51                proof_block.extend(self.create_package_script(
52                    borrows_state,
53                    *i,
54                    &mut old_outer,
55                )?);
56            }
57        }
58        let mut wand_lhs = Vec::new();
59        for i in outputs {
60            let i = i.expect_lifetime_projection();
61            let exprs = self.encode_lifetime_projection(i, &mut old_outer)?;
62            wand_lhs.extend(exprs);
63        }
64        let wand = self.vcx.mk_wand(
65            self.vcx.mk_conj(self.vcx.alloc_slice(&wand_lhs)),
66            self.vcx.mk_conj(self.vcx.alloc_slice(&wand_rhs)),
67        );
68        if package {
69            let proof_block = self.vcx.alloc_slice(&proof_block);
70            self.stmt(self.vcx.mk_package_stmt(wand, proof_block));
71        } else {
72            self.stmt(self.vcx.mk_apply_stmt(wand));
73        }
74        Ok(())
75    }
76
77    fn create_package_script(
78        &mut self,
79        borrows_state: &BorrowsState<'_, 'vir>,
80        rhs: impl Into<PcgNode<'vir>>,
81        old_outer: &mut WandOldOuter<'vir>,
82    ) -> EncodeResult<'vir, Vec<vir::Stmt<'vir>>, E> {
83        let ug = UnblockGraph::for_node(rhs, borrows_state, self.pcg_ctxt());
84
85        let WandOldOuter::Label(label) = old_outer else {
86            unreachable!()
87        };
88        let label = *label.get_or_insert_with(|| self.new_label("outer_package"));
89        let actions = ug.actions(self.pcg_ctxt()).unwrap();
90
91        self.block(|visitor| visitor.pcs_unblock_actions(borrows_state, &actions, Some(label)))
92    }
93}
