15/// Encodes the builtin MIR binary operations (e.g. `Add`, `Sub`, `Mul`, `Div`,
16/// etc.) as Viper functions with the correct semantics.
17pub struct MirBuiltinBinOpEnc;
18
19#[derive(Clone, Debug)]
20pub enum MirBuiltinBinOpEncError {
21    Unsupported(#[allow(dead_code)] String),
22}
23
24type EncodeResult<'vir, T> = Result<T, EncodeFullError<'vir, MirBuiltinBinOpEnc>>;
25
26#[derive(Debug, Clone, Copy, Hash, PartialEq, Eq)]
27pub struct MirBuiltinBinOpTask<'vir> {
28    result_ty: RustTyDecomposition<'vir>,
29    op: mir::BinOp,
30    lhs_ty: RustTy<'vir>,
31    rhs_ty: RustTy<'vir>,
32}
33
34impl<'vir> MirBuiltinBinOpTask<'vir> {
35    pub fn new(
36        mut result_ty: RustTyDecomposition<'vir>,
37        op: mir::BinOp,
38        lhs_ty: RustTyDecomposition<'vir>,
39        rhs_ty: RustTyDecomposition<'vir>,
40    ) -> Self {
41        // The result type is always the concrete `<int>` or `(<int>, bool)`),
42        // remove the context to avoid duplicate keys.
43        result_ty.args = result_ty.args.with_empty_context();
44        Self {
45            result_ty,
46            op,
47            lhs_ty: lhs_ty.ty,
48            rhs_ty: rhs_ty.ty,
49        }
50    }
51}

53impl TaskEncoder for MirBuiltinBinOpEnc {
54    task_encoder::encoder_cache!(MirBuiltinBinOpEnc);
55    const ENCODER_NAME: &'static str = "MIR builtin binary op encoder";
56
57    type TaskDescription<'vir> = MirBuiltinBinOpTask<'vir>;
58
59    type OutputFullDependency<'vir> = vir::FunctionIdn<'vir, (vir::CSnap, vir::CSnap), vir::CSnap>;
60    type OutputFullLocal<'vir> = vir::Function<'vir>;
61
62    type EncodingError = MirBuiltinBinOpEncError;
63
64    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
65        *task
66    }
67
68    fn do_encode_full<'vir>(
69        task_key: &Self::TaskKey<'vir>,
70        deps: &mut TaskEncoderDependencies<'vir, Self>,
71    ) -> EncodeFullResult<'vir, Self> {
72        deps.emit_output_ref(*task_key, ())?;
73        vir::with_vcx(|vcx| {
74            let MirBuiltinBinOpTask {
75                result_ty,
76                op,
77                lhs_ty,
78                rhs_ty,
79            } = *task_key;
80            let mut encode = |ty| {
81                let ty = RustTyDecomposition::identity(ty);
82                deps.require_dep::<TyUsePureEnc>(ty)
83            };
84            let lhs_ty_data = encode(lhs_ty)?;
85            let l_ty_snap = lhs_ty_data.snapshot.downcast_ty();
86            let rhs_ty_data = encode(rhs_ty)?;
87            let r_ty_snap = rhs_ty_data.snapshot.downcast_ty();
88            let res_ty_data = encode(result_ty.ty)?;
89            let res_ty_snap = res_ty_data.snapshot.downcast_ty();
90
91            let name = vir::vir_format_identifier!(
92                vcx,
93                "mir_binop_{op:?}_{}_{}",
94                lhs_ty.name(),
95                rhs_ty.name()
96            );
97            let fn_idn = FunctionIdn::new(name, (l_ty_snap, r_ty_snap), res_ty_snap);
98
99            let lhs_decl = vcx.mk_local_decl("arg1", l_ty_snap);
100            let rhs_decl = vcx.mk_local_decl("arg2", r_ty_snap);
101            let lhs = vcx.mk_local_ex(lhs_decl);
102            let rhs = vcx.mk_local_ex(rhs_decl);
103            let (pres, body) = match lhs_ty_data.specifics {
104                TySpecifics::Raw(..) => {
105                    let body = Self::handle_bin_op_raw(vcx, lhs, rhs, op)?;
106                    (Vec::new(), body)
107                }
108                TySpecifics::Primitive(
109                    l_ty_prim @ TyPurePrimData {
110                        kind: TyPurePrimDataKind::Bool | TyPurePrimDataKind::Int(_),
111                    },
112                ) => {
113                    let r_ty_prim = rhs_ty_data.expect_primitive();
114                    let lhs = l_ty_prim.snap_to_prim(lhs);
115                    let rhs = r_ty_prim.snap_to_prim(rhs);
116                    // `l_ty` is the type the operation is performed in. The operands
117                    // do not always share a type (e.g. a shift's amount may be a
118                    // different integer type than the shifted value), so we do not
119                    // require `lhs_ty == rhs_ty` here.
120                    let l_ty = *lhs_ty.expect_primitive();
121
122                    if op.is_overflowing() {
123                        let val =
124                            Self::handle_bin_op_overflowing(vcx, deps, result_ty, op, lhs, rhs)?;
125                        (Vec::new(), val)
126                    } else {
127                        let res_ty = *result_ty.ty.expect_primitive();
128                        let (pres, val) =
129                            Self::handle_bin_op_native(vcx, lhs, rhs, res_ty, op, l_ty, deps)?;
130                        (pres, res_ty_data.expect_primitive().prim_to_snap(val))
131                    }
132                }
133                TySpecifics::Primitive(TyPurePrimData {
134                    kind: TyPurePrimDataKind::Float(float),
135                }) => {
136                    let r_ty_prim = rhs_ty_data.expect_primitive();
137                    assert!(matches!(r_ty_prim.kind, TyPurePrimDataKind::Float(_)));
138                    let body = Self::handle_bin_op_float(vcx, lhs, rhs, op, float);
139                    (Vec::new(), body)
140                }
141                _ => unreachable!(),
142            };
143            let pres = vcx.alloc_slice(&pres);
144            let function =
145                vcx.mk_function(fn_idn, (lhs_decl, rhs_decl), pres, &[], None, Some(body));
146            Ok((function, fn_idn))
147        })
148    }
149
150    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
151        for function in Self::all_outputs_local_no_errors(program) {
152            program.add_function(function);
153        }
154    }
155}

157impl MirBuiltinBinOpEnc {
158    fn handle_bin_op_native<'vir>(
159        vcx: &'vir vir::VirCtxt<'vir>,
160        lhs: vir::ExprPrim<'vir>,
161        mut rhs: vir::ExprPrim<'vir>,
162        res_ty: ty::Ty<'vir>,
163        op: mir::BinOp,
164        in_ty: ty::Ty<'vir>,
165        deps: &mut TaskEncoderDependencies<'vir, Self>,
166    ) -> EncodeResult<'vir, (Vec<vir::ExprBool<'vir>>, vir::ExprPrim<'vir>)> {
167        use mir::BinOp::*;
168        if matches!(op, BitXor | BitAnd | BitOr | Shl | Shr) && !in_ty.is_bool() {
169            if matches!(op, Shl | Shr) {
170                // RHS must be smaller than the bit width of the LHS, this is
171                // implicit in the `Shl` and `Shr` operators.
172                rhs = vcx.mk_bin_op_expr(
173                    vir::BinOpKind::Mod,
174                    rhs.downcast_ty(),
175                    vcx.get_bit_width_int(in_ty.kind()),
176                );
177            }
178
179            let (size, is_signed) = vir::VirCtxt::get_int_data(in_ty.kind());
180            let bv_domain = deps.require_dep::<BitVecEnc>(match size {
181                8 => BitVecSize::BitVec8,
182                16 => BitVecSize::BitVec16,
183                32 => BitVecSize::BitVec32,
184                64 => BitVecSize::BitVec64,
185                128 => BitVecSize::BitVec128,
186                _ => unreachable!("Invalid bitvec size"),
187            })?;
188            let lhs_bv = (bv_domain.from_int)(lhs);
189            let rhs_bv = (bv_domain.from_int)(rhs);
190            let res_bv = match op {
191                BitXor => (bv_domain.bv_xor)(lhs_bv, rhs_bv),
192                BitAnd => (bv_domain.bv_and)(lhs_bv, rhs_bv),
193                BitOr => (bv_domain.bv_or)(lhs_bv, rhs_bv),
194                Shl => (bv_domain.bv_shl)(lhs_bv, rhs_bv),
195                Shr => (bv_domain.bv_shr)(lhs_bv, rhs_bv),
196                _ => unreachable!(),
197            };
198            let to_int = if is_signed {
199                bv_domain.sbv_to_int
200            } else {
201                bv_domain.ubv_to_int
202            };
203            Ok((vec![], (to_int)(res_bv)))
204        } else if matches!(op, Cmp) {
205            // Cmp does not have a direct analogue to VIR binary operations,
206            // so we treat it specially.
207            // a > b ? 1 : (b > a ? -1 : 0)
208            let a_gt_b = vcx
209                .mk_bin_op_expr(vir::BinOpKind::CmpGt, lhs, rhs)
210                .downcast_ty();
211            let b_gt_a = vcx
212                .mk_bin_op_expr(vir::BinOpKind::CmpGt, rhs, lhs)
213                .downcast_ty();
214            let val = vcx
215                .mk_ternary_expr(
216                    a_gt_b,
217                    vcx.mk_int::<1>(),
218                    vcx.mk_ternary_expr(b_gt_a, vcx.mk_int::<-1>(), vcx.mk_int::<0>()),
219                )
220                .upcast_ty();
221            Ok((vec![], val))
222        } else {
223            let op_kind = vir::BinOpKind::from(op);
224            let viper_val = vcx
225                .mk_bin_op_expr_inner(op_kind, lhs.as_dyn(), rhs.as_dyn())
226                .downcast_ty();
227            Ok(match op {
228                // Overflow well defined as wrapping (implicit) and for the shifts
229                // the RHS will be masked to the bit width.
230                Add | Sub | Mul | Shl | Shr => (
231                    Vec::new(),
232                    vcx.get_wrapped_val(viper_val.downcast_ty(), res_ty.kind())
233                        .upcast_ty(),
234                ),
235                // Undefined behavior to overflow (need precondition)
236                AddUnchecked | SubUnchecked | MulUnchecked => {
237                    let min = vcx.get_min_int(res_ty.kind());
238                    // `(arg1 op arg2) >= -iN::MIN`
239                    let lower_bound = vcx
240                        .mk_bin_op_expr(vir::BinOpKind::CmpGe, viper_val.downcast_ty(), min)
241                        .downcast_ty::<vir::Bool>();
242                    let max = vcx.get_max_int(res_ty.kind());
243                    // `(arg1 op arg2) <= iN::MAX`
244                    let upper_bound = vcx
245                        .mk_bin_op_expr(vir::BinOpKind::CmpLe, viper_val.downcast_ty(), max)
246                        .downcast_ty::<vir::Bool>();
247                    (vec![lower_bound, upper_bound], viper_val)
248                }
249                // Overflow is well defined as wrapping (implicit), but shifting by
250                // more than the bit width (or less than 0) is undefined behavior.
251                ShlUnchecked | ShrUnchecked => {
252                    let min = vcx.mk_int::<0>();
253                    // `arg2 >= 0`
254                    let lower_bound = vcx
255                        .mk_bin_op_expr(vir::BinOpKind::CmpGe, rhs.downcast_ty(), min)
256                        .downcast_ty::<vir::Bool>();
257                    let max = vcx.get_bit_width_int(in_ty.kind());
258                    // `arg2 < bit_width(arg1)`
259                    let upper_bound = vcx
260                        .mk_bin_op_expr(vir::BinOpKind::CmpLt, rhs.downcast_ty(), max)
261                        .downcast_ty::<vir::Bool>();
262                    (
263                        vec![lower_bound, upper_bound],
264                        vcx.get_wrapped_val(viper_val.downcast_ty(), res_ty.kind())
265                            .upcast_ty(),
266                    )
267                }
268                // Could divide by zero or overflow if divisor is `-1`
269                Div | Rem => {
270                    // `0 != arg2 `
271                    let pre = vcx
272                        .mk_bin_op_expr(vir::BinOpKind::CmpNe, vcx.mk_int::<0>(), rhs.downcast_ty())
273                        .downcast_ty::<vir::Bool>();
274                    let mut pres = vec![pre];
275                    let mut val = viper_val;
276                    if res_ty.is_signed() {
277                        let min = vcx.get_min_int(res_ty.kind());
278                        // `arg1 != -iN::MIN`
279                        let arg1_cond =
280                            vcx.mk_bin_op_expr(vir::BinOpKind::CmpNe, lhs.downcast_ty(), min);
281                        // `-1 != arg2 `
282                        let arg2_cond = vcx.mk_bin_op_expr(
283                            vir::BinOpKind::CmpNe,
284                            vcx.mk_int::<-1>(),
285                            rhs.downcast_ty(),
286                        );
287                        // `-1 != arg2 || arg1 != -iN::MIN`
288                        let pre = vcx
289                            .mk_bin_op_expr(vir::BinOpKind::Or, arg1_cond, arg2_cond)
290                            .downcast_ty::<vir::Bool>();
291                        pres.push(pre);
292                        // The Rust and Viper (SMT) semantics for `\` and `%` do
293                        // not match up when `arg1 < 0`: Rust truncates (`\`
294                        // rounds toward zero, `%` takes the sign of the
295                        // dividend), whereas Viper's `\`/`%` are euclidean
296                        // (always `0 <= arg1 % arg2 < |arg2|`).
297                        // `-arg1`
298                        let neg_lhs = vcx.mk_unary_op_expr(vir::UnOpKind::Neg, lhs);
299                        // `(-arg1) op arg2`
300                        let neg_op = vcx.mk_bin_op_expr(op_kind, neg_lhs, rhs);
301                        // `-((-arg1) op arg2)`
302                        let neg_result = vcx.mk_unary_op_expr(vir::UnOpKind::Neg, neg_op);
303                        // `arg1 == 0`
304                        let lhs_zero = vcx.mk_eq_expr(lhs.downcast_ty(), vcx.mk_int::<0>());
305                        // `arg1 == 0 ? 0 : -((-arg1) op arg2)`
306                        let zero_or_neg = vcx.mk_ternary_expr(
307                            lhs_zero,
308                            vcx.mk_int::<0>().upcast_ty(),
309                            neg_result,
310                        );
311                        // `arg1 > 0`
312                        let lhs_pos = vcx
313                            .mk_bin_op_expr(
314                                vir::BinOpKind::CmpGt,
315                                lhs.downcast_ty(),
316                                vcx.mk_int::<0>(),
317                            )
318                            .downcast_ty();
319                        // `arg1 > 0 ? arg1 op arg2 : (arg1 == 0 ? 0 : -((-arg1) op arg2))`
320                        val = vcx.mk_ternary_expr(lhs_pos, val, zero_or_neg);
321                    }
322                    (pres, val)
323                }
324                // Cannot overflow and no undefined behavior (the bitwise
325                // operations are `bool`-only, rejected above for integers)
326                BitXor | BitAnd | BitOr | Eq | Lt | Le | Ne | Ge | Gt | Offset => {
327                    (Vec::new(), viper_val)
328                }
329
330                // these are handled in `handle_checked_bin_op`
331                AddWithOverflow | SubWithOverflow | MulWithOverflow => unreachable!(),
332
333                // this is handled separately, earlier
334                Cmp => unreachable!(),
335            })
336        }
337    }

339    fn handle_bin_op_raw<'vir>(
340        vcx: &'vir vir::VirCtxt<'vir>,
341        lhs: vir::ExprCSnap<'vir>,
342        rhs: vir::ExprCSnap<'vir>,
343        op: mir::BinOp,
344    ) -> EncodeResult<'vir, vir::ExprCSnap<'vir>> {
345        use mir::BinOp::*;
346        match op {
347            Eq => Ok(vcx.mk_eq_expr(lhs, rhs).upcast_ty()),
348            Ne => {
349                let eq = vcx.mk_eq_expr(lhs, rhs).upcast_ty();
350                let ne = vcx.mk_unary_op_expr(vir::UnOpKind::Not, eq);
351                Ok(ne.downcast_ty::<vir::Bool>().upcast_ty())
352            }
353            // We do not encode addresses as Ints yet
354            // We cannot compare Viper refs using <, <=, >=, >, or use offset
355            Lt | Le | Ge | Gt => Err(EncodeFullError::EncodingError(
356                MirBuiltinBinOpEncError::Unsupported(format!(
357                    "ordering comparison `{op:?}` on raw pointers"
358                )),
359                None,
360            )),
361            Offset => Err(EncodeFullError::EncodingError(
362                MirBuiltinBinOpEncError::Unsupported("pointer offset".to_string()),
363                None,
364            )),
365            // Raw pointers support no arithmetic, bitwise, or shift operators
366            _ => unreachable!("`{op:?}` cannot occur on raw pointer operands"),
367        }
368    }

370    fn handle_bin_op_float<'vir>(
371        vcx: &'vir vir::VirCtxt<'vir>,
372        lhs: vir::ExprCSnap<'vir>,
373        rhs: vir::ExprCSnap<'vir>,
374        op: mir::BinOp,
375        float: FloatDomain<'vir>,
376    ) -> vir::ExprCSnap<'vir> {
377        use mir::BinOp::*;
378        match op {
379            Add => (float.fp_add)(lhs, rhs),
380            AddUnchecked | AddWithOverflow => unreachable!(),
381            Sub => (float.fp_sub)(lhs, rhs),
382            SubUnchecked | SubWithOverflow => unreachable!(),
383            Mul => (float.fp_mul)(lhs, rhs),
384            MulUnchecked | MulWithOverflow => unreachable!(),
385            Div => (float.fp_div)(lhs, rhs),
386            Rem => {
387                // SMT uses n = x / y -> round to nearest
388                // Rust truncates
389                // Therefore we cannot use SMT rem
390                let div_res = (float.fp_div)(lhs, rhs);
391                let div_trunc = (float.fp_trunc)(div_res);
392                let mul_res = (float.fp_mul)(div_trunc, rhs);
393                (float.fp_sub)(lhs, mul_res)
394            }
395            BitXor | BitAnd | BitOr | Shl | ShlUnchecked | Shr | ShrUnchecked => unreachable!(),
396            Eq => {
397                let prim_res = (float.fp_eq)(lhs, rhs);
398                prim_res.upcast_ty()
399            }
400            Lt => {
401                let prim_res = (float.fp_lt)(lhs, rhs);
402                prim_res.upcast_ty()
403            }
404            Le => {
405                let prim_res = (float.fp_leq)(lhs, rhs);
406                prim_res.upcast_ty()
407            }
408            Ne => {
409                let prim_res = (float.fp_eq)(lhs, rhs);
410                let neq = vcx.mk_unary_op_expr(vir::UnOpKind::Not, prim_res.upcast_ty());
411                neq.downcast_ty::<vir::Bool>().upcast_ty()
412            }
413            Ge => {
414                let prim_res = (float.fp_geq)(lhs, rhs);
415                prim_res.upcast_ty()
416            }
417            Gt => {
418                let prim_res = (float.fp_gt)(lhs, rhs);
419                prim_res.upcast_ty()
420            }
421            Cmp => todo!(), // maybe don't implement here but as a stdlib specification
422            Offset => unreachable!(),
423        }
424    }

426    fn handle_bin_op_overflowing<'vir>(
427        vcx: &'vir vir::VirCtxt<'vir>,
428        deps: &mut TaskEncoderDependencies<'vir, Self>,
429        result_ty: RustTyDecomposition<'vir>,
430        op: mir::BinOp,
431        lhs: vir::ExprPrim<'vir>,
432        rhs: vir::ExprPrim<'vir>,
433    ) -> Result<vir::ExprCSnap<'vir>, EncodeFullError<'vir, Self>> {
434        // `op` can only be the overflowing version of `Add`, `Sub` or `Mul`
435        assert!(op.is_overflowing());
436
437        // The result of a checked add will always be `(T, bool)`, get the `T`
438        // type
439        assert_eq!(result_ty.ty.name(), "2_Tuple");
440        let res_ty_int = result_ty.args.args()[0].expect_ty();
441        let bool_ty = result_ty.args.args()[1].expect_ty();
442        assert!(bool_ty.is_bool());
443
444        // Re-encode the result tuple with its concrete arguments (always
445        // `(<some int>, bool)`) so that the tuple constructor casts the concrete
446        // `int`/`bool` field snapshots to the tuple's generic (`Param`) fields.
447        // (Encoding it via `identity` would give identity field casters that
448        // expect already-generic `Param` snapshots.)
449        let e_res_ty = deps.require_dep::<TyUsePureEnc>(result_ty)?;
450
451        let ty = RustTyDecomposition::from_prim_ty(res_ty_int);
452        let e_res_ty_int = deps.require_dep::<TyUsePureEnc>(ty)?.expect_primitive();
453        assert_eq!(vir::TYPE_INT.upcast_ty(), e_res_ty_int.prim_type());
454        let prim_type = e_res_ty_int.prim_type().downcast_ty::<vir::Int>();
455
456        // Unbounded value
457        let val_exp = vcx
458            .mk_bin_op_expr(vir::BinOpKind::from(op), lhs, rhs)
459            .downcast_ty();
460        let val_decl = vcx.mk_local_decl("val", prim_type);
461        let val = vcx.mk_local_ex(val_decl);
462        // Wrapped value
463        //
464        // The eager `mod` below is costly when `val` is nonlinear: under Z3's
465        // legacy arithmetic solver (`smt.arith.solver=2`) the checks around it
466        // can exhaust their resource budget, so the server selects solver 6.
467        // Should that ever need reverting, an encoding that stayed robust
468        // under solver 2 is to keep `mod` out of the in-range path entirely:
469        //   wrapped_val = in_range(val) ? val : wrap_N(val)
470        //   overflowed  = !in_range(val)
471        // with `wrap_N` an uninterpreted domain function axiomatised as
472        // `forall x :: {wrap_N(x)} wrap_N(x) == get_wrapped_val(x)`. The flag
473        // must not mention `wrapped_val`, or `mod` is pulled back in; any
474        // form that leaves a `mod` term on the in-range path stays fragile.
475        let wrapped_val_decl = vcx.mk_local_decl("wrapped_val", prim_type);
476        let wrapped_val_exp = vcx.get_wrapped_val(val, res_ty_int.kind());
477        let wrapped_val = vcx.mk_local_ex(wrapped_val_decl);
478        let wrapped_val_snap = e_res_ty_int.prim_to_snap(wrapped_val.upcast_ty());
479        // Overflowed?
480        let overflowed = vcx
481            .mk_bin_op_expr(vir::BinOpKind::CmpNe, wrapped_val, val)
482            .downcast_ty::<vir::Bool>();
483        // `tuple(prim_to_snap(wrapped_val), wrapped_val != val)`
484        let tuple = e_res_ty
485            .expect_structlike()
486            .field_snaps_to_snap(vec![wrapped_val_snap.upcast_ty(), overflowed.upcast_ty()]);
487        // `let wrapped_val == (val ..) in $tuple`
488        let inner_let = vcx.mk_let_expr(wrapped_val_decl, wrapped_val_exp, tuple);
489        Ok(vcx.mk_let_expr(val_decl, val_exp, inner_let))
490    }
