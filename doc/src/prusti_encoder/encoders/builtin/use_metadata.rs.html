15/// Emits the metadata rewrite for a concrete `[T; N] -> [T]` (array-to-slice)
16/// coercion: an axiom defining `metadata_cast(_, [T; N], [T])` as the array's
17/// static length `N` - the fat-pointer metadata of the resulting slice. Like
18/// `ValueCastAxiomEnc`, it is keyed on the *generic* array/slice shapes (the
19/// `RustTy`, shared by every `[T; N]`), so it is encoded once and is itself
20/// generic over the element type and length. The abstract `metadata_cast`
21/// function and its identity axiom live in `MetadataCastEnc`; this encoder
22/// attaches only the array->slice specialization (so `MetadataCastEnc` no longer
23/// has to assume that array/slice types exist in the program).
24pub struct MetadataCastAxiomEnc;
25
26impl TaskEncoder for MetadataCastAxiomEnc {
27    task_encoder::encoder_cache!(MetadataCastAxiomEnc);
28    const ENCODER_NAME: &'static str = "metadata cast axiom encoder";
29
30    type TaskDescription<'vir> = (RustTy<'vir>, RustTy<'vir>);
31    type OutputFullLocal<'vir> = Option<vir::Domain<'vir>>;
32
33    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
34        *task
35    }
36
37    fn do_encode_full<'vir>(
38        task_key: &Self::TaskKey<'vir>,
39        deps: &mut TaskEncoderDependencies<'vir, Self>,
40    ) -> EncodeFullResult<'vir, Self> {
41        deps.emit_output_ref(*task_key, ())?;
42        let (array_ty, slice_ty) = *task_key;
43        match (&array_ty.specifics, &slice_ty.specifics) {
44            (TySpecifics::ArrayLike(ad), TySpecifics::ArrayLike(sd)) if !ad.slice && sd.slice => {}
45            _ => return Ok((None, ())),
46        }
47
48        let metadata_cast = deps.require_dep::<MetadataCastEnc>(())?;
49        let array_ctor = deps.require_ref::<TyConstructorEnc>(array_ty)?;
50        let slice_ctor = deps.require_ref::<TyConstructorEnc>(slice_ty)?;
51        // Quantify over the array's own params (`T`, `N`), so the axiom is generic
52        // over every `[T; N]` rather than baking in a concrete element/length.
53        let array_gen = deps.require_dep::<GenericParamsEnc>(array_ty.params)?;
54        // The slice metadata is the array's length `N`, held generically (as a
55        // `p_Param` snapshot); cast the concrete `usize` length into the generic
56        // context via `usize`'s pure caster.
57        let usize = RustTyDecomposition::from_prim_ty(vir::with_vcx(|vcx| vcx.tcx().types.usize));
58        let len_caster = deps.require_dep::<GArgsCastEnc<Pure>>(Some(RustTyNormalized {
59            param: RustTyDecomposition::param(),
60            concrete: usize,
61        }))?;
62
63        vir::with_vcx(|vcx| {
64            // The (generic) array/slice referent type values `[T; N]` / `[T]`, in the
65            // same form the `cast` fn passes as `metadata_cast`'s `[U, V]`.
66            let u_ty = (array_ctor.ty_constructor)(array_gen.ty_exprs(), array_gen.const_exprs());
67            let v_ty = (slice_ctor.ty_constructor)(array_gen.ty_exprs(), &[]);
68            // The array's single const param `N`, made generic.
69            let len = array_gen.const_exprs()[0];
70            let len_generic = len_caster
71                .cast_to_callee_ctx(len.upcast_ty())
72                .downcast_ty::<vir::PSnap>();
73
74            let input_decl = vcx.mk_local_decl("input", vir::TYPE_PSNAP);
75            let input = vcx.mk_local_ex(input_decl);
76            let call = metadata_cast(input, u_ty, v_ty);
77            let body = vcx.mk_eq_expr(call, len_generic);
78            // forall input, T, N :: { metadata_cast(input, [T; N], [T]) }
79            //     metadata_cast(input, [T; N], [T]) == make_generic_usize(N)
80            let axiom_expr = vir::expr! {
81                forall [input_decl], ..[array_gen.ty_decls()], ..[array_gen.const_decls()]
82                    :: { [call] }
83                    [body]
84            };
85            let axiom = vcx.mk_domain_axiom(
86                vir::ViperIdent::new("metadata_cast_array_slice"),
87                axiom_expr,
88            );
89            let domain = vcx.mk_domain(
90                vir::ViperIdent::new("MetadataCastArraySlice"),
91                &[],
92                vcx.alloc_slice(&[axiom]),
93                &[],
94                None,
95            );
96            Ok((Some(domain), ()))
97        })
98    }
99
100    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
101        for domain in Self::all_outputs_local_no_errors(program) {
102            let Some(domain) = domain else {
103                continue;
104            };
105            program.add_domain(domain);
106        }
107    }
108}

110/// Emits the functional specification of `unsize_value_cast` for a concrete
111/// `[T; N] -> [T]` (array-to-slice) coercion: an axiom stating that the coerced
112/// slice has the same elements as the original array (the length is carried by
113/// the fat-pointer metadata, set by `metadata_cast`). The coercion's referent
114/// value is held as a generic `p_Param`, so the relation is expressed by
115/// `make_concrete`-ing both sides to their `[T; N]`/`[T]` snapshots and comparing
116/// elements. Keyed on the *generic* array/slice shapes (the `RustTy`, shared by
117/// every `[T; N]`), so it is encoded once and is itself generic over the element
118/// type `T` and length `N`. The abstract `unsize_value_cast` function and its
119/// identity axiom live in `ValueCastEnc`; this encoder attaches only the
120/// array->slice specialization.
121pub struct ValueCastAxiomEnc;
122
123impl TaskEncoder for ValueCastAxiomEnc {
124    task_encoder::encoder_cache!(ValueCastAxiomEnc);
125    const ENCODER_NAME: &'static str = "value cast axiom encoder";
126
127    type TaskDescription<'vir> = (RustTy<'vir>, RustTy<'vir>);
128    type OutputFullLocal<'vir> = Option<vir::Domain<'vir>>;
129
130    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
131        *task
132    }
133
134    fn do_encode_full<'vir>(
135        task_key: &Self::TaskKey<'vir>,
136        deps: &mut TaskEncoderDependencies<'vir, Self>,
137    ) -> EncodeFullResult<'vir, Self> {
138        deps.emit_output_ref(*task_key, ())?;
139        let (array_ty, slice_ty) = *task_key;
140        match (&array_ty.specifics, &slice_ty.specifics) {
141            (TySpecifics::ArrayLike(ad), TySpecifics::ArrayLike(sd)) if !ad.slice && sd.slice => {}
142            _ => return Ok((None, ())),
143        }
144
145        // The most-generic array/slice (identity args), so the axiom quantifies
146        // over the element/length params rather than baking in a concrete `[T; N]`.
147        let array_inner = RustTyDecomposition::identity(array_ty);
148        let mut slice_inner = RustTyDecomposition::identity(slice_ty);
149
150        // Re-base the slice's element onto the array's element type parameter: a
151        // valid `[T; N] -> [T]` coercion always has matching element types, so
152        // the axiom quantifies over the array's params only (`U`, `M`) and the
153        // slice reuses the array's element `U`. This is also what lets the
154        // concrete-keyed axiom below trigger without a free slice-element var.
155        slice_inner.args = GArgs::new(
156            array_ty.params,
157            vir::with_vcx(|vcx| vcx.tcx().mk_args(&[array_ty.params.rust_params()[1]])),
158        );
159
160        let value_cast = deps.require_dep::<ValueCastEnc>(())?;
161        let array_use = deps.require_dep::<TyUsePureEnc>(array_inner)?;
162        let slice_use = deps.require_dep::<TyUsePureEnc>(slice_inner)?;
163        let array_ctor = deps.require_ref::<TyConstructorEnc>(array_ty)?;
164        let slice_ctor = deps.require_ref::<TyConstructorEnc>(slice_ty)?;
165        let array_gen = deps.require_dep::<GenericParamsEnc>(array_ty.params)?;
166        let param = RustTyDecomposition::param();
167        let array_caster = deps.require_dep::<GArgsCastEnc<Pure>>(Some(RustTyNormalized {
168            param,
169            concrete: array_inner,
170        }))?;
171        let slice_caster = deps.require_dep::<GArgsCastEnc<Pure>>(Some(RustTyNormalized {
172            param,
173            concrete: slice_inner,
174        }))?;
175
176        vir::with_vcx(|vcx| {
177            let array_pure = array_use.expect_array();
178            let slice_pure = slice_use.expect_array();
179            // The (generic) referent type values `[T; N]` and `[T]`, in the same
180            // form `MirBuiltinUseCastEnc` passes as the `unsize`/`undo` method's
181            // `U`/`V` type arguments. The slice element is the array's element `U`
182            // (see `slice_args`).
183            let u_ty = (array_ctor.ty_constructor)(array_gen.ty_exprs(), array_gen.const_exprs());
184            let v_ty = (slice_ctor.ty_constructor)(array_gen.ty_exprs(), &[]);
185
186            let arr_decl = vcx.mk_local_decl("arr", array_use.snapshot.downcast_ty());
187            let arr = vcx.mk_local_ex(arr_decl);
188            let arr_generic = array_caster
189                .cast_to_callee_ctx(arr.upcast_ty())
190                .downcast_ty::<vir::PSnap>();
191            let arr_vc = value_cast(arr_generic, u_ty, v_ty);
192            let sli = slice_caster
193                .cast_to_caller_ctx(arr_vc.upcast_ty())
194                .downcast_ty::<vir::CSnap>();
195
196            let idx_decl = vcx.mk_local_decl("idx", vir::TYPE_INT);
197            let idx = vcx.mk_local_ex(idx_decl);
198            // Compare the elements in their generic (`p_Param`) form: the array and
199            // slice share the same element type, so the concrete element conversion
200            // would be redundant.
201            let idx_body = vcx.mk_eq_expr(
202                array_pure.index_generic(arr, idx),
203                slice_pure.index_generic(sli, idx),
204            );
205            let idx_forall = vcx.mk_forall_expr(
206                vcx.alloc_slice(&[idx_decl]),
207                vcx.alloc_slice(&[vcx.mk_trigger(&[array_pure.index_generic(arr, idx)])]),
208                idx_body,
209            );
210            let axiom_expr = vir::expr! {
211                forall [arr_decl], ..[array_gen.ty_decls()], ..[array_gen.const_decls()]
212                    :: { [arr_vc] }
213                    [idx_forall]
214            };
215            let axiom = vcx.mk_domain_axiom(
216                vir::ViperIdent::new("unsize_value_cast_array_slice"),
217                axiom_expr,
218            );
219
220            let domain = vcx.mk_domain(
221                vir::ViperIdent::new("UnsizeValueCastArraySlice"),
222                &[],
223                vcx.alloc_slice(&[axiom]),
224                &[],
225                None,
226            );
227            Ok((Some(domain), ()))
228        })
229    }
230
231    fn emit_outputs<'vir>(program: &mut task_encoder::Program<'vir>) {
232        for domain in Self::all_outputs_local_no_errors(program) {
233            let Some(domain) = domain else {
234                continue;
235            };
236            program.add_domain(domain);
237        }
238    }
