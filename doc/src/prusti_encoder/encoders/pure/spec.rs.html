26/// The VIR expression and span corresponding to either the lhs or rhs of a
27/// pledge. It will be conjoined to the permission expression of the
28/// corresponding side of the wand for the encoded pledge.
29#[derive(Debug, Clone, Copy)]
30pub struct PledgeExpr<'vir> {
31    did: DefId,
32    expr: vir::ExprGenBool<'vir, ExprInput<'vir>, vir::ExprKind<'vir>>,
33}

36pub struct PledgeArgs<'vir>(&'vir FxHashMap<mir::Local, vir::ExprSnap<'vir>>, mir::Local);

38impl<'vir> std::ops::Index<mir::Local> for PledgeArgs<'vir> {
39    type Output = vir::ExprSnap<'vir>;
40
41    fn index(&self, index: mir::Local) -> &Self::Output {
42        if index == mir::RETURN_PLACE {
43            &self.0[&self.1]
44        } else {
45            &self.0[&index]
46        }
47    }

50impl<'vir> PledgeExpr<'vir> {
51    pub fn new(
52        did: DefId,
53        expr: vir::ExprGenBool<'vir, ExprInput<'vir>, vir::ExprKind<'vir>>,
54    ) -> Self {
55        Self { did, expr }
56    }
57
58    pub fn pledge_args<T: Borrow<vir::ExprSnap<'vir>>>(
59        result: vir::ExprSnap<'vir>,
60        args: impl IntoIterator<Item = T>,
61    ) -> PledgeArgs<'vir> {
62        let mut all_args: FxHashMap<mir::Local, _> = args
63            .into_iter()
64            .enumerate()
65            .map(|(idx, a)| ((idx + 1).into(), *a.borrow()))
66            .collect();
67        let result_local = (all_args.len() + 1).into();
68        all_args.insert(result_local, result);
69        vir::with_vcx(|vcx| PledgeArgs(vcx.alloc(all_args), result_local))
70    }
71
72    pub fn expr(&self, args: PledgeArgs<'vir>) -> vir::ExprBool<'vir> {
73        vir::with_vcx(|vcx| {
74            self.expr
75                .reify(vcx, (self.did, args.0, vir::OldLabel::None))
76        })
77    }
78
79    pub fn expr_at_label(&self, args: PledgeArgs<'vir>, label: &'vir str) -> vir::ExprBool<'vir> {
80        vir::with_vcx(|vcx| {
81            self.expr
82                .reify(vcx, (self.did, args.0, vir::OldLabel::Label(label)))
83        })
84    }
85
86    pub fn span(&self) -> Span {
87        vir::with_vcx(|vcx| vcx.tcx().def_span(self.did))
88    }
89}

91/// VIR expressions for a pledge, including a user-written `assert_on_expiry`
92/// predicate if present.
93#[derive(Clone, Copy, Debug)]
94pub struct EncodedPledge<'vir> {

96    /// predicate, if present.

98    /// The pure rhs of the wand.
99    pub expiry_postcondition: PledgeExpr<'vir>,

102#[derive(Clone)]
103pub struct MirSpecEncOutput<'vir> {
104    /// Each precondition paired with the source span of its spec item.
105    pub pres: Vec<(vir::ExprBool<'vir>, Span)>,
106    /// Each postcondition paired with the source span of its spec item.
107    pub posts: Vec<(vir::ExprBool<'vir>, Span)>,
108    pub pledges: Vec<EncodedPledge<'vir>>,
109    pub pre_args: &'vir FxHashMap<mir::Local, vir::ExprSnap<'vir>>,
110    #[allow(dead_code)]
111    pub post_args: &'vir FxHashMap<mir::Local, vir::ExprSnap<'vir>>,
112}

114impl<'vir> MirSpecEncOutput<'vir> {
115    /// The precondition expressions, discarding their spans.
116    pub fn pre_exprs(&self) -> impl Iterator<Item = vir::ExprBool<'vir>> + '_ {
117        self.pres.iter().map(|(pre, _)| *pre)
118    }
119
120    /// The postcondition expressions, discarding their spans.
121    pub fn post_exprs(&self) -> impl Iterator<Item = vir::ExprBool<'vir>> + '_ {
122        self.posts.iter().map(|(post, _)| *post)
123    }
124}

126/// State shared by every spec encoded within one `MirSpecEnc` task.
127#[derive(Clone, Copy)]
128struct SpecEncCtx<'vir> {
129    extern_spec: Option<ExternSpecKind>,
130    enc_mode: MirSpecEncMode,
131    context_def_id: DefId,
132    substs: ty::GenericArgsRef<'vir>,
133}
134
135#[derive(Clone, Copy, Debug, PartialEq, Eq, Hash)]
136pub enum MirSpecEncMode {
137    /// Assumes the arguments and the result are available in local variables
138    /// `_1p`, ... `_np`, and `_0p`, respectively, all of type `Ref``, i.e.,
139    /// their snapshot is taken first.
140    Impure,
141
142    /// Assumes the arguments are available in local varialbes `_1s`, ... `_ns`,
143    /// all of snapshot types, and the result is the result of the current
144    /// function, i.e., `result` in Viper syntax.
145    PureWithResult,
146
147    /// Assumes the arguments and the result are available in local variables
148    /// `_1s`, ... `_ns`, and `_0s`, respectively, all of snapshot types.
149    PureWithoutResult,
150}

152impl TaskEncoder for MirSpecEnc {
153    task_encoder::encoder_cache!(MirSpecEnc);
154    const ENCODER_NAME: &'static str = "MIR spec encoder";
155
156    type TaskDescription<'tcx> = (
157        DefId, // The function annotated with specs
158        DefId, // Context, i.e., where the specs are emitted
159        MirSpecEncMode,
160    );
161
162    type OutputFullDependency<'vir> = MirSpecEncOutput<'vir>;
163
164    type EncodingError = <MirPureEnc as TaskEncoder>::EncodingError;
165
166    fn task_to_key<'vir>(task: &Self::TaskDescription<'vir>) -> Self::TaskKey<'vir> {
167        *task
168    }
169
170    fn do_encode_full<'vir>(
171        task_key: &Self::TaskKey<'vir>,
172        deps: &mut TaskEncoderDependencies<'vir, Self>,
173    ) -> EncodeFullResult<'vir, Self> {
174        let (def_id, context_def_id, enc_mode) = *task_key;
175        deps.emit_output_ref(*task_key, ())?;
176
177        vir::with_vcx(|vcx| {
178            let base_params = GParams::from(def_id);
179            let context_params = GParams::from(context_def_id);
180            let substs =
181                find_trait_method_substs(vcx.tcx(), context_def_id, context_params.rust_params())
182                    .map(|s| s.1)
183                    .unwrap_or(base_params.rust_params());
184            let substs = vcx.tcx().mk_args(substs);
185            let substs_for = |inherited: bool| {
186                // if the spec is not inherited and we are currently building the spec for the actual function we're encoding,
187                // do not use substitution
188                if !inherited && def_id == context_def_id {
189                    vcx.tcx().mk_args_from_iter(
190                        context_params
191                            .rust_params()
192                            .iter()
193                            .filter(|arg| arg.as_region().is_none()),
194                    )
195                } else {
196                    substs
197                }
198            };
199
200            let local_defs = deps.require_dep::<crate::encoders::local_def::MirLocalDefEnc>(
201                MirLocalDefEncTask::LocalSubsts {
202                    def_id,
203                    context_def_id,
204                    substs: if def_id == context_def_id {
205                        context_params.rust_params()
206                    } else {
207                        substs
208                    },
209                    all_locals: false,
210                },
211            )?;
212            let specs = deps
213                .require_dep::<crate::encoders::SpecEnc>(crate::encoders::SpecEncTask { def_id })?;
214            let (pres, pres_inherited) = crate::encoders::spec_items(&specs.pres);
215            let (posts, posts_inherited) = crate::encoders::spec_items(&specs.posts);
216            let (pledges, pledges_inherited) = crate::encoders::spec_items(&specs.pledges);
217            let pre_ctx = SpecEncCtx {
218                extern_spec: specs.extern_spec,
219                enc_mode,
220                context_def_id,
221                substs: substs_for(pres_inherited),
222            };
223            let post_ctx = SpecEncCtx {
224                extern_spec: specs.extern_spec,
225                enc_mode,
226                context_def_id,
227                substs: substs_for(posts_inherited),
228            };
229            let pledge_ctx = SpecEncCtx {
230                extern_spec: specs.extern_spec,
231                enc_mode,
232                context_def_id,
233                substs: substs_for(pledges_inherited),
234            };
235
236            let local_iter = (1..=local_defs.arg_count).map(mir::Local::from);
237            let all_args: FxHashMap<mir::Local, _> = match enc_mode {
238                MirSpecEncMode::Impure => local_iter
239                    .map(|local| (local, local_defs[local].impure_snap))
240                    .collect(),
241                MirSpecEncMode::PureWithResult => {
242                    let result_ty = local_defs[mir::RETURN_PLACE].local_snap.ty();
243                    local_iter
244                        .map(|local| (local, vcx.mk_local_ex(local_defs[local].local_snap)))
245                        .chain([((local_defs.arg_count + 1).into(), vcx.mk_result(result_ty))])
246                        .collect()
247                }
248                MirSpecEncMode::PureWithoutResult => local_iter
249                    .map(|local| (local, vcx.mk_local_ex(local_defs[local].local_snap)))
250                    .chain([(
251                        (local_defs.arg_count + 1).into(),
252                        vcx.mk_local_ex(local_defs[mir::RETURN_PLACE].local_snap),
253                    )])
254                    .collect(),
255            };
256            let all_args = vcx.alloc(all_args);
257            let pre_args = all_args; // it should be ok to provide more keys than required
258
259            // Encode each functional precondition; if one cannot be encoded (e.g.
260            // it uses an unsupported feature), report the error at *that spec's*
261            // span and skip only it, keeping the permission contract and the other
262            // specs intact.
263            let pres: Vec<(vir::ExprBool<'_>, Span)> = pres
264                .iter()
265                .filter_map(|spec_def_id| {
266                    let spec = Self::encode_pure(vcx, deps, pre_ctx, *spec_def_id, "precondition")?;
267                    let expr = spec.expr.downcast_ty::<vir::Bool>();
268                    let span = vcx.tcx().def_span(*spec_def_id);
269                    // Reify *inside* the span scope: the nodes created by the
270                    // reification pick up the ambient span, which makes error
271                    // positions inside this precondition point at the spec.
272                    let expr = vcx.with_span(span, |vcx| {
273                        expr.reify(vcx, (*spec_def_id, pre_args, vir::OldLabel::None))
274                    });
275                    Some((expr, span))
276                })
277                .collect();
278
279            let post_args = match enc_mode {
280                MirSpecEncMode::Impure => {
281                    let post_args: FxHashMap<mir::Local, vir::ExprSnap<'vir>> = pre_args
282                        .iter()
283                        .map(|(local, arg)| (*local, vcx.mk_old_expr(arg)))
284                        .chain([(
285                            (local_defs.arg_count + 1).into(),
286                            local_defs[mir::RETURN_PLACE].impure_snap,
287                        )])
288                        .collect();
289                    vcx.alloc(post_args)
290                }
291                MirSpecEncMode::PureWithResult | MirSpecEncMode::PureWithoutResult => all_args,
292            };
293            let posts: Vec<(vir::ExprBool<'_>, Span)> = posts
294                .iter()
295                .filter_map(|spec_def_id| {
296                    let span = vcx.tcx().def_span(*spec_def_id);
297                    vcx.with_span(span, |vcx| {
298                        let spec =
299                            Self::encode_pure(vcx, deps, post_ctx, *spec_def_id, "postcondition")?;
300                        vcx.handle_error("postcondition.violated:assertion.false", move |_| {
301                            Some(vec![PrustiError::verification(
302                                "postcondition might not hold",
303                                span.into(),
304                            )])
305                        });
306                        let expr = spec.expr.downcast_ty::<vir::Bool>();
307                        let expr = expr.reify(vcx, (*spec_def_id, post_args, vir::OldLabel::None));
308                        let expr = expr.realloc_span();
309                        Some((expr, span))
310                    })
311                })
312                .collect();
313            let pledges = pledges
314                .iter()
315                .filter_map(
316                    |Pledge {
317                         lhs: lhs_def_id,
318                         rhs: rhs_def_id,
319                         ..
320                     }| {
321                        // Optional expiry obligation (lhs). If it cannot be encoded,
322                        // report at its span and skip the whole pledge.
323                        let lhs_expr = match *lhs_def_id {
324                            Some(lhs_def_id) => {
325                                let spec = Self::encode_pure(
326                                    vcx,
327                                    deps,
328                                    pledge_ctx,
329                                    lhs_def_id,
330                                    "pledge lhs",
331                                )?;
332                                let lhs = spec.expr.downcast_ty::<vir::Bool>();
333                                Some(PledgeExpr::new(lhs_def_id, lhs))
334                            }
335                            None => None,
336                        };
337                        let spec =
338                            Self::encode_pure(vcx, deps, pledge_ctx, *rhs_def_id, "pledge rhs")?;
339                        let rhs = spec.expr.downcast_ty::<vir::Bool>();
340                        let rhs_span = vcx.tcx().def_span(*rhs_def_id);
341                        let rhs_expr = vcx.with_span(rhs_span, move |vcx| {
342                            vcx.handle_error("exhale.failed:assertion.false", move |_| {
343                                Some(vec![PrustiError::verification(
344                                    "pledge postcondition might not hold",
345                                    rhs_span.into(),
346                                )])
347                            });
348                            rhs
349                        });
350                        let rhs_expr = PledgeExpr::new(*rhs_def_id, rhs_expr);
351                        Some(EncodedPledge {
352                            expiry_obligation: lhs_expr,
353                            expiry_postcondition: rhs_expr,
354                        })
355                    },
356                )
357                .collect::<Vec<_>>();
358            let data = MirSpecEncOutput {
359                pres,
360                posts,
361                pledges,
362                pre_args,
363                post_args,
364            };
365            Ok(((), data))
366        })
367    }
368}

370impl MirSpecEnc {
371    fn encode_pure<'vir>(
372        vcx: &'vir vir::VirCtxt<'vir>,
373        deps: &mut TaskEncoderDependencies<'vir, Self>,
374        ctx: SpecEncCtx<'vir>,
375        def_id: DefId,
376        type_: &str,
377    ) -> Option<MirPureEncOutput<'vir>> {
378        let span = vcx.tcx().def_span(def_id);
379        let spec = deps.require_dep::<MirPureEnc>(crate::encoders::MirPureEncTask {
380            encoding_depth: 0,
381            kind: PureKind::Spec {
382                context: ctx.context_def_id,
383                mode: ctx.enc_mode,
384            },
385            parent_def_id: def_id,
386            gargs: GArgs::new(
387                GParams::new_maybe_extern(ctx.context_def_id, ctx.extern_spec),
388                ctx.substs,
389            ),
390        });
391        spec.inspect_err(|err| {
392            let (message, err_span) = crate::encoders::mir_fn::dep_error(err);
393            vcx.emit_early_error(PrustiError::unsupported(
394                format!("cannot encode {type_}: {message}"),
395                err_span.unwrap_or(span).into(),
396            ))
397        })
398        .ok()
399    }
