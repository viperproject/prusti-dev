38// TODO: find a better way of handling selective verification.
39// Currently, this thread local static is used to converst the initial list of defpaths from the
40// `VERIFY_ONLY_DEFPATHS` option from `Vec<String>` to `Vec<DefId>`. This is done,
41// so the encoder (impure/pure_function_enc) can check elements for containment.
42// Because currently, it does not have the crate name available to it. But that crate name
43// is part of the defpaths passed through the option.
44thread_local!(
45    pub static SELECTIVE_TASKS: std::cell::OnceCell<Vec<DefId>> = const { std::cell::OnceCell::new() }
46);
47
48pub fn is_selected(def_id: &DefId) -> bool {
49    SELECTIVE_TASKS.with(|selective_tasks| {
50        selective_tasks
51            .get()
52            .as_ref()
53            .is_none_or(|procs| procs.contains(def_id))
54    })
55}

57pub fn test_entrypoint<'tcx>(
58    tcx: ty::TyCtxt<'tcx>,
59    body: EnvBody<'tcx>,
60    def_spec: prusti_interface::specs::typed::DefSpecificationMap,
61    // this is None if the verification is not selective - all procedures should be encoded.
62    // if the verification is selective, only the procedures in this vector should be encoded with body
63    procedures: Option<Vec<DefId>>,
64    env_diagnostic: &EnvDiagnostic<'tcx>,
65) -> request::RequestWithContext {
66    let ident_style = if config::short_viper_names() {
67        vir::IdentStyle::ItemName
68    } else {
69        vir::IdentStyle::DefPath
70    };
71    vir::init_vcx(vir::VirCtxt::new(tcx, body, def_spec, ident_style));
72    SELECTIVE_TASKS.with(|selective_tasks| {
73        if let Some(procs) = procedures {
74            selective_tasks
75                .set(procs)
76                .expect("Selective tasks were already set");
77        }
78    });
79
80    crate::encoders::encode_all_in_crate(tcx);
81    task_encoder::drain_triggers();
82
83    if config::show_ide_info() {
84        vir::with_vcx(|vcx| vcx.emit_contract_spans(env_diagnostic));
85    }
86    let mut program = task_encoder::Program::default();
87
88    // We output results from both monomorphic and polymorphic encoding of
89    // functions, because even when Prusti is configured to use the monomorphic
90    // it will still use `MirPolyImpureEnc` directly sometimes (see usages
91    // earlier in this file).
92    program.header("user functions");
93    crate::encoders::FunctionCallEnc::emit_outputs(&mut program);
94
95    program.header("user methods");
96    crate::encoders::MethodCallEnc::emit_outputs(&mut program);
97
98    program.header("MIR builtins");
99    crate::encoders::MirBuiltinUnOpEnc::emit_outputs(&mut program);
100    crate::encoders::MirBuiltinBinOpEnc::emit_outputs(&mut program);
101    crate::encoders::MirBuiltinUseCastEnc::emit_outputs(&mut program);
102    // Unsize casts call `metadata_cast`; the `&mut` methods call
103    // `value_cast`. Both domains must be emitted too.
104    crate::encoders::MetadataCastEnc::emit_outputs(&mut program);
105    crate::encoders::ValueCastEnc::emit_outputs(&mut program);
106    crate::encoders::MetadataCastAxiomEnc::emit_outputs(&mut program);
107    crate::encoders::ValueCastAxiomEnc::emit_outputs(&mut program);
108    crate::encoders::PrustiBuiltinEnc::emit_outputs(&mut program);
109
110    program.header("pure generic casts");
111    GArgsCastEnc::<Pure>::emit_outputs(&mut program);
112
113    program.header("impure generic casts");
114    GArgsCastEnc::<Impure>::emit_outputs(&mut program);
115
116    program.header("snapshots");
117    crate::encoders::TyUsePureEnc::emit_outputs(&mut program);
118    BitVecEnc::emit_outputs(&mut program);
119    FloatBitVecConvEnc::emit_outputs(&mut program);
120
121    program.header("predicates");
122    crate::encoders::TyUseImpureEnc::emit_outputs(&mut program);
123
124    program.header("type constructors");
125    TyConstructorEnc::emit_outputs(&mut program);
126    crate::encoders::TyInhabitedEnc::emit_outputs(&mut program);
127
128    program.header("constants");
129    ConstEnc::emit_outputs(&mut program);
130
131    program.header("custom");
132    PairUseEnc::emit_outputs(&mut program);
133    ReadPermEnc::emit_outputs(&mut program);
134    RefDataEnc::emit_outputs(&mut program);
135
136    program.header("traits");
137    TraitEnc::emit_outputs(&mut program);
138    SizednessEnc::emit_outputs(&mut program);
139    TraitFnEnc::emit_outputs(&mut program);
140    TraitImplEnc::emit_outputs(&mut program);
141    TraitImplConditionEnc::emit_outputs(&mut program);
142    TraitImplItemEnc::emit_outputs(&mut program);
143    TraitImplDefaultFnEnc::emit_outputs(&mut program);
144
145    if config::simplify_encoding() {
146        program.simplify();
147    }
148
149    if std::env::var("LOCAL_TESTING").is_ok() {
150        std::fs::write("local-testing/simple.vpr", program.code()).unwrap();
151    }
152
153    for (error_msg, span) in program.encoder_errors().drain(..) {
154        PrustiError::internal(error_msg, span.into()).emit(env_diagnostic);
155    }
156
157    // Errors raised during encoding (e.g. an unsupported feature that was
158    // replaced by an abstract stub) so that they surface rather than silently
159    // degrading.
160    for error in early_errors() {
161        error.emit(env_diagnostic);
162    }
163
164    let program = program.mk_program();
165
166    request::RequestWithContext {
167        program: program.to_ref(),
168    }
169}
