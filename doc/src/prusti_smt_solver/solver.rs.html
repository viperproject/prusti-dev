3use crate::subprocess::{communicate, get_version, pass_error};
4use async_std::process::{Command, Stdio};
5use context::Context;
6use futures::try_join;

8mod context;
9mod subprocess;

11#[async_std::main]
12async fn main() -> Result<(), std::io::Error> {
13    let z3_path = std::env::var("PRUSTI_ORIGINAL_SMT_SOLVER_PATH").unwrap();
14    let context = Context::new().await;
15    for arg in std::env::args() {
16        context.write_to_log("init", &format!("{arg}\n")).await?;
17    }
18    context.write_config_to_log().await?;
19    let args: Vec<_> = std::env::args().skip(1).collect();
20    let mut cmd = Command::new(z3_path);
21    cmd.args(&args)
22        .stdin(Stdio::piped())
23        .stdout(Stdio::piped())
24        .stderr(Stdio::piped());
25    if let Some(z3_trace_path) = &context.z3_trace_path {
26        cmd.arg("trace=true")
27            .arg("proof=true")
28            .arg(format!("trace_file_name={z3_trace_path}"))
29            .arg("smt.qi.profile=true")
30            .arg("smt.qi.profile_freq=10000");
31    }
32    context.write_to_log("z3", &format!("{cmd:?}\n")).await?;
33    let solver = cmd.spawn()?;
34    let solver_stdin = solver.stdin.expect("failed to create stdin pipe with Z3");
35    let solver_stdout = solver.stdout.expect("failed to create stdout pipe with Z3");
36    let solver_stderr = solver.stderr.expect("failed to create stderr pipe with Z3");
37    let error = pass_error(&context, solver_stderr);
38    if args
39        .iter()
40        .any(|arg| arg == "--version" || arg == "-version")
41    {
42        let communicate = get_version(&context, solver_stdout);
43        try_join!(communicate, error)?;
44    } else {
45        let communicate = communicate(&context, solver_stdin, solver_stdout);
46        try_join!(communicate, error)?;
47    };
48    Ok(())
49}
