7use proc_macro::TokenStream;
8use proc_macro2::TokenStream as TokenStream2;
9use quote::quote;
10use syn::{token::Paren, ItemFn, ReturnType, Type, TypeParen};
11
12// Using `tracing::instrument` without this crate (from the vanilla tracing crate)
13// causes RA to not pick up the return type properly atm - it colours wrong and
14// appears like a local variable on hover. Somehow this is fixed by wrapping the
15// return type in brackets (i.e. `-> TokenStream` to `-> (TokenStream)`) and so we
16// do that here.
17#[proc_macro_attribute]
18pub fn instrument(attr: TokenStream, tokens: TokenStream) -> TokenStream {
19    let (attr, tokens): (TokenStream2, TokenStream2) = (attr.into(), tokens.into());
20    if let Ok(mut item) = syn::parse2::<ItemFn>(tokens.clone()) {
21        if let ReturnType::Type(a, ty) = item.sig.output {
22            let new_ty = Type::Paren(TypeParen {
23                paren_token: Paren::default(),
24                elem: ty,
25            });
26            item.sig.output = ReturnType::Type(a, Box::new(new_ty));
27        }
28        quote! {
29            #[tracing::tracing_instrument(#attr)]
30            #item
31        }
32        .into()
33    } else {
34        quote! {
35            #[tracing::tracing_instrument(#attr)]
36            #tokens
37        }
38        .into()
39    }
40}
