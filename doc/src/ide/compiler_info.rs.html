9/// This struct will be passed to prusti-assistant containing information
10/// about the program that is currently being verified
11#[derive(Serialize)]
12pub struct IdeInfo {
13    procedure_defs: Vec<ProcDef>,
14    function_calls: Vec<ProcDef>,
15    queried_source: Option<String>,
16}
17
18impl IdeInfo {
19    pub fn collect(
20        env: &Environment<'_>,
21        procedures: &Vec<DefId>,
22        def_spec: &typed::DefSpecificationMap,
23    ) -> Self {
24        let procedure_defs = collect_procedures(env, procedures, def_spec);
25        let source_map = env.tcx().sess.source_map();
26        let function_calls = collect_fncalls(env)
27            .into_iter()
28            .map(|(name, defid, span)| ProcDef {
29                name,
30                defid,
31                // span,
32                span: VscSpan::from_span(&span, source_map),
33            })
34            .collect::<Vec<_>>();
35
36        // For declaring external specifications:
37        let queried_source = query_signature::collect_queried_signature(env.tcx(), &function_calls);
38        Self {
39            procedure_defs,
40            function_calls,
41            queried_source,
42        }
43    }
44}

46/// A struct that contains either a reference to a procedure that can be verified
47/// (for selective verification) or a function call (so a user can query
48/// external_spec blocks for it). The name contains the defpath.
49#[derive(Serialize, Debug, Clone)]
50pub struct ProcDef {
51    pub name: String,
52    #[serde(skip)]
53    pub defid: DefId,
54    pub span: VscSpan,
55}

57/// collect information about the program that will be passed to IDE.
58/// This should find all non-trusted functions that can be verified
59fn collect_procedures(
60    env: &Environment<'_>,
61    procedures: &Vec<DefId>,
62    def_spec: &typed::DefSpecificationMap,
63) -> Vec<ProcDef> {
64    let sourcemap: &SourceMap = env.tcx().sess.source_map();
65    let mut procs = Vec::new();
66    for defid in procedures {
67        let defpath = env.name.get_unique_item_name(*defid);
68        let span = env.query.get_def_span(defid);
69        let vscspan = VscSpan::from_span(&span, sourcemap);
70
71        // Filter out the predicates and trusted methods,
72        // since we don't want to allow selective verification
73        // for them
74        let mut is_predicate = false;
75        let mut is_trusted = false;
76
77        let proc_spec_opt = def_spec.get_proc_spec(defid);
78        if let Some(proc_spec) = proc_spec_opt {
79            let kind_spec = proc_spec
80                .base_spec
81                .kind
82                .extract_with_selective_replacement();
83            let trusted_spec = proc_spec
84                .base_spec
85                .trusted
86                .extract_with_selective_replacement();
87            if let Some(typed::ProcedureSpecificationKind::Predicate(..)) = kind_spec {
88                is_predicate = true;
89            }
90            if let Some(true) = trusted_spec {
91                is_trusted = true;
92            }
93        }
94
95        if !is_trusted && !is_predicate {
96            procs.push(ProcDef {
97                name: defpath,
98                defid: *defid,
99                span: vscspan,
100            });
101        }
102    }
103    procs
104}

106/// collect all the function calls, so the extension can query external_spec
107/// templates for it
108fn collect_fncalls(env: &Environment<'_>) -> Vec<(String, DefId, Span)> {
109    let mut fnvisitor = call_finder::CallSpanFinder::new(env);
110    env.tcx().hir_visit_all_item_likes_in_crate(&mut fnvisitor);
111
112    fnvisitor.called_functions
