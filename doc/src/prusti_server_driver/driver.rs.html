8use clap::Parser;
9use prusti_utils::config;
10
11/// A verification server to handle Prusti verification requests.
12#[derive(Parser, Debug)]
13#[clap(version, about, long_about = None)]
14struct Args {
15    /// Sets the port on which to listen for incoming verification requests.
16    /// Pass 0 to get a free one assigned by the OS.
17    #[clap(short, long, value_name = "PORT", default_value_t = 0)]
18    port: u16,
19}
20
21fn main() {
22    env_logger::init_from_env(
23        env_logger::Env::new()
24            .filter_or("PRUSTI_LOG", config::log())
25            .write_style_or("PRUSTI_LOG_STYLE", config::log_style()),
26    );
27
28    let args = Args::parse();
29
30    prusti_server::start_server_on_port(args.port);
31}
