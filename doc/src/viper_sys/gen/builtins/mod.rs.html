1use jni::{objects::JObject, JNIEnv};
2use jni::strings::JavaStr;
3use std::ffi::CStr;

6pub fn java_str_to_string(string: &JavaStr) -> Result<String> {
7    unsafe { Ok(CStr::from_ptr(string.get_raw()).to_str().unwrap_or_else(|err| panic!("java_str_to_string - failed to convert string: {err:?}")).to_string()) }
8}

10fn get_jobject_class_name_inner<'a, 'c, O>(env: &'c JNIEnv<'c>, object: O) -> Result<String>
11where
12    O: Into<JObject<'a>>
13{
14    let object_class = env.get_object_class(object)?;
15    java_str_to_string(
16        &env.get_string(
17            env.call_method(object_class, "getName", "()Ljava/lang/String;", &[])?
18                 .l()?
19                 .into(),
20        )?,
21    )
22}
23
24pub fn get_jobject_class_name<'a, O>(env: &'a JNIEnv<'a>, object: O) -> String
25where
26    O: Into<JObject<'a>>
27{
28    get_jobject_class_name_inner(env, object).unwrap_or("<unknown>".to_string()).replace('.', "/")
