3/// Determines how data should be treated during reification, as well as
4/// de/serialisation.
5pub(crate) enum ReifyKind {
6    // There is no `ReifyString`, since there is nothing to reify.
7    /// String reference, `&'vir str`.
8    ///
9    /// Reify: passthrough.
10    ///   Ser: serialise as `String`.
11    //  Deser: allocate into arena.
12    PassString,
13
14    /// Slice of a reifiable type, e.g. `&'vir [Expr<'vir>]`.
15    ///
16    /// Reify: reify every element.
17    ///   Ser: serialise every element as owned data.
18    /// Deser: allocate every element into arena.
19    ReifySlice,
20
21    /// Slice of a non-refiable type, e.g. `&'vir [LocalDecl<'vir>]`.
22    ///
23    /// Reify: passthrough.
24    ///   Ser: serialise every element as owned data.
25    /// Deser: allocate every element into arena.
26    PassSlice,
27
28    /// Option of a reifiable type, e.g. `Option<Expr<'vir>>`.
29    ///
30    /// Reify: reify value, if present.
31    ///   Ser: serialise value as owned data, if present.
32    /// Deser: allocate value into arena.
33    ReifyOption,
34
35    /// Option of a non-reifiable type, e.g. `Option<VirSpan<'vir>>`.
36    ///
37    /// Reify: passthrough.
38    ///   Ser: serialise value as owned data, if present.
39    /// Deser: allocate value into arena.
40    PassOption,
41
42    /// Reifiable owned type.
43    ///
44    /// Reify: reify value.
45    ///   Ser: default serde implementation.
46    /// Deser: default serde implementation.
47    ReifyOwned,
48
49    /// Non-reifiable owned type.
50    ///
51    /// Reify: passthrough.
52    ///   Ser: default serde implementation.
53    /// Deser: default serde implementation.
54    PassOwned,
55
56    /// Reference to a non-reifiable type.
57    ///
58    /// Reify: passthrough.
59    ///   Ser: serialise as owned data.
60    /// Deser: allocate value into arena.
61    PassRef,
62}
63
64impl ReifyKind {
65    pub(crate) fn of_field(field: &syn::Field) -> ReifyKind {
66        // parse flags from `#[vir(..)]` attributes
67        enum VirFlag {
68            ReifyPass,
69            IsRef,
70        }
71        let flags = field
72            .attrs
73            .iter()
74            .filter(|attr| match &attr.meta {
75                syn::Meta::List(l) => {
76                    l.path.segments.len() == 1 && l.path.segments[0].ident == "vir"
77                }
78                _ => false,
79            })
80            .flat_map(|attr| {
81                attr.parse_args_with(Punctuated::<syn::Ident, syn::Token![,]>::parse_terminated)
82                    .expect("invalid vir attribute arguments")
83                    .into_iter()
84            })
85            .map(|ident| match ident.to_string().as_str() {
86                "reify_pass" => VirFlag::ReifyPass,
87                "is_ref" => VirFlag::IsRef,
88                _ => panic!("invalid vir flag: {ident}"),
89            })
90            .collect::<Vec<_>>();
91        let is_reify_pass = flags.iter().any(|flag| matches!(flag, VirFlag::ReifyPass));
92        let is_ref = flags.iter().any(|flag| matches!(flag, VirFlag::IsRef));
93
94        // is this a string? `&'vir str`
95        if matches!(&field.ty, syn::Type::Reference(syn::TypeReference {
96            mutability: None,
97            elem: box syn::Type::Path(syn::TypePath {
98                qself: None,
99                path: syn::Path {
100                    leading_colon: None,
101                    segments,
102                },
103            }),
104            ..
105        }) if segments.len() == 1 && matches!(&segments[0], syn::PathSegment {
106            ident,
107            arguments: syn::PathArguments::None,
108        } if ident == "str"))
109        {
110            assert!(!is_ref, "invalid flag on string: is_ref");
111            assert!(!is_reify_pass, "invalid flag on string: is_reify_pass");
112            return ReifyKind::PassString;
113        }
114
115        // is this a slice? `&'vir [..]`
116        if matches!(
117            &field.ty,
118            syn::Type::Reference(syn::TypeReference {
119                mutability: None,
120                elem: box syn::Type::Slice(_),
121                ..
122            })
123        ) {
124            assert!(!is_ref, "invalid flag on slice: is_ref");
125            if is_reify_pass {
126                return ReifyKind::PassSlice;
127            } else {
128                return ReifyKind::ReifySlice;
129            }
130        }
131
132        // is this an option? `Option<..>`
133        if matches!(&field.ty, syn::Type::Path(syn::TypePath {
134            qself: None,
135            path: syn::Path {
136                leading_colon: None,
137                segments,
138            },
139        }) if segments.len() == 1 && matches!(&segments[0], syn::PathSegment {
140            ident,
141            arguments: syn::PathArguments::AngleBracketed(..),
142        } if ident == "Option"))
143        {
144            assert!(!is_ref, "invalid flag on Option: is_ref");
145            if is_reify_pass {
146                return ReifyKind::PassOption;
147            } else {
148                return ReifyKind::ReifyOption;
149            }
150        }
151
152        if is_reify_pass {
153            if is_ref {
154                ReifyKind::PassRef
155            } else {
156                ReifyKind::PassOwned
157            }
158        } else {
159            assert!(!is_ref);
160            ReifyKind::ReifyOwned
161        }
162    }
163
164    /// Should `reify` be called on this field when reifying its container?
165    pub(crate) fn should_reify(&self) -> bool {
166        matches!(
167            self,
168            Self::ReifySlice | Self::ReifyOption | Self::ReifyOwned
169        )
170    }
171}
