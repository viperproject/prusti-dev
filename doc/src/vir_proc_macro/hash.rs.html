1use proc_macro::TokenStream;
2use quote::quote;
3use syn::{parse_macro_input, spanned::Spanned, DeriveInput};
4
5pub fn derive_hash(input: TokenStream) -> TokenStream {
6    let input = parse_macro_input!(input as DeriveInput);
7    let name = input.ident;
8    let (generic_args, generic_params) = crate::params_to_args_and_params(&input.generics);
9
10    TokenStream::from(match input.data {
11        syn::Data::Struct(syn::DataStruct {
12            fields: syn::Fields::Named(syn::FieldsNamed { named, .. }),
13            ..
14        }) => {
15            let field_idents = named
16                .iter()
17                .map(|field| field.ident.as_ref().unwrap())
18                .collect::<Vec<_>>();
19
20            quote! {
21                impl<#(#generic_params),*> std::hash::Hash for #name<#(#generic_args),*> {
22                    fn hash<H>(&self, state: &mut H)
23                    where
24                        H: std::hash::Hasher,
25                    {
26                        use std::hash::Hash;
27                        #(self.#field_idents.hash(state);)*
28                    }
29                }
30            }
31        }
32
33        syn::Data::Enum(syn::DataEnum { variants, .. }) => {
34            let variant_hash_arms = variants
35                .iter()
36                .enumerate()
37                .map(|(variant_index, variant)| {
38                    let variant_ident = &variant.ident;
39                    match &variant.fields {
40                        syn::Fields::Unnamed(syn::FieldsUnnamed { unnamed, .. }) => {
41                            let variant_fields_pat = (0..unnamed.len())
42                                .map(|idx| syn::Ident::new(&format!("f{idx}"), unnamed.span()))
43                                .collect::<Vec<_>>();
44
45                            quote! {
46                                Self::#variant_ident(#(#variant_fields_pat),*) => {
47                                    state.write_u8(#variant_index as u8);
48                                    #(#variant_fields_pat.hash(state);)*
49                                }
50                            }
51                        }
52                        syn::Fields::Unit => quote! {
53                            Self::#variant_ident => state.write_u8(#variant_index as u8),
54                        },
55                        _ => panic!("VirHash can only handle tuple and unit variants"),
56                    }
57                })
58                .collect::<Vec<_>>();
59
60            quote! {
61                impl<#(#generic_params),*> std::hash::Hash for #name<#(#generic_args),*> {
62                    fn hash<H>(&self, state: &mut H)
63                    where
64                        H: std::hash::Hasher,
65                    {
66                        use std::hash::Hash;
67                        match self {
68                            #(#variant_hash_arms)*
69                            _ => unreachable!(),
70                        }
71                    }
72                }
73            }
74        }
75
76        _ => unreachable!(),
77    })
78}
