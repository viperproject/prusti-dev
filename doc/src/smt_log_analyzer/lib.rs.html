21pub struct Settings {
22    pub write_statistics: bool,
23    pub quantifier_instantiations_ignore_builtin: bool,
24    pub quantifier_instantiations_bound_global_kind: Option<u64>,
25    pub quantifier_instantiations_bound_trace: Option<u64>,
26    pub quantifier_instantiations_bound_trace_kind: Option<u64>,
27    pub unique_triggers_bound: Option<u64>,
28    pub unique_triggers_bound_total: Option<u64>,
29    pub check_active_scopes_count: Option<u32>,
30    pub pop_scopes_by_one: bool,
31    /// If Some, dumps all triggers that match the specified quantifier.
32    pub trace_quantifier_triggers: Option<QuantifierId>,

35fn process_line(settings: &Settings, state: &mut State, line: &str) -> Result<(), Error> {
36    let mut parser = Parser::from_line(line);
37    match parser.parse_event_kind()? {
38        EventKind::Pop => {
39            let scopes_to_pop = parser.parse_number()?;
40            let active_scopes_count = parser.parse_number()?;
41            parser.check_eof()?;
42            assert_eq!(state.active_scopes_count(), active_scopes_count);
43            if settings.pop_scopes_by_one {
44                let mut scopes_to_pop = scopes_to_pop;
45                while scopes_to_pop > 0 {
46                    state.pop_scopes(1);
47                    scopes_to_pop -= 1;
48                }
49            } else {
50                state.pop_scopes(scopes_to_pop);
51            }
52        }
53        EventKind::Push => {
54            let active_scopes_count = parser.parse_number()?;
55            parser.check_eof()?;
56            assert_eq!(state.active_scopes_count(), active_scopes_count);
57            state.push_scope();
58        }
59        EventKind::MkApp => {
60            if let Some(term_id) = parser.try_parse_id()? {
61                let ident = parser.parse_name()?;
62                if ident.starts_with("basic_block_marker") {
63                    state.register_label(ident.to_string());
64                } else {
65                    let name = ident.to_string();
66                    let mut args = Vec::new();
67                    while let Some(arg) = parser.try_parse_id()? {
68                        args.push(arg);
69                    }
70                    state.register_term_function_application(term_id, name, args);
71                }
72            }
73        }
74        EventKind::MkVar => {
75            if let Some(term_id) = parser.try_parse_id()? {
76                let index = parser.parse_number()?;
77                state.register_term_bound_variable(term_id, index);
78            }
79        }
80        EventKind::AttachMeaning => {
81            let term_id = parser.parse_id()?;
82            let ident = parser.parse_name()?;
83            let ident = ident.to_string();
84            parser.skip_whitespace();
85            let remaining = parser.remaining();
86            state.register_term_attach_meaning(term_id, ident, remaining);
87        }
88        EventKind::MkQuant => {
89            if let Ok(id) = parser.parse_id() {
90                // We are ignoring the builtin quantifiers that have non-numeric
91                // ids such as datatype#6
92                let name = parser.parse_name()?;
93                state.register_quantifier(id, name.to_string());
94            }
95        }
96        EventKind::NewMatch => {
97            let fingerprint = parser.parse_hex_number()?;
98            if fingerprint != 0 {
99                // Built-in quantifiers have fingerprint 0; ignore them.
100                let quantifier_id = parser.parse_id()?;
101                state.register_matched_quantifier(quantifier_id)?;
102                let _trigger_id = parser.parse_id()?;
103                while let Some(_variable_instantiation) = parser.try_parse_id()? {}
104                parser.consume(';')?;
105                while let Some(matched_term) = parser.try_parse_quant_term()? {
106                    match matched_term {
107                        QuantTerm::Single(matched) => {
108                            state.register_matched_trigger_term(quantifier_id, matched)?;
109                        }
110                        QuantTerm::Pair(original, matched) => {
111                            state.register_matched_trigger_term(quantifier_id, original)?;
112                            state.register_matched_trigger_term(quantifier_id, matched)?;
113                        }
114                    }
115                }
116                parser.check_eof()?;
117            } else {
118                state.register_matched_quantifier(BUILTIN_QUANTIFIER_ID)?;
119            }
120        }
121        EventKind::InstDiscovered => {
122            let name = parser.parse_name()?;
123            assert_eq!(name, "theory-solving");
124            let fingerprint = parser.parse_hex_number()?;
125            assert_eq!(fingerprint, 0);
126            let theory = parser.parse_theory()?;
127            state.register_inst_discovered(theory)?;
128        }
129        EventKind::Instance => {
130            state.register_instance()?;
131        }
132        EventKind::Unrecognized => {}
133    }
134    Ok(())
135}

137pub fn analyze(
138    z3_trace_path: &std::path::PathBuf,
139    settings: Settings,
140) -> Result<(), std::io::Error> {
141    // TODO: Collect the quantifier definitions from the smt file.
142
143    let file = File::open(z3_trace_path)?;
144    let mut reader = BufReader::new(file);
145
146    let mut state = State::default();
147
148    // Builtin quantifiers have fingerprint 0.
149    state.register_quantifier(BUILTIN_QUANTIFIER_ID, "builtin quantifier".to_string());
150
151    // Theories.
152    state.register_theory(parser::TheoryKind::Arith);
153    state.register_theory(parser::TheoryKind::Basic);
154    state.register_theory(parser::TheoryKind::Datatype);
155    state.register_theory(parser::TheoryKind::UserSort);
156
157    // Tracing triggers.
158    state.mark_quantifier_for_tracing(settings.trace_quantifier_triggers);
159
160    let mut line = String::new();
161    let mut prev_line = String::new();
162    let mut line_number = 0;
163    assert!(reader.read_line(&mut prev_line)? > 0);
164    while reader.read_line(&mut line)? > 0 {
165        // We use this `prev_line` to avoid processing the last line, which is
166        // very likely to be invalid because Silicon kills Z3.
167        std::mem::swap(&mut line, &mut prev_line);
168        line_number += 1;
169        process_line(&settings, &mut state, &line)
170            .unwrap_or_else(|error| panic!("Error {error} on line {line_number}"));
171        line.clear();
172    }
173    let scopes_left = state.active_scopes_count();
174    state.pop_all_scopes();
175    let input_file = z3_trace_path.to_str().unwrap();
176    if settings.write_statistics {
177        state.write_statistics(input_file);
178    }
179    if let Some(expected_scopes_count) = settings.check_active_scopes_count {
180        assert_eq!(scopes_left, expected_scopes_count);
181    }
182    state.check_bounds(
183        input_file,
184        settings.quantifier_instantiations_ignore_builtin,
185        settings.quantifier_instantiations_bound_global_kind,
186        settings.quantifier_instantiations_bound_trace,
187        settings.quantifier_instantiations_bound_trace_kind,
188        settings.unique_triggers_bound,
189        settings.unique_triggers_bound_total,
190    );
191    Ok(())
192}
