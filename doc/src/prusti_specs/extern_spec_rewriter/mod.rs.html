1//! Parsing of `#[extern_spec]` attributed structures
2
3pub mod impls;
4pub mod mods;
5pub mod traits;
6pub mod functions;
7pub mod foreign_mods;
8mod common;
9
10#[derive(Debug, Clone, Copy)]
11pub enum ExternSpecKind {
12    InherentImpl,
13    TraitImpl,
14    Trait,
15    Method,
16}
17
18impl ExternSpecKind {
19    const INHERENT_IMPL_IDENT: &'static str = "inherent_impl";
20    const TRAIT_IMPL_IDENT: &'static str = "trait_impl";
21    const TRAIT_IDENT: &'static str = "trait";
22    const METHOD_IDENT: &'static str = "method";
23}
24
25impl TryFrom<String> for ExternSpecKind {
26    type Error = String;
27
28    fn try_from(string: String) -> Result<Self, Self::Error> {
29        match string.as_str() {
30            Self::INHERENT_IMPL_IDENT => Ok(Self::InherentImpl),
31            Self::TRAIT_IMPL_IDENT => Ok(Self::TraitImpl),
32            Self::TRAIT_IDENT => Ok(Self::Trait),
33            Self::METHOD_IDENT => Ok(Self::Method),
34            _ => Err(string),
35        }
36    }
37}
38
39impl From<ExternSpecKind> for String {
40    fn from(spec_type: ExternSpecKind) -> Self {
41        String::from(match spec_type {
42            ExternSpecKind::InherentImpl => ExternSpecKind::INHERENT_IMPL_IDENT,
43            ExternSpecKind::TraitImpl => ExternSpecKind::TRAIT_IMPL_IDENT,
44            ExternSpecKind::Trait => ExternSpecKind::TRAIT_IDENT,
45            ExternSpecKind::Method => ExternSpecKind::METHOD_IDENT,
46        })
47    }
48}
