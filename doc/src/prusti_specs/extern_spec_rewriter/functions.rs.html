1//! Process external specifications on free functions.
2//! In practice, these will be combined with a module argument to extern_spec
3//! e.g. `#[extern_spec(core::mem)] fn swap`
4
5use super::common::generate_extern_spec_function_stub;
6use crate::ExternSpecKind;
7use proc_macro2::{Group, TokenStream, TokenTree};
8use quote::{quote, ToTokens};
9use syn::{parse_quote_spanned, spanned::Spanned};
10
11pub fn rewrite_stub(
12    stub_tokens: &TokenStream,
13    path: &syn::Path,
14    is_unsafe: bool,
15) -> syn::Result<TokenStream> {
16    // Transforms function stubs (functions with a `;` after the
17    // signature instead of the body) into functions, then
18    // processes them.
19    let mut new_tokens = TokenStream::new();
20    for mut token in stub_tokens.clone().into_iter() {
21        if let TokenTree::Punct(punct) = &mut token {
22            if punct.as_char() == ';' {
23                new_tokens.extend(
24                    Group::new(proc_macro2::Delimiter::Brace, TokenStream::new()).to_token_stream(),
25                );
26                continue;
27            }
28        }
29        new_tokens.extend(token.to_token_stream());
30    }
31    let res: Result<syn::Item, _> = syn::parse2(new_tokens);
32    if res.is_err() {
33        return Err(syn::Error::new(
34            stub_tokens.span(),
35            "invalid function signature",
36        ));
37    }
38
39    let mut item = res.unwrap();
40    if let syn::Item::Fn(item_fn) = &mut item {
41        Ok(rewrite_fn(item_fn, path, is_unsafe))
42    } else {
43        Ok(quote!(#item))
44    }
45}

47/// Rewrite a specification function to a call to the specified function.
48/// The result of this rewriting is then parsed in `ExternSpecResolver`.
49pub fn rewrite_fn(item_fn: &syn::ItemFn, path: &syn::Path, is_unsafe: bool) -> TokenStream {
50    let ident = &item_fn.sig.ident;
51    let path_span = item_fn.sig.ident.span();
52    let path = parse_quote_spanned!(path_span=> #path :: #ident);
53
54    generate_extern_spec_function_stub(item_fn, &path, ExternSpecKind::Method, true, is_unsafe)
55}
