10    fn parse(input: ParseStream) -> syn::Result<Self> {
11        let mut attrs = input.call(syn::Attribute::parse_outer)?;
12        let mut cl: syn::ExprClosure = input.parse()?;
13
14        let mut pres: Vec<syn::Result<syn::Expr>> = vec![];
15        let mut posts: Vec<syn::Result<syn::Expr>> = vec![];
16
17        // collect and remove any specification attributes
18        // leave other attributes intact
19        attrs.retain(|attr| {
20            if let Some(id) = attr.path.get_ident() {
21                match id.to_string().as_ref() {
22                    "requires" => pres.push(syn::parse2(attr.tokens.clone())),
23                    "ensures" => posts.push(syn::parse2(attr.tokens.clone())),
24                    _ => return true,
25                }
26                false
27            } else {
28                true
29            }
30        });
31        cl.attrs = attrs;
32
33        Ok(Self {
34            pres: pres.into_iter().collect::<syn::Result<Vec<_>>>()?,
35            posts: posts.into_iter().collect::<syn::Result<Vec<_>>>()?,
36            cl,
37        })
