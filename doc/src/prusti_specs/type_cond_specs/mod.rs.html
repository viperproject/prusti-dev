8pub fn generate(attr: TokenStream, item: &untyped::AnyFnItem) -> GeneratedResult {
9    let tokens_span = attr.span();
10
11    // Parse type-conditional spec refinements information
12    let type_cond_spec = parse_type_cond_spec(attr)?;
13
14    let mut new_items = vec![];
15    let mut new_attrs = vec![];
16
17    for nested_spec in type_cond_spec.specs {
18        let (mut generated_items, generated_attrs) = match nested_spec {
19            NestedSpec::Ensures(tokens) => {
20                let span = tokens.span();
21                generate_for_ensures(tokens, span, item)?
22            }
23            NestedSpec::Requires(tokens) => {
24                let span = tokens.span();
25                generate_for_requires(tokens, span, item)?
26            }
27            NestedSpec::Pure => generate_for_pure_refinements(item)?,
28        };
29
30        for generated_item in generated_items.iter_mut() {
31            let item_fn = match generated_item {
32                syn::Item::Fn(item_fn) => item_fn,
33                _ => unreachable!(),
34            };
35
36            // Add bounds as a where clause
37            item_fn.sig.generics.where_clause = Some(generate_where_clause_for_spec(
38                &type_cond_spec.trait_bounds,
39                item_fn.sig.generics.where_clause.as_ref(),
40            ));
41
42            // Add attribute to mark this as a "specification with constraint" (used for processing the contract in `SpecCollector`)
43            item_fn.attrs.push(parse_quote_spanned! {tokens_span=>
44                #[prusti::type_cond_spec_trait_bounds_in_where_clause]
45            });
46        }
47
48        new_items.extend(generated_items);
49        new_attrs.extend(generated_attrs);
50    }
51
52    Ok((new_items, new_attrs))
53}
54
55fn generate_where_clause_for_spec(
56    trait_bounds: &[syn::PredicateType],
57    existing_where_clause: Option<&syn::WhereClause>,
58) -> syn::WhereClause {
59    let mut where_clause = existing_where_clause
60        .cloned()
61        .unwrap_or_else(|| syn::parse_quote!(where));
62    where_clause
63        .predicates
64        .extend(trait_bounds.iter().map(|bound| -> syn::WherePredicate {
65            let span = bound.span();
66            parse_quote_spanned! {span=>
67                #bound
68            }
69        }));
70    where_clause
71}
