1//! Processes `#[model]` attributed types.
2//!
3//! Usage documentation can be found in the corresponding macro definition.
4//!
5//! Given a `#[model]` attributed type `T`, this logic creates the following three items:
6//! * A struct `M` which holds the model's fields
7//! * A trait which provides a `model` method to be used in specifications
8//! * An implementation of the aforementioned trait for `T`.
9//!   The implementation is `unimplemented!()`, `#[pure]` and `#[trusted]`
10//!
11//! The model struct `M` must be copyable.
12//!
13//! # Note
14//! This macro always generates a trait with a `model` method on the fly for every modelled type.
15//! With this design, one can even model external types which are not present in the local crate.
16

30/// See module level documentation
31pub fn rewrite(item_struct: syn::ItemStruct) -> syn::Result<Vec<syn::Item>> {
32    let res = rewrite_internal(item_struct);
33    match res {
34        Ok(result) => Ok(vec![
35            syn::Item::Struct(result.model_struct),
36            syn::Item::Trait(result.to_model_trait),
37            syn::Item::Impl(result.model_impl),
38        ]),
39        Err(err) => Err(err.into()),
40    }
41}
42
43type TypeModelGenerationResult<R> = Result<R, TypeModelGenerationError>;
44
45fn rewrite_internal(item_struct: syn::ItemStruct) -> TypeModelGenerationResult<TypeModel> {
46    let idents = GeneratedIdents::generate(&item_struct);
47
48    let model_struct = ModelStruct::create(&item_struct, &idents)?;
49    let to_model_trait = ToModelTrait::create(&item_struct, &model_struct, &idents);
50    let model_impl = create_model_impl(&item_struct, &model_struct, &to_model_trait)?;
51
52    Ok(TypeModel {
53        model_struct: model_struct.item,
54        to_model_trait: to_model_trait.item,
55        model_impl,
56    })
57}
58
59struct ModelStruct {
60    item: syn::ItemStruct,
61
62    /// The path to the generated struct, e.g. to be used as a return type
63    path: syn::Path,
64}
65
66impl ModelStruct {
67    fn create(
68        item_struct: &syn::ItemStruct,
69        idents: &GeneratedIdents,
70    ) -> TypeModelGenerationResult<Self> {
71        if item_struct.fields.is_empty() {
72            return Err(TypeModelGenerationError::MissingStructFields(
73                item_struct.span(),
74            ));
75        }
76
77        let model_struct_ident = &idents.model_struct_ident;
78        let mut model_struct: syn::ItemStruct = parse_quote_spanned! {item_struct.span()=>
79            #[derive(Copy, Clone)]
80            #[allow(non_camel_case_types)]
81            struct #model_struct_ident {}
82        };
83
84        let params = item_struct
85            .parse_user_annotated_type_params()
86            .map_err(TypeModelGenerationError::NonParsableTypeParam)?;
87        model_struct.generics.params.extend(
88            params
89                .iter()
90                .filter_map(UserAnnotatedTypeParam::as_generic_type_param)
91                .cloned()
92                .map(syn::GenericParam::Type),
93        );
94
95        model_struct.fields = item_struct.fields.clone();
96        add_phantom_data_for_generic_params(&mut model_struct);
97
98        let generic_idents = model_struct
99            .generics
100            .params
101            .iter()
102            .filter_map(|generic_param| match generic_param {
103                syn::GenericParam::Type(type_param) => Some(&type_param.ident),
104                _ => None,
105            });
106        let model_path: syn::Path = parse_quote!(
107            #model_struct_ident < #(#generic_idents),* >
108        );
109
110        Ok(Self {
111            item: model_struct,
112            path: model_path,
113        })
114    }
115}

117struct ToModelTrait {
118    item: syn::ItemTrait,
119
120    /// The path to the generated trait, e.g. to be used as a return type
121    path: syn::Path,
122}
123
124impl ToModelTrait {
125    fn create(
126        item_struct: &syn::ItemStruct,
127        model_struct: &ModelStruct,
128        idents: &GeneratedIdents,
129    ) -> Self {
130        let generic_params: Vec<syn::GenericParam> =
131            model_struct.item.generics.params.iter().cloned().collect();
132        let generic_params_idents: Vec<Ident> = generic_params
133            .iter()
134            .filter_map(|generic_param| match generic_param {
135                syn::GenericParam::Type(type_param) => Some(type_param.ident.clone()),
136                _ => None,
137            })
138            .collect();
139
140        let model_path = &model_struct.path;
141
142        let to_model_trait_ident = &idents.to_model_trait_ident;
143        let item = parse_quote_spanned! {item_struct.span()=>
144            #[allow(non_camel_case_types)]
145            trait #to_model_trait_ident<#(#generic_params),*> {
146                #[pure]
147                #[trusted]
148                #[prusti::type_models_to_model_fn]
149                fn model(&self) -> #model_path;
150            }
151        };
152
153        let path: syn::Path = parse_quote!(
154            #to_model_trait_ident<#(#generic_params_idents),*>
155        );
156
157        Self { item, path }
158    }
159}

161fn create_model_impl(
162    item_struct: &syn::ItemStruct,
163    model_struct: &ModelStruct,
164    to_model_trait: &ToModelTrait,
165) -> TypeModelGenerationResult<syn::ItemImpl> {
166    let ident = &item_struct.ident;
167
168    let mut rewritten_generics: Vec<syn::GenericParam> = Vec::new();
169    for param in &item_struct.generics.params {
170        match param {
171            syn::GenericParam::Lifetime(_) => rewritten_generics.push(parse_quote!('_)),
172            syn::GenericParam::Type(type_param) => {
173                let mut cloned = type_param.clone();
174                cloned.attrs.clear();
175                cloned.bounds = Punctuated::default();
176                rewritten_generics.push(syn::GenericParam::Type(cloned))
177            }
178            syn::GenericParam::Const(const_param) => {
179                return Err(TypeModelGenerationError::ConstParamDisallowed(
180                    const_param.span(),
181                ))
182            }
183        }
184    }
185
186    let generic_params: Vec<syn::GenericParam> =
187        model_struct.item.generics.params.iter().cloned().collect();
188
189    let impl_path: syn::Path = parse_quote!(
190        #ident < #(#rewritten_generics),* >
191    );
192
193    let to_model_trait_path = &to_model_trait.path;
194    let model_struct_path = &model_struct.path;
195    let to_model_trait_str = &to_model_trait.item.ident.to_string();
196
197    Ok(parse_quote_spanned! {item_struct.span()=>
198        #[prusti::type_models_to_model_impl]
199        #[prusti::specs_version = #SPECS_VERSION]
200        impl<#(#generic_params),*> #to_model_trait_path for #impl_path {
201            #[trusted]
202            #[pure]
203            #[prusti::type_models_to_model_fn = #to_model_trait_str]
204            fn model(&self) -> #model_struct_path {
205                unimplemented!("Models can only be used in specifications")
206            }
207        }
208    })
209}

211/// [syn::Ident]s which are used for the generated items
212struct GeneratedIdents {
213    model_struct_ident: syn::Ident,
214    to_model_trait_ident: syn::Ident,
215}
216
217impl GeneratedIdents {
218    fn generate(item_struct: &syn::ItemStruct) -> Self {
219        let mut name = item_struct.ident.to_string();
220
221        for param in item_struct.generics.params.iter() {
222            if let syn::GenericParam::Type(ty_param) = param {
223                name.push_str(ty_param.ident.to_string().as_str());
224            }
225        }
226
227        let uuid = Uuid::new_v4().simple();
228
229        GeneratedIdents {
230            model_struct_ident: Ident::new(
231                format!("Prusti{name}Model_{uuid}").as_str(),
232                item_struct.ident.span(),
233            ),
234            to_model_trait_ident: Ident::new(
235                format!("Prusti{name}ToModel_{uuid}").as_str(),
236                item_struct.ident.span(),
237            ),
238        }
239    }
240}

242/// Errors that can happen during model generation.
243/// These are mostly wrong usages of the macro by the user.
244#[derive(Debug)]
245enum TypeModelGenerationError {
246    /// Thrown when the model contains no fields
247    MissingStructFields(proc_macro2::Span),
248
249    /// Thrown when using a const type param in the model
250    ConstParamDisallowed(proc_macro2::Span),
251
252    /// Thrown when user annotated generics could not be parsed
253    NonParsableTypeParam(UserAnnotatedTypeParamParserError),
254}
255
256impl std::convert::From<TypeModelGenerationError> for syn::Error {
257    fn from(err: TypeModelGenerationError) -> Self {
258        match err {
259            TypeModelGenerationError::MissingStructFields(span) => {
260                syn::Error::new(span, "Type model must have at least one field")
261            }
262            TypeModelGenerationError::ConstParamDisallowed(span) => {
263                syn::Error::new(span, "Const generics are disallowed for models")
264            }
265            TypeModelGenerationError::NonParsableTypeParam(parse_err) => parse_err.into(),
266        }
267    }
268}

270/// Type to represent generated code during expansion of the `#[model]` macro
271struct TypeModel {
272    /// The struct which represents the model
273    model_struct: syn::ItemStruct,
274
275    /// A trait which will be implemented on the modelled type
276    /// to return the [TypeModel::model_struct]
277    to_model_trait: syn::ItemTrait,
278
279    /// The implementation of the [TypeModel::model_trait] on the modelled type.
280    model_impl: syn::ItemImpl,
281}
282
283impl ToTokens for TypeModel {
284    fn to_tokens(&self, tokens: &mut TokenStream) {
285        self.to_model_trait.to_tokens(tokens);
286        self.model_struct.to_tokens(tokens);
287        self.model_impl.to_tokens(tokens);
288    }
289}

297        let to_model_trait: syn::ItemTrait = parse_quote!(
298            trait PrustiFooToModel {}
299        );
300        let model_struct: syn::ItemStruct = parse_quote!(

302        );
303        let trait_impl: syn::ItemImpl = parse_quote!(impl ToModel for Foo {});
304
305        let rewritten_model = TypeModel {
306            to_model_trait: to_model_trait.clone(),
