49pub enum FunctionCallOrLoop<FunctionCallData, LoopData> {
50    FunctionCall(FunctionCallData),
51    Loop(LoopData),
52}

54impl<FunctionCallData, LoopData> FunctionCallOrLoop<FunctionCallData, LoopData> {
55    pub(crate) fn bimap<R>(
56        self,
57        f: impl FnOnce(FunctionCallData) -> R,
58        g: impl FnOnce(LoopData) -> R,
59    ) -> R {
60        match self {
61            FunctionCallOrLoop::FunctionCall(data) => f(data),
62            FunctionCallOrLoop::Loop(data) => g(data),
63        }
64    }
65}

77impl<'tcx> AbstractionEdge<'tcx> {
78    /// Creates a singleton coupling hyperedge from this edge.
79    ///
80    /// This is presumably NOT what you want, as there is no coupling logic
81    /// involved.  Instead, consider [`BorrowsGraph::coupling_results`].
82    /// However, Prusti is currently using this function for loops.
83    #[must_use]
84    pub fn into_singleton_coupled_edge(self) -> PcgCoupledEdgeKind<'tcx> {
85        match self {
86            AbstractionEdge::FunctionCall(function_call) => {
87                PcgCoupledEdgeKind::function_call(function_call.into_singleton_coupled_edge())
88            }
89            AbstractionEdge::Loop(loop_abstraction) => {
90                PcgCoupledEdgeKind::loop_(loop_abstraction.into_singleton_coupled_edge())
91            }
92        }
93    }
94}

165impl<
166    'tcx,
167    Ctxt: Copy,
168    P: PcgPlace<'tcx, Ctxt>,
169    Input: LabelLifetimeProjection<'tcx> + PcgNodeLike<'tcx, Ctxt, P>,
170    Output: LabelLifetimeProjection<'tcx> + PcgNodeLike<'tcx, Ctxt, P>,
171> LabelEdgeLifetimeProjections<'tcx, Ctxt, P> for AbstractionBlockEdge<'tcx, Input, Output>
172{
173    fn label_lifetime_projections(
174        &mut self,
175        predicate: &LabelNodePredicate<'tcx, P>,
176        label: Option<LifetimeProjectionLabel>,
177        ctxt: Ctxt,
178    ) -> LabelLifetimeProjectionResult {
179        let source_context =
180            LabelNodeContext::new(SourceOrTarget::Source, BorrowPcgEdgeType::Abstraction);
181        let target_context =
182            LabelNodeContext::new(SourceOrTarget::Target, BorrowPcgEdgeType::Abstraction);
183        let mut changed = LabelLifetimeProjectionResult::Unchanged;
184        if predicate.applies_to(self.input.to_pcg_node(ctxt), source_context) {
185            changed |= self.input.label_lifetime_projection(label);
186        }
187        if predicate.applies_to(self.output.to_pcg_node(ctxt), target_context) {
188            changed |= self.output.label_lifetime_projection(label);
189        }
190        changed
191    }
192}

200impl<'tcx, Ctxt, P: PcgPlace<'tcx, Ctxt>> AbstractionInputLike<'tcx, Ctxt, P>
201    for LoopAbstractionInput<'tcx, P>
202{
203    fn blocks(&self, node: BlockedNode<'tcx, P>, _ctxt: Ctxt) -> bool {
204        match node {
205            PcgNode::Place(p) => **self == p.into(),
206            PcgNode::LifetimeProjection(region_projection) => match region_projection.base {
207                PlaceOrConst::Place(_) => **self == PcgNode::LifetimeProjection(region_projection),
208                PlaceOrConst::Const(_) => false,
209            },
210        }
211    }
212
213    fn to_abstraction_input(self, _ctxt: Ctxt) -> AbstractionInputTarget<'tcx, P> {
214        AbstractionInputTarget(self.0)
215    }
216}

236impl<
237    'tcx,
238    Ctxt,
239    P: PcgPlace<'tcx, Ctxt>,
240    Input: AbstractionInputLike<'tcx, Ctxt, P>,
241    Output: Copy + PcgNodeLike<'tcx, Ctxt, P>,
242> EdgeData<'tcx, Ctxt, P> for AbstractionBlockEdge<'tcx, Input, Output>
243{
244    fn blocks_node<'slf>(&self, node: BlockedNode<'tcx, P>, ctxt: Ctxt) -> bool {
245        self.input.blocks(node, ctxt)
246    }
247
248    fn blocked_nodes<'slf>(
249        &'slf self,
250        ctxt: Ctxt,
251    ) -> Box<dyn Iterator<Item = BlockedNode<'tcx, P>> + 'slf>
252    where
253        'tcx: 'slf,
254    {
255        Box::new(std::iter::once(
256            self.input.to_abstraction_input(ctxt).to_pcg_node(ctxt),
257        ))
258    }
259
260    fn blocked_by_nodes<'slf>(
261        &'slf self,
262        ctxt: Ctxt,
263    ) -> Box<dyn Iterator<Item = LocalNode<'tcx, P>> + 'slf>
264    where
265        'tcx: 'slf,
266    {
267        Box::new(std::iter::once(
268            self.output
269                .to_pcg_node(ctxt)
270                .try_to_local_node(ctxt)
271                .unwrap(),
272        ))
273    }
274}
