41/// A borrow that is explicit in the MIR (e.g. `let x = &mut y;`)
42#[derive(PartialEq, Eq, Clone, Debug, Hash)]
43pub struct BorrowEdge<'tcx, P = Place<'tcx>> {
44    /// The place that is blocked by the borrow, e.g. the y in `let x = &mut y;`
45    pub(crate) blocked_place: MaybeLabelledPlace<'tcx, P>,
46    /// The place that is assigned by the borrow, e.g. the x in `let x = &mut y;`
47    pub(crate) assigned_ref: MaybeLabelledPlace<'tcx, P>,
48    kind: mir::BorrowKind,
49
50    /// The location when the borrow was created
51    reserve_location: Location,
52
53    pub(crate) region: ty::Region<'tcx>,
54
55    // For some reason this may not be defined for certain shared borrows
56    borrow_index: Option<BorrowIndex>,
57
58    assigned_lifetime_projection_label: Option<LifetimeProjectionLabel>,
59}

61impl<'tcx, Ctxt: DebugCtxt, P: PcgPlace<'tcx, Ctxt>> LabelEdgeLifetimeProjections<'tcx, Ctxt, P>
62    for BorrowEdge<'tcx, P>
63{
64    fn label_lifetime_projections(
65        &mut self,
66        predicate: &LabelNodePredicate<'tcx, P>,
67        label: Option<LifetimeProjectionLabel>,
68        ctxt: Ctxt,
69    ) -> LabelLifetimeProjectionResult {
70        let mut changed = LabelLifetimeProjectionResult::Unchanged;
71        if predicate.applies_to(
72            PcgNode::LifetimeProjection(self.assigned_lifetime_projection(ctxt).rebase()),
73            LabelNodeContext::new(SourceOrTarget::Target, BorrowPcgEdgeType::Borrow),
74        ) {
75            self.assigned_lifetime_projection_label = label;
76            changed = LabelLifetimeProjectionResult::Changed;
77        }
78        changed
79    }
80}

82impl<'tcx, Ctxt: DebugCtxt, P: PcgPlace<'tcx, Ctxt>> LabelEdgePlaces<'tcx, Ctxt, P>
83    for BorrowEdge<'tcx, P>
84where
85    MaybeLabelledPlace<'tcx, P>: LabelPlace<'tcx, Ctxt, P>,
86    MaybeLabelledPlace<'tcx, P>: PcgNodeLike<'tcx, Ctxt, P>,
87    LifetimeProjection<'tcx, MaybeLabelledPlace<'tcx, P>>: PcgNodeLike<'tcx, Ctxt, P>,
88{
89    fn label_blocked_places(
90        &mut self,
91        predicate: &LabelNodePredicate<'tcx, P>,
92        labeller: &impl PlaceLabeller<'tcx, Ctxt, P>,
93        ctxt: Ctxt,
94    ) -> HashSet<NodeReplacement<'tcx, P>> {
95        conditionally_label_places(
96            vec![&mut self.blocked_place],
97            predicate,
98            labeller,
99            LabelNodeContext::new(SourceOrTarget::Source, BorrowPcgEdgeType::Borrow),
100            ctxt,
101        )
102    }
103
104    fn label_blocked_by_places(
105        &mut self,
106        predicate: &LabelNodePredicate<'tcx, P>,
107        labeller: &impl PlaceLabeller<'tcx, Ctxt, P>,
108        ctxt: Ctxt,
109    ) -> HashSet<NodeReplacement<'tcx, P>> {
110        let mut result = HashSet::default();
111        let initial_assigned_lifetime_projection = self.assigned_lifetime_projection(ctxt);
112        let from: PcgNodeWithPlace<'tcx, P> =
113            PcgNode::LifetimeProjection(initial_assigned_lifetime_projection.rebase());
114        let node_context = LabelNodeContext::new(SourceOrTarget::Target, BorrowPcgEdgeType::Borrow);
115        if predicate.applies_to(from, node_context) {
116            let changed = self.assigned_ref.label_place(labeller, ctxt);
117            if changed {
118                result.insert(NodeReplacement::new(
119                    from,
120                    self.assigned_lifetime_projection(ctxt).to_pcg_node(ctxt),
121                ));
122            }
123        }
124        result
125    }
126}

138impl<'a, 'tcx: 'a, Ctxt: HasCompilerCtxt<'a, 'tcx>> DisplayWithCtxt<Ctxt> for BorrowEdge<'tcx> {
139    fn display_output(&self, ctxt: Ctxt, mode: OutputMode) -> DisplayOutput {
140        let rp_part = if let Some(rp) = self.assigned_lifetime_projection_label {
141            format!(" <{}>", rp.display_output((), mode).into_text())
142        } else {
143            String::new()
144        };
145        DisplayOutput::Text(
146            format!(
147                "borrow: {}{} = &{} {}",
148                self.assigned_ref.display_output(ctxt, mode).into_text(),
149                rp_part,
150                if self.kind.mutability() == Mutability::Mut {
151                    "mut "
152                } else {
153                    ""
154                },
155                self.blocked_place.display_output(ctxt, mode).into_text(),
156            )
157            .into(),
158        )
159    }
160}

162impl<'tcx, Ctxt: Copy, P: PcgPlace<'tcx, Ctxt>> EdgeData<'tcx, Ctxt, P> for BorrowEdge<'tcx, P> {
163    fn blocks_node<'slf>(&self, node: BlockedNode<'tcx, P>, _ctxt: Ctxt) -> bool {
164        match node {
165            PcgNode::Place(p) => self.blocked_place == p,
166            _ => false,
167        }
168    }
169
170    fn is_blocked_by<'slf>(&self, node: LocalNode<'tcx, P>, ctxt: Ctxt) -> bool {
171        match node {
172            PcgNode::Place(_) => false,
173            PcgNode::LifetimeProjection(region_projection) => {
174                region_projection == self.assigned_lifetime_projection(ctxt)
175            }
176        }
177    }
178
179    fn blocked_nodes<'slf>(
180        &'slf self,
181        _ctxt: Ctxt,
182    ) -> Box<dyn Iterator<Item = BlockedNode<'tcx, P>> + 'slf>
183    where
184        'tcx: 'slf,
185    {
186        Box::new(std::iter::once(self.blocked_place.into()))
187    }
188
189    fn blocked_by_nodes<'slf>(
190        &'slf self,
191        ctxt: Ctxt,
192    ) -> Box<dyn Iterator<Item = LocalNode<'tcx, P>> + 'slf>
193    where
194        'tcx: 'slf,
195    {
196        let rp = self.assigned_lifetime_projection(ctxt);
197        Box::new(std::iter::once(LocalNode::LifetimeProjection(rp)))
198    }
199}

201impl<'tcx> BorrowEdge<'tcx> {
202    pub(crate) fn borrow_index(&self) -> Option<BorrowIndex> {
203        self.borrow_index
204    }
205
206    #[must_use]
207    pub fn region(&self) -> ty::Region<'tcx> {
208        self.region
209    }
210
211    #[must_use]
212    pub fn kind(&self) -> mir::BorrowKind {
213        self.kind
214    }
215
216    #[must_use]
217    pub fn assigned_ref(&self) -> MaybeLabelledPlace<'tcx> {
218        self.assigned_ref
219    }
220
221    #[must_use]
222    pub fn blocked_place(&self) -> MaybeLabelledPlace<'tcx> {
223        self.blocked_place
224    }
225
226    pub(crate) fn new<'a>(
227        blocked_place: MaybeLabelledPlace<'tcx>,
228        assigned_place: MaybeLabelledPlace<'tcx>,
229        kind: mir::BorrowKind,
230        reservation_location: Location,
231        region: ty::Region<'tcx>,
232        ctxt: impl HasSettings<'a> + HasBorrowCheckerCtxt<'a, 'tcx>,
233    ) -> Self
234    where
235        'tcx: 'a,
236    {
237        assert!(assigned_place.ty(ctxt).ty.ref_mutability().is_some());
238        let borrow = Self {
239            blocked_place,
240            assigned_ref: assigned_place,
241            kind,
242            reserve_location: reservation_location,
243            region,
244            assigned_lifetime_projection_label: None,
245            borrow_index: ctxt.bc().region_to_borrow_index(region.into()),
246        };
247        borrow.assert_validity(ctxt);
248        borrow
249    }
250
251    pub(crate) fn reserve_location(&self) -> Location {
252        self.reserve_location
253    }
254
255    #[must_use]
256    pub fn is_mut(&self) -> bool {
257        self.kind.mutability() == Mutability::Mut
258    }
259
260    /// The deref of the assigned place of the borrow. For example, if the borrow is
261    /// `let x = &mut y;`, then the deref place is `*x`.
262    #[must_use]
263    pub fn deref_place(&self, ctxt: CompilerCtxt<'_, 'tcx>) -> MaybeLabelledPlace<'tcx> {
264        self.assigned_ref.project_deref(ctxt)
265    }
266}
267impl<'tcx, P: Copy + std::fmt::Debug> BorrowEdge<'tcx, P> {
268    /// The region projection associated with the *type* of the assigned place
269    /// of the borrow. For example in `let x: &'x mut i32 = ???`, the assigned
270    /// region projection is `x↓'x`.
271    pub(crate) fn assigned_lifetime_projection<Ctxt: Copy>(
272        &self,
273        ctxt: Ctxt,
274    ) -> LifetimeProjection<'tcx, MaybeLabelledPlace<'tcx, P>>
275    where
276        P: PcgPlace<'tcx, Ctxt>,
277    {
278        match self.assigned_ref.place().rust_ty(ctxt).kind() {
279            ty::TyKind::Ref(region, _, _) => LifetimeProjection::new(
280                self.assigned_ref,
281                (*region).into(),
282                self.assigned_lifetime_projection_label,
283                ctxt,
284            )
285            .unwrap(),
286            other => unreachable!("{:?}", other),
287        }
288    }
289}
