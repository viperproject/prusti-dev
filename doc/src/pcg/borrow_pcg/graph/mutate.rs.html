27/// Insert `edge` into `edges`, joining validity conditions if an identical
28/// edge kind is already present. Edge mutations (e.g. unlabelling lifetime
29/// projections) can make two conditional edges identical; their conditions
30/// must be joined, not overwritten.
31fn insert_joining_conditions<EdgeKind: Eq + std::hash::Hash, VC, Ctxt: Copy>(
32    edges: &mut HashMap<EdgeKind, VC>,
33    edge: Conditioned<EdgeKind, VC>,
34    ctxt: Ctxt,
35) where
36    VC: ValidityConditionOps<Ctxt>,
37{
38    match edges.entry(edge.value) {
39        hash_map::Entry::Occupied(mut existing) => {
40            existing.get_mut().join(&edge.conditions, ctxt);
41        }
42        hash_map::Entry::Vacant(slot) => {
43            slot.insert(edge.conditions);
44        }
45    }
46}

48impl<'tcx, EdgeKind> BorrowsGraph<'tcx, EdgeKind> {
49    fn mut_edge_conditions(&mut self, mut f: impl FnMut(&mut ValidityConditions) -> bool) -> bool {
50        let mut changed = false;
51        for conditions in self.edges.values_mut() {
52            if f(conditions) {
53                changed = true;
54            }
55        }
56        changed
57    }
58
59    pub(crate) fn add_path_condition(
60        &mut self,
61        pc: PathCondition,
62        ctxt: CompilerCtxt<'_, 'tcx>,
63    ) -> bool {
64        self.mut_edge_conditions(|conditions| conditions.insert(pc, ctxt.body()))
65    }
66}

68impl<'tcx, EdgeKind, VC> BorrowsGraph<'tcx, EdgeKind, VC> {
69    pub(crate) fn mut_edges<Ctxt: Copy>(
70        &mut self,
71        mut f: impl FnMut(&mut Conditioned<EdgeKind, VC>) -> bool,
72        ctxt: Ctxt,
73    ) -> bool
74    where
75        EdgeKind: Eq + std::hash::Hash + PartialEq,
76        VC: ValidityConditionOps<Ctxt>,
77    {
78        let mut changed = false;
79        let mut new_edges = HashMap::default();
80        for (kind, conditions) in self.edges.drain() {
81            let mut edge = Conditioned::new(kind, conditions);
82            if f(&mut edge) {
83                changed = true;
84            }
85            insert_joining_conditions(&mut new_edges, edge, ctxt);
86        }
87        self.edges = new_edges;
88        changed
89    }
90
91    pub(crate) fn filter_mut_edges<Ctxt: Copy>(
92        &mut self,
93        mut f: impl FnMut(&mut Conditioned<EdgeKind, VC>) -> FilterMutResult,
94        ctxt: Ctxt,
95    ) -> bool
96    where
97        EdgeKind: Eq + std::hash::Hash + PartialEq,
98        VC: ValidityConditionOps<Ctxt>,
99    {
100        let mut changed = false;
101        let mut new_edges = HashMap::default();
102        for (kind, conditions) in self.edges.drain() {
103            let mut edge = Conditioned::new(kind, conditions);
104            match f(&mut edge) {
105                FilterMutResult::Changed => {
106                    changed = true;
107                    insert_joining_conditions(&mut new_edges, edge, ctxt);
108                }
109                FilterMutResult::Unchanged => {
110                    insert_joining_conditions(&mut new_edges, edge, ctxt);
111                }
112                FilterMutResult::Remove => {}
113            }
114        }
115        self.edges = new_edges;
116        changed
117    }
118    pub(crate) fn label_place<P: PcgPlace<'tcx, Ctxt>, Ctxt: Copy>(
119        &mut self,
120        place: P,
121        reason: LabelPlaceReason,
122        labeller: &impl PlaceLabeller<'tcx, Ctxt, P>,
123        ctxt: Ctxt,
124    ) -> HashSet<NodeReplacement<'tcx, P>>
125    where
126        EdgeKind: LabelEdgePlaces<'tcx, Ctxt, P> + Eq + std::hash::Hash,
127        VC: ValidityConditionOps<Ctxt>,
128    {
129        let mut all_replacements = HashSet::default();
130        self.mut_edges(
131            |edge| {
132                let replacements = reason.apply_to_edge(place, &mut edge.value, labeller, ctxt);
133                let changed = !replacements.is_empty();
134                all_replacements.extend(replacements);
135                changed
136            },
137            ctxt,
138        );
139        all_replacements
140    }
