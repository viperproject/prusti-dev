63    pub(crate) fn render_debug_graph<'a>(
64        &self,
65        block: BasicBlock,
66        debug_imgcat: Option<DebugImgcat>,
67        capabilities: &impl PlaceCapabilitiesReader<'tcx>,
68        comment: &str,
69        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
70    ) where
71        'tcx: 'a,
72    {
73        #[cfg(feature = "visualization")]
74        if borrows_imgcat_debug(block, debug_imgcat)
75            && let Ok(dot_graph) = generate_borrows_dot_graph(ctxt.bc_ctxt(), capabilities, self)
76        {
77            DotGraph::render_with_imgcat(&dot_graph, comment).unwrap_or_else(|e| {
78                eprintln!("Error rendering self graph: {e}");
79            });
80        }
81    }

96    fn apply_placeholder_labels<'mir, Ctxt>(
97        &mut self,
98        _capabilities: &impl PlaceCapabilitiesReader<'tcx>,
99        ctxt: Ctxt,
100    ) where
101        'tcx: 'mir,
102        Ctxt: HasBorrowCheckerCtxt<'mir, 'tcx> + HasSettings<'mir>,
103    {
104        let nodes = self.nodes(ctxt.bc_ctxt());
105        for node in nodes {
106            if let PcgNode::LifetimeProjection(rp) = node
107                && rp.is_future()
108                && let Some(PcgNode::LifetimeProjection(local_rp)) =
109                    rp.try_to_local_node(ctxt.bc_ctxt())
110            {
111                let orig_rp = local_rp.with_label(None, ctxt.bc_ctxt());
112                self.filter_mut_edges(
113                    |edge| {
114                        edge.value
115                            .label_lifetime_projections(
116                                &LabelNodePredicate::equals_lifetime_projection(orig_rp),
117                                Some(LifetimeProjectionLabel::Future),
118                                ctxt,
119                            )
120                            .to_filter_mut_result()
121                    },
122                    ctxt,
123                );
124            }
125        }
126    }

128    pub(crate) fn merge<'a>(
129        &mut self,
130        other_graph: &BorrowsGraph<'tcx>,
131        ctxt: AnalysisCtxt<'a, 'tcx>,
132    ) {
133        pcg_validity_assert!(
134            other_graph.is_valid(ctxt.bc_ctxt()),
135            [ctxt],
136            "Other graph is invalid"
137        );
138        for other_edge in other_graph.edges() {
139            self.insert(other_edge.to_owned_edge(), ctxt);
140        }
141    }

143    /// Finalize the merged graph, constructing a loop abstraction only after
144    /// every incoming graph and its validity conditions have been merged.
145    pub(crate) fn finish_join<'a>(
146        &mut self,
147        validity_conditions: &ValidityConditions,
148        mut args: JoinBorrowsArgs<'_, 'a, 'tcx>,
149        ctxt: AnalysisCtxt<'a, 'tcx>,
150    ) {
151        let self_block = args.self_block;
152        let old_self = self.clone();
153
154        if let Some(used_places) = args
155            .body_analysis
156            .get_places_used_in_loop_with_head(self_block)
157        {
158            self.join_loop(used_places, validity_conditions, args.reborrow(), ctxt);
159            #[cfg(feature = "visualization")]
160            if borrows_imgcat_debug(self_block, Some(DebugImgcat::JoinLoop))
161                && let Ok(dot_graph) =
162                    generate_borrows_dot_graph(ctxt.ctxt, args.capabilities, self)
163            {
164                DotGraph::render_with_imgcat(&dot_graph, "After join (loop):").unwrap_or_else(
165                    |e| {
166                        eprintln!("Error rendering self graph: {e}");
167                    },
168                );
169            }
170            pcg_validity_assert!(
171                self.is_valid(ctxt.bc_ctxt()),
172                [ctxt],
173                "Graph became invalid after join"
174            );
175            return;
176        }
177        for edge in self
178            .edges()
179            .map(BorrowPcgEdgeLike::to_owned_edge)
180            .collect::<Vec<_>>()
181        {
182            if let BorrowPcgEdgeKind::Abstraction(_) = edge.kind() {
183                continue;
184            }
185            if self.is_encapsulated_by_abstraction(&edge.value, ctxt.ctxt) {
186                self.remove(edge.kind());
187            }
188        }
189
190        self.apply_placeholder_labels(args.capabilities, ctxt);
191
192        if validity_checks_enabled() && !self.is_valid(ctxt.bc_ctxt()) {
193            pcg_validity_assert!(
194                false,
195                [ctxt],
196                "Graph became invalid after join at {self_block:?}"
197            );
198            #[cfg(feature = "visualization")]
199            {
200                if let Ok(dot_graph) =
201                    generate_borrows_dot_graph(ctxt.ctxt, args.capabilities, self)
202                {
203                    DotGraph::render_with_imgcat(&dot_graph, "Invalid self graph").unwrap_or_else(
204                        |e| {
205                            eprintln!("Error rendering self graph: {e}");
206                        },
207                    );
208                }
209                if let Ok(dot_graph) =
210                    generate_borrows_dot_graph(ctxt.ctxt, args.capabilities, &old_self)
211                {
212                    DotGraph::render_with_imgcat(&dot_graph, "Old self graph").unwrap_or_else(
213                        |e| {
214                            eprintln!("Error rendering old self graph: {e}");
215                        },
216                    );
217                }
218            }
219        }
220    }

222    fn join_loop<'mir>(
223        &mut self,
224        used_places: &PlaceUsages<'tcx>,
225        validity_conditions: &ValidityConditions,
226        mut args: JoinBorrowsArgs<'_, 'mir, 'tcx>,
227        ctxt: AnalysisCtxt<'mir, 'tcx>,
228    ) {
229        let loop_head = args.self_block;
230        logging::log!(
231            &LogPredicate::DebugBlock,
232            ctxt,
233            "used places: {}",
234            used_places.display_string(ctxt.ctxt)
235        );
236
237        #[cfg(feature = "visualization")]
238        let mut dot_graphs: Vec<(String, String)> = vec![];
239
240        #[cfg(feature = "visualization")]
241        dot_graphs.push((
242            "Pre-Loop".to_owned(),
243            self.debug_graph(args.capabilities, ctxt),
244        ));
245
246        // p_loop
247        let live_loop_places = used_places.usages_where(|p| {
248            args.body_analysis.is_live_and_initialized_at(
249                mir::Location {
250                    block: args.self_block,
251                    statement_index: 0,
252                },
253                p.place,
254            )
255        });
256
257        let live_loop_places = live_loop_places
258            .iter()
259            .map(|p| {
260                if p.place.contains_unsafe_deref(ctxt.ctxt) {
261                    let edges = self.edges_blocking(p.place.into(), ctxt);
262                    let raw_ptr_edge = edges
263                        .into_iter()
264                        .filter_map(|e| match e.kind {
265                            BorrowPcgEdgeKind::Delegation(raw_ptr_edge) => Some(raw_ptr_edge),
266                            _ => None,
267                        })
268                        .collect::<Vec<_>>();
269                    assert_eq!(raw_ptr_edge.len(), 1);
270                    PlaceUsage {
271                        place: raw_ptr_edge[0].aliased_place.place(),
272                        usage: p.usage,
273                    }
274                } else {
275                    p
276                }
277            })
278            .collect::<PlaceUsages<'_>>();
279
280        logging::log!(
281            &LogPredicate::DebugBlock,
282            ctxt,
283            "live loop places: {}",
284            live_loop_places.display_string(ctxt.ctxt)
285        );
286
287        let loop_blocked_places = live_loop_places.usages_where(|p| {
288            ctxt.ctxt.borrow_checker.is_directly_blocked(
289                p.place,
290                mir::Location {
291                    block: args.self_block,
292                    statement_index: 0,
293                },
294                ctxt.ctxt,
295            )
296        });
297
298        logging::log!(
299            &LogPredicate::DebugBlock,
300            ctxt,
301            "loop_blocked_places: {}",
302            loop_blocked_places.display_string(ctxt.ctxt)
303        );
304
305        let loop_blocker_places = live_loop_places
306            .usages_where(|p| !p.usage.is_read() && !p.place.regions(ctxt.ctxt).is_empty());
307
308        logging::log!(
309            &LogPredicate::DebugBlock,
310            ctxt,
311            "loop_blocker_places: {}",
312            loop_blocker_places.display_string(ctxt.ctxt)
313        );
314
315        // Label mutated places before expansion
316        #[cfg(feature = "visualization")]
317        let mut place_labels: Vec<(String, Vec<PlaceLabelReplacement>)> = Vec::new();
318        for pu in used_places.iter() {
319            let owned: &_ = args.owned;
320            let pcg_ref = PcgRef {
321                borrow: BorrowStateRef::new(self, validity_conditions),
322                place_capabilities: args.capabilities,
323                owned,
324            };
325            if !pcg_ref.is_leaf_place(pu.place, ctxt) && !pu.usage.is_read() {
326                let replacements = self.label_place(
327                    pu.place,
328                    LabelPlaceReason::JoinLoop,
329                    &SetLabel(SnapshotLocation::BeforeJoin(loop_head)),
330                    ctxt,
331                );
332                #[cfg(feature = "visualization")]
333                if !replacements.is_empty() {
334                    place_labels.push((
335                        pu.place.display_string(ctxt),
336                        replacements
337                            .into_iter()
338                            .map(|r| {
339                                PlaceLabelReplacement::new(
340                                    r.from.display_string(ctxt),
341                                    r.to.display_string(ctxt),
342                                )
343                            })
344                            .collect(),
345                    ));
346                }
347                #[cfg(not(feature = "visualization"))]
348                let _ = replacements;
349            }
350        }
351
352        #[cfg(feature = "visualization")]
353        dot_graphs.push((
354            "Place-Label".to_owned(),
355            self.debug_graph(args.capabilities, ctxt),
356        ));
357
358        let loop_id = ctxt
359            .body_analysis
360            .loop_analysis
361            .loop_head_of(loop_head)
362            .unwrap();
363        let invariant_capabilities = ctxt.body_analysis.loop_invariant_capabilities(loop_id);
364
365        logging::log!(
366            &LogPredicate::DebugBlock,
367            ctxt,
368            "loop invariant capabilities: {}",
369            invariant_capabilities.display_string(ctxt.ctxt)
370        );
371
372        self.unpack_for_loop_invariant_capabilities(
373            &loop_blocked_places,
374            invariant_capabilities,
375            validity_conditions,
376            &mut args.reborrow(),
377            ctxt,
378        );
379
380        let expand_places = loop_blocker_places.joined_with(&loop_blocked_places);
381
382        self.expand_places_for_abstraction(
383            &loop_blocked_places,
384            &expand_places,
385            validity_conditions,
386            &mut args.reborrow(),
387            ctxt,
388        );
389        let capabilities = args.capabilities;
390        self.render_debug_graph(
391            loop_head,
392            Some(DebugImgcat::JoinLoop),
393            capabilities,
394            "G_Pre'",
395            ctxt.ctxt,
396        );
397
398        #[cfg(feature = "visualization")]
399        dot_graphs.push((
400            "Post-Expand".to_owned(),
401            self.debug_graph(capabilities, ctxt),
402        ));
403
404        // p_roots: compute per-place ancestors
405        let ancestors_of_live_places: Vec<_> = live_loop_places
406            .iter()
407            .map(|p| {
408                let roots = self.get_borrow_roots(p.place, loop_head, ctxt.ctxt);
409                (p, roots)
410            })
411            .collect();
412
413        let live_roots: HashSet<_> = ancestors_of_live_places
414            .iter()
415            .flat_map(|(_, roots)| roots.iter().copied())
416            .collect();
417
418        logging::log!(
419            &LogPredicate::DebugBlock,
420            ctxt,
421            "live roots: {}",
422            live_roots.display_string(ctxt.ctxt)
423        );
424
425        let root_places = live_roots
426            .iter()
427            .filter_map(PcgNode::related_maybe_remote_current_place)
428            .filter(|p| {
429                !(p.is_local() && live_loop_places.contains(p.relevant_place_for_blocking()))
430            })
431            .collect::<HashSet<_>>();
432
433        logging::log!(
434            &LogPredicate::DebugBlock,
435            ctxt,
436            "root places: {}",
437            root_places.display_string(ctxt.ctxt)
438        );
439
440        let ConstructAbstractionGraphResult {
441            graph: mut abstraction_graph,
442            to_label,
443            capability_updates,
444            blocked_root_places,
445        } = self.get_loop_abstraction_graph(
446            &loop_blocked_places,
447            &root_places,
448            &loop_blocker_places,
449            loop_head,
450            validity_conditions,
451            ctxt,
452        );
453
454        abstraction_graph.render_debug_graph(
455            loop_head,
456            Some(DebugImgcat::JoinLoop),
457            capabilities,
458            "Abstraction graph",
459            ctxt.ctxt,
460        );
461
462        for rp in &to_label {
463            self.filter_mut_edges(
464                |edge| {
465                    edge.value
466                        .label_lifetime_projections(
467                            rp,
468                            Some(LifetimeProjectionLabel::Location(SnapshotLocation::Loop(
469                                loop_head,
470                            ))),
471                            ctxt,
472                        )
473                        .to_filter_mut_result()
474                },
475                ctxt,
476            );
477        }
478
479        for (place, cap_option) in capability_updates {
480            if let Some(cap) = cap_option {
481                capabilities.insert(place, cap, ctxt);
482            } else {
483                capabilities.remove(place, ctxt);
484            }
485        }
486
487        // For each root place that is actually blocked by the loop abstraction,
488        // downgrade mutable-reference ancestors from E to W.  Without this,
489        // `variant: E` and `(*variant).fields: E` can coexist once the
490        // abstraction edge is removed (e.g. in unreachable `otherwise` branches),
491        // violating the parent-child capability invariant.
492        for root_place in &blocked_root_places {
493            let mut p = *root_place;
494            while let Some(parent) = p.parent_place() {
495                if capabilities.get(parent, ctxt.ctxt) == Some(CapabilityKind::Exclusive)
496                    && (parent.ref_mutability(ctxt.ctxt).is_some() || parent.is_raw_ptr(ctxt.ctxt))
497                {
498                    capabilities.insert(parent, CapabilityKind::Write, ctxt.ctxt);
499                }
500                p = parent;
501            }
502        }
503
504        let abstraction_graph_pcg_nodes = abstraction_graph.nodes(ctxt.ctxt);
505        let to_cut = self.identify_subgraph_to_cut(loop_head, &abstraction_graph_pcg_nodes, ctxt);
506        self.reconnect_cut_outputs(
507            &to_cut,
508            &mut abstraction_graph,
509            loop_head,
510            validity_conditions,
511            ctxt,
512        );
513
514        #[cfg(feature = "visualization")]
515        dot_graphs.push((
516            "Abstraction".to_owned(),
517            abstraction_graph.debug_graph(capabilities, ctxt),
518        ));
519        #[cfg(feature = "visualization")]
520        dot_graphs.push(("To Cut".to_owned(), to_cut.debug_graph(capabilities, ctxt)));
521
522        to_cut.render_debug_graph(
523            loop_head,
524            Some(DebugImgcat::JoinLoop),
525            capabilities,
526            "To cut",
527            ctxt.ctxt,
528        );
529        self.render_debug_graph(
530            loop_head,
531            Some(DebugImgcat::JoinLoop),
532            capabilities,
533            "Self before cut",
534            ctxt.ctxt,
535        );
536
537        // Store the loop debug data for the visualization
538        #[cfg(feature = "visualization")]
539        ctxt.set_debug_loop_data(PcgLoopDebugData {
540            used_places: used_places.to_debug_repr(ctxt),
541            live_loop_places: live_loop_places.to_debug_repr(ctxt),
542            loop_blocked_places: loop_blocked_places.to_debug_repr(ctxt),
543            loop_blocker_places: loop_blocker_places.to_debug_repr(ctxt),
544            ancestors_of_live_places: ancestors_of_live_places
545                .iter()
546                .map(|(p, ancestors)| {
547                    (
548                        p.display_string(ctxt),
549                        ancestors.iter().map(|n| n.display_string(ctxt)).collect(),
550                    )
551                })
552                .collect(),
553            root_places: root_places
554                .iter()
555                .map(|p| (p.display_string(ctxt), Vec::<String>::new()))
556                .collect(),
557            dot_graphs,
558            place_labels,
559        });
560
561        for edge in to_cut.edges() {
562            self.remove(edge.kind());
563        }
564        self.render_debug_graph(
565            loop_head,
566            Some(DebugImgcat::JoinLoop),
567            capabilities,
568            "Self after cut",
569            ctxt.ctxt,
570        );
571        for edge in abstraction_graph.into_edges() {
572            self.insert(edge, ctxt.ctxt);
573        }
574        let self_places = self.places(ctxt.ctxt);
575        for place in to_cut.places(ctxt.ctxt) {
576            if place.as_borrowed_place(ctxt.ctxt).is_some()
577                && capabilities.get(place, ctxt.ctxt).is_some()
578                && !self_places.contains(&place)
579            {
580                capabilities.remove(place, ctxt);
581            }
582        }
583        self.render_debug_graph(
584            loop_head,
585            Some(DebugImgcat::JoinLoop),
586            capabilities,
587            "Final graph",
588            ctxt.ctxt,
589        );
590    }
