34/// A data structure used for querying the Borrow PCG
35///
36/// It contains a reference to a Borrow PCG and mutable data structures for
37/// caching intermediate query results.
38pub struct FrozenGraphRef<
39    'graph,
40    'tcx,
41    P = Place<'tcx>,
42    EdgeKind = BorrowPcgEdgeKind<'tcx>,
43    VC = ValidityConditions,
44> {
45    graph: &'graph BorrowsGraph<'tcx, EdgeKind, VC>,
46    nodes_cache: RefCell<Option<FxHashSet<PcgNodeWithPlace<'tcx, P>>>>,
47    edges_blocking_cache: RefCell<FxHashMap<PcgNode<'tcx>, CachedBlockingEdges<'graph, 'tcx>>>,
48    leaf_edges_cache: RefCell<Option<CachedLeafEdges<'graph, 'tcx>>>,
49    roots_cache: RefCell<Option<FxHashSet<PcgNode<'tcx>>>>,
50}
51
52impl<'tcx, P: PcgNodeComponent, VC> FrozenGraphRef<'_, 'tcx, P, BorrowPcgEdgeKind<'tcx, P>, VC> {}
53
54impl<'graph, 'tcx, EdgeKind, VC> FrozenGraphRef<'graph, 'tcx, Place<'tcx>, EdgeKind, VC> {
55    pub(crate) fn new(graph: &'graph BorrowsGraph<'tcx, EdgeKind, VC>) -> Self {
56        Self {
57            graph,
58            nodes_cache: RefCell::new(None),
59            edges_blocking_cache: RefCell::new(FxHashMap::default()),
60            leaf_edges_cache: RefCell::new(None),
61            roots_cache: RefCell::new(None),
62        }
63    }
64}

66impl<'tcx, P: PcgNodeComponent, EdgeKind: std::hash::Hash + Eq, VC>
67    FrozenGraphRef<'_, 'tcx, P, EdgeKind, VC>
68{
69    pub fn leaf_nodes<'slf, Ctxt: Copy + DebugCtxt>(
70        &'slf self,
71        ctxt: Ctxt,
72    ) -> Vec<LocalNode<'tcx, P>>
73    where
74        EdgeKind: std::hash::Hash + Eq + EdgeData<'tcx, Ctxt, P>,
75        P: PcgPlace<'tcx, Ctxt>,
76    {
77        self.nodes(ctxt)
78            .iter()
79            .filter_map(|node| node.try_to_local_node(ctxt))
80            .filter(|node| self.is_leaf(*node, ctxt))
81            .collect()
82    }
83
84    pub(crate) fn is_leaf<'slf, Ctxt: Copy + DebugCtxt>(
85        &'slf self,
86        node: LocalNode<'tcx, P>,
87        ctxt: Ctxt,
88    ) -> bool
89    where
90        EdgeKind: std::hash::Hash + Eq + EdgeData<'tcx, Ctxt, P>,
91    {
92        self.graph
93            .edges()
94            .all(|edge| !edge.kind.blocks_node(node.into(), ctxt))
95    }
96
97    pub fn nodes<'slf, Ctxt: Copy + DebugCtxt>(
98        &'slf self,
99        ctxt: Ctxt,
100    ) -> Ref<'slf, FxHashSet<PcgNodeWithPlace<'tcx, P>>>
101    where
102        EdgeKind: std::hash::Hash + Eq + EdgeData<'tcx, Ctxt, P>,
103        P: PcgPlace<'tcx, Ctxt>,
104    {
105        if let Ok(nodes) = Ref::filter_map(self.nodes_cache.borrow(), Option::as_ref) {
106            return nodes;
107        }
108        let nodes = self.graph.nodes(ctxt);
109        self.nodes_cache.replace(Some(nodes));
110        Ref::map(self.nodes_cache.borrow(), |o| o.as_ref().unwrap())
111    }
112}

114impl<'graph, 'tcx> FrozenGraphRef<'graph, 'tcx> {
115    pub(crate) fn is_acyclic<'mir: 'graph, 'bc: 'graph>(
116        &self,
117        ctxt: CompilerCtxt<'mir, 'tcx>,
118    ) -> bool {
119        enum PushResult<'tcx, 'graph> {
120            ExtendPath(Path<'tcx, 'graph>),
121            Cycle,
122        }
123
124        #[derive(Clone, Debug, Eq, PartialEq, Hash)]
125        struct Path<'tcx, 'graph>(Vec<BorrowPcgEdgeRef<'tcx, 'graph>>);
126
127        impl<'tcx, 'graph> Path<'tcx, 'graph> {
128            fn try_push(
129                mut self,
130                edge: BorrowPcgEdgeRef<'tcx, 'graph>,
131                _ctxt: CompilerCtxt<'_, 'tcx>,
132            ) -> PushResult<'tcx, 'graph> {
133                if self.0.contains(&edge) {
134                    PushResult::Cycle
135                } else {
136                    self.0.push(edge);
137                    PushResult::ExtendPath(self)
138                }
139            }
140
141            fn last(&self) -> BorrowPcgEdgeRef<'tcx, 'graph> {
142                *self.0.last().unwrap()
143            }
144
145            fn new(edge: BorrowPcgEdgeRef<'tcx, 'graph>) -> Self {
146                Self(vec![edge])
147            }
148
149            #[must_use]
150            fn leads_to_cycle<'mir: 'graph, 'bc: 'graph>(
151                &self,
152                graph: &FrozenGraphRef<'graph, 'tcx>,
153                ctxt: CompilerCtxt<'mir, 'tcx>,
154            ) -> bool {
155                let curr = self.last();
156                let blocking_edges = curr
157                    .blocked_by_nodes(ctxt)
158                    .flat_map(|node| graph.get_edges_blocking(node.into(), ctxt))
159                    .unique();
160                for edge in blocking_edges {
161                    match self.clone().try_push(edge, ctxt) {
162                        PushResult::Cycle => {
163                            tracing::info!("Cycle: {}", self.0.display_string(ctxt));
164                            return true;
165                        }
166                        PushResult::ExtendPath(next_path) => {
167                            if next_path.leads_to_cycle(graph, ctxt) {
168                                return true;
169                            }
170                        }
171                    }
172                }
173                false
174            }
175        }
176
177        for node in self.nodes(ctxt).iter() {
178            for edge in self.get_edges_blocking(*node, ctxt) {
179                if Path::new(edge).leads_to_cycle(self, ctxt) {
180                    return false;
181                }
182            }
183        }
184
185        true
186    }

188    pub fn roots<'slf, 'bc: 'graph>(
189        &'slf self,
190        ctxt: CompilerCtxt<'_, 'tcx>,
191    ) -> Ref<'slf, FxHashSet<PcgNode<'tcx>>> {
192        if let Ok(roots) = Ref::filter_map(self.roots_cache.borrow(), Option::as_ref) {
193            return roots;
194        }
195        let roots = self.graph.roots(ctxt);
196        self.roots_cache.replace(Some(roots));
197        Ref::map(self.roots_cache.borrow(), |o| o.as_ref().unwrap())
198    }

200    pub fn leaf_edges<'slf, 'a: 'graph, 'bc: 'graph>(
201        &'slf self,
202        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
203    ) -> CachedLeafEdges<'graph, 'tcx>
204    where
205        'tcx: 'a,
206    {
207        {
208            let edges = self.leaf_edges_cache.borrow();
209            if let Some(edges) = edges.as_ref() {
210                return edges.clone();
211            }
212        }
213        let edges: CachedLeafEdges<'graph, 'tcx> = self.graph.leaf_edges_set(ctxt, self);
214        self.leaf_edges_cache.replace(Some(edges.clone()));
215        edges
216    }
217
218    pub fn get_edges_blocking<'slf, 'mir: 'graph, 'bc: 'graph>(
219        &'slf self,
220        node: PcgNode<'tcx>,
221        ctxt: CompilerCtxt<'mir, 'tcx>,
222    ) -> CachedBlockingEdges<'graph, 'tcx> {
223        {
224            let map = self.edges_blocking_cache.borrow();
225            if let Some(edges) = map.get(&node) {
226                return edges.clone();
227            }
228        }
229        let edges = CachedBlockingEdges::new(self.graph.edges_blocking_set(node, ctxt));
230        self.edges_blocking_cache
231            .borrow_mut()
232            .insert(node, edges.clone());
233        edges
234    }
235
236    pub fn has_edge_blocking<'slf, 'a: 'graph, 'bc: 'graph>(
237        &'slf self,
238        node: PcgNode<'tcx>,
239        ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
240    ) -> bool
241    where
242        'tcx: 'a,
243    {
244        {
245            let map = self.edges_blocking_cache.borrow();
246            if let Some(edges) = map.get(&node) {
247                return !edges.is_empty();
248            }
249        }
250        let edges = CachedBlockingEdges::new(self.graph.edges_blocking_set(node, ctxt));
251        let result = !edges.is_empty();
252        self.edges_blocking_cache.borrow_mut().insert(node, edges);
253        result
254    }
