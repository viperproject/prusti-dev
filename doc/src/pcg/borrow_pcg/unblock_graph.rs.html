18type UnblockEdge<'tcx> = BorrowPcgEdge<'tcx>;

20/// A subgraph of the Borrow PCG including the edges that should be removed
21/// in order to unblock a given node.
22#[derive(Clone, Debug)]
23pub struct UnblockGraph<'tcx, Edge = UnblockEdge<'tcx>> {
24    edges: HashSet<Edge>,
25    _marker: PhantomData<&'tcx ()>,
26}

28/// An action that removes an edge from the Borrow PCG
29#[derive(Clone, Debug, Eq, PartialEq, From)]
30pub struct BorrowPcgUnblockAction<'tcx, Edge = BorrowPcgEdge<'tcx>> {
31    pub(super) edge: Edge,
32    _marker: PhantomData<&'tcx ()>,
33}

35impl<Edge> BorrowPcgUnblockAction<'_, Edge> {
36    pub fn new(edge: Edge) -> Self {
37        Self {
38            edge,
39            _marker: PhantomData,
40        }
41    }
42    pub fn edge(&self) -> &Edge {
43        &self.edge
44    }
45}

47impl<'tcx, Edge: std::fmt::Debug + Clone + Eq + std::hash::Hash> UnblockGraph<'tcx, Edge> {
48    /// Returns an ordered list of actions to unblock the edges in the graph.
49    /// This is essentially a topological sort of the edges.
50    pub fn actions<'a, Ctxt: HasBorrowCheckerCtxt<'a, 'tcx>>(
51        self,
52        ctxt: Ctxt,
53    ) -> Result<Vec<BorrowPcgUnblockAction<'tcx, Edge>>, PcgInternalError>
54    where
55        'tcx: 'a,
56        Edge: EdgeData<'tcx, Ctxt>,
57    {
58        let mut edges = self.edges;
59        let mut actions = vec![];
60
61        while !edges.is_empty() {
62            let mut to_keep = edges.clone();
63
64            let should_kill_edge = |edge: &Edge| {
65                edge.blocked_by_nodes(ctxt)
66                    .all(|node| edges.iter().all(|e| !e.blocks_node(node.into(), ctxt)))
67            };
68            for edge in &edges {
69                if should_kill_edge(edge) {
70                    actions.push(BorrowPcgUnblockAction::new(edge.clone()));
71                    to_keep.remove(edge);
72                }
73            }
74            if to_keep.len() >= edges.len() {
75                return Err(PcgInternalError::new(format!(
76                    "Didn't remove any leaves {edges:#?}"
77                )));
78            }
79            edges = to_keep;
80        }
81        Ok(actions)
82    }
83}

85impl<'tcx> UnblockGraph<'tcx> {
86    pub(crate) fn new() -> Self {
87        Self {
88            edges: HashSet::default(),
89            _marker: PhantomData,
90        }
91    }
92
93    pub fn for_node(
94        node: impl Into<BlockedNode<'tcx>>,
95        state: &BorrowsState<'_, 'tcx>,
96        ctxt: CompilerCtxt<'_, 'tcx>,
97    ) -> Self {
98        Self::for_nodes(vec![node], state, ctxt)
99    }
100
101    pub fn for_nodes(
102        nodes: impl IntoIterator<Item = impl Into<BlockedNode<'tcx>>>,
103        state: &BorrowsState<'_, 'tcx>,
104        ctxt: CompilerCtxt<'_, 'tcx>,
105    ) -> Self {
106        let mut ug = Self::new();
107        for node in nodes {
108            ug.unblock_node(node.into(), state, ctxt);
109        }
110        ug
111    }
112
113    #[must_use]
114    pub fn is_empty(&self) -> bool {
115        self.edges.is_empty()
116    }
117
118    pub(crate) fn edges(&self) -> impl Iterator<Item = &UnblockEdge<'tcx>> {
119        self.edges.iter()
120    }
121
122    fn add_dependency(&mut self, unblock_edge: UnblockEdge<'tcx>) -> bool {
123        self.edges.insert(unblock_edge)
124    }
125
126    pub(crate) fn kill_edge<
127        'a,
128        Edge: BorrowPcgEdgeLike<'tcx> + EdgeData<'tcx, Ctxt>,
129        Ctxt: Copy + HasBorrowCheckerCtxt<'a, 'tcx>,
130    >(
131        &mut self,
132        edge: &Edge,
133        borrows: &BorrowsState<'a, 'tcx>,
134        ctxt: Ctxt,
135    ) where
136        'tcx: 'a,
137    {
138        if self.add_dependency(edge.clone().to_owned_edge()) {
139            for blocking_node in edge.blocked_by_nodes(ctxt) {
140                self.unblock_node(blocking_node.into(), borrows, ctxt.bc_ctxt());
141            }
142        }
143    }
144
145    #[tracing::instrument(skip(self, borrows, ctxt))]
146    pub(crate) fn unblock_node<'a: 'slf, 'slf, Ctxt: Copy + HasBorrowCheckerCtxt<'a, 'tcx>>(
147        &'slf mut self,
148        node: BlockedNode<'tcx>,
149        borrows: &'slf BorrowsState<'a, 'tcx>,
150        ctxt: Ctxt,
151    ) where
152        'tcx: 'a,
153        BorrowPcgEdgeRef<'tcx, 'slf>: EdgeData<'tcx, Ctxt>,
154    {
155        for edge in borrows.edges_blocking(node, ctxt.bc_ctxt()) {
156            self.kill_edge(&edge, borrows, ctxt);
157        }
158    }
