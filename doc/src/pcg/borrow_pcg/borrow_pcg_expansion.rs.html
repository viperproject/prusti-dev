51/// The projections resulting from an expansion of a place.
52///
53/// This representation is preferred to a `Vec<PlaceElem>` because it ensures
54/// it enables a more reasonable notion of equality between expansions. Directly
55/// storing the place elements in a `Vec` could lead to different representations
56/// for the same expansion, e.g. `{*x.f.a, *x.f.b}` and `{*x.f.b, *x.f.a}`.
57#[derive(PartialEq, Eq, Clone, Debug, Hash, From)]
58pub enum PlaceExpansion<'tcx, D = ()> {
59    /// Fields from e.g. a struct or tuple, e.g. `{*x.f} -> {*x.f.a, *x.f.b}`
60    /// Note that for region projections, not every field of the base type may
61    /// be included. For example consider the following:
62    /// ```ignore
63    /// struct S<'a, 'b> { x: &'a mut i32, y: &'b mut i32 }
64    ///
65    /// let s: S<'a, 'b> = S { x: &mut 1, y: &mut 2 };
66    /// ```
67    /// The projection of `s↓'a` contains only `{s.x↓'a}` because nothing under
68    /// `'a` is accessible via `s.y`.
69    Fields(BTreeMap<FieldIdx, (ty::Ty<'tcx>, D)>),
70    /// See [`PlaceElem::Deref`]
71    Deref(D),
72    Guided(RequiredGuide<D>),
73}
74
75impl<Ctxt: DebugCtxt> HasValidityCheck<Ctxt> for PlaceExpansion<'_> {
76    fn check_validity(&self, _ctxt: Ctxt) -> Result<(), String> {
77        Ok(())
78    }
79}

91impl<'tcx> PlaceExpansion<'tcx> {
92    /// Construct a `Fields` variant from the simple map that callers
93    /// conventionally build (without per-child data).
94    pub(crate) fn fields(map: BTreeMap<FieldIdx, ty::Ty<'tcx>>) -> Self {
95        PlaceExpansion::Fields(map.into_iter().map(|(idx, ty)| (idx, (ty, ()))).collect())
96    }
97
98    /// Construct a `Deref` variant with no per-child data.
99    pub(crate) fn deref() -> Self {
100        PlaceExpansion::Deref(())
101    }
102
103    pub(crate) fn block_type<'a>(
104        &self,
105        base_place: Place<'tcx>,
106        obtain_type: ObtainType,
107        ctxt: impl HasCompilerCtxt<'a, 'tcx>,
108    ) -> BlockType
109    where
110        'tcx: 'a,
111    {
112        if matches!(
113            obtain_type,
114            ObtainType::Capability(CapabilityKind::Read)
115                | ObtainType::TwoPhaseExpand
116                | ObtainType::LoopInvariant {
117                    usage_type: PlaceUsageType::Read,
118                    ..
119                }
120        ) {
121            BlockType::Read
122        } else if self.is_deref() {
123            if base_place.is_shared_ref(ctxt) {
124                BlockType::DerefSharedRef
125            } else if base_place.is_mut_ref(ctxt) {
126                if base_place.projects_shared_ref(ctxt) {
127                    BlockType::DerefMutRefUnderSharedRef
128                } else {
129                    BlockType::DerefMutRefForExclusive
130                }
131            } else {
132                BlockType::Other
133            }
134        } else {
135            BlockType::Other
136        }
137    }
138
139    pub(crate) fn guide(&self) -> Option<RepackGuide> {
140        match self {
141            PlaceExpansion::Guided(guide) => Some(RepackGuide::from(*guide)),
142            _ => None,
143        }
144    }
145
146    pub(crate) fn from_places<'a>(
147        places: Vec<Place<'tcx>>,
148        ctxt: impl HasCompilerCtxt<'a, 'tcx>,
149    ) -> Self
150    where
151        'tcx: 'a,
152    {
153        let mut fields = BTreeMap::new();
154
155        for place in places {
156            let corrected_place = CorrectedPlace::new(place, ctxt);
157            let last_projection = corrected_place.last_projection();
158            if let Some(elem) = last_projection {
159                match *elem {
160                    PlaceElem::Field(field_idx, ty) => {
161                        fields.insert(field_idx, (ty, ()));
162                    }
163                    PlaceElem::Deref => return PlaceExpansion::Deref(()),
164                    other => {
165                        let repack_guide: RepackGuide = other
166                            .try_into()
167                            .unwrap_or_else(|()| todo!("unsupported place elem: {:?}", other));
168                        // Default variant is excluded by construction from TryFrom above.
169                        let required = repack_guide
170                            .into_non_default()
171                            .expect("TryFrom<PlaceElem> never yields RepackGuide::Default");
172                        return PlaceExpansion::Guided(required);
173                    }
174                }
175            }
176        }
177
178        if fields.is_empty() {
179            unreachable!()
180        }
181        PlaceExpansion::Fields(fields)
182    }
183
184    pub(crate) fn elems(&self) -> Vec<PlaceElem<'tcx>> {
185        match self {
186            PlaceExpansion::Fields(fields) => fields
187                .iter()
188                .sorted_by_key(|(idx, _)| *idx)
189                .map(|(idx, (ty, ()))| PlaceElem::Field(*idx, *ty))
190                .collect(),
191            PlaceExpansion::Deref(()) => vec![PlaceElem::Deref],
192            PlaceExpansion::Guided(RepackGuide::ConstantIndex(c, ())) => {
193                let mut elems = vec![(*c).into()];
194                elems.extend(c.other_elems());
195                elems
196            }
197            PlaceExpansion::Guided(guided) => vec![(*guided).into()],
198        }
199    }
200}

249impl<'tcx, P: PcgNodeComponent> BorrowPcgExpansion<'tcx, P> {
250    pub fn base(&self) -> LocalNode<'tcx, P> {
251        match self {
252            BorrowPcgExpansion::Place(expansion) => expansion.base.into(),
253            BorrowPcgExpansion::LifetimeProjection(expansion) => {
254                PcgNode::LifetimeProjection(expansion.base)
255            }
256        }
257    }
258
259    pub fn expansion(&self) -> Vec<LocalNode<'tcx, P>> {
260        match self {
261            BorrowPcgExpansion::Place(expansion) => {
262                expansion.expansion.iter().map(|p| (*p).into()).collect()
263            }
264            BorrowPcgExpansion::LifetimeProjection(expansion) => expansion
265                .expansion
266                .iter()
267                .map(|p| PcgNode::LifetimeProjection(*p))
268                .collect(),
269        }
270    }
271}

355impl<'tcx, Ctxt: Copy, P: PcgPlace<'tcx, Ctxt>> LabelEdgeLifetimeProjections<'tcx, Ctxt, P>
356    for LifetimeProjectionExpansion<'tcx, P>
357where
358    LocalLifetimeProjection<'tcx, P>: LabelLifetimeProjection<'tcx>,
359{
360    fn label_lifetime_projections(
361        &mut self,
362        predicate: &LabelNodePredicate<'tcx, P>,
363        label: Option<LifetimeProjectionLabel>,
364        ctxt: Ctxt,
365    ) -> LabelLifetimeProjectionResult {
366        let source_context = LabelNodeContext::new(
367            SourceOrTarget::Source,
368            BorrowPcgEdgeType::BorrowPcgExpansion,
369        );
370        let target_context = LabelNodeContext::new(
371            SourceOrTarget::Target,
372            BorrowPcgEdgeType::BorrowPcgExpansion,
373        );
374        let mut changed = LabelLifetimeProjectionResult::Unchanged;
375        if predicate.applies_to(self.base.to_pcg_node(ctxt), source_context) {
376            changed |= self.base.label_lifetime_projection(label);
377        }
378        for p in &mut self.expansion {
379            if predicate.applies_to(p.to_pcg_node(ctxt), target_context) {
380                changed |= p.label_lifetime_projection(label);
381            }
382        }
383        changed
384    }
385}

413impl<'a, 'tcx: 'a, Ctxt: DebugCtxt + HasCompilerCtxt<'a, 'tcx>> HasValidityCheck<Ctxt>
414    for BorrowPcgExpansionData<MaybeLabelledPlace<'tcx>>
415{
416    fn check_validity(&self, ctxt: Ctxt) -> Result<(), String> {
417        if self.expansion.contains(&self.base) {
418            return Err(format!("expansion contains base: {self:?}"));
419        }
420        for p in &self.expansion {
421            if let Some(PcgNode::Place(node)) = p.try_to_local_node(ctxt)
422                && node.place().is_owned(ctxt)
423            {
424                return Err(format!(
425                    "Expansion of {:?} contains owned place {}",
426                    self,
427                    node.place().display_string(ctxt)
428                ));
429            }
430        }
431        Ok(())
432    }
433}

490impl<'tcx> BorrowPcgExpansion<'tcx> {
491    #[must_use]
492    pub fn guide(&self) -> Option<RepackGuide> {
493        match self {
494            BorrowPcgExpansion::Place(place_expansion) => place_expansion.guide,
495            BorrowPcgExpansion::LifetimeProjection(lifetime_projection_expansion) => {
496                lifetime_projection_expansion.guide
497            }
498        }
499    }
500
501    /// Returns true iff the expansion is packable, i.e. without losing any
502    /// information. This is the case when the expansion node labels (for
503    /// places, and for region projections) are the same as the base node
504    /// labels, and the expansion nodes all hold the same capability.
505    ///
506    /// A [`CapabilityKind::Write`] expansion node is never packable: packing
507    /// e.g. `x.f: W, x.g: W` into `x: W` would lose the information that the
508    /// fields are individually accessible.
509    pub(crate) fn is_packable(
510        &self,
511        capabilities: &impl PlaceCapabilitiesReader<'tcx>,
512        ctxt: impl HasCompilerCtxt<'_, 'tcx>,
513    ) -> bool {
514        let BorrowPcgExpansion::Place(place_expansion) = self else {
515            return false;
516        };
517        let mut fst_cap = None;
518        place_expansion.expansion.iter().all(|p| {
519            if let MaybeLabelledPlace::Current(place) = p {
520                let cap = capabilities.get(*place, ctxt);
521                if cap == Some(CapabilityKind::Write) {
522                    return false;
523                }
524                if let Some(fst_cap) = fst_cap {
525                    if fst_cap != cap {
526                        return false;
527                    }
528                } else {
529                    fst_cap = Some(cap);
530                }
531            }
532            place_expansion.base.place().is_prefix_exact(p.place())
533                && p.location() == place_expansion.base.location()
534        })
535    }
536}

538impl<'tcx, Node: PcgNodeComponent + 'tcx> BorrowPcgExpansionData<Node> {
539    pub fn base(&self) -> Node {
540        self.base
541    }
542
543    pub fn expansion(&self) -> &[Node] {
544        &self.expansion
545    }
546
547    pub(crate) fn new<
548        'a,
549        Ctxt: HasSettings<'a> + DebugCtxt,
550        P: PlaceLike<'tcx, Ctxt> + DisplayWithCtxt<Ctxt>,
551    >(
552        base: Node,
553        expansion: &PlaceExpansion<'tcx>,
554        ctxt: Ctxt,
555    ) -> Result<Self, PcgError>
556    where
557        Node: Ord + HasPlace<'tcx, P> + PlaceProjectable<'tcx, Ctxt>,
558        Self: HasValidityCheck<Ctxt>,
559    {
560        pcg_validity_assert!(
561            !(base.is_place() && base.place().is_ref(ctxt) && expansion.is_deref()),
562            [ctxt],
563            "Deref expansion of {} should be a Deref edge, not an expansion",
564            base.place().display_string(ctxt)
565        );
566        let result = Self {
567            base,
568            guide: expansion.guide(),
569            expansion: expansion
570                .elems()
571                .into_iter()
572                .map(|elem| base.project_deeper(elem, ctxt))
573                .collect::<Result<Vec<_>, _>>()?,
574        };
575        result.assert_validity(ctxt);
576        Ok(result)
577    }
