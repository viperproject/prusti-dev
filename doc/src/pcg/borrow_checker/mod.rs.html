54    BorrowCheckerInterface<'tcx> for T
55{
56    fn is_dead(&self, node: PcgNode<'tcx>, location: Location) -> bool {
57        !self.is_live(node, location)
58    }

60    fn twophase_borrow_activations(&self, location: Location) -> BTreeSet<Location> {
61        #[rustversion::since(2026-07-28)]
62        let borrow_idxs: &[BorrowIndex] = self.borrow_set().activations_at_location(&location);
63        #[rustversion::before(2026-07-28)]
64        let borrow_idxs: &[BorrowIndex] = self
65            .borrow_set()
66            .activation_map()
67            .get(&location)
68            .map_or(&[], |idxs| &idxs[..]);
69        borrow_idxs
70            .iter()
71            .map(|idx| get_reserve_location(&self.borrow_set()[*idx]))
72            .collect()
73    }

75    fn borrows_blocking(
76        &self,
77        blocked_place: Place<'tcx>,
78        location: Location,
79        ctxt: CompilerCtxt<'_, 'tcx>,
80    ) -> Vec<BorrowData<'tcx>> {
81        let mut borrows = vec![];
82        each_borrow_involving_path(
83            &mut (),
84            ctxt,
85            blocked_place,
86            self.borrow_set(),
87            |borrow_index| self.borrow_in_scope_at(borrow_index, location),
88            |_this, _, borrow| {
89                borrows.push(borrow.clone());
90                ControlFlow::Continue(())
91            },
92        );
93        borrows
94    }
95
96    fn region_to_borrow_index(&self, region: PcgRegion<'tcx>) -> Option<BorrowIndex> {
97        RustBorrowCheckerInterface::region_to_borrow_index(self, region)
98    }

100    fn is_directly_blocked(
101        &self,
102        blocked_place: Place<'tcx>,
103        location: Location,
104        ctxt: CompilerCtxt<'_, 'tcx>,
105    ) -> bool {
106        let mut conflict = false;
107        each_borrow_involving_path(
108            &mut (),
109            ctxt,
110            blocked_place,
111            self.borrow_set(),
112            |borrow_index| self.borrow_in_scope_at(borrow_index, location),
113            |_this, _borrow_index, borrow| {
114                if borrow.get_borrowed_place() == blocked_place {
115                    conflict = true;
116                    ControlFlow::Break(())
117                } else {
118                    ControlFlow::Continue(())
119                }
120            },
121        );
122        conflict
123    }
124
125    fn blocks(
126        &self,
127        blocking_place: Place<'tcx>,
128        blocked_place: Place<'tcx>,
129        location: Location,
130        ctxt: CompilerCtxt<'_, 'tcx>,
131    ) -> bool {
132        let mut conflict = false;
133        each_borrow_involving_path(
134            &mut (),
135            ctxt,
136            blocked_place,
137            self.borrow_set(),
138            |borrow_index| self.borrow_in_scope_at(borrow_index, location),
139            |_this, borrow_index, _borrow| {
140                tracing::debug!(
141                    "Checking if {} contains {:?} at {:?}",
142                    blocking_place.display_string(ctxt),
143                    borrow_index,
144                    location
145                );
146                if blocking_place
147                    .regions(ctxt)
148                    .iter()
149                    .any(|region| self.origin_contains_loan_at(*region, borrow_index, location))
150                {
151                    conflict = true;
152                    ControlFlow::Break(())
153                } else {
154                    ControlFlow::Continue(())
155                }
156            },
157        );
158        conflict
159    }

182pub trait BorrowCheckerInterface<'tcx>: DisplayCtxtFor<RegionVid> {
183    /* Main Interface Start */
184
185    /// Answers the question: Does `node` contain borrow extents that are not
186    /// in scope at `location`?
187    fn is_dead(&self, node: PcgNode<'tcx>, location: Location) -> bool;
188
189    fn is_directly_blocked(
190        &self,
191        blocked_place: Place<'tcx>,
192        location: Location,
193        ctxt: CompilerCtxt<'_, 'tcx>,
194    ) -> bool;
195
196    fn blocks(
197        &self,
198        blocking_place: Place<'tcx>,
199        blocked_place: Place<'tcx>,
200        location: Location,
201        ctxt: CompilerCtxt<'_, 'tcx>,
202    ) -> bool;
203
204    /// Returns true iff `sup` is required to outlive `sub` at `location`.
205    fn outlives(&self, sup: PcgRegion<'tcx>, sub: PcgRegion<'tcx>, location: Location) -> bool;
206
207    fn borrows_blocking(
208        &self,
209        blocked_place: Place<'tcx>,
210        location: Location,
211        ctxt: CompilerCtxt<'_, 'tcx>,
212    ) -> Vec<BorrowData<'tcx>>;
213
214    /// Returns the set of two-phase borrows that activate at `location`.
215    /// Each borrow in the returned set is represented by the MIR location
216    /// that it was created at.
217    fn twophase_borrow_activations(&self, location: Location) -> BTreeSet<Location>;
218
219    /* Main Interface End */
220
221    /// Returns true iff `reg1` outlives `reg2` and `reg2` outlives `reg1`.
222    fn same_region(
223        &self,
224        reg1: PcgRegion<'tcx>,
225        reg2: PcgRegion<'tcx>,
226        location: Location,
227    ) -> bool {
228        self.outlives(reg1, reg2, location) && self.outlives(reg2, reg1, location)
229    }
230
231    // Implementors of this trait should be dyn-safe
232    fn as_dyn(&self) -> &dyn BorrowCheckerInterface<'tcx>;
233
234    /*
235     * The remaining methods are only used for debugging / visualization.
236     */
237
238    /// If this is a Rust borrow checker, return its interface.
239    /// TODO: Get rid of this at some point
240    fn rust_borrow_checker(&self) -> Option<&dyn RustBorrowCheckerInterface<'tcx>>;
241
242    // DEBUG ONLY
243    /// If the borrow checker is based on Polonius, it can define this method to
244    /// expose its output facts. This is only used for debugging /
245    /// visualization.
246    /// TODO: Remove
247    fn polonius_output(&self) -> Option<&PoloniusOutput>;
248
249    /// Currently only used for associating borrows with their indexes for
250    /// visualization purposes.
251    /// TODO: Remove
252    fn region_to_borrow_index(&self, region: PcgRegion<'tcx>) -> Option<BorrowIndex>;
253
254    // TODO: Remove, only for visualization
255    fn input_facts(&self) -> &PoloniusInput;
256
257    /// Returns an iterator over all `RegionVid`s for this body.
258    ///
259    /// The implementation is a hack: it extracts region vids from the
260    /// constraint SCC indices in the region inference context. This is
261    /// currently used in
262    /// [`DefinedFnCallWithCallTys::normalize_sig`](crate::borrow_pcg::edge::abstraction::function::DefinedFnCallWithCallTys)
263    /// to pre-populate an inference context with the correct number of region
264    /// variables before performing type normalization. This method should be
265    /// removed if we find a better way to perform that normalization.
266    fn iter_region_vids<'slf>(&'slf self) -> Box<dyn Iterator<Item = RegionVid> + 'slf>
267    where
268        'tcx: 'slf,
269    {
270        Box::new(
271            self.rust_borrow_checker()
272                .unwrap()
273                .region_infer_ctxt()
274                .constraint_sccs()
275                .scc_indices()
276                .iter_enumerated()
277                .map(move |(region_vid, _)| region_vid),
278        )
279    }
280}

298/// An interface to the results of the borrow-checker analysis. The PCG queries
299/// this interface as part of its analysis, for example, to identify when borrows
300/// expire.
301pub trait RustBorrowCheckerInterface<'tcx> {
302    /// Returns true iff the node is live *before* `location`. A node is live
303    /// iff:
304    /// - it is a place node, and the place is live
305    /// - it is a lifetime projection node, and the lifetime is live
306    ///
307    /// For lifetimes, we use the notion of liveness defined here in
308    /// <https://rust-lang.github.io/rfcs/2094-nll.html#liveness>:
309    ///
310    /// > A lifetime `L` is live at a point `P` if there is some variable `p` which is
311    /// > live at `P`, and `L` appears in the type of `p`
312    ///
313    /// However, as referenced in the above link, there are some subtleties
314    /// related to places that will be dropped. Follow the link for more details.
315    ///
316    /// Note that the return place and its lifetime projections will always be
317    /// live.
318    fn is_live(&self, node: PcgNode<'tcx>, location: Location) -> bool;
319
320    fn borrow_set(&self) -> &BorrowSet<'tcx>;
321
322    /// Returns the set of borrows that are live either before or after
323    /// `location`.
324    fn borrows_in_scope_at(&self, location: Location, before: bool) -> InScopeBorrows;
325
326    fn borrow_in_scope_at(&self, borrow_index: BorrowIndex, location: Location) -> bool {
327        self.borrows_in_scope_at(location, true)
328            .contains(borrow_index)
329    }
330
331    fn origin_contains_loan_at(
332        &self,
333        region: PcgRegion<'tcx>,
334        loan: BorrowIndex,
335        location: Location,
336    ) -> bool;
337
338    fn region_to_borrow_index(&self, region: PcgRegion<'tcx>) -> Option<BorrowIndex> {
339        #[rustversion::since(2026-07-28)]
340        {
341            self.borrow_set()
342                .iter_enumerated()
343                .find_map(|(index, data)| (data.pcg_region() == region).then_some(index))
344        }
345        #[rustversion::before(2026-07-28)]
346        {
347            self.borrow_set()
348                .location_map()
349                .iter()
350                .enumerate()
351                .find_map(move |(index, (_, data))| {
352                    (data.pcg_region() == region).then_some(index.into())
353                })
354        }
355    }
356
357    fn borrow_index_to_region(&self, borrow_index: BorrowIndex) -> RegionVid {
358        self.borrow_set()[borrow_index].region()
359    }
360
361    /// If the borrow checker is based on Polonius, it can define this method to
362    /// expose its output facts. This is only used for debugging /
363    /// visualization.
364    fn polonius_output(&self) -> Option<&PoloniusOutput>;
365
366    fn borrows_blocking(
367        &self,
368        blocked_place: Place<'tcx>,
369        location: Location,
370        ctxt: CompilerCtxt<'_, 'tcx>,
371    ) -> Vec<BorrowData<'tcx>> {
372        let mut borrows = vec![];
373        each_borrow_involving_path(
374            &mut (),
375            ctxt,
376            blocked_place,
377            self.borrow_set(),
378            |borrow_index| self.borrow_in_scope_at(borrow_index, location),
379            |_this, _, borrow| {
380                borrows.push(borrow.clone());
381                ControlFlow::Continue(())
382            },
383        );
384        borrows
385    }
386
387    fn loans_killed_at(&self, location: Location) -> BTreeSet<RegionVid> {
388        let location_indices = [
389            self.location_table().start_index(location),
390            self.location_table().mid_index(location),
391        ];
392        self.input_facts()
393            .loan_killed_at
394            .iter()
395            .filter(|(_, point)| location_indices.contains(point))
396            .map(|(loan, _)| self.borrow_index_to_region(*loan))
397            .collect()
398    }
399
400    fn input_facts(&self) -> &PoloniusInput;
401
402    fn region_infer_ctxt(&self) -> &RegionInferenceContext<'tcx>;
403
404    fn location_table(&self) -> &LocationTable;
405
406    /// Returns true iff `sup` is required to outlive `sub` everywhere.
407    /// This can be useful e.g. for determining outlives relations of
408    /// arguments to a function based on its signature.
409    fn outlives_everywhere(&self, sup: PcgRegion<'tcx>, sub: PcgRegion<'tcx>) -> bool {
410        match (sup, sub) {
411            (PcgRegion::RegionVid(sup), PcgRegion::RegionVid(sub)) => {
412                self.region_infer_ctxt().eval_outlives(sup, sub)
413            }
414            (PcgRegion::ReStatic, _) => true,
415            _ => false,
416        }
417    }
418}

437pub(super) fn each_borrow_involving_path<'tcx, F, I, S>(
438    s: &mut S,
439    ctxt: CompilerCtxt<'_, 'tcx>,
440    borrowed_place: Place<'tcx>,
441    borrow_set: &BorrowSet<'tcx>,
442    is_candidate: I,
443    mut op: F,
444) where
445    F: FnMut(&mut S, BorrowIndex, &BorrowData<'tcx>) -> ControlFlow<()>,
446    I: Fn(BorrowIndex) -> bool,
447{
448    // The number of candidates can be large, but borrows for different locals cannot conflict with
449    // each other, so we restrict the working set a priori.
450    let Some(borrows_for_place_base) = borrow_set.get_borrows_for_local(borrowed_place.local)
451    else {
452        return;
453    };
454
455    // check for loan restricting path P being used. Accounts for
456    // borrows of P, P.a.b, etc.
457    for &i in borrows_for_place_base {
458        if !is_candidate(i) {
459            continue;
460        }
461        let borrowed = &borrow_set[i];
462
463        if places_conflict(
464            ctxt.tcx(),
465            ctxt.body(),
466            borrowed.get_borrowed_place().to_rust_place(ctxt),
467            borrowed_place.to_rust_place(ctxt),
468            PlaceConflictBias::Overlap,
469        ) {
470            let ctrl = op(s, i, borrowed);
471            if matches!(ctrl, ControlFlow::Break(())) {
472                return;
473            }
474        }
475    }
