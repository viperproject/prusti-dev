70#[derive(Clone, Debug, PartialEq, Eq, Hash)]
71pub struct GraphNode {
72    id: NodeId,
73    node_type: NodeType,
74}
75
76impl GraphNode {
77    pub(crate) fn label_text(&self) -> Cow<'_, str> {
78        match &self.node_type {
79            NodeType::PlaceNode { label, .. } => label.into(),
80            NodeType::RegionProjectionNode { label, .. } => label.text(),
81        }
82    }
83
84    fn to_dot_node(&self) -> DotNode {
85        match &self.node_type {
86            NodeType::PlaceNode {
87                owned,
88                capability,
89                location,
90                label,
91                ty,
92            } => {
93                let location_html: Html = match location {
94                    Some(l) => Html::Seq(vec![
95                        Html::space(),
96                        l.display_output((), OutputMode::Short).into_html(),
97                    ]),
98                    None => Html::empty(),
99                };
100                let color = if location.is_some()
101                    || matches!(capability, None | Some(CapabilityKind::Write))
102                {
103                    "gray"
104                } else if *owned {
105                    "black"
106                } else {
107                    "darkgreen"
108                };
109                let (style, penwidth) = if *owned {
110                    (None, None)
111                } else {
112                    (
113                        Some(DotStringAttr("rounded".into())),
114                        Some(DotFloatAttr(1.5)),
115                    )
116                };
117                let capability_text = match capability {
118                    Some(k) => format!(": {k:?}"),
119                    None => String::new(),
120                };
121                let label_html = Html::Font(
122                    "courier",
123                    Box::new(Html::Seq(vec![
124                        Html::Text(format!("{label}{capability_text}").into()),
125                        location_html,
126                    ])),
127                );
128                DotNode {
129                    id: self.id.to_string(),
130                    label: DotLabel::Html(label_html),
131                    color: DotStringAttr(color.into()),
132                    font_color: DotStringAttr(color.into()),
133                    shape: DotStringAttr("rect".into()),
134                    style,
135                    penwidth,
136                    tooltip: Some(DotStringAttr(ty.clone().into())),
137                }
138            }
139            NodeType::RegionProjectionNode {
140                label,
141                loans,
142                base_ty: place_ty,
143            } => DotNode {
144                id: self.id.to_string(),
145                label: DotLabel::Html(label.clone()),
146                color: DotStringAttr("blue".into()),
147                font_color: DotStringAttr("blue".into()),
148                shape: DotStringAttr("octagon".into()),
149                style: None,
150                penwidth: None,
151                tooltip: Some(DotStringAttr(format!("{place_ty}\\\n{loans}").into())),
152            },
153        }
154    }
155}

173#[derive(Clone, Debug, PartialEq, Eq, Hash)]
174pub(crate) enum GraphEdge<'a> {
175    Abstract {
176        blocked: NodeId,
177        blocking: NodeId,
178        label: String,
179    },
180    // For literal borrows
181    Alias {
182        blocked_place: NodeId,
183        blocking_place: NodeId,
184    },
185    Borrow {
186        borrowed_place: NodeId,
187        assigned_region_projection: NodeId,
188        kind: String,
189        location: Location,
190        region: String,
191        validity_conditions: &'a ValidityConditions,
192        borrow_index: Option<String>,
193    },
194    Projection {
195        source: NodeId,
196        target: NodeId,
197    },
198    DerefExpansion {
199        source: NodeId,
200        target: NodeId,
201        validity_conditions: &'a ValidityConditions,
202    },
203    BorrowFlow {
204        source: NodeId,
205        target: NodeId,
206        kind: BorrowFlowEdgeKind<'a, String>,
207    },
208    Coupled {
209        source: NodeId,
210        target: NodeId,
211    },
212    Delegation {
213        source: NodeId,
214        target: NodeId,
215    },
216    RawPtrDeref {
217        source: NodeId,
218        target: NodeId,
219    },
220}
221
222impl<'a> GraphEdge<'a> {
223    pub(crate) fn validity_conditions(&self) -> Option<&'a ValidityConditions> {
224        match self {
225            GraphEdge::Borrow {
226                validity_conditions,
227                ..
228            }
229            | GraphEdge::DerefExpansion {
230                validity_conditions,
231                ..
232            } => Some(validity_conditions),
233            GraphEdge::Projection { .. }
234            | GraphEdge::Alias { .. }
235            | GraphEdge::Abstract { .. }
236            | GraphEdge::BorrowFlow { .. }
237            | GraphEdge::Coupled { .. }
238            | GraphEdge::Delegation { .. }
239            | GraphEdge::RawPtrDeref { .. } => None,
240        }
241    }
242    pub(super) fn to_dot_edge<'tcx: 'a>(
243        &self,
244        edge_id: Option<DotEdgeId>,
245        ctxt: impl HasCompilerCtxt<'a, 'tcx>,
246    ) -> DotEdge {
247        match self {
248            GraphEdge::Projection { source, target } => DotEdge {
249                id: edge_id,
250                from: source.to_string(),
251                to: target.to_string(),
252                options: EdgeOptions::directed(EdgeDirection::Forward),
253            },
254            GraphEdge::Alias {
255                blocked_place,
256                blocking_place,
257            } => DotEdge {
258                id: edge_id,
259                from: blocked_place.to_string(),
260                to: blocking_place.to_string(),
261                options: EdgeOptions::directed(EdgeDirection::Forward)
262                    .with_color("grey".into())
263                    .with_style("dashed"),
264            },
265            GraphEdge::Borrow {
266                borrowed_place,
267                assigned_region_projection: assigned_place,
268                location: _,
269                region,
270                kind,
271                validity_conditions,
272                borrow_index,
273            } => DotEdge {
274                id: edge_id,
275                to: assigned_place.to_string(),
276                from: borrowed_place.to_string(),
277                options: EdgeOptions::directed(EdgeDirection::Forward)
278                    .with_color("orange".into())
279                    .with_label(format!(
280                        "{}{} {}",
281                        if let Some(borrow_index) = borrow_index {
282                            format!("{borrow_index}: ")
283                        } else {
284                            String::new()
285                        },
286                        kind,
287                        region
288                    ))
289                    .with_tooltip(
290                        validity_conditions
291                            .display_output(ctxt, OutputMode::Short)
292                            .into_text(),
293                    ),
294            },
295            GraphEdge::DerefExpansion {
296                source,
297                target,
298                validity_conditions,
299            } => DotEdge {
300                id: edge_id,
301                from: source.to_string(),
302                to: target.to_string(),
303                options: EdgeOptions::directed(EdgeDirection::Forward)
304                    .with_color("green".into())
305                    .with_tooltip(
306                        validity_conditions
307                            .display_output(ctxt, OutputMode::Short)
308                            .into_text(),
309                    ),
310            },
311            GraphEdge::Abstract {
312                blocked,
313                blocking,
314                label,
315            } => DotEdge {
316                id: edge_id,
317                from: blocked.to_string(),
318                to: blocking.to_string(),
319                options: EdgeOptions::directed(EdgeDirection::Forward)
320                    .with_label(label.clone())
321                    .with_penwidth(3.0),
322            },
323            GraphEdge::BorrowFlow {
324                source,
325                target,
326                kind,
327            } => {
328                let options = EdgeOptions::directed(EdgeDirection::Forward)
329                    .with_label(kind.to_string())
330                    .with_color("purple".into());
331                let options = match kind {
332                    BorrowFlowEdgeKind::BorrowOutlives { regions_equal } => {
333                        if *regions_equal {
334                            options.with_penwidth(2.0)
335                        } else {
336                            options.with_style("dashed")
337                        }
338                    }
339                    _ => options,
340                };
341                DotEdge {
342                    id: edge_id,
343                    from: source.to_string(),
344                    to: target.to_string(),
345                    options,
346                }
347            }
348            GraphEdge::Coupled { source, target } => DotEdge {
349                id: edge_id,
350                from: source.to_string(),
351                to: target.to_string(),
352                options: EdgeOptions::directed(EdgeDirection::Forward)
353                    .with_color("red".into())
354                    .with_style("dashed"),
355            },
356            GraphEdge::Delegation { source, target } => DotEdge {
357                id: edge_id,
358                from: source.to_string(),
359                to: target.to_string(),
360                options: EdgeOptions::directed(EdgeDirection::Forward)
361                    .with_color("blue".into())
362                    .with_style("dashed"),
363            },
364            GraphEdge::RawPtrDeref { source, target } => DotEdge {
365                id: edge_id,
366                from: source.to_string(),
367                to: target.to_string(),
368                options: EdgeOptions::directed(EdgeDirection::Forward)
369                    .with_color("green".into())
370                    .with_style("dashed"),
371            },
372        }
373    }
374}

376pub struct Graph<'a> {
377    nodes: Vec<GraphNode>,
378    edges: HashSet<GraphEdge<'a>>,
379}
380
381impl<'a> Graph<'a> {
382    fn new(nodes: Vec<GraphNode>, edges: HashSet<GraphEdge<'a>>) -> Self {
383        Self { nodes, edges }
384    }
385
386    pub fn has_edge_between_labelled_nodes<'tcx: 'a>(
387        &self,
388        label1: &str,
389        label2: &str,
390        ctxt: impl HasCompilerCtxt<'a, 'tcx>,
391    ) -> bool {
392        let Some(label_1_id) = self
393            .nodes
394            .iter()
395            .find(|n| n.label_text() == label1)
396            .map(|n| n.id)
397        else {
398            return false;
399        };
400        let Some(label_2_id) = self
401            .nodes
402            .iter()
403            .find(|n| n.label_text() == label2)
404            .map(|n| n.id)
405        else {
406            return false;
407        };
408        self.edges.iter().any(|edge| {
409            let dot_edge = edge.to_dot_edge(None, ctxt);
410            dot_edge.from == label_1_id.to_string() && dot_edge.to == label_2_id.to_string()
411        })
412    }
413}

415pub(crate) fn generate_borrows_dot_graph<'a, 'tcx: 'a>(
416    ctxt: CompilerCtxt<'a, 'tcx>,
417    capabilities: &impl PlaceCapabilitiesReader<'tcx>,
418    borrows_domain: &'a BorrowsGraph<'tcx>,
419) -> io::Result<String> {
420    let constructor = BorrowsGraphConstructor::new(borrows_domain, capabilities, ctxt.bc_ctxt());
421    let graph = constructor.construct_graph();
422    let mut buf = vec![];
423    let drawer = GraphDrawer::new(&mut buf, None);
424    drawer.draw(&graph, ctxt)?;
425    Ok(String::from_utf8(buf).unwrap())
426}

428pub(crate) fn generate_unblock_dot_graph<'a, 'tcx: 'a>(
429    ctxt: CompilerCtxt<'a, 'tcx>,
430    capabilities: &impl PlaceCapabilitiesReader<'tcx>,
431    unblock_graph: &UnblockGraph<'tcx>,
432) -> io::Result<String> {
433    let mut borrows_graph = BorrowsGraph::default();
434    for edge in unblock_graph.edges() {
435        borrows_graph.insert(edge.clone(), ctxt.bc_ctxt());
436    }
437    let constructor = BorrowsGraphConstructor::new(&borrows_graph, capabilities, ctxt.bc_ctxt());
438    let graph = constructor.construct_graph();
439    let mut buf = vec![];
440    let drawer = GraphDrawer::new(&mut buf, None);
441    drawer.draw(&graph, ctxt)?;
442    Ok(String::from_utf8(buf).unwrap())
443}

458pub(crate) fn write_pcg_dot_graph_to_file<'a, 'tcx: 'a>(
459    pcg: PcgRef<'a, 'tcx>,
460    ctxt: impl HasBorrowCheckerCtxt<'a, 'tcx>,
461    location: Location,
462    file_path: &Path,
463) -> io::Result<()> {
464    let constructor = PcgGraphConstructor::new(pcg, ctxt.bc_ctxt(), location);
465    let graph = constructor.construct_graph();
466    let dot_file = File::create(file_path).unwrap();
467    let ctxt_file = File::create(file_path.with_extension("json")).unwrap();
468    let drawer = GraphDrawer::new(dot_file, Some(ctxt_file));
469    drawer.draw(&graph, ctxt)
470}
