13impl<T> CompilerCtxt<'_, '_, T> {
14    pub(crate) fn function_metadata_slug(&self) -> FunctionSlug {
15        FunctionSlug::new(self.def_id(), self.tcx)
16    }
17
18    pub(crate) fn function_metadata(&self) -> FunctionMetadata {
19        let start = SourcePos::new(self.mir.span.lo(), self.tcx);
20        let def_id = self.def_id();
21        let fn_sig = skip_normalization(self.tcx.fn_sig(def_id).instantiate_identity());
22        let fn_sig = self.tcx.liberate_late_bound_regions(def_id.into(), fn_sig);
23        let signature = fn_sig.to_string();
24        let debug_signature = format!("{fn_sig:#?}");
25        FunctionMetadata::new(
26            self.body_def_path_str(),
27            signature,
28            debug_signature,
29            self.source().unwrap(),
30            start,
31        )
32    }
33}

35impl PcgCtxt<'_, '_> {
36    pub fn update_debug_visualization_metadata(&self) {
37        if let Some(metadata) = self.visualization_function_metadata() {
38            self.settings.write_new_debug_visualization_metadata(
39                self.compiler_ctxt.function_metadata_slug(),
40                &metadata,
41            );
42        }
43    }
44
45    pub(crate) fn visualization_function_metadata(&self) -> Option<FunctionMetadata> {
46        self.settings
47            .visualization
48            .then(|| self.compiler_ctxt.function_metadata())
49    }
50
51    pub fn visualization_output_path(&self) -> Option<PathBuf> {
52        self.settings.visualization.then(|| {
53            self.settings
54                .visualization_data_dir
55                .join(self.compiler_ctxt.function_metadata_slug().path())
56        })
57    }
58}
