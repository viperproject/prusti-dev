108fn format_bin_op(op: BinOp) -> &'static str {
109    match op {
110        BinOp::Add | BinOp::AddWithOverflow => "+",
111        BinOp::Sub | BinOp::SubWithOverflow => "-",
112        BinOp::Mul | BinOp::MulWithOverflow => "*",
113        BinOp::Div => "/",
114        BinOp::Rem => "%",
115        BinOp::AddUnchecked => todo!(),
116        BinOp::SubUnchecked => todo!(),
117        BinOp::MulUnchecked => todo!(),
118        BinOp::BitXor => "^",
119        BinOp::BitAnd => "&",
120        BinOp::BitOr => "|",
121        BinOp::Shl | BinOp::ShlUnchecked => "<<",
122        BinOp::Shr | BinOp::ShrUnchecked => ">>",
123        BinOp::Eq => "==",
124        BinOp::Lt => "<",
125        BinOp::Le => "<=",
126        BinOp::Ne => "!=",
127        BinOp::Ge => ">=",
128        BinOp::Gt => ">",
129        BinOp::Offset => todo!(),
130        BinOp::Cmp => todo!(),
131    }
132}

181#[allow(unreachable_patterns)]
182fn format_rvalue<'tcx>(rvalue: &Rvalue<'tcx>, ctxt: CompilerCtxt<'_, 'tcx>) -> String {
183    #[rustversion::before(2026-02-18)]
184    if let Rvalue::ShallowInitBox(operand, _) = rvalue {
185        return format!("Box({})", format_operand(operand, ctxt));
186    }
187    match rvalue {
188        Rvalue::Use(operand, ..) => format_operand(operand, ctxt),
189        Rvalue::Repeat(operand, c) => format!("repeat {} {c}", format_operand(operand, ctxt)),
190        Rvalue::Ref(_region, kind, place) => {
191            let kind = match kind {
192                mir::BorrowKind::Shared => "",
193                mir::BorrowKind::Mut { .. } => "mut",
194                mir::BorrowKind::Fake(_) => "fake",
195            };
196            format!("&{kind} {}", format_place(place, ctxt))
197        }
198        Rvalue::RawPtr(kind, place) => format_raw_ptr(*kind, place, ctxt),
199        Rvalue::ThreadLocalRef(_) => todo!(),
200        Rvalue::Cast(_, operand, ty) => format!("{} as {ty}", format_operand(operand, ctxt)),
201        Rvalue::BinaryOp(op, box (lhs, rhs)) => {
202            format!(
203                "{} {} {}",
204                format_operand(lhs, ctxt),
205                format_bin_op(*op),
206                format_operand(rhs, ctxt)
207            )
208        }
209        Rvalue::UnaryOp(op, val) => {
210            format!("{op:?} {}", format_operand(val, ctxt))
211        }
212        Rvalue::Discriminant(place) => format!("Discriminant({})", format_place(place, ctxt)),
213        Rvalue::Aggregate(kind, ops) => {
214            format!(
215                "Aggregate {:?} {}",
216                kind,
217                ops.iter().map(|op| format_operand(op, ctxt)).join(", ")
218            )
219        }
220        Rvalue::CopyForDeref(place) => format!("CopyForDeref({})", format_place(place, ctxt)),
221        _ => todo!(),
222    }
223}
224fn format_terminator<'tcx>(
225    terminator: &TerminatorKind<'tcx>,
226    ctxt: CompilerCtxt<'_, 'tcx>,
227) -> String {
228    match terminator {
229        TerminatorKind::Drop {
230            place,
231            target,
232            unwind,
233            ..
234        } => {
235            format!(
236                "drop({}) -> [return: {target:?}, unwind: {unwind:?}]",
237                format_place(place, ctxt)
238            )
239        }
240        TerminatorKind::Call {
241            func,
242            args,
243            destination,
244            target: _,
245            unwind: _,
246            call_source: _,
247            fn_span: _,
248        } => {
249            format!(
250                "{} = {}({})",
251                format_place(destination, ctxt),
252                format_operand(func, ctxt),
253                args.iter()
254                    .map(|arg| format_operand(&arg.node, ctxt))
255                    .join(", ")
256            )
257        }
258        _ => format!("{terminator:?}"),
259    }
260}

262fn format_stmt<'tcx>(stmt: &Statement<'tcx>, ctxt: CompilerCtxt<'_, 'tcx>) -> String {
263    match &stmt.kind {
264        mir::StatementKind::Assign(box (place, rvalue)) => {
265            format!(
266                "{} = {}",
267                format_place(place, ctxt),
268                format_rvalue(rvalue, ctxt)
269            )
270        }
271        mir::StatementKind::FakeRead(box (_, place)) => {
272            format!("FakeRead({})", format_place(place, ctxt))
273        }
274        mir::StatementKind::SetDiscriminant {
275            place,
276            variant_index,
277        } => format!(
278            "SetDiscriminant({} {:?})",
279            format_place(place, ctxt),
280            variant_index
281        ),
282        mir::StatementKind::StorageLive(local) => {
283            format!("StorageLive({})", format_local(*local, ctxt))
284        }
285        mir::StatementKind::StorageDead(local) => {
286            format!("StorageDead({})", format_local(*local, ctxt))
287        }
288        mir::StatementKind::PlaceMention(place) => {
289            format!("PlaceMention({})", format_place(place, ctxt))
290        }
291        mir::StatementKind::AscribeUserType(_, _) => "AscribeUserType(...)".to_owned(),
292        mir::StatementKind::Coverage(_) => "coverage".to_owned(),
293        mir::StatementKind::Intrinsic(non_diverging_intrinsic) => {
294            format!("Intrinsic({non_diverging_intrinsic:?})")
295        }
296        mir::StatementKind::ConstEvalCounter => todo!(),
297        mir::StatementKind::Nop => todo!(),
298        _ => todo!(),
299    }
300}

302fn mk_mir_stmt(
303    stmt: String,
304    debug_stmt: String,
305    span: Span,
306    location: mir::Location,
307    ctxt: CompilerCtxt<'_, '_>,
308) -> MirStmt {
309    let bc = ctxt.borrow_checker.rust_borrow_checker().unwrap();
310    let location_table = ctxt
311        .borrow_checker
312        .rust_borrow_checker()
313        .unwrap()
314        .location_table();
315    let invalidated_at = &bc.input_facts().loan_invalidated_at;
316    let loans_invalidated_start = invalidated_at
317        .iter()
318        .filter_map(|(point, idx)| {
319            if *point == location_table.start_index(location) {
320                let borrow_region = bc.borrow_index_to_region(*idx);
321                Some(format!("{borrow_region:?}"))
322            } else {
323                None
324            }
325        })
326        .collect::<Vec<_>>();
327    let loans_invalidated_mid = invalidated_at
328        .iter()
329        .filter_map(|(point, idx)| {
330            if *point == location_table.mid_index(location) {
331                let borrow_region = bc.borrow_index_to_region(*idx);
332                Some(format!("{borrow_region:?}"))
333            } else {
334                None
335            }
336        })
337        .collect::<Vec<_>>();
338    let borrows_in_scope_start = bc
339        .borrows_in_scope_at(location, true)
340        .iter()
341        .map(|bi| format!("{bi:?}"))
342        .collect::<Vec<_>>();
343    let borrows_in_scope_mid = bc
344        .borrows_in_scope_at(location, false)
345        .iter()
346        .map(|bi| format!("{bi:?}"))
347        .collect::<Vec<_>>();
348    let source_pos_low = SourcePos::new(span.lo(), ctxt.tcx());
349    let source_pos_high = SourcePos::new(span.hi(), ctxt.tcx());
350    MirStmt {
351        stmt,
352        debug_stmt,
353        span: MirStmtSpan {
354            low: source_pos_low,
355            high: source_pos_high,
356        },
357        loans_invalidated_start,
358        loans_invalidated_mid,
359        borrows_in_scope_start,
360        borrows_in_scope_mid,
361    }
362}

378/// Resolves region names for all region projections within a set of input types
379/// and an output type, together with optional argument names.
380///
381/// Used to produce human-readable labels for DOT nodes in function/call shape
382/// graphs.
383pub struct ShapeLabelFormatter {
384    /// Per-argument region label lists (indexed by `ArgIdx`, then by
385    /// `LifetimeProjectionIdx`).
386    input_regions: Vec<Vec<String>>,
387    /// Region labels for the output type.
388    output_regions: Vec<String>,
389    /// Actual parameter names when available (indexed by `ArgIdx`).
390    arg_labels: Option<Vec<String>>,
391}
392
393impl ShapeLabelFormatter {
394    /// Build a formatter from the types at a call site (lifetime projections).
395    fn new<'tcx>(
396        input_tys: &[ty::Ty<'tcx>],
397        output_ty: ty::Ty<'tcx>,
398        arg_labels: Vec<String>,
399        tcx: TyCtxt<'tcx>,
400    ) -> Self {
401        let input_regions = input_tys
402            .iter()
403            .map(|ty| {
404                extract_regions(*ty)
405                    .iter()
406                    .map(|r| r.to_short_string(tcx))
407                    .collect()
408            })
409            .collect();
410        let output_regions = extract_regions(output_ty)
411            .iter()
412            .map(|r| r.to_short_string(tcx))
413            .collect();
414        Self {
415            input_regions,
416            output_regions,
417            arg_labels: Some(arg_labels),
418        }
419    }

421    /// Build a formatter from the types at a fn sig (generalized lifetime
422    /// projections), which includes `RegionsIn(T)` entries for type parameters.
423    #[must_use]
424    pub fn for_generalized<'tcx>(
425        input_tys: &[ty::Ty<'tcx>],
426        output_ty: ty::Ty<'tcx>,
427        arg_labels: Vec<String>,
428        tcx: TyCtxt<'tcx>,
429    ) -> Self {
430        use crate::borrow_pcg::visitor::{GeneralizedLifetime, extract_generalized_lifetimes};
431
432        let format_gl = |gl: &GeneralizedLifetime<'tcx>| match gl {
433            GeneralizedLifetime::Region(r) => r.to_short_string(tcx),
434            GeneralizedLifetime::RegionsIn(opaque) => format!("RegionsIn({opaque})"),
435        };
436
437        let input_regions = input_tys
438            .iter()
439            .map(|ty| {
440                extract_generalized_lifetimes(*ty, None)
441                    .iter()
442                    .map(format_gl)
443                    .collect()
444            })
445            .collect();
446        let output_regions = extract_generalized_lifetimes(output_ty, None)
447            .iter()
448            .map(format_gl)
449            .collect();
450        Self {
451            input_regions,
452            output_regions,
453            arg_labels: Some(arg_labels),
454        }
455    }

457    /// Format a function-shape input node as a human-readable label.
458    #[must_use]
459    pub fn format_input<Kind: RegionIdxMarker>(&self, input: &FunctionShapeInput<Kind>) -> String {
460        let base_name = self.arg_base_name(input.base);
461        let region_name = self.resolve_input_region(input.base, input.region_idx.index());
462        format!("{base_name}↓{region_name}")
463    }
464
465    /// Format a function-shape output node as a human-readable label.
466    #[must_use]
467    pub fn format_output<Kind: RegionIdxMarker>(
468        &self,
469        output: &FunctionShapeOutput<Kind>,
470    ) -> String {
471        let (base_name, region_name) = match output.base {
472            ArgIdxOrResult::Argument(arg) => {
473                let name = self.arg_base_name(arg);
474                let region = self.resolve_input_region(arg, output.region_idx.index());
475                (name, region)
476            }
477            ArgIdxOrResult::Result => {
478                let region = self
479                    .output_regions
480                    .get(output.region_idx.index())
481                    .cloned()
482                    .unwrap_or_else(|| format!("i{}", output.region_idx.index()));
483                ("result".to_owned(), region)
484            }
485        };
486        format!("{base_name}↓{region_name}")
487    }

489    /// Returns the human-readable name for the given argument, falling back to
490    /// the `ArgIdx` display format when no names are available.
491    fn arg_base_name(&self, arg: ArgIdx) -> String {
492        self.arg_labels
493            .as_ref()
494            .and_then(|names| names.get(*arg).cloned())
495            .unwrap_or_else(|| arg.to_string())
496    }

498    /// Build a formatter for a function's generalized sig shape, looking up
499    /// argument names, types, and trait-bound regions from the `def_id`.
500    #[must_use]
501    pub fn for_fn_sig<'tcx>(def_id: hir::def_id::DefId, tcx: TyCtxt<'tcx>) -> Self {
502        use crate::borrow_pcg::{
503            edge::abstraction::function::TraitBoundRegions, visitor::GeneralizedLifetime,
504        };
505
506        let sig = borrow_pcg::abstraction::FunctionData::new(def_id).identity_fn_sig(tcx);
507        let arg_labels = callee_arg_labels(def_id, tcx);
508        let param_env = tcx.param_env(def_id);
509        let tbr = TraitBoundRegions::new(param_env);
510        let tbr_map = tbr.as_map();
511
512        let format_gl = |gl: &GeneralizedLifetime<'tcx>| match gl {
513            GeneralizedLifetime::Region(r) => r.to_short_string(tcx),
514            GeneralizedLifetime::RegionsIn(opaque) => format!("RegionsIn({opaque})"),
515        };
516
517        let input_regions = sig
518            .inputs()
519            .iter()
520            .map(|ty| {
521                extract_generalized_lifetimes(*ty, Some(tbr_map))
522                    .iter()
523                    .map(&format_gl)
524                    .collect()
525            })
526            .collect();
527        let mut output_gls = extract_generalized_lifetimes(sig.output(), Some(tbr_map))
528            .into_iter()
529            .collect::<Vec<_>>();
530
531        // Mirror the logic in result_projections: for type parameter return
532        // types, include all signature lifetimes.
533        if matches!(sig.output().kind(), ty::TyKind::Param(_)) {
534            for input_ty in sig.inputs() {
535                for gl in extract_generalized_lifetimes(*input_ty, Some(tbr_map)) {
536                    if matches!(gl, GeneralizedLifetime::Region(_)) && !output_gls.contains(&gl) {
537                        output_gls.push(gl);
538                    }
539                }
540            }
541        }
542
543        let output_regions = output_gls.iter().map(&format_gl).collect();
544        Self {
545            input_regions,
546            output_regions,
547            arg_labels: Some(arg_labels),
548        }
549    }

551    /// Returns `(raw_display, pretty_label)` pairs for every input and output
552    /// node in `shape`. The raw display matches the format used by
553    /// [`FunctionShape`]'s `Display` impl (e.g. `Arg(0)|2`), and the pretty
554    /// label uses visualization-style names (e.g. `self↓'a`).
555    #[must_use]
556    pub fn node_labels<Kind: RegionIdxMarker>(
557        &self,
558        shape: &borrow_pcg::abstraction::FunctionShape<Kind>,
559    ) -> Vec<(String, String)> {
560        use crate::borrow_pcg::abstraction::{FunctionShapeInput, FunctionShapeOutput};
561
562        fn raw_input<K: RegionIdxMarker>(input: &FunctionShapeInput<K>) -> String {
563            format!("Arg({})↓{}", *input.base, input.region_idx.index())
564        }
565        fn raw_output<K: RegionIdxMarker>(output: &FunctionShapeOutput<K>) -> String {
566            match output.base {
567                ArgIdxOrResult::Argument(arg) => {
568                    format!("Arg({})↓{}", *arg, output.region_idx.index())
569                }
570                ArgIdxOrResult::Result => format!("Result↓{}", output.region_idx.index()),
571            }
572        }
573
574        let mut labels = Vec::new();
575        let shape_clone = shape.clone();
576        let (inputs, outputs) = shape_clone.take_inputs_and_outputs();
577        for input in &inputs {
578            labels.push((raw_input(input), self.format_input(input)));
579        }
580        for output in &outputs {
581            labels.push((raw_output(output), self.format_output(output)));
582        }
583        labels
584    }

586    /// Resolves the lifetime name for the given region index within an argument
587    /// type, falling back to the debug format when out of bounds.
588    fn resolve_input_region(&self, arg: ArgIdx, idx: usize) -> String {
589        self.input_regions
590            .get(*arg)
591            .and_then(|regions| regions.get(idx))
592            .cloned()
593            .unwrap_or_else(|| format!("i{idx}"))
594    }
595}

597/// Render a [`FunctionShape`] as a DOT bipartite graph with the given name.
598///
599/// When `formatter` is provided, nodes display human-readable labels (e.g.
600/// `x↓'a`). Otherwise falls back to the `Display` impl (e.g. `a0↓0`).
601fn function_shape_to_dot<Kind: RegionIdxMarker>(
602    shape: borrow_pcg::abstraction::FunctionShape<Kind>,
603    graph_name: &str,
604    formatter: Option<&ShapeLabelFormatter>,
605) -> String {
606    use super::dot_graph::{DotEdge, DotGraph, DotNode, DotSubgraph, EdgeDirection, EdgeOptions};
607
608    let shape_edges: Vec<_> = shape.edges().collect();
609    let (inputs, outputs) = shape.take_inputs_and_outputs();
610
611    let fmt_input = |input: &FunctionShapeInput<Kind>| -> String {
612        formatter.map_or_else(|| input.to_string(), |f| f.format_input(input))
613    };
614    let fmt_output = |output: &FunctionShapeOutput<Kind>| -> String {
615        formatter.map_or_else(|| output.to_string(), |f| f.format_output(output))
616    };
617
618    let input_node_id = |input: &FunctionShapeInput<Kind>| format!("in_{input}");
619    let output_node_id = |output: &FunctionShapeOutput<Kind>| format!("out_{output}");
620
621    let input_nodes = inputs
622        .iter()
623        .map(|input| DotNode::simple(input_node_id(input), fmt_input(input)))
624        .collect();
625
626    // Collect explicit output nodes plus any edge targets that reference
627    // argument positions (e.g. self-loops like Arg(0)|1 -> Arg(0)|1). These
628    // get their own node in the outputs cluster with the same label as the
629    // corresponding input.
630    let mut output_node_ids: Vec<String> = Vec::new();
631    let mut output_nodes: Vec<DotNode> = outputs
632        .iter()
633        .map(|output| {
634            let id = output_node_id(output);
635            output_node_ids.push(id.clone());
636            DotNode::simple(id, fmt_output(output))
637        })
638        .collect();
639    for edge in &shape_edges {
640        let id = output_node_id(&edge.output);
641        if !output_node_ids.contains(&id) {
642            output_node_ids.push(id.clone());
643            output_nodes.push(DotNode::simple(id, fmt_output(&edge.output)));
644        }
645    }
646
647    let edges = shape_edges
648        .iter()
649        .map(|edge| DotEdge {
650            id: None,
651            from: input_node_id(&edge.input),
652            to: output_node_id(&edge.output),
653            options: EdgeOptions::directed(EdgeDirection::Forward),
654        })
655        .collect();
656
657    let graph = DotGraph {
658        name: Cow::Owned(graph_name.to_owned()),
659        nodes: Vec::new(),
660        edges,
661        subgraphs: vec![
662            DotSubgraph {
663                name: "inputs".into(),
664                label: "Inputs".into(),
665                nodes: input_nodes,
666            },
667            DotSubgraph {
668                name: "outputs".into(),
669                label: "Outputs".into(),
670                nodes: output_nodes,
671            },
672        ],
673    };
674    graph.to_string()
675}

677/// Returns the argument names for a function, using `"_"` for unnamed args.
678#[must_use]
679pub fn callee_arg_labels(def_id: hir::def_id::DefId, tcx: TyCtxt<'_>) -> Vec<String> {
680    tcx.fn_arg_idents(def_id)
681        .iter()
682        .map(|ident| {
683            if let Some(ident) = ident {
684                ident.to_string()
685            } else {
686                "_".to_owned()
687            }
688        })
689        .collect()
690}

692/// Generate a DOT graph representing the function shape (bipartite graph of
693/// lifetime outlives relationships) for the callee at a call site.
694fn generate_function_shape_dot(
695    def_id: hir::def_id::DefId,
696    ctxt: CompilerCtxt<'_, '_>,
697) -> Option<String> {
698    use borrow_pcg::abstraction::FunctionShape;
699    let shape = FunctionShape::for_fn_sig(def_id, ctxt).ok()?;
700    let fn_name = ctxt.tcx().def_path_str(def_id);
701
702    let formatter = ShapeLabelFormatter::for_fn_sig(def_id, ctxt.tcx());
703    Some(function_shape_to_dot(shape, &fn_name, Some(&formatter)))
704}
705
706/// Generate a DOT graph representing the call shape derived solely from the
707/// types of the operands at the call site (ignoring the callee's fn def
708/// signature).
709fn generate_call_shape_dot<'tcx>(
710    input_tys: Vec<ty::Ty<'tcx>>,
711    output_ty: ty::Ty<'tcx>,
712    location: mir::Location,
713    ctxt: CompilerCtxt<'_, 'tcx>,
714    callee_def_id: Option<hir::def_id::DefId>,
715) -> String {
716    use crate::borrow_pcg::{
717        abstraction::FunctionShape, edge::abstraction::function::FnCallDataSource,
718    };
719
720    let arg_labels = callee_def_id.map(|did| callee_arg_labels(did, ctxt.tcx()));
721    let formatter =
722        ShapeLabelFormatter::new(&input_tys, output_ty, arg_labels.unwrap(), ctxt.tcx());
723
724    let data_source = FnCallDataSource::new(location, input_tys, output_ty);
725    let shape = FunctionShape::new(&data_source, ctxt);
726    function_shape_to_dot(shape, "call_shape", Some(&formatter))
727}

729fn mk_mir_graph(ctxt: CompilerCtxt<'_, '_>) -> MirGraph {
730    let mut nodes = Vec::new();
731    let mut edges = Vec::new();
732
733    for (bb, data) in ctxt.body().basic_blocks.iter_enumerated() {
734        let stmts = data.statements.iter().enumerate().map(|(idx, stmt)| {
735            let stmt_text = format_stmt(stmt, ctxt);
736            let location = mir::Location {
737                block: bb,
738                statement_index: idx,
739            };
740            mk_mir_stmt(
741                stmt_text,
742                format!("{stmt:?}"),
743                stmt.source_info.span,
744                location,
745                ctxt,
746            )
747        });
748
749        let terminator_text = format_terminator(&data.terminator().kind, ctxt);
750        let terminator_location = mir::Location {
751            block: bb,
752            statement_index: data.statements.len(),
753        };
754        let terminator = mk_mir_stmt(
755            terminator_text,
756            format!("{:?}", data.terminator()),
757            data.terminator().source_info.span,
758            terminator_location,
759            ctxt,
760        );
761
762        // TODO: In principle, we can also create graphs for call shapes for
763        // undefined functions (e.g. function pointers, closures).
764        let (sig_shape_dot, call_shape_dot) = if let TerminatorKind::Call {
765            func,
766            args,
767            destination,
768            ..
769        } = &data.terminator().kind
770        {
771            let callee_def_id = resolve_callee(func, ctxt.body(), ctxt.tcx());
772            let sig_shape =
773                callee_def_id.and_then(|def_id| generate_function_shape_dot(def_id, ctxt));
774            let input_tys = args
775                .iter()
776                .map(|arg| arg.node.ty(ctxt.body(), ctxt.tcx()))
777                .collect();
778            let output_ty = destination.ty(ctxt.body(), ctxt.tcx()).ty;
779            let call_shape = generate_call_shape_dot(
780                input_tys,
781                output_ty,
782                terminator_location,
783                ctxt,
784                callee_def_id,
785            );
786            (sig_shape, Some(call_shape))
787        } else {
788            (None, None)
789        };
790
791        nodes.push(MirNode {
792            id: format!("{bb:?}"),
793            block: bb.as_usize(),
794            stmts: stmts.collect(),
795            terminator,
796            sig_shape_dot,
797            call_shape_dot,
798        });
799
800        match &data.terminator().kind {
801            TerminatorKind::Goto { target } => {
802                edges.push(MirEdge {
803                    source: format!("{bb:?}"),
804                    target: format!("{target:?}"),
805                    label: Cow::Borrowed("goto"),
806                });
807            }
808            TerminatorKind::SwitchInt { discr: _, targets } => {
809                for (val, target) in targets.iter() {
810                    edges.push(MirEdge {
811                        source: format!("{bb:?}"),
812                        target: format!("{target:?}"),
813                        label: Cow::Owned(val.to_string()),
814                    });
815                }
816                edges.push(MirEdge {
817                    source: format!("{bb:?}"),
818                    target: format!("{:?}", targets.otherwise()),
819                    label: Cow::Borrowed("otherwise"),
820                });
821            }
822            TerminatorKind::UnwindResume | TerminatorKind::Return | TerminatorKind::Unreachable => {
823            }
824            TerminatorKind::UnwindTerminate(_) => todo!(),
825            TerminatorKind::Drop { target, .. } => {
826                edges.push(MirEdge {
827                    source: format!("{bb:?}"),
828                    target: format!("{target:?}"),
829                    label: Cow::Borrowed("drop"),
830                });
831            }
832            TerminatorKind::Call {
833                func: _,
834                args: _,
835                destination: _,
836                target,
837                unwind,
838                call_source: _,
839                fn_span: _,
840            } => {
841                if let Some(target) = target {
842                    edges.push(MirEdge {
843                        source: format!("{bb:?}"),
844                        target: format!("{target:?}"),
845                        label: Cow::Borrowed("call"),
846                    });
847                    match unwind {
848                        UnwindAction::Continue => todo!(),
849                        UnwindAction::Unreachable => todo!(),
850                        UnwindAction::Terminate(_) => todo!(),
851                        UnwindAction::Cleanup(cleanup) => {
852                            edges.push(MirEdge {
853                                source: format!("{bb:?}"),
854                                target: format!("{cleanup:?}"),
855                                label: Cow::Borrowed("unwind"),
856                            });
857                        }
858                    }
859                }
860            }
861            TerminatorKind::Assert {
862                cond: _,
863                expected: _,
864                msg: _,
865                target,
866                unwind,
867            } => {
868                match unwind {
869                    UnwindAction::Continue => todo!(),
870                    UnwindAction::Unreachable => todo!(),
871                    UnwindAction::Terminate(_) => todo!(),
872                    UnwindAction::Cleanup(cleanup) => {
873                        edges.push(MirEdge {
874                            source: format!("{bb:?}"),
875                            target: format!("{cleanup:?}"),
876                            label: Cow::Borrowed("unwind"),
877                        });
878                    }
879                }
880                edges.push(MirEdge {
881                    source: format!("{bb:?}"),
882                    target: format!("{target:?}"),
883                    label: Cow::Borrowed("success"),
884                });
885            }
886            TerminatorKind::Yield {
887                value: _,
888                resume: _,
889                resume_arg: _,
890                drop: _,
891            } => todo!(),
892            TerminatorKind::FalseEdge {
893                real_target,
894                imaginary_target: _,
895            }
896            | TerminatorKind::FalseUnwind {
897                real_target,
898                unwind: _,
899            } => {
900                edges.push(MirEdge {
901                    source: format!("{bb:?}"),
902                    target: format!("{real_target:?}"),
903                    label: Cow::Borrowed("real"),
904                });
905            }
906            TerminatorKind::InlineAsm { .. } => todo!(),
907            TerminatorKind::CoroutineDrop => todo!(),
908            _ => todo!(),
909        }
910    }
911
912    MirGraph { nodes, edges }
913}
914pub(crate) fn generate_json_from_mir(path: &Path, ctxt: CompilerCtxt<'_, '_>) -> io::Result<()> {
915    let mir_graph = mk_mir_graph(ctxt);
916    let mut file = File::create(path)?;
917    serde_json::to_writer(&mut file, &mir_graph)?;
918    Ok(())
919}
