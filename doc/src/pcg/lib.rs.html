85/// Instructs that the current capability to the place (first [`CapabilityKind`]) should
86/// be weakened to the second given capability. We guarantee that `_.1 > _.2`.
87/// If `_.2` is `None`, the capability is removed.

91pub struct Weaken<'tcx, Place = utils::Place<'tcx>, ToCap = Option<CapabilityKind>> {
92    pub(crate) place: Place,
93    pub(crate) from: CapabilityKind,
94    pub(crate) to: ToCap,
95    /// Temporary flag indicating this Weaken was emitted due to a `StorageDead`.
96    /// Required until <https://github.com/prusti/pcg/issues/137> is resolved.
97    pub(crate) for_storage_dead: bool,
98    #[serde(skip)]
99    _marker: PhantomData<&'tcx ()>,
100}

332    #[must_use]
333    pub fn with_settings(tcx: TyCtxt<'tcx>, settings: PcgSettings) -> Self {
334        Self {
335            tcx,
336            arena: bumpalo::Bump::new(),
337            settings,
338            #[cfg(feature = "visualization")]
339            debug_function_metadata: RefCell::new(visualization::FunctionsMetadata::new()),

344        self.arena.alloc(val)
345    }
346
347    pub fn new_ctxt<'slf: 'a, 'a>(
348        &'slf self,
349        body: &'a impl BodyAndBorrows<'tcx>,
350        bc: &'a impl BorrowCheckerInterface<'tcx>,
351    ) -> &'a PcgCtxt<'a, 'tcx> {
352        let pcg_ctxt: PcgCtxt<'a, 'tcx> =
353            PcgCtxt::with_settings(body.body(), self.tcx, bc, Cow::Borrowed(&self.settings));
354        #[cfg(feature = "visualization")]
355        if let Some(identifier) = pcg_ctxt.visualization_function_metadata() {
356            self.debug_function_metadata
357                .borrow_mut()
358                .insert(pcg_ctxt.compiler_ctxt.function_metadata_slug(), identifier);
359        }
360        self.alloc(pcg_ctxt)
361    }
362
363    pub fn new_nll_ctxt<'slf: 'a, 'a>(
364        &'slf self,
365        body: &'a impl BodyAndBorrows<'tcx>,
366    ) -> &'a PcgCtxt<'a, 'tcx> {
367        let bc = self.arena.alloc(NllBorrowCheckerImpl::new(self.tcx, body));
368        self.new_ctxt(body, bc)
369    }

487/// The main entrypoint for running the PCG.
488///
489/// # Arguments
490///
491/// * `pcg_ctxt` - The context the PCG will use for its analysis. Use [`PcgCtxt::new`] to create this.
492pub fn run_pcg<'a, 'tcx>(pcg_ctxt: &'a PcgCtxt<'_, 'tcx>) -> PcgOutput<'a, 'tcx> {
493    let settings = pcg_ctxt.settings.borrow();
494    tracing::info!(
495        "Running PCG (visualization: {})",
496        pcg_ctxt.settings.visualization
497    );
498    let engine = PcgEngine::new(
499        pcg_ctxt.compiler_ctxt,
500        &pcg_ctxt.move_data,
501        PcgArena::new(&pcg_ctxt.arena),
502        settings,
503        #[cfg(feature = "visualization")]
504        pcg_ctxt.visualization_output_path(),
505    );
506    let body = pcg_ctxt.compiler_ctxt.body();
507    let tcx = pcg_ctxt.compiler_ctxt.tcx();
508    let mut analysis = compute_fixpoint(AnalysisEngine(engine), tcx, body);
509    for block in body.basic_blocks.indices() {
510        let engine = analysis.get_analysis();
511        let ctxt = engine.analysis_ctxt(block);
512        let state = analysis.entry_state_for_block_mut(block);
513        state.complete(ctxt);
514    }
515
516    let mut analysis_results = results::PcgAnalysisResults::new(analysis.into_results_cursor(body));
517
518    #[cfg(feature = "visualization")]
519    if let Some(dir_path) = pcg_ctxt.visualization_output_path() {
520        generate_json_from_mir(&dir_path.join("mir.json"), pcg_ctxt.compiler_ctxt)
521            .expect("Failed to generate JSON from MIR");
522        {
523            use std::fmt::Write;
524            let loop_analysis = analysis_results.analysis().loop_analysis();
525            let mut loop_text = String::new();
526            for bb in body.basic_blocks.indices() {
527                let depth = loop_analysis.loop_depth(bb);
528                let loops: Vec<_> = loop_analysis.loops(bb).collect();
529                let loop_head = loop_analysis.loop_head_of(bb);
530                writeln!(
531                    &mut loop_text,
532                    "{bb:?} depth: {depth}, loops: {loops:?}, loop head?: {loop_head:?}"
533                )
534                .unwrap();
535            }
536            std::fs::write(dir_path.join("loop_analysis.txt"), &loop_text)
537                .expect("Failed to write loop analysis text");
538        }
539        let mut visualization_data = PcgVisualizationData::new();
540        for block in body.basic_blocks.indices() {
541            let Ok(Some(pcg_block)) = analysis_results.get_all_for_bb(block) else {
542                continue;
543            };
544            let ctxt = analysis_results.analysis().analysis_ctxt(block);
545            let loop_analysis = analysis_results.analysis().loop_analysis();
546            let loop_invariant_capabilities = loop_analysis
547                .loops(block)
548                .map(|loop_id| {
549                    let capabilities = pcg_block.loop_invariant_place_capabilities(
550                        analysis_results.analysis().loop_place_usages(loop_id),
551                        pcg_ctxt.compiler_ctxt,
552                    );
553                    (
554                        loop_analysis[loop_id].into(),
555                        capabilities
556                            .into_iter()
557                            .map(|(place, capability)| {
558                                (place.display_string(pcg_ctxt.compiler_ctxt), capability)
559                            })
560                            .collect(),
561                    )
562                })
563                .collect();
564            let (loop_data, debug_graphs) = if let Some(graphs) = ctxt.graphs {
565                let block_data = graphs.dot_graphs.borrow();
566                (block_data.loop_data.clone(), block_data.graphs.clone())
567            } else {
568                (None, Vec::new())
569            };
570
571            let terminator_stmt_index = body[block].statements.len();
572            let is_return = matches!(&body[block].terminator().kind, mir::TerminatorKind::Return);
573            let return_unblock = if is_return {
574                let filename = format!("{block:?}_remote_lifetime_projection_unblock.dot");
575                let post_main = &pcg_block.terminator.states[EvalStmtPhase::PostMain];
576                let remote_lifetime_projections = post_main
577                    .borrow
578                    .graph()
579                    .nodes(pcg_ctxt.compiler_ctxt)
580                    .into_iter()
581                    .filter(|node| {
582                        node.try_into_lifetime_projection().is_ok_and(|rp| {
583                            rp.base()
584                                .maybe_remote_current_place()
585                                .is_some_and(MaybeRemoteCurrentPlace::is_remote)
586                        })
587                    })
588                    .collect::<Vec<_>>();
589                let unblock_graph = borrow_pcg::unblock_graph::UnblockGraph::for_nodes(
590                    remote_lifetime_projections,
591                    &post_main.borrow,
592                    pcg_ctxt.compiler_ctxt,
593                );
594                let unblock_actions = unblock_graph
595                    .clone()
596                    .actions(pcg_ctxt.compiler_ctxt)
597                    .expect("Failed to generate return unblock actions")
598                    .into_iter()
599                    .map(|action| action.edge().display_string(pcg_ctxt.compiler_ctxt))
600                    .collect::<Vec<_>>();
601                let dot_graph = visualization::generate_unblock_dot_graph(
602                    pcg_ctxt.compiler_ctxt,
603                    &post_main.place_capabilities,
604                    &unblock_graph,
605                )
606                .expect("Failed to generate return unblock graph");
607                std::fs::write(dir_path.join(&filename), dot_graph)
608                    .expect("Failed to write return unblock graph");
609                Some((filename, unblock_actions))
610            } else {
611                None
612            };
613
614            let statements = pcg_block
615                .statements()
616                .map(|stmt| {
617                    let actions: EvalStmtData<Vec<AppliedActionDebugRepr>> =
618                        stmt.actions.debug_repr(pcg_ctxt.compiler_ctxt);
619                    PcgStmtVisualizationData {
620                        actions,
621                        graphs: debug_graphs
622                            .get(stmt.location.statement_index)
623                            .cloned()
624                            .unwrap_or_default(),
625                        return_unblock_graph: if stmt.location.statement_index
626                            == terminator_stmt_index
627                        {
628                            return_unblock
629                                .as_ref()
630                                .map(|(filename, _)| filename.clone())
631                        } else {
632                            None
633                        },
634                        return_unblock_actions: if stmt.location.statement_index
635                            == terminator_stmt_index
636                        {
637                            return_unblock.as_ref().map(|(_, actions)| actions.clone())
638                        } else {
639                            None
640                        },
641                    }
642                })
643                .collect();
644
645            let successors = pcg_block
646                .successors()
647                .map(|succ| {
648                    (
649                        succ.block().into(),
650                        PcgSuccessorVisualizationData {
651                            actions: succ.actions().debug_repr(pcg_ctxt.compiler_ctxt),
652                        },
653                    )
654                })
655                .collect();
656
657            visualization_data.insert(
658                block.into(),
659                PcgBlockVisualizationData {
660                    statements,
661                    successors,
662                    loop_data,
663                    loop_invariant_capabilities,
664                },
665            );
666        }
667
668        let pcg_data_file_path = dir_path.join("pcg_data.json");
669        let pcg_data_json = serde_json::to_string(&visualization_data).unwrap();
670        std::fs::write(&pcg_data_file_path, pcg_data_json)
671            .expect("Failed to write pcg data to JSON file");
672    }
673
674    if validity_checks_enabled() {
675        for (block, _data) in body.basic_blocks.iter_enumerated() {
676            let Ok(Some(pcg_block)) = analysis_results.get_all_for_bb(block) else {
677                continue;
678            };
679            for (statement_index, statement) in pcg_block.statements.iter().enumerate() {
680                statement.assert_validity_at_location(
681                    mir::Location {
682                        block,
683                        statement_index,
684                    },
685                    pcg_ctxt,
686                );
687            }
688        }
689    }
690
691    analysis_results
692}

779macro_rules! pcg_validity_assert {
780    // Entry point with brackets - parse using token trees
781    ($cond:expr, [$($ctxt_and_loc:tt)*]) => {
782        pcg_validity_assert!(@parse_context $cond, [$($ctxt_and_loc)*], "{}", stringify!($cond))
783    };
784    ($cond:expr, [$($ctxt_and_loc:tt)*], $($arg:tt)*) => {
785        pcg_validity_assert!(@parse_context $cond, [$($ctxt_and_loc)*], $($arg)*)
786    };
787
788    // Parse context patterns - match the entire token sequence
789    (@parse_context $cond:expr, [$ctxt:tt at $loc:tt], $($arg:tt)*) => {
790        {
791            let ctxt = $ctxt;
792            let loc = $loc;
793            let func_name = $crate::utils::ctxt::DebugCtxt::func_name(&ctxt);
794            let crate_part = std::env::var("CARGO_CRATE_NAME").map(|s| format!(" (Crate: {s})")).unwrap_or_default();
795            pcg_validity_assert!(@with_test_case $cond, ctxt, func_name, "PCG Assertion Failed {crate_part}: [{func_name} at {loc:?}] {}", format!($($arg)*));
796        }
797    };
798    (@parse_context $cond:expr, [$ctxt:tt], $($arg:tt)*) => {
799        {
800            let ctxt = $ctxt;
801            let func_name = $crate::utils::ctxt::DebugCtxt::func_name(&ctxt);
802            let crate_part = std::env::var("CARGO_CRATE_NAME").map(|s| format!(" (Crate: {s})")).unwrap_or_default();
803            pcg_validity_assert!(@with_test_case $cond, ctxt, func_name, "PCG Assertion Failed {crate_part}: [{func_name}] {}", format!($($arg)*));
804        }
805    };
806
807    // Helper branch that generates test case format when context is available
808    (@with_test_case $cond:expr, $ctxt:expr, $func_name:expr, $($arg:tt)*) => {
809        if $ctxt.settings().validity_checks {
810            #[allow(clippy::neg_cmp_op_on_partial_ord)]
811            if !$cond {
812                tracing::error!($($arg)*);
813                // Generate test case format if we're in a crate
814                if let Ok(crate_name) = std::env::var("CARGO_CRATE_NAME") {
815                    let crate_version = std::env::var("CARGO_PKG_VERSION").unwrap_or_else(|_| "unknown".to_owned());
816                    let num_bbs = $crate::utils::ctxt::DebugCtxt::num_basic_blocks(&$ctxt);
817                    let test_case = format!("{};{};2025-03-13;{};{}",
818                        crate_name, crate_version, $func_name, num_bbs);
819                    tracing::error!("To reproduce this failure, use test case: {test_case}");
820                }
821                if !$crate::validity_checks_warn_only() {
822                    assert!($cond, $($arg)*);
823                }
824            }
825        }
826    };
827
828    // Without brackets
829    ($cond:expr) => {
830        pcg_validity_assert!($cond, "PCG Assertion Failed: {}", stringify!($cond))
831    };
832    ($cond:expr, $($arg:tt)*) => {
833        if $crate::validity_checks_enabled() {
834            #[allow(clippy::neg_cmp_op_on_partial_ord)]
835            if !$cond {
836                tracing::error!($($arg)*);
837                if !$crate::validity_checks_warn_only() {
838                    assert!($cond, $($arg)*);
839                }
840            }
841        }
842    };
843}
