24/// A guide for an expansion that isn't a simple struct-field unpack.
25///
26/// The `Default` type parameter is used in context where no guide is required
27/// (e.g. for a box deref, where only one expansion is possible).
28/// If a guide is required (c.f. [`RequiredGuide`]), `Default`
29/// is instantiated to `!` so the variant is statically unreachable.
30#[derive(Clone, Copy, Eq, PartialEq, Hash)]
31pub enum RepackGuide<Local = mir::Local, D = (), Default = D> {
32    /// No non-trivial guide (e.g. an ordinary field/tuple expansion).
33    Default(Default),
34    Downcast(Option<Symbol>, VariantIdx, D),
35    ConstantIndex(ConstantIndex, D),
36    Index(Local, D),
37    Subslice {
38        from: u64,
39        to: u64,
40        from_end: bool,
41        data: D,
42    },
43}
44
45/// A `RepackGuide` guaranteed to not be [`RepackGuide::Default`].
46/// Used for enum downcasts, array accesses etc.
47pub(crate) type RequiredGuide<D = ()> = RepackGuide<mir::Local, D, !>;
48
49/// Custom `Debug` that does not print the per-child data payload, so
50/// that test-visible annotation strings stay stable (`Index(_5)` rather
51/// than `Index(_5, ())`) across the generic-parameter introduction.
52/// The `Default` variant's payload is still shown because it is
53/// distinctive and not merely scaffolding.
54impl<Local: std::fmt::Debug, D, Default: std::fmt::Debug> std::fmt::Debug
55    for RepackGuide<Local, D, Default>
56{
57    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
58        match self {
59            RepackGuide::Default(d) => f.debug_tuple("Default").field(d).finish(),
60            RepackGuide::Downcast(sym, idx, _) => {
61                f.debug_tuple("Downcast").field(sym).field(idx).finish()
62            }
63            RepackGuide::ConstantIndex(ci, _) => f.debug_tuple("ConstantIndex").field(ci).finish(),
64            RepackGuide::Index(local, _) => f.debug_tuple("Index").field(local).finish(),
65            RepackGuide::Subslice {
66                from,
67                to,
68                from_end,
69                data: _,
70            } => f
71                .debug_struct("Subslice")
72                .field("from", from)
73                .field("to", to)
74                .field("from_end", from_end)
75                .finish(),
76        }
77    }
78}

80impl<Local, D, Default> RepackGuide<Local, D, Default> {
81    #[allow(dead_code)]
82    pub(crate) fn map_data<R>(self, f: impl FnOnce(D) -> R) -> RepackGuide<Local, R, Default> {
83        match self {
84            RepackGuide::Default(d) => RepackGuide::Default(d),
85            RepackGuide::Downcast(s, v, data) => RepackGuide::Downcast(s, v, f(data)),
86            RepackGuide::ConstantIndex(c, data) => RepackGuide::ConstantIndex(c, f(data)),
87            RepackGuide::Index(local, data) => RepackGuide::Index(local, f(data)),
88            RepackGuide::Subslice {
89                from,
90                to,
91                from_end,
92                data,
93            } => RepackGuide::Subslice {
94                from,
95                to,
96                from_end,
97                data: f(data),
98            },
99        }
100    }
101
102    pub(crate) fn into_non_default(self) -> Option<RequiredGuide<D>>
103    where
104        Local: Into<mir::Local>,
105    {
106        match self {
107            RepackGuide::Default(_) => None,
108            RepackGuide::Downcast(s, v, d) => Some(RepackGuide::Downcast(s, v, d)),
109            RepackGuide::ConstantIndex(c, d) => Some(RepackGuide::ConstantIndex(c, d)),
110            RepackGuide::Index(local, d) => Some(RepackGuide::Index(local.into(), d)),
111            RepackGuide::Subslice {
112                from,
113                to,
114                from_end,
115                data,
116            } => Some(RepackGuide::Subslice {
117                from,
118                to,
119                from_end,
120                data,
121            }),
122        }
123    }
124}

143            RepackGuide::Default(never) => match never {},
144            RepackGuide::Downcast(s, v, ()) => RepackGuide::Downcast(s, v, ()),
145            RepackGuide::ConstantIndex(c, ()) => RepackGuide::ConstantIndex(c, ()),
146            RepackGuide::Index(local, ()) => RepackGuide::Index(local, ()),
147            RepackGuide::Subslice {
148                from,
149                to,
150                from_end,
151                data: (),
152            } => RepackGuide::Subslice {
153                from,
154                to,
155                from_end,
156                data: (),
157            },
158        }
159    }

162impl From<RequiredGuide> for PlaceElem<'_> {
163    fn from(val: RequiredGuide) -> Self {
164        match val {
165            RepackGuide::Default(never) => match never {},
166            RepackGuide::Index(local, ()) => PlaceElem::Index(local),
167            RepackGuide::Downcast(symbol, variant_idx, ()) => {
168                PlaceElem::Downcast(symbol, variant_idx)
169            }
170            RepackGuide::ConstantIndex(constant_index, ()) => PlaceElem::ConstantIndex {
171                offset: constant_index.offset,
172                min_length: constant_index.min_length,
173                from_end: constant_index.from_end,
174            },
175            RepackGuide::Subslice {
176                from,
177                to,
178                from_end,
179                data: (),
180            } => PlaceElem::Subslice { from, to, from_end },
181        }
182    }
183}

185impl From<RepackGuide> for PlaceElem<'_> {
186    fn from(val: RepackGuide) -> Self {
187        // Safe because ambient `RepackGuide` *shouldn't* carry `Default`
188        // at points where we reify it to a `PlaceElem`.
189        val.into_non_default()
190            .expect("unexpected RepackGuide::Default in PlaceElem conversion")
191            .into()
192    }
193}

195impl TryFrom<PlaceElem<'_>> for RepackGuide {
196    type Error = ();
197    fn try_from(elem: PlaceElem<'_>) -> Result<Self, Self::Error> {
198        match elem {
199            PlaceElem::Index(local) => Ok(RepackGuide::Index(local, ())),
200            PlaceElem::Downcast(symbol, variant_idx) => {
201                Ok(RepackGuide::Downcast(symbol, variant_idx, ()))
202            }
203            PlaceElem::ConstantIndex {
204                offset,
205                min_length,
206                from_end,
207            } => Ok(RepackGuide::ConstantIndex(
208                ConstantIndex {
209                    offset,
210                    min_length,
211                    from_end,
212                },
213                (),
214            )),
215            PlaceElem::Subslice { from, to, from_end } => Ok(RepackGuide::Subslice {
216                from,
217                to,
218                from_end,
219                data: (),
220            }),
221            _ => Err(()),
222        }
223    }
224}

363#[serde(tag = "type", content = "data")]
364pub enum RepackOp<'tcx, Local = mir::Local, Place = crate::utils::Place<'tcx>, Guide = RepackGuide>
365{
366    /// Rust will sometimes join two `BasicBlocks` where a local is live in one and dead in the other.
367    /// Our analysis will join these two into a state where the local is dead, and this Op marks the
368    /// edge from where it was live.
369    ///
370    /// This is not an issue in the MIR since it generally has a
371    /// [`mir::StatementKind::StorageDead`](https://doc.rust-lang.org/nightly/nightly-rustc/rustc_middle/mir/enum.StatementKind.html#variant.StorageDead)
372    /// right after the merge point, which is fine in Rust semantics, since
373    /// [`mir::StatementKind::StorageDead`](https://doc.rust-lang.org/nightly/nightly-rustc/rustc_middle/mir/enum.StatementKind.html#variant.StorageDead)
374    /// is a no-op if the local is already (conditionally) dead.
375    ///
376    /// This Op only appears for edges between basic blocks. It is often emitted for edges to panic
377    /// handling blocks, but can also appear in regular code for example in the MIR of
378    /// [this function](https://github.com/dtolnay/syn/blob/3da56a712abf7933b91954dbfb5708b452f88504/src/attr.rs#L623-L628).
379    StorageDead(Local),
380    /// This Op only appears within a `BasicBlock` and is attached to a
381    /// [`mir::StatementKind::StorageDead`](https://doc.rust-lang.org/nightly/nightly-rustc/rustc_middle/mir/enum.StatementKind.html#variant.StorageDead)
382    /// statement. We emit it for any such statement where the local may already be dead. We
383    /// guarantee to have inserted a [`RepackOp::StorageDead`] before this Op so that one can
384    /// safely ignore the statement this is attached to.
385    IgnoreStorageDead(Local),
386    /// Instructs that the current capability to the place (first [`CapabilityKind`]) should
387    /// be weakened to the second given capability. We guarantee that `_.1 > _.2`.
388    ///
389    /// This Op is used prior to a [`RepackOp::Collapse`] to ensure that all packed up places have
390    /// the same capability. It can also appear at basic block join points, where one branch has
391    /// a weaker capability than the other.
392    Weaken(Weaken<'tcx, Place, CapabilityKind>),
393    /// Instructs that one should unpack `place` with the capability.
394    /// We guarantee that the current state holds exactly the given capability for the given place.
395    /// `guide` denotes e.g. the enum variant to unpack to. One can use
396    /// [`Place::expand_one_level(_.0, _.1, ..)`](Place::expand_one_level) to get the set of all
397    /// places (except as noted in the documentation for that fn) which will be obtained by unpacking.
398    Expand(RepackExpand<'tcx, Place, Guide>),
399    /// Instructs that one should pack up `place` with the given capability.
400    /// `guide` denotes e.g. the enum variant to pack from. One can use
401    /// [`Place::expand_one_level(_.0, _.1, ..)`](Place::expand_one_level) to get the set of all
402    /// places which should be packed up. We guarantee that the current state holds exactly the
403    /// given capability for all places in this set.
404    Collapse(RepackCollapse<'tcx, Place, Guide>),
405    /// TODO
406    DerefShallowInit(Place, Place),
407    /// This place should have its capability changed from `Lent` (for mutably
408    /// borrowed places) or `Read` (for shared borrow places), to the given
409    /// capability, because it is no longer lent out.
410    RegainLoanedCapability(RegainedCapability<Place>),
411}

450impl<'a, 'tcx: 'a, Ctxt: HasCompilerCtxt<'a, 'tcx>> DebugRepr<Ctxt> for RepackOp<'tcx> {
451    type Repr = RepackOp<'static, String, String, String>;
452    fn debug_repr(&self, ctxt: Ctxt) -> Self::Repr {
453        match self {
454            RepackOp::StorageDead(local) => RepackOp::StorageDead(local.display_string(ctxt)),
455            RepackOp::IgnoreStorageDead(local) => {
456                RepackOp::IgnoreStorageDead(local.display_string(ctxt))
457            }
458            RepackOp::RegainLoanedCapability(regained_capability) => {
459                RepackOp::RegainLoanedCapability(regained_capability.debug_repr(ctxt))
460            }
461            RepackOp::Weaken(weaken) => RepackOp::Weaken(weaken.debug_repr(ctxt)),
462            RepackOp::Expand(expand) => RepackOp::Expand(expand.debug_repr(ctxt)),
463            RepackOp::Collapse(collapse) => RepackOp::Collapse(collapse.debug_repr(ctxt)),
464            RepackOp::DerefShallowInit(place, place2) => {
465                RepackOp::DerefShallowInit(place.display_string(ctxt), place2.display_string(ctxt))
466            }
467        }
468    }
469}

491            .into(),
492        )
493    }

496impl<'tcx> RepackOp<'tcx> {
497    /// Create a Weaken, checking that it strictly reduces the capability.
498    pub(crate) fn weaken(place: Place<'tcx>, from: CapabilityKind, to: CapabilityKind) -> Self {
499        pcg_validity_assert!(
500            from > to,
501            "a weaken must strictly reduce the capability ({from:?} -> {to:?})"
502        );
503        Self::Weaken(Weaken::new(place, from, to))
504    }
505
506    /// Temporary: create a Weaken for `StorageDead`.
507    /// Required until <https://github.com/prusti/pcg/issues/137> is resolved.
508    pub(crate) fn weaken_for_storage_dead(
509        place: Place<'tcx>,
510        from: CapabilityKind,
511        to: CapabilityKind,
512    ) -> Self {
513        pcg_validity_assert!(
514            from > to,
515            "a weaken must strictly reduce the capability ({from:?} -> {to:?})"
516        );
517        Self::Weaken(Weaken::new_for_storage_dead(place, from, to))
518    }
519    pub(crate) fn expand<'a>(
520        from: Place<'tcx>,
521        guide: Option<RepackGuide>,
522        for_cap: CapabilityKind,
523        _ctxt: impl HasCompilerCtxt<'a, 'tcx>,
524    ) -> Self {
525        // Note that we might generate expand annotations with `Write` capability for
526        // the `bridge` operation to generate annotations between basic blocks.
527        Self::Expand(RepackExpand {
528            from,
529            guide,
530            capability: for_cap,
531            _marker: PhantomData,
532        })
533    }
534
535    #[must_use]
536    pub fn affected_place(&self) -> Place<'tcx> {
537        match *self {
538            RepackOp::StorageDead(local) | RepackOp::IgnoreStorageDead(local) => local.into(),
539            RepackOp::Weaken(Weaken { place, .. })
540            | RepackOp::Collapse(RepackCollapse { to: place, .. })
541            | RepackOp::Expand(RepackExpand { from: place, .. })
542            | RepackOp::RegainLoanedCapability(RegainedCapability { place, .. })
543            | RepackOp::DerefShallowInit(place, _) => place,
544        }
545    }
