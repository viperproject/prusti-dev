58/// Whether expansion edges in the PCG are packed up as soon as they can be
59/// packed without losing information, or only at the point where a capability
60/// to the packed place is required.
61#[derive(Debug, PartialEq, Eq, Clone, Copy, Hash, Default)]
62pub enum PackStrategy {
63    /// Pack an expansion as soon as it becomes packable.
64    #[default]
65    Eager,
66    /// Pack an expansion of `p` only when a capability that `p` cannot hold
67    /// while it is expanded is required.
68    Lazy,
69}

85impl GlobalPcgSettings {
86    #[must_use]
87    pub fn new() -> Self {
88        Self::new_returning_vars().0
89    }
90    pub(crate) fn new_returning_vars() -> (Self, HashSet<String>) {
91        let mut processed_vars = HashSet::new();
92        let skip_bodies_with_loops =
93            PcgSettings::process_bool_var(&mut processed_vars, "PCG_SKIP_BODIES_WITH_LOOPS", false);
94        let max_basic_blocks =
95            PcgSettings::process_usize_var(&mut processed_vars, "PCG_MAX_BASIC_BLOCKS");
96        let test_crates_start_from =
97            PcgSettings::process_usize_var(&mut processed_vars, "PCG_TEST_CRATES_START_FROM");
98        let allow_borrowck_errors =
99            PcgSettings::process_bool_var(&mut processed_vars, "PCG_ALLOW_BORROWCK_ERRORS", false);
100        let be_rustc = PcgSettings::process_bool_var(&mut processed_vars, "PCG_BE_RUSTC", false);
101        (
102            Self {
103                allow_borrowck_errors,
104                skip_bodies_with_loops,
105                max_basic_blocks,
106                test_crates_start_from,
107                be_rustc,
108            },
109            processed_vars,
110        )
111    }
112}

142    pub(crate) fn create_visualization_data_directory(path: &Path, erase_contents: bool) {
143        if erase_contents {
144            std::fs::remove_dir_all(path)
145                .expect("Failed to delete visualization directory contents");
146        }
147        std::fs::create_dir_all(path).expect("Failed to create visualization directory");
148
149        // Log the absolute path after directory creation
150        if let Ok(absolute_path) = std::fs::canonicalize(path) {
151            tracing::info!("Visualization directory: {absolute_path:?}");
152        } else {
153            tracing::info!("Visualization directory: {path:?}");
154        }
155    }

157    pub(crate) fn functions_json_path(&self) -> PathBuf {
158        self.visualization_data_dir.join("functions.json")
159    }

161    #[must_use]
162    pub fn new() -> Self {
163        // Hack just to ensure that we don't raise an error when seeing a global var
164        let mut processed_vars = GlobalPcgSettings::new_returning_vars().1;
165
166        // Process all known settings
167        let check_cycles = Self::process_bool_var(&mut processed_vars, "PCG_CHECK_CYCLES", false);
168        let validity_checks = Self::process_bool_var(
169            &mut processed_vars,
170            "PCG_VALIDITY_CHECKS",
171            cfg!(debug_assertions),
172        );
173        let pcg_debug_block = Self::process_debug_block(&mut processed_vars);
174        let debug_imgcat = Self::process_debug_imgcat(&mut processed_vars);
175        let validity_checks_warn_only =
176            Self::process_bool_var(&mut processed_vars, "PCG_VALIDITY_CHECKS_WARN_ONLY", false);
177        let panic_on_error =
178            Self::process_bool_var(&mut processed_vars, "PCG_PANIC_ON_ERROR", false);
179        let polonius = Self::process_bool_var(&mut processed_vars, "PCG_POLONIUS", false);
180        let dump_mir_dataflow =
181            Self::process_bool_var(&mut processed_vars, "PCG_DUMP_MIR_DATAFLOW", false);
182
183        let visualization = Self::process_bool_var(&mut processed_vars, "PCG_VISUALIZATION", false);
184        let visualization_data_dir = Self::process_path_with_default(
185            &mut processed_vars,
186            "PCG_VISUALIZATION_DATA_DIR",
187            PathBuf::from("../visualization/data"),
188        );
189
190        let check_annotations =
191            Self::process_bool_var(&mut processed_vars, "PCG_CHECK_ANNOTATIONS", false);
192        let emit_annotations =
193            Self::process_bool_var(&mut processed_vars, "PCG_EMIT_ANNOTATIONS", false);
194        let check_function = Self::process_string_var(&mut processed_vars, "PCG_CHECK_FUNCTION");
195        let skip_function = Self::process_string_var(&mut processed_vars, "PCG_SKIP_FUNCTION");
196        let coupling = Self::process_bool_var(&mut processed_vars, "PCG_COUPLING", true);
197        let pack_strategy = Self::process_pack_strategy(&mut processed_vars);
198
199        // Check for unknown PCG_ environment variables
200        Self::check_for_unknown_vars(&processed_vars);
201
202        Self {
203            check_cycles,
204            validity_checks,
205            debug_block: pcg_debug_block,
206            debug_imgcat,
207            validity_checks_warn_only,
208            panic_on_error,
209            polonius,
210            dump_mir_dataflow,
211            visualization,
212            visualization_data_dir,
213            check_annotations,
214            emit_annotations,
215            check_function,
216            skip_function,
217            coupling,
218            pack_strategy,
219        }
220    }

222    fn process_bool_var(processed: &mut HashSet<String>, var_name: &str, default: bool) -> bool {
223        processed.insert(var_name.to_owned());
224        env_feature_enabled(var_name).unwrap_or(default)
225    }
226
227    fn process_usize_var(processed: &mut HashSet<String>, var_name: &str) -> Option<usize> {
228        processed.insert(var_name.to_owned());
229        match std::env::var(var_name) {
230            Ok(val) => Some(
231                val.parse()
232                    .unwrap_or_else(|_| panic!("{var_name} must be a valid usize, got: '{val}'")),
233            ),
234            Err(_) => None,
235        }
236    }
237
238    fn process_path_with_default(
239        processed: &mut HashSet<String>,
240        var_name: &str,
241        default: PathBuf,
242    ) -> PathBuf {
243        if let Some(user_path) = Self::process_string_var(processed, var_name) {
244            let user_path = PathBuf::from(user_path);
245            tracing::info!("Using user path for {var_name}: {user_path:?}");
246            match user_path.canonicalize() {
247                Ok(path) => tracing::info!("Absolute path: {path:?}"),
248                Err(e) => tracing::warn!(
249                    "User path {} cannot be canonicalized from current directory {}: {e}",
250                    user_path.display(),
251                    std::env::current_dir().unwrap().display()
252                ),
253            }
254            user_path
255        } else {
256            tracing::info!(
257                "Using default path for {var_name}: {:?}",
258                default.canonicalize()
259            );
260            default
261        }
262    }
263
264    fn process_string_var(processed: &mut HashSet<String>, var_name: &str) -> Option<String> {
265        processed.insert(var_name.to_owned());
266        std::env::var(var_name).ok().filter(|val| !val.is_empty())
267    }
268
269    fn process_debug_block(processed: &mut HashSet<String>) -> Option<BasicBlock> {
270        processed.insert("PCG_DEBUG_BLOCK".to_owned());
271        match std::env::var("PCG_DEBUG_BLOCK") {
272            Ok(val) => {
273                assert!(
274                    val.starts_with("bb"),
275                    "PCG_DEBUG_BLOCK must start with 'bb'"
276                );
277                let block_id: usize = val[2..].parse().unwrap_or_else(|_| {
278                    panic!(
279                        "PCG_DEBUG_BLOCK must be in format 'bbN' where N is a number, got: '{val}'"
280                    )
281                });
282                Some(block_id.into())
283            }
284            Err(_) => None,
285        }
286    }
287
288    fn process_debug_imgcat(processed: &mut HashSet<String>) -> Vec<DebugImgcat> {
289        processed.insert("PCG_DEBUG_IMGCAT".to_owned());
290        match std::env::var("PCG_DEBUG_IMGCAT") {
291            Ok(val) => val
292                .split(',')
293                .map(str::trim)
294                .flat_map(|s| match s.to_lowercase().as_str() {
295                    "true" | "all" => DebugImgcat::all(),
296                    "join_loop" => vec![DebugImgcat::JoinLoop],
297                    "join_owned" => vec![DebugImgcat::JoinOwned],
298                    "join_borrows" => vec![DebugImgcat::JoinBorrows],
299                    _ => panic!("Unexpected value for PCG_DEBUG_IMGCAT: {s}"),
300                })
301                .collect(),
302            Err(_) => vec![],
303        }
304    }
305
306    fn process_pack_strategy(processed: &mut HashSet<String>) -> PackStrategy {
307        processed.insert("PCG_PACK_STRATEGY".to_owned());
308        match std::env::var("PCG_PACK_STRATEGY") {
309            Ok(val) => match val.trim().to_lowercase().as_str() {
310                "eager" => PackStrategy::Eager,
311                "lazy" => PackStrategy::Lazy,
312                other => panic!(
313                    "Environment variable PCG_PACK_STRATEGY has unexpected value: '{other}'. Expected one of: eager, lazy"
314                ),
315            },
316            Err(_) => PackStrategy::default(),
317        }
318    }
319
320    fn check_for_unknown_vars(processed: &HashSet<String>) {
321        let unknown_vars: Vec<String> = std::env::vars()
322            .filter_map(|(key, _)| {
323                (key.starts_with("PCG_") && !processed.contains(&key)).then_some(key)
324            })
325            .collect();
326
327        assert!(
328            unknown_vars.is_empty(),
329            "Unknown PCG_ environment variable(s) found: {}. Known variables are: {}",
330            unknown_vars.iter().join(", "),
331            processed.iter().join(", ")
332        );
333    }
334}

356fn env_feature_enabled(feature: &str) -> Option<bool> {
357    match std::env::var(feature) {
358        Ok(val) => {
359            if val.is_empty() {
360                None
361            } else {
362                match val.as_str() {
363                    "true" | "1" => Some(true),
364                    "false" | "0" => Some(false),
365                    other => panic!(
366                        "Environment variable {feature} has unexpected value: '{other}'. Expected one of: true, false, 1, 0, or empty string"
367                    ),
368                }
369            }
370        }
371        Err(_) => None,
372    }
