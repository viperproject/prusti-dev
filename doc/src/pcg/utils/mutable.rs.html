19struct Upvar<'tcx> {
20    pub(crate) place: CapturedPlace<'tcx>,
21    pub(crate) by_ref: bool,
22}
23
24#[derive(Debug, Clone, Copy, PartialEq, Eq)]
25pub enum LocalMutationIsAllowed {
26    Yes,
27    ExceptUpvars,
28    No,
29}
30
31impl<'a, 'tcx: 'a, BC: Copy> CompilerCtxt<'a, 'tcx, BC> {
32    #[allow(unreachable_patterns)]
33    fn upvars(self) -> Vec<Upvar<'tcx>> {
34        let def = self.body().source.def_id().expect_local();
35        self.tcx
36            .closure_captures(def)
37            .iter()
38            .map(|&captured_place| {
39                let capture = captured_place.info.capture_kind;
40                let by_ref = match capture {
41                    UpvarCapture::ByValue => false,
42                    UpvarCapture::ByRef(..) => true,
43                    _ => todo!(),
44                };
45                Upvar {
46                    place: captured_place.clone(),
47                    by_ref,
48                }
49            })
50            .collect()
51    }
52}

54impl<'tcx> Place<'tcx> {
55    pub fn is_mutable<'a>(
56        self,
57        is_local_mutation_allowed: LocalMutationIsAllowed,
58        ctxt: impl HasCompilerCtxt<'a, 'tcx>,
59    ) -> Result<RootPlace<'tcx>, Self>
60    where
61        'tcx: 'a,
62    {
63        let upvars = ctxt.ctxt().upvars();
64        self.is_mutable_helper(is_local_mutation_allowed, &upvars, ctxt)
65    }
66
67    /// Whether this value can be written or borrowed mutably.
68    /// Returns the root place if the place passed in is a projection.
69    fn is_mutable_helper<'a>(
70        self,
71        is_local_mutation_allowed: LocalMutationIsAllowed,
72        upvars: &[Upvar<'tcx>],
73        ctxt: impl HasCompilerCtxt<'a, 'tcx>,
74    ) -> Result<RootPlace<'tcx>, Self>
75    where
76        'tcx: 'a,
77    {
78        match self.last_projection() {
79            None => {
80                let local = &ctxt.body().local_decls[self.local];
81                match local.mutability {
82                    Mutability::Not => match is_local_mutation_allowed {
83                        LocalMutationIsAllowed::Yes => Ok(RootPlace {
84                            place: self,
85                            is_local_mutation_allowed: LocalMutationIsAllowed::Yes,
86                        }),
87                        LocalMutationIsAllowed::ExceptUpvars => Ok(RootPlace {
88                            place: self,
89                            is_local_mutation_allowed: LocalMutationIsAllowed::ExceptUpvars,
90                        }),
91                        LocalMutationIsAllowed::No => Err(self),
92                    },
93                    Mutability::Mut => Ok(RootPlace {
94                        place: self,
95                        is_local_mutation_allowed,
96                    }),
97                }
98            }
99            Some((place_base, elem)) => {
100                match elem {
101                    ProjectionElem::Deref => {
102                        let base_ty = place_base.ty(ctxt).ty;
103
104                        // Check the kind of deref to decide
105                        match base_ty.kind() {
106                            TyKind::Ref(_, _, mutbl) => {
107                                match mutbl {
108                                    // Shared borrowed data is never mutable
109                                    Mutability::Not => Err(self),
110                                    // Mutably borrowed data is mutable, but only if we have a
111                                    // unique path to the `&mut`
112                                    Mutability::Mut => {
113                                        let mode =
114                                            match self.is_upvar_field_projection(upvars, ctxt) {
115                                                Some(field) if upvars[field.index()].by_ref => {
116                                                    is_local_mutation_allowed
117                                                }
118                                                _ => LocalMutationIsAllowed::Yes,
119                                            };
120
121                                        place_base.is_mutable_helper(mode, upvars, ctxt)
122                                    }
123                                }
124                            }
125                            TyKind::RawPtr(_, mutbl) => {
126                                match mutbl {
127                                    // `*const` raw pointers are not mutable
128                                    Mutability::Not => Err(self),
129                                    // `*mut` raw pointers are always mutable, regardless of
130                                    // context. The users have to check by themselves.
131                                    Mutability::Mut => Ok(RootPlace {
132                                        place: self,
133                                        is_local_mutation_allowed,
134                                    }),
135                                }
136                            }
137                            // `Box<T>` owns its content, so mutable if its location is mutable
138                            _ if base_ty.is_box() => place_base.is_mutable_helper(
139                                is_local_mutation_allowed,
140                                upvars,
141                                ctxt,
142                            ),
143                            // Deref should only be for reference, pointers or boxes
144                            _ => panic!("Deref of unexpected type: {base_ty:?}"),
145                        }
146                    }
147                    // All other projections are owned by their base path, so mutable if
148                    // base path is mutable
149                    ProjectionElem::Field(..)
150                    | ProjectionElem::Index(..)
151                    | ProjectionElem::ConstantIndex { .. }
152                    | ProjectionElem::Subslice { .. }
153                    | ProjectionElem::OpaqueCast { .. }
154                    | ProjectionElem::Downcast(..) => {
155                        let upvar_field_projection = self.is_upvar_field_projection(upvars, ctxt);
156                        if let Some(field) = upvar_field_projection {
157                            let upvar = &upvars[field.index()];
158                            match (upvar.place.mutability, is_local_mutation_allowed) {
159                                (
160                                    Mutability::Not,
161                                    LocalMutationIsAllowed::No
162                                    | LocalMutationIsAllowed::ExceptUpvars,
163                                ) => Err(self),
164                                (Mutability::Not, LocalMutationIsAllowed::Yes)
165                                | (Mutability::Mut, _) => {
166                                    // Subtle: this is an upvar
167                                    // reference, so it looks like
168                                    // `self.foo` -- we want to double
169                                    // check that the location `*self`
170                                    // is mutable (i.e., this is not a
171                                    // `Fn` closure). But if that
172                                    // check succeeds, we want to
173                                    // *blame* the mutability on
174                                    // `place` (that is,
175                                    // `self.foo`). This is used to
176                                    // propagate the info about
177                                    // whether mutability declarations
178                                    // are used outwards, so that we register
179                                    // the outer variable as mutable. Otherwise a
180                                    // test like this fails to record the `mut`
181                                    // as needed:
182                                    //
183                                    // ```
184                                    // fn foo<F: FnOnce()>(_f: F) { }
185                                    // fn main() {
186                                    //     let var = Vec::new();
187                                    //     foo(move || {
188                                    //         var.push(1);
189                                    //     });
190                                    // }
191                                    // ```
192                                    let _ = place_base.is_mutable_helper(
193                                        is_local_mutation_allowed,
194                                        upvars,
195                                        ctxt,
196                                    )?;
197                                    Ok(RootPlace {
198                                        place: self,
199                                        is_local_mutation_allowed,
200                                    })
201                                }
202                            }
203                        } else {
204                            place_base.is_mutable_helper(is_local_mutation_allowed, upvars, ctxt)
205                        }
206                    }
207                    _ => todo!(),
208                }
209            }
210        }
211    }

213    /// If `place` is a field projection, and the field is being projected from a closure type,
214    /// then returns the index of the field being projected. Note that this closure will always
215    /// be `self` in the current MIR, because that is the only time we directly access the fields
216    /// of a closure type.
217    fn is_upvar_field_projection<'a>(
218        self,
219        upvars: &[Upvar<'tcx>],
220        ctxt: impl HasCompilerCtxt<'a, 'tcx>,
221    ) -> Option<FieldIdx>
222    where
223        'tcx: 'a,
224    {
225        let mut place_ref = *self;
226        let mut by_ref = false;
227
228        if let Some((place_base, ProjectionElem::Deref)) = place_ref.last_projection() {
229            place_ref = place_base;
230            by_ref = true;
231        }
232
233        match place_ref.last_projection() {
234            Some((place_base, ProjectionElem::Field(field, _ty))) => {
235                let base_ty = place_base.ty(ctxt.body(), ctxt.tcx()).ty;
236                (base_ty.is_closure() && (!by_ref || upvars[field.index()].by_ref)).then_some(field)
237            }
238            _ => None,
239        }
240    }
