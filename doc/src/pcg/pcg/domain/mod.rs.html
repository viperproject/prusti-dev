63impl<'a, 'tcx> PcgDomainData<'a, 'tcx> {
64    pub(crate) fn from_incoming(
65        self_block: BasicBlock,
66        other: &DomainDataWithCtxt<'a, 'tcx, AnalysisCtxt<'a, 'tcx>>,
67    ) -> Self {
68        pcg_validity_assert!(self_block != other.ctxt.block);
69        let mut entry_state = (*other.data.pcg.states.0.post_main).clone();
70        entry_state
71            .borrow
72            .add_cfg_edge(other.ctxt.block, self_block, other.ctxt);
73        let domain_data = DomainData::new(Rc::new_in(entry_state, other.ctxt.arena));
74        Self {
75            pcg: domain_data,
76            actions: EvalStmtData::default(),
77        }
78    }
79
80    pub(crate) fn start_block(ctxt: AnalysisCtxt<'a, 'tcx>) -> Self {
81        let pcg = Pcg::start_block(ctxt);
82        Self {
83            pcg: DomainData::new(Rc::new_in(pcg, ctxt.arena)),
84            actions: EvalStmtData::default(),
85        }
86    }
87}

96pub(crate) struct BodyAnalysis<'a, 'tcx> {
97    pub(crate) definitely_initialized: DefinitelyInitialized<'a, 'tcx>,
98    pub(crate) place_liveness: PlaceLiveness<'a, 'tcx>,
99    pub(crate) loop_place_usage_analysis: LoopPlaceUsageAnalysis<'tcx>,
100    pub(crate) loop_analysis: LoopAnalysis,
101    loop_invariant_capabilities: IndexVec<LoopId, PlaceUsages<'tcx>>,
102}
103
104impl<'a, 'tcx> BodyAnalysis<'a, 'tcx> {
105    pub(crate) fn new(ctxt: CompilerCtxt<'a, 'tcx>, move_data: &'a MoveData<'tcx>) -> Self {
106        let definitely_initialized = DefinitelyInitialized::new(ctxt.tcx(), ctxt.body(), move_data);
107        let place_liveness = PlaceLiveness::new(ctxt);
108        let loop_analysis = LoopAnalysis::find_loops(ctxt.body());
109        let loop_place_usage_analysis =
110            LoopPlaceUsageAnalysis::new(ctxt.tcx(), ctxt.body(), &loop_analysis);
111        let mut analysis = Self {
112            definitely_initialized,
113            place_liveness,
114            loop_place_usage_analysis,
115            loop_analysis,
116            loop_invariant_capabilities: IndexVec::new(),
117        };
118        let loops = analysis.loop_analysis.all_loops().collect::<Vec<_>>();
119        analysis.loop_invariant_capabilities = loops
120            .into_iter()
121            .map(|loop_id| analysis.compute_loop_invariant_capabilities(loop_id, ctxt))
122            .collect();
123        analysis
124    }
125
126    /// The loop invariant capabilities of `loop_id`: the places used in the
127    /// loop that are live and initialized at its head, consolidated (c.f.
128    /// [`PlaceUsages::consolidate`]) such that no place in the result is a
129    /// prefix of another.
130    ///
131    /// Places behind a raw pointer dereference are excluded: their capabilities
132    /// are governed by the delegation edges of the borrow PCG rather than by
133    /// unpacking the owned PCG.
134    fn compute_loop_invariant_capabilities(
135        &self,
136        loop_id: LoopId,
137        ctxt: CompilerCtxt<'a, 'tcx>,
138    ) -> PlaceUsages<'tcx> {
139        let loop_head_location = mir::Location {
140            block: self.loop_analysis.loop_head_block(loop_id),
141            statement_index: 0,
142        };
143        self.get_places_used_in_loop(loop_id)
144            .usages_where(|usage| {
145                self.is_live_and_initialized_at(loop_head_location, usage.place)
146                    && !usage.place.contains_unsafe_deref(ctxt)
147            })
148            .consolidate()
149    }
150
151    pub(crate) fn loop_invariant_capabilities(&self, loop_id: LoopId) -> &PlaceUsages<'tcx> {
152        &self.loop_invariant_capabilities[loop_id]
153    }
154
155    pub(crate) fn is_read_only_in_loop(&self, block: BasicBlock, place: Place<'tcx>) -> bool {
156        self.loop_analysis.loops(block).any(|loop_id| {
157            let mut overlapping_usages = self
158                .loop_invariant_capabilities(loop_id)
159                .iter()
160                .filter(|usage| usage.place.is_prefix_of(place) || place.is_prefix_of(usage.place))
161                .peekable();
162            if overlapping_usages.peek().is_none() {
163                // No usages in the loop
164                return false;
165            }
166            overlapping_usages.all(|usage| usage.usage.is_read())
167        })
168    }
169
170    /// The places whose expansions must be preserved while `block` executes,
171    /// because the invariant capabilities of a loop containing `block` require
172    /// them to be individually accessible.
173    pub(crate) fn places_required_by_enclosing_loops(
174        &self,
175        block: BasicBlock,
176    ) -> impl Iterator<Item = Place<'tcx>> + '_ {
177        self.loop_analysis
178            .loops(block)
179            .flat_map(|loop_id| self.loop_invariant_capabilities(loop_id).iter_places())
180    }
181
182    pub(crate) fn is_loop_head(&self, block: BasicBlock) -> bool {
183        self.loop_place_usage_analysis.is_loop_head(block)
184    }
185
186    pub(crate) fn get_places_used_in_loop(&self, loop_id: LoopId) -> &PlaceUsages<'tcx> {
187        let loop_head = self.loop_analysis.loop_head_block(loop_id);
188        self.loop_place_usage_analysis
189            .get_used_places(loop_head)
190            .unwrap()
191    }
192
193    pub(crate) fn get_places_used_in_loop_with_head(
194        &self,
195        block: BasicBlock,
196    ) -> Option<&PlaceUsages<'tcx>> {
197        self.loop_analysis
198            .loop_head_of(block)
199            .map(|loop_id| self.get_places_used_in_loop(loop_id))
200    }
201
202    pub(crate) fn is_live_and_initialized_at(
203        &self,
204        location: mir::Location,
205        place: Place<'tcx>,
206    ) -> bool {
207        self.place_liveness.is_live(place, location)
208            && self.is_definitely_initialized_at(location, place)
209    }
210
211    pub(crate) fn is_definitely_initialized_at(
212        &self,
213        location: mir::Location,
214        place: Place<'tcx>,
215    ) -> bool {
216        self.definitely_initialized
217            .is_definitely_initialized(location, place)
218    }
219}

363impl<'a, 'tcx> PendingDataflowState<'a, 'tcx, AnalysisCtxt<'a, 'tcx>> {
364    pub(crate) fn join(
365        self,
366        ctxt: AnalysisCtxt<'a, 'tcx>,
367    ) -> Result<DomainDataWithCtxt<'a, 'tcx, AnalysisCtxt<'a, 'tcx>>, PcgError> {
368        let (mut first, mut rest) = self.take_first();
369        rest.retain(|other| ctxt.should_join_from(other.ctxt.block));
370
371        if !rest.is_empty() {
372            // Determine the common owned state before merging borrow graphs
373            // or applying the loop abstraction.
374            let mut target = (*first.data.pcg.states.0.post_main).clone();
375            for other in &rest {
376                target.join_owned(
377                    &other.data.pcg.states.0.post_main,
378                    SnapshotLocation::BeforeJoin(ctxt.block),
379                    SnapshotLocation::After(other.ctxt.block),
380                    ctxt,
381                )?;
382            }
383
384            // Preserve each predecessor's values before its edge repacks.
385            for incoming in std::iter::once(&mut first).chain(rest.iter_mut()) {
386                Rc::make_mut(&mut incoming.data.pcg.states.0.post_main).join_owned(
387                    &target,
388                    SnapshotLocation::After(incoming.ctxt.block),
389                    SnapshotLocation::BeforeJoin(ctxt.block),
390                    ctxt,
391                )?;
392            }
393        }
394
395        let mut result: DomainDataWithCtxt<'a, 'tcx, AnalysisCtxt<'a, 'tcx>> =
396            DomainDataWithCtxt::new(PcgDomainData::from_incoming(ctxt.block, &first), ctxt);
397        let curr = Rc::make_mut(&mut result.data.pcg.entry_state);
398        for other in &rest {
399            curr.merge_borrows(&other.data.pcg.states.0.post_main, other.ctxt.block, ctxt);
400        }
401        if !rest.is_empty() || ctxt.body_analysis.is_loop_head(ctxt.block) {
402            curr.finish_join(ctxt);
403        }
404        Ok(result)
405    }
406}

414impl<'a, 'tcx> DataflowState<'a, 'tcx, AnalysisCtxt<'a, 'tcx>> {
415    pub(crate) fn ensure_transfer(
416        &mut self,
417        ctxt: AnalysisCtxt<'a, 'tcx>,
418    ) -> Result<&mut DomainDataWithCtxt<'a, 'tcx, AnalysisCtxt<'a, 'tcx>>, PcgError> {
419        match self {
420            DataflowState::Pending(pending_state) => {
421                let pending_state = pending_state.take();
422                *self = DataflowState::Transfer(pending_state.join(ctxt)?);
423                Ok(self.expect_transfer_mut())
424            }
425            DataflowState::Transfer(domain_data_with_ctxt) => Ok(domain_data_with_ctxt),
426        }
427    }
428
429    pub(crate) fn take_into_data(
430        self,
431        ctxt: AnalysisCtxt<'a, 'tcx>,
432    ) -> Result<DomainDataWithCtxt<'a, 'tcx, AnalysisCtxt<'a, 'tcx>>, PcgError> {
433        match self {
434            DataflowState::Pending(pending) => Ok(pending.join(ctxt)?),
435            DataflowState::Transfer(transfer) => Ok(transfer),
436        }
437    }
438}

532#[derive(From, Clone)]
533pub enum PcgDomain<'a, 'tcx> {
534    Bottom,
535    Error(PcgError),
536    Analysis(DataflowState<'a, 'tcx, AnalysisCtxt<'a, 'tcx>>),
537    Results(DomainDataWithCtxt<'a, 'tcx, ResultsCtxt<'a, 'tcx>>),
538}
539
540impl Eq for PcgDomain<'_, '_> {}
541
542impl PartialEq for PcgDomain<'_, '_> {
543    fn eq(&self, other: &Self) -> bool {
544        match (self, other) {
545            (PcgDomain::Bottom, PcgDomain::Bottom) => true,
546            (PcgDomain::Error(e1), PcgDomain::Error(e2)) => e1 == e2,
547            (PcgDomain::Analysis(d1), PcgDomain::Analysis(d2)) => d1 == d2,
548            (PcgDomain::Results(d1), PcgDomain::Results(d2)) => d1 == d2,
549            _ => false,
550        }
551    }
552}

554impl<'a, 'tcx: 'a> PcgDomain<'a, 'tcx> {
555    pub(crate) fn is_results(&self) -> bool {
556        matches!(self, PcgDomain::Results(_))
557    }
558
559    pub(crate) fn take(&mut self) -> PcgDomain<'a, 'tcx> {
560        std::mem::replace(self, PcgDomain::Bottom)
561    }
562
563    pub(crate) fn bottom() -> Self {
564        Self::Bottom
565    }
566    pub(crate) fn record_error(&mut self, error: PcgError) {
567        *self = Self::Error(error);
568    }
569    pub(crate) fn data_mut(&mut self) -> &mut PcgDomainData<'a, 'tcx> {
570        match self {
571            PcgDomain::Analysis(dataflow_state) => &mut dataflow_state.expect_transfer_mut().data,
572            PcgDomain::Results(data) => &mut data.data,
573            _ => panic!("Expected analysis or results domain, got {self:?}"),
574        }
575    }
576    pub(crate) fn complete(&mut self, ctxt: AnalysisCtxt<'a, 'tcx>) {
577        if self.is_bottom() || self.is_error() {
578            return;
579        }
580        let domain = self.take();
581        *self = PcgDomain::Results(
582            domain
583                .take_analysis()
584                .take_into_data(ctxt)
585                .unwrap()
586                .into_results(),
587        );
588    }
589}

614        &self.data().pcg[phase.into()]
615    }
616
617    #[cfg(feature = "visualization")]
618    fn generate_dot_graph(
619        &self,
620        phase: DataflowStmtPhase,
621        location: mir::Location,
622        ctxt: AnalysisCtxt<'a, 'tcx>,
623    ) {
624        let index = match phase {
625            DataflowStmtPhase::EvalStmt(phase) => DomainDataIndex::Eval(phase),
626            _ => DomainDataIndex::Initial,
627        };
628        ctxt.generate_pcg_debug_visualization_graph(
629            location,
630            stmt_graphs::ToGraph::Phase(phase),
631            self.pcg(index).into(),
632        );
633    }

641impl ErrorState {

643        self.error.as_ref()
644    }

646    pub(crate) fn record_error(&mut self, error: PcgError) {
647        assert!(!*PANIC_ON_ERROR, "PCG Error: {error:?}");
648        tracing::error!("PCG Error: {error:?}");
649        self.error = Some(error);
650    }

653impl<'a, 'tcx> PcgDomain<'a, 'tcx> {
654    pub(crate) fn error(&self) -> Option<&PcgError> {
655        match self {
656            PcgDomain::Error(e) => Some(e),
657            _ => None,
658        }
659    }
660    pub(crate) fn is_bottom(&self) -> bool {
661        matches!(self, PcgDomain::Bottom)
662    }
663
664    pub(crate) fn is_error(&self) -> bool {
665        matches!(self, PcgDomain::Error(_))
666    }
667
668    pub(crate) fn take_analysis(self) -> DataflowState<'a, 'tcx, AnalysisCtxt<'a, 'tcx>> {
669        match self {
670            PcgDomain::Analysis(dataflow_state) => dataflow_state,
671            _ => panic!("Expected analysis domain, got {self:?}"),
672        }
673    }
674
675    pub(crate) fn expect_results_or_error(
676        &self,
677    ) -> Result<&DomainDataWithCtxt<'a, 'tcx, ResultsCtxt<'a, 'tcx>>, PcgError> {
678        match self {
679            PcgDomain::Results(dataflow_state) => Ok(dataflow_state),
680            PcgDomain::Error(pcg_error) => Err(pcg_error.clone()),
681            _ => panic!("Expected results or error domain, got {self:?}"),
682        }
683    }
684}
685
686impl JoinSemiLattice for PcgDomain<'_, '_> {
687    fn join(&mut self, other: &Self) -> bool {
688        match other {
689            PcgDomain::Bottom => false,
690            PcgDomain::Error(e) => {
691                self.record_error(e.clone());
692                false
693            }
694            PcgDomain::Analysis(dataflow_state) => {
695                let other = dataflow_state.expect_transfer();
696                match self {
697                    PcgDomain::Bottom => {
698                        *self =
699                            PcgDomain::Analysis(DataflowState::Pending(vec![other.clone()].into()));
700                        true
701                    }
702                    PcgDomain::Error(_) => false,
703                    PcgDomain::Analysis(dataflow_state) => {
704                        let self_dataflow_state = dataflow_state.expect_pending_mut();
705                        self_dataflow_state.push(other.clone());
706                        false
707                    }
708                    PcgDomain::Results(_) => unreachable!(),
709                }
710            }
711            PcgDomain::Results(_) => unreachable!(),
712        }
713    }
714}
