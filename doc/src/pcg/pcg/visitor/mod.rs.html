49impl<'pcg, 'a, 'tcx: 'a, Ctxt: DataflowCtxt<'a, 'tcx>> PcgVisitor<'pcg, 'a, 'tcx, Ctxt> {
50    pub(crate) fn visit(
51        pcg: &'pcg mut Pcg<'a, 'tcx>,
52        tw: &'pcg TripleWalker<'a, 'tcx>,
53        analysis_location: AnalysisLocation,
54        analysis_object: AnalysisObject<'_, 'tcx>,
55        ctxt: Ctxt,
56    ) -> Result<AppliedActions<'tcx>, PcgError> {
57        let visitor = Self::new(pcg, ctxt, tw, analysis_location);
58        let actions = visitor.apply(analysis_object)?;
59        Ok(actions)
60    }

62    pub(crate) fn new(
63        pcg: &'pcg mut Pcg<'a, 'tcx>,
64        ctxt: Ctxt,
65        tw: &'pcg TripleWalker<'a, 'tcx>,
66        analysis_location: AnalysisLocation,
67    ) -> Self {
68        Self {
69            pcg,
70            ctxt,
71            actions: AppliedActions::default(),
72            analysis_location,
73            tw,
74        }
75    }

89    fn perform_borrow_initial_pre_operand_actions(&mut self) -> Result<(), PcgError> {
90        self.place_obtainer()
91            .pack_old_and_dead_borrow_leaves(None)?;
92        for created_location in self.ctxt.bc().twophase_borrow_activations(self.location()) {
93            self.activate_twophase_borrow_created_at(created_location)?;
94        }
95        let frozen_graph = self.pcg.borrow.graph.frozen_graph();
96        let leaf_nodes = frozen_graph.leaf_nodes(self.ctxt.bc_ctxt());
97        let leaf_future_node_places = leaf_nodes
98            .iter()
99            .filter_map(|node| match node {
100                PcgNode::Place(_) => None,
101                PcgNode::LifetimeProjection(region_projection) => {
102                    if region_projection.is_future() {
103                        region_projection.base.as_current_place()
104                    } else {
105                        None
106                    }
107                }
108            })
109            .collect::<HashSet<_>>();
110        for place in leaf_future_node_places {
111            // If the place is a leaf, and it's not borrowed, then we remove the future label
112            if self.pcg.is_expansion_leaf(place, self.ctxt)
113                && !self
114                    .ctxt
115                    .bc()
116                    .is_directly_blocked(place, self.location(), self.ctxt.bc_ctxt())
117                && self.pcg.capability_of(place, self.ctxt).is_none()
118            {
119                let action = PcgAction::restore_capability(
120                    place,
121                    CapabilityKind::Exclusive,
122                    "Leaf future node restore cap",
123                    self.ctxt,
124                );
125                self.record_and_apply_action(action)?;
126            }
127        }
128        Ok(())
129    }
130
131    pub(crate) fn apply(
132        mut self,
133        object: AnalysisObject<'_, 'tcx>,
134    ) -> Result<AppliedActions<'tcx>, PcgError> {
135        match self.phase() {
136            EvalStmtPhase::PreOperands => {
137                self.perform_borrow_initial_pre_operand_actions()?;
138                self.place_obtainer().collapse_owned_places()?;
139                for triple in &self.tw.operand_triples {
140                    tracing::debug!("Require triple {triple:?}");
141                    self.require_triple(*triple)?;
142                }
143            }
144            EvalStmtPhase::PostOperands => {
145                for triple in &self.tw.operand_triples {
146                    self.ensure_triple(*triple)?;
147                }
148            }
149            EvalStmtPhase::PreMain => {
150                for triple in &self.tw.main_triples {
151                    tracing::debug!("Require triple {triple:?}");
152                    self.require_triple(*triple)?;
153                }
154            }
155            EvalStmtPhase::PostMain => {
156                for triple in &self.tw.main_triples {
157                    self.ensure_triple(*triple)?;
158                }
159            }
160        }
161        let location = self.location();
162        match object {
163            AnalysisObject::Statement(statement) => {
164                self.visit_statement_fallable(statement, location)?;
165            }
166            AnalysisObject::Terminator(terminator) => {
167                self.visit_terminator_fallable(terminator, location)?;
168                if self.phase() == EvalStmtPhase::PostMain {
169                    // when the analysis object is a terminator, this step ensures
170                    // the owned PCG is in the most-packed state for the join.
171                    self.place_obtainer().collapse_owned_places()?;
172                }
173            }
174        }
175        Ok(self.actions)
176    }
177
178    fn outlives(&self, sup: PcgRegion<'tcx>, sub: PcgRegion<'tcx>) -> bool {
179        self.ctxt.bc().outlives(sup, sub, self.location())
180    }
181
182    fn connect_outliving_projections(
183        &mut self,
184        source_proj: LifetimeProjection<'tcx, PlaceOrConst<'tcx, MaybeLabelledPlace<'tcx>>>,
185        target: Place<'tcx>,
186        kind: impl Fn(PcgRegion) -> BorrowFlowEdgeKind<'tcx>,
187    ) -> Result<(), PcgError> {
188        let ctxt = self.ctxt;
189        for target_proj in target.lifetime_projections(self.ctxt) {
190            if self.outlives(
191                source_proj.region(ctxt.ctxt()),
192                target_proj.region(ctxt.ctxt()),
193            ) {
194                self.record_and_apply_action(
195                    BorrowPcgAction::add_edge(
196                        BorrowPcgEdge::new(
197                            BorrowFlowEdge::new(
198                                source_proj.to_pcg_lifetime_projection(),
199                                target_proj.into(),
200                                kind(target_proj.region(self.ctxt.ctxt())),
201                            )
202                            .into(),
203                            self.pcg.borrow.validity_conditions.clone(),
204                        ),
205                        "connect_outliving_projections",
206                    )
207                    .into(),
208                )?;
209            }
210            if let PlaceOrConst::Place(base) = source_proj.base {
211                let local_source_proj = source_proj.with_base(base);
212                self.place_obtainer().redirect_source_of_future_edges(
213                    local_source_proj,
214                    target_proj.into(),
215                    ctxt,
216                )?;
217            }
218        }
219        Ok(())
220    }
221}

223impl<'a, 'tcx: 'a, Ctxt: DataflowCtxt<'a, 'tcx>> FallableVisitor<'tcx>
224    for PcgVisitor<'_, 'a, 'tcx, Ctxt>
225{
226    #[tracing::instrument(skip(self, _location))]
227    fn visit_statement_fallable(
228        &mut self,
229        statement: &Statement<'tcx>,
230        _location: Location,
231    ) -> Result<(), PcgError> {
232        self.perform_statement_actions(statement)
233    }
234
235    fn visit_operand_fallable(
236        &mut self,
237        operand: &Operand<'tcx>,
238        location: Location,
239    ) -> Result<(), PcgError> {
240        if let Operand::Move(place) = operand
241            && self.phase() == EvalStmtPhase::PostOperands
242        {
243            let snapshot_location = self.prev_snapshot_location();
244            self.record_and_apply_action(
245                BorrowPcgAction::label_place_and_update_related_capabilities(
246                    (*place).into(),
247                    snapshot_location,
248                    LabelPlaceReason::MoveOut,
249                )
250                .into(),
251            )?;
252        }
253        self.super_operand_fallable(operand, location);
254        Ok(())
255    }
256
257    fn visit_terminator_fallable(
258        &mut self,
259        terminator: &Terminator<'tcx>,
260        location: Location,
261    ) -> Result<(), PcgError> {
262        self.super_terminator_fallable(terminator, location)?;
263        if self.phase() == EvalStmtPhase::PostMain
264            && let mir::TerminatorKind::Call {
265                func,
266                args,
267                destination,
268                fn_span,
269                ..
270            } = &terminator.kind
271        {
272            let destination: Place<'tcx> = (*destination).into();
273            self.make_function_call_abstraction(
274                func,
275                *fn_span,
276                &args.iter().map(|arg| &arg.node).collect::<Vec<_>>(),
277                destination,
278                location,
279            )?;
280        }
281        Ok(())
282    }
283
284    fn visit_place_fallable(
285        &mut self,
286        _place: Place<'tcx>,
287        _context: mir::visit::PlaceContext,
288        _location: Location,
289    ) {
290    }
291
292    fn visit_rvalue_fallable(
293        &mut self,
294        rvalue: &Rvalue<'tcx>,
295        location: Location,
296    ) -> Result<(), PcgError> {
297        if matches!(rvalue, Rvalue::Ref(_, mir::BorrowKind::Fake(_), _)) {
298            return Ok(());
299        }
300        self.super_rvalue_fallable(rvalue, location)?;
301        Ok(())
302    }
303}

305impl<'state, 'a: 'state, 'tcx: 'a, Ctxt: DataflowCtxt<'a, 'tcx>>
306    PlaceObtainer<'state, 'a, 'tcx, Ctxt>
307{
308    /// The places that must not be collapsed here because they are part of the
309    /// loop invariant. For example, if the loop invariant requires `x.f` but
310    /// NOT `x.g`, then `x` must not be collapsed (because the loop invariant
311    /// does not provide permission to `x.g`).
312    fn places_required_by_enclosing_loops(&self) -> Vec<Place<'tcx>> {
313        self.ctxt
314            .body_analysis()
315            .places_required_by_enclosing_loops(self.location().block)
316            .collect()
317    }
318
319    fn collapse_iteration(
320        &mut self,
321        local: mir::Local,
322        iteration: usize,
323        places_to_keep_unpacked: &[Place<'tcx>],
324    ) -> Result<bool, PcgError> {
325        let local_expansions = self.pcg.owned[local].expansions();
326        let leaf_expansions = local_expansions.leaf_expansions(self.ctxt);
327        let parent_places = leaf_expansions
328            .iter()
329            .map(|pe| pe.place)
330            .collect::<HashSet<_>>();
331        let places_to_collapse = parent_places
332            .into_iter()
333            .filter(|place| {
334                !places_to_keep_unpacked
335                    .iter()
336                    .any(|required| place.is_strict_prefix_of(*required))
337            })
338            .filter_map(|place| {
339                let expansion_places = local_expansions.all_children_of(place, self.ctxt);
340                if expansion_places
341                    .iter()
342                    .all(|p| !self.pcg.borrow.graph.contains(*p, self.ctxt.bc_ctxt()))
343                    && let Some(candidate_cap) = self
344                        .pcg
345                        .place_capabilities
346                        .uniform_capability(expansion_places.into_iter(), self.ctxt)
347                {
348                    Some((place, candidate_cap))
349                } else {
350                    None
351                }
352            })
353            .collect::<HashSet<_>>();
354        if places_to_collapse.is_empty() {
355            return Ok(false);
356        }
357        for (place, candidate_cap) in places_to_collapse {
358            self.collapse_owned_places_and_lifetime_projections_to(
359                place,
360                candidate_cap,
361                format!(
362                    "Collapse owned place {} (iteration {})",
363                    place.display_string(self.ctxt),
364                    iteration
365                ),
366                self.ctxt,
367            )?;
368            if place.projection.is_empty()
369                && self.pcg.capability_of(place, self.ctxt) == Some(CapabilityKind::Read)
370                && !self
371                    .ctxt
372                    .body_analysis()
373                    .is_read_only_in_loop(self.location().block, place)
374            {
375                self.pcg
376                    .place_capabilities
377                    .insert(place, CapabilityKind::Exclusive, self.ctxt);
378            }
379        }
380        Ok(true)
381    }
382}

384impl<'state, 'a: 'state, 'tcx: 'a, Ctxt: DataflowCtxt<'a, 'tcx>>
385    PlaceObtainer<'state, 'a, 'tcx, Ctxt>
386{
387    pub(crate) fn collapse_owned_places(&mut self) -> Result<(), PcgError> {
388        let allocated_locals = self.pcg.owned.allocated_locals();
389        let places_to_keep_unpacked = self.places_required_by_enclosing_loops();
390        for local in allocated_locals {
391            self.iterations_for_local(local, &places_to_keep_unpacked)?;
392        }
393        Ok(())
394    }
395
396    fn iterations_for_local(
397        &mut self,
398        local: mir::Local,
399        places_to_keep_unpacked: &[Place<'tcx>],
400    ) -> Result<(), PcgError> {
401        let mut iteration = 1;
402        while self.collapse_iteration(local, iteration, places_to_keep_unpacked)? {
403            iteration += 1;
404        }
405        Ok(())
406    }
407}

409impl<'a, 'tcx: 'a, Ctxt: DataflowCtxt<'a, 'tcx>> PcgVisitor<'_, 'a, 'tcx, Ctxt> {
410    fn activate_twophase_borrow_created_at(
411        &mut self,
412        created_location: Location,
413    ) -> Result<(), PcgError> {
414        let Some(borrow) = self.pcg.borrow_created_at(created_location) else {
415            return Ok(());
416        };
417        tracing::info!(
418            "{:?} activate twophase borrow: {}",
419            self.location(),
420            borrow.display_string(self.ctxt)
421        );
422        let blocked_place = borrow.blocked_place.place();
423        self.place_obtainer()
424            .remove_read_permission_upwards_and_label_rps(
425                blocked_place,
426                AdjustCapabilityReason::TwoPhaseBorrowActivation,
427            )?;
428        Ok(())
429    }
430}
