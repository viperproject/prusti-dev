34impl<'a, 'tcx: 'a, Ctxt: DataflowCtxt<'a, 'tcx>> PcgVisitor<'_, 'a, 'tcx, Ctxt> {
35    pub(crate) fn settings(&self) -> &'a PcgSettings {
36        self.ctxt.settings()
37    }
38
39    fn node_for_input(
40        &self,
41        call: &FunctionCall<'_, 'tcx>,
42        input: LifetimeProjection<'tcx, ArgIdx>,
43    ) -> FunctionCallAbstractionInput<'tcx> {
44        let operand = call.inputs[*input.base];
45        let operand = self.maybe_labelled_operand(operand);
46        FunctionCallAbstractionInput(
47            LifetimeProjection::from_index(operand, input.region_idx)
48                .with_label(Some(self.prev_snapshot_location().into()), self.ctxt),
49        )
50    }
51
52    fn node_for_output(
53        &self,
54        call: &FunctionCall<'_, 'tcx>,
55        output: LifetimeProjection<'tcx, ArgIdxOrResult>,
56    ) -> FunctionCallAbstractionOutput<'tcx> {
57        match output.base {
58            ArgIdxOrResult::Argument(arg_idx) => {
59                let operand = call.inputs[*arg_idx];
60                let place = self.maybe_labelled_operand(operand).expect_place();
61                LifetimeProjection::from_index(place, output.region_idx)
62                    .with_label(
63                        Some(SnapshotLocation::After(self.location().block).into()),
64                        self.ctxt,
65                    )
66                    .into()
67            }
68            ArgIdxOrResult::Result => {
69                LifetimeProjection::from_index(call.output, output.region_idx).into()
70            }
71        }
72    }
73
74    fn create_edges_for_shape(
75        &mut self,
76        shape: &FunctionShape,
77        call: &FunctionCall<'_, 'tcx>,
78    ) -> Result<(), PcgError> {
79        let metadata = FunctionCallAbstractionEdgeMetadata {
80            location: call.location,
81            defined_fn_call: call.defined,
82        };
83        let abstraction_edges: HashSet<AbstractionBlockEdge<'_, _, _>> = shape
84            .edges()
85            .map(|AbstractionBlockEdge { input, output, .. }| {
86                AbstractionBlockEdge::new_checked(
87                    self.node_for_input(call, input),
88                    self.node_for_output(call, output),
89                    self.ctxt,
90                )
91            })
92            .collect();
93        if self.settings().coupling {
94            let coupled_edges = CoupledEdgesData::new(abstraction_edges.iter().copied());
95            if !coupled_edges.is_empty() {
96                tracing::debug!("Coupled edges: {coupled_edges:?}");
97            }
98            for edge in coupled_edges {
99                let pcg_coupled_edge = PcgCoupledEdgeKind::function_call(
100                    FunctionCallCoupledEdgeKind::new(metadata, edge),
101                );
102                self.record_and_apply_action(
103                    BorrowPcgAction::add_edge(
104                        BorrowPcgEdge::new(
105                            pcg_coupled_edge.into(),
106                            self.pcg.borrow.validity_conditions.clone(),
107                        ),
108                        "Function call",
109                    )
110                    .into(),
111                )?;
112            }
113        } else {
114            for edge in abstraction_edges {
115                self.record_and_apply_action(
116                    BorrowPcgAction::add_edge(
117                        BorrowPcgEdge::new(
118                            AbstractionEdge::FunctionCall(FunctionCallAbstraction::new(
119                                metadata, edge,
120                            ))
121                            .into(),
122                            self.pcg.borrow.validity_conditions.clone(),
123                        ),
124                        "Function call",
125                    )
126                    .into(),
127                )?;
128            }
129        }
130        Ok(())
131    }
132    #[tracing::instrument(skip(self, func, args, destination))]
133    pub(super) fn make_function_call_abstraction(
134        &mut self,
135        func: &Operand<'tcx>,
136        call_span: Span,
137        args: &[&Operand<'tcx>],
138        destination: utils::Place<'tcx>,
139        location: Location,
140    ) -> Result<(), PcgError> {
141        let call = FunctionCall::new(
142            location,
143            args,
144            destination,
145            match func.ty(self.ctxt.body(), self.ctxt.tcx()).kind() {
146                ty::TyKind::FnDef(def_id, substs) => Some(DefinedFnCall::new(
147                    FunctionData::new(*def_id),
148                    fn_def_args(substs),
149                    self.ctxt.ctxt().def_id(),
150                    call_span,
151                )),
152                _ => None,
153            },
154        );
155
156        let ctxt = self.ctxt;
157
158        // The versions of the region projections for the function inputs just
159        // before they were moved out, labelled with their last modification
160        // time
161        let arg_region_projections = args
162            .iter()
163            .map(|arg| self.maybe_labelled_operand(arg))
164            .flat_map(|input_place| self.ctxt.extract_lifetime_projections(input_place))
165            .collect::<Vec<_>>();
166
167        let pre_rps = arg_region_projections
168            .iter()
169            .map(|rp| {
170                rp.with_label(
171                    Some(self.place_obtainer().prev_snapshot_location().into()),
172                    self.ctxt,
173                )
174            })
175            .collect::<Vec<_>>();
176
177        let post_rps = arg_region_projections
178            .iter()
179            .map(|rp| {
180                rp.with_label(
181                    Some(SnapshotLocation::After(self.location().block).into()),
182                    self.ctxt,
183                )
184            })
185            .collect::<Vec<_>>();
186
187        for (rp, post_rp) in arg_region_projections.iter().zip(post_rps.iter()) {
188            if let (Some(rp), Some(post_rp)) = (
189                rp.try_to_local_lifetime_projection(),
190                post_rp.try_to_local_lifetime_projection(),
191            ) {
192                self.place_obtainer()
193                    .redirect_source_of_future_edges(rp, post_rp, ctxt)?;
194            }
195        }
196
197        for (rp, pre_rp) in arg_region_projections.iter().zip(pre_rps.iter()) {
198            if let Some(rp) = rp.try_to_local_lifetime_projection() {
199                self.record_and_apply_action(
200                    BorrowPcgAction::label_lifetime_projection(
201                        LabelNodePredicate::equals_lifetime_projection(rp),
202                        pre_rp.label(),
203                        format!(
204                            "Function call:Label Pre version of {}",
205                            rp.display_string(self.ctxt),
206                        ),
207                    )
208                    .into(),
209                )?;
210            }
211        }
212        let shape = call.shape(self.ctxt);
213        self.create_edges_for_shape(&shape, &call)?;
214
215        Ok(())
216    }
