109    fn restore_place(&mut self, place: Place<'tcx>, context: &str) -> Result<(), PcgError> {
110        // The place to restore could come from a local that was conditionally
111        // allocated and therefore we can't get back to it, and certainly
112        // shouldn't give it any capability
113        // TODO: Perhaps the join should label such places?
114        if !self.pcg.owned.is_allocated(place.local) {
115            return Ok(());
116        }
117        let blocked_cap = self.pcg.capability_of(place, self.ctxt);
118
119        // TODO: If the place projects a shared ref, do we even need to restore a capability?
120        //
121        // For owned places we consult the initialisation state so we
122        // don't unsoundly restore to `Exclusive` when the place was
123        // moved out on some incoming path (#137): the join of the init
124        // trees would then leave the place as `Uninit`, which maps to
125        // `Write`.
126        let restore_cap = if place.place().projects_shared_ref(self.ctxt)
127            || self
128                .ctxt
129                .body_analysis()
130                .is_read_only_in_loop(self.location().block, place)
131        {
132            CapabilityKind::Read
133        } else if let Some(owned) = place.as_owned_place(self.ctxt)
134            && let Some(init_cap) = self.pcg.owned.owned_capability(owned)
135        {
136            init_cap
137        } else {
138            CapabilityKind::Exclusive
139        };
140
141        // The blocked capability would be None if the place was mutably
142        // borrowed. The capability would be ShallowExclusive if the place is
143        // a mutable reference (when dereferencing a mutable ref, the ref
144        // place retains shallow-exclusive capability).
145        if matches!(
146            blocked_cap,
147            None | Some(CapabilityKind::Write | CapabilityKind::ShallowExclusive)
148        ) && blocked_cap != Some(restore_cap)
149        {
150            self.record_and_apply_action(PcgAction::restore_capability(
151                place,
152                restore_cap,
153                context.to_owned(),
154                self.ctxt,
155            ))?;
156        }
157        for rp in place.lifetime_projections(self.ctxt) {
158            self.record_and_apply_action(
159                    BorrowPcgAction::remove_lifetime_projection_label(
160                        rp.with_placeholder_label(self.ctxt).into(),
161                        format!(
162                            "Place {} unblocked: remove placeholder label of rps of newly unblocked nodes",
163                            place.display_string(self.ctxt.bc_ctxt())
164                        ),
165                    )
166                    .into(),
167                )?;
168        }
169        Ok(())
170    }
171    fn update_unblocked_node_capabilities_and_remove_placeholder_projections(
172        &mut self,
173        edge: &BorrowPcgEdgeKind<'tcx>,
174    ) -> Result<(), PcgError> {
175        // Snapshot the borrow-graph places *before* acquiring `fg` so the
176        // owned HashSet doesn't hold a borrow of the graph across the
177        // restore_place call below.
178        let borrow_places = self.pcg.borrow.graph.places(self.ctxt.bc_ctxt());
179
180        let fg = self.pcg.borrow.graph.frozen_graph();
181        let blocked_nodes = edge.blocked_nodes(self.ctxt.bc_ctxt());
182
183        // After removing an edge, some nodes may become accessible, their capabilities should be restored
184        let to_restore = blocked_nodes
185            .into_iter()
186            .filter(|node| !fg.has_edge_blocking(*node, self.ctxt.bc_ctxt()))
187            .collect::<Vec<_>>();
188
189        for node in to_restore {
190            if let Some(place) = node.as_current_place() {
191                // Only restore if the place is still tracked somewhere.  After
192                // a loop abstraction edge is removed, the blocked place (e.g.
193                // `(*variant).fields`) may no longer appear in either the owned
194                // PCG or the borrow graph; restoring a capability there would
195                // violate the invariant that every capability-holding place must
196                // be tracked.
197                if !borrow_places.contains(&place)
198                    && !self.pcg.owned.contains_place(place, self.ctxt.bc_ctxt())
199                {
200                    continue;
201                }
202                self.restore_place(
203                    place,
204                    "update_unblocked_node_capabilities_and_remove_placeholder_projections",
205                )?;
206            }
207        }
208        Ok(())
209    }

211    /// If the following conditions apply:
212    /// 1. `expansion` is a dereference of a place `p`
213    /// 2. `*p` does not contain any borrows
214    /// 3. The target of this expansion is not labelled
215    ///
216    /// Then we perform an optimization where instead of connecting the blocked
217    /// lifetime projection to the current one, we instead remove the label of
218    /// the blocked lifetime projection.
219    ///
220    /// This is sound because the lifetime projection only contains the single
221    /// borrow that `p` refers to and therefore the set of borrows cannot be
222    /// changed. In other words, the set of borrows in the lifetime projection
223    /// at the point it was dereferenced is the same as the current set of
224    /// borrows in the lifetime projection.
225    ///
226    /// Note the third condition: if the expansion is labelled, that indicates
227    /// that the expansion occurred at a point where `p` had a different value
228    /// than the current one. We don't want to perform this optimization because
229    /// the it is referring to this different value.
230    /// For test case see rustls-pki-types@1.11.0 `server_name::parser::Parser::`<'`a>::read_char`
231    ///
232    /// TODO: In the above test case, should the parent place also be labelled?
233    fn unlabel_blocked_region_projections_if_applicable(
234        &mut self,
235        deref: &DerefEdge<'tcx>,
236        context: &str,
237    ) -> Result<(), PcgError> {
238        if deref.deref_place.is_current()
239            && deref.deref_place.lifetime_projections(self.ctxt).is_empty()
240        {
241            self.unlabel_source_lifetime_projections(deref, context)
242        } else {
243            Ok(())
244        }
245    }

247    pub(crate) fn remove_deref_edges_to(
248        &mut self,
249        deref_place: MaybeLabelledPlace<'tcx>,
250        edges: HashSet<Conditioned<DerefEdge<'tcx>>>,
251        context: &str,
252    ) -> Result<(), PcgError> {
253        for edge in edges {
254            let borrow_edge: BorrowPcgEdge<'tcx> =
255                BorrowPcgEdge::new(edge.value.into(), edge.conditions);
256            self.apply_action(
257                BorrowPcgAction::remove_edge(borrow_edge, context.to_owned()).into(),
258            )?;
259            self.unlabel_blocked_region_projections_if_applicable(&edge.value, context)?;
260        }
261        if let Some(deref_place) = deref_place.as_current_place() {
262            self.apply_action(
263                BorrowPcgAction::label_place_and_update_related_capabilities(
264                    deref_place,
265                    self.prev_snapshot_location(),
266                    LabelPlaceReason::Collapse,
267                )
268                .into(),
269            )?;
270            let ref_place = deref_place.parent_place().unwrap();
271            // Perhaps the place isn't yet a leaf(e.g. the ref itself is conditionally borrowed)
272            // In that case we shouldn't restore its caps
273            if self.pcg.is_leaf_place(ref_place, self.ctxt) {
274                self.restore_place(
275                    ref_place,
276                    &format!("{context}: remove_deref_edges_to: restore parent place"),
277                )?;
278            }
279        }
280        Ok(())
281    }

283    #[tracing::instrument(skip(self, edge))]
284    pub(crate) fn remove_edge_and_perform_associated_state_updates(
285        &mut self,
286        edge: &BorrowPcgEdge<'tcx>,
287        context: &str,
288    ) -> Result<(), PcgError> {
289        if let BorrowPcgEdgeKind::Deref(deref) = edge.kind() {
290            return self.remove_deref_edges_to(
291                deref.deref_place,
292                vec![Conditioned::new(*deref, edge.conditions.clone())]
293                    .into_iter()
294                    .collect(),
295                context,
296            );
297        }
298        self.record_and_apply_action(
299            BorrowPcgAction::remove_edge(edge.clone(), context.to_owned()).into(),
300        )?;
301
302        // This is true iff the expansion is for a place (not a region projection), and changes
303        // could have been made to the root place via the expansion
304        // We check that the base is place and either:
305        // - The base has no capability, meaning it was previously expanded mutably
306        // - The base has write capability, it is a mutable ref
307        let is_mutable_place_expansion = if let BorrowPcgEdgeKind::BorrowPcgExpansion(expansion) =
308            edge.kind()
309            && let Some(place) = expansion.base().as_current_place()
310        {
311            matches!(
312                self.pcg.capability_of(place, self.ctxt),
313                Some(CapabilityKind::Write) | None
314            )
315        } else {
316            false
317        };
318
319        self.update_unblocked_node_capabilities_and_remove_placeholder_projections(&edge.value)?;
320
321        match &edge.value {
322            BorrowPcgEdgeKind::Deref(deref) => {
323                self.unlabel_blocked_region_projections_if_applicable(deref, context)?;
324                if deref.deref_place.is_current() {
325                    self.apply_action(
326                        BorrowPcgAction::label_place_and_update_related_capabilities(
327                            deref.deref_place.place(),
328                            self.prev_snapshot_location(),
329                            LabelPlaceReason::Collapse,
330                        )
331                        .into(),
332                    )?;
333                }
334            }
335            BorrowPcgEdgeKind::BorrowPcgExpansion(expansion) => {
336                if is_mutable_place_expansion {
337                    // If the expansion contained region projections, we need to
338                    // label them, they will flow into the now unblocked
339                    // projection (i.e. the one obtained by removing the
340                    // placeholder label)
341
342                    // For example, if we a are packing *s.i into *s at l
343                    // we need to label *s.i|'s to  *s|'s at l
344                    // because we will remove the label from *s|'s at l'
345                    // to become *s|'s. Otherwise we'd have both *s|'s and *s.i|'s
346                    for exp_node in expansion.expansion() {
347                        if let PcgNode::Place(place) = exp_node {
348                            for rp in place.lifetime_projections(self.ctxt) {
349                                tracing::debug!(
350                                    "labeling region projection: {}",
351                                    rp.display_string(self.ctxt)
352                                );
353                                self.record_and_apply_action(
354                                    BorrowPcgAction::label_lifetime_projection(
355                                        LabelNodePredicate::equals_lifetime_projection(rp),
356                                        Some(self.prev_snapshot_location().into()),
357                                        format!(
358                                            "{}: {}",
359                                            context, "Label region projections of expansion"
360                                        ),
361                                    )
362                                    .into(),
363                                )?;
364                            }
365                        }
366                    }
367                }
368            }
369            BorrowPcgEdgeKind::Borrow(borrow) => {
370                if self.ctxt.bc().is_dead(
371                    borrow
372                        .assigned_lifetime_projection(self.ctxt)
373                        .to_pcg_node(self.ctxt.bc_ctxt()),
374                    self.location(),
375                ) && let MaybeLabelledPlace::Current(place) = borrow.assigned_ref()
376                    && let Some(existing_cap) = self.pcg.capability_of(place, self.ctxt)
377                {
378                    self.record_and_apply_action(
379                        BorrowPcgAction::weaken(
380                            place,
381                            existing_cap,
382                            Some(CapabilityKind::Write),
383                            "remove borrow edge",
384                        )
385                        .into(),
386                    )?;
387                }
388            }
389            _ => {}
390        }
391        Ok(())
392    }

394    /// As an optimization, for expansions of the form {y, y|'y at l} -> *y,
395    /// if *y doesn't contain any borrows, we currently don't introduce placeholder
396    /// projections for y|'y: the set of borrows is guaranteed not to change as long as *y
397    /// is in the graph.
398    ///
399    /// Accordingly, when we want to remove *y in such cases, we just remove the
400    /// label rather than use the normal logic (of renaming the placeholder
401    /// projection to the current one).
402    fn unlabel_source_lifetime_projections(
403        &mut self,
404        deref: &DerefEdge<'tcx>,
405        context: &str,
406    ) -> Result<(), PcgError> {
407        self.record_and_apply_action(
408            BorrowPcgAction::remove_lifetime_projection_label(
409                deref.blocked_lifetime_projection,
410                format!("{context}: unlabel blocked_region_projections"),
411            )
412            .into(),
413        )?;
414        Ok(())
415    }

417    pub(crate) fn remove_read_permission_downwards(
418        &mut self,
419        upgraded_place: Place<'tcx>,
420    ) -> Result<(), PcgError> {
421        let to_remove = self
422            .pcg
423            .place_capabilities
424            .iter()
425            .filter(|(p, _)| {
426                p.projection.len() > upgraded_place.projection.len()
427                    && upgraded_place.is_prefix_of(*p)
428            })
429            .collect::<Vec<_>>();
430        for (p, cap) in to_remove {
431            self.record_and_apply_action(
432                BorrowPcgAction::weaken(p, cap, None, "Remove read permission downwards").into(),
433            )?;
434        }
435        Ok(())
436    }

438    pub(crate) fn upgrade_read_to_exclusive(&mut self, place: Place<'tcx>) -> Result<(), PcgError> {
439        tracing::debug!(
440            "upgrade_read_to_exclusive: {}",
441            place.display_string(self.ctxt.bc_ctxt())
442        );
443        self.record_and_apply_action(
444            BorrowPcgAction::restore_capability(
445                place,
446                CapabilityKind::Exclusive,
447                "upgrade_read_to_exclusive",
448            )
449            .into(),
450        )?;
451        self.remove_read_permission_downwards(place)?;
452        if let Some(parent) = place.parent_place() {
453            self.remove_read_permission_upwards_and_label_rps(
454                parent,
455                AdjustCapabilityReason::UpgradeReadToExclusive,
456            )?;
457        }
458        Ok(())
459    }

461    // This function is called if we want to obtain non-read capability to `place`
462    // If the closest ancestor of `place` has read capability, then we are allowed to
463    // upgrade the capability of the ancestor to `E` in exchange for downgrading all
464    // other pre / postfix places to None.
465    //
466    // This is sound because if we need to obtain non-read capability to
467    // `place`, and there are any ancestors of `place` in the graph with R
468    // capability, one such ancestor originally had E capability was
469    // subsequently downgraded. This function finds such an ancestor (if one
470    // exists), and performs the capability exchange.
471    pub(crate) fn upgrade_closest_read_ancestor_to_exclusive_and_update_rps(
472        &mut self,
473        place: Place<'tcx>,
474    ) -> Result<(), PcgError> {
475        let mut expand_root = place;
476        loop {
477            if let Some(cap) = self.pcg.capability_of(expand_root, self.ctxt) {
478                if cap.is_read() {
479                    self.upgrade_read_to_exclusive(expand_root)?;
480                }
481                return Ok(());
482            }
483            if let Some(parent) = expand_root.parent_place() {
484                expand_root = parent;
485            } else {
486                return Ok(());
487            }
488        }
489    }

491    pub(crate) fn record_and_apply_action(
492        &mut self,
493        action: PcgAction<'tcx>,
494    ) -> Result<(), PcgError>
495    where
496        Ctxt: HasBorrowCheckerCtxt<'a, 'tcx> + HasSettings<'a>,
497    {
498        tracing::debug!("Applying Action: {}", action.debug_line(self.ctxt));
499        let result = match &action {
500            PcgAction::Borrow(action) => {
501                self.pcg
502                    .borrow
503                    .apply_action(action.clone(), self.pcg.place_capabilities, self.ctxt)
504            }
505            PcgAction::Owned(owned_action) => match owned_action.kind {
506                RepackOp::RegainLoanedCapability(regained_capability) => {
507                    self.pcg.place_capabilities.regain_loaned_capability(
508                        regained_capability.place,
509                        regained_capability.capability,
510                        self.pcg.borrow.as_mut_ref(),
511                        self.ctxt,
512                    );
513                    ApplyActionResult::changed_no_display()
514                }
515                RepackOp::Expand(expand) => {
516                    self.pcg.owned.perform_expand_action(
517                        expand,
518                        self.pcg.place_capabilities,
519                        self.ctxt,
520                    );
521                    ApplyActionResult::changed_no_display()
522                }
523                RepackOp::DerefShallowInit(from, to) => {
524                    let target_places = from.expand_one_level(to, self.ctxt)?.expansion();
525                    let capability_projections = self.pcg.owned[from.local].expansions_mut();
526                    capability_projections.insert_expansion(
527                        from,
528                        PlaceExpansion::from_places(target_places.clone(), self.ctxt),
529                    );
530                    for target_place in target_places {
531                        self.pcg.place_capabilities.insert(
532                            target_place,
533                            CapabilityKind::Read,
534                            self.ctxt,
535                        );
536                    }
537                    ApplyActionResult::changed_no_display()
538                }
539                RepackOp::Collapse(collapse) => {
540                    let capability_projections = self.pcg.owned[collapse.local()].expansions_mut();
541                    capability_projections.perform_collapse_action(
542                        collapse,
543                        self.pcg.place_capabilities,
544                        self.ctxt,
545                    );
546                    ApplyActionResult::changed_no_display()
547                }
548                RepackOp::Weaken(weaken) => {
549                    pcg_validity_assert!(
550                        self.pcg
551                            .place_capability_equals(weaken.place, weaken.from, self.ctxt)
552                            || weaken.place.contains_unsafe_deref(self.ctxt)
553                    );
554                    self.pcg
555                        .place_capabilities
556                        .insert(weaken.place, weaken.to, self.ctxt);
557                    ApplyActionResult::changed_no_display()
558                }
559                _ => unreachable!(),
560            },
561        };
562        #[cfg(feature = "visualization")]
563        let location = self.location();
564        #[cfg(feature = "visualization")]
565        let phase = self.phase();
566        if let Some(actions) = &mut self.actions {
567            #[cfg(feature = "visualization")]
568            if let Some(analysis_ctxt) = self.ctxt.try_into_analysis_ctxt()
569                && let Some(phase) = phase
570            {
571                let pcg_ref = PcgRef {
572                    borrow: self.pcg.borrow.as_ref(),
573                    place_capabilities: self.pcg.place_capabilities,
574                    owned: self.pcg.owned,
575                };
576                analysis_ctxt.generate_pcg_debug_visualization_graph(
577                    location,
578                    stmt_graphs::ToGraph::Action(phase, actions.len()),
579                    pcg_ref,
580                );
581            }
582            actions.push(AppliedAction::new(action, result));
583        }
584        Ok(())
585    }
586    pub(crate) fn phase(&self) -> Option<EvalStmtPhase> {
587        match self.prev_snapshot_location {
588            SnapshotLocation::Before(analysis_location) => Some(analysis_location.eval_stmt_phase),
589            _ => None,
590        }
591    }

602/// Whether the place must be collapsed before it can hold `obtain_cap`.
603fn requires_collapse(current_cap: Option<CapabilityKind>, obtain_cap: CapabilityKind) -> bool {
604    current_cap
605        .is_none_or(|cap| matches!(cap.partial_cmp(&obtain_cap), Some(Ordering::Less) | None))

618        place: Place<'tcx>,
619        obtain_type: ObtainType,
620    ) -> Result<(), PcgError> {
621        let obtain_cap = obtain_type.capability(place, self.ctxt);
622
623        // Under `PackStrategy::Lazy` the expansions below `place` were not
624        // packed as soon as they became packable, so they are packed here,
625        // where a capability that `place` cannot hold while it is expanded is
626        // about to be required.
627        if self.ctxt.settings().pack_strategy == PackStrategy::Lazy
628            && requires_collapse(self.pcg.capability_of(place, self.ctxt), obtain_cap)
629        {
630            self.pack_expansions_for_obtain_of(place)?;
631        }
632
633        if obtain_cap.is_write() {
634            tracing::debug!(
635                "labeling and removing capabilities for deref projections of postfix places"
636            );
637            self.record_and_apply_action(
638                BorrowPcgAction::label_place_and_update_related_capabilities(
639                    place,
640                    self.prev_snapshot_location(),
641                    LabelPlaceReason::Write,
642                )
643                .into(),
644            )?;
645            self.record_and_apply_action(
646                BorrowPcgAction::label_lifetime_projection(
647                    LabelNodePredicate::And(vec![
648                        LabelNodePredicate::PlaceEquals(place),
649                        LabelNodePredicate::InSourceNodes,
650                        LabelNodePredicate::EdgeType(BorrowPcgEdgeType::BorrowFlow {
651                            future_edge_kind: Some(FutureEdgeKind::FromExpansion),
652                        }),
653                    ]),
654                    None,
655                    "label place and update related capabilities",
656                )
657                .into(),
658            )?;
659            self.render_debug_graph(None, "after step 1");
660        }
661
662        let current_cap = self.pcg.capability_of(place, self.ctxt);
663        tracing::debug!(
664            "Obtain {:?} to place {} in phase {:?}: Current cap: {:?}, Obtain cap: {:?}",
665            obtain_type,
666            place.display_string(self.ctxt.bc_ctxt()),
667            self.phase(),
668            current_cap,
669            obtain_cap
670        );
671
672        // STEP 2
673        if requires_collapse(current_cap, obtain_cap) {
674            // If we want to get e.g. write permission but we currently have
675            // read permission, we will obtain read with the collapse and then
676            // upgrade in the subsequent step
677            let collapse_cap = if current_cap == Some(CapabilityKind::Read) {
678                CapabilityKind::Read
679            } else {
680                obtain_cap
681            };
682            // When collapsing to write capability (an overwrite or a
683            // StorageDead of e.g. a partially-moved local), leaves that still
684            // hold exclusive capability must be explicitly weakened first, so
685            // that consumers see the capability drop; see
686            // <https://github.com/prusti/pcg/issues/137>.
687            if collapse_cap.is_write() {
688                let to_weaken = self
689                    .pcg
690                    .place_capabilities
691                    .owned_capabilities(place.local, self.ctxt)
692                    .filter_map(|(p, k)| {
693                        (place.is_prefix_of(p) && (k.is_exclusive() || k.is_shallow_exclusive()))
694                            .then_some((p, *k))
695                    })
696                    .collect::<Vec<_>>();
697                for (p, from_cap) in to_weaken {
698                    let weaken = if obtain_type == ObtainType::ForStorageDead {
699                        RepackOp::weaken_for_storage_dead(p, from_cap, CapabilityKind::Write)
700                    } else {
701                        RepackOp::weaken(p, from_cap, CapabilityKind::Write)
702                    };
703                    self.record_and_apply_action(PcgAction::Owned(OwnedPcgAction::new(
704                        weaken, None,
705                    )))?;
706                }
707            }
708            tracing::debug!(
709                "Collapsing owned places to {}",
710                place.display_string(self.ctxt.bc_ctxt())
711            );
712            self.collapse_owned_places_and_lifetime_projections_to(
713                place,
714                collapse_cap,
715                format!("Obtain {}", place.display_string(self.ctxt.bc_ctxt())),
716                self.ctxt,
717            )?;
718            self.render_debug_graph(
719                None,
720                &format!(
721                    "after step 2 (collapse owned places and lifetime projections to {})",
722                    place.display_string(self.ctxt.bc_ctxt())
723                ),
724            );
725        }
726
727        // STEP 3
728        if !obtain_cap.is_read() {
729            tracing::debug!(
730                "Obtain {:?} to place {} in phase {:?}",
731                obtain_type,
732                place.display_string(self.ctxt.bc_ctxt()),
733                self.phase()
734            );
735            // It's possible that we want to obtain exclusive or write permission to
736            // a field that we currently only have read access for. For example,
737            // consider the following case:
738            //
739            // There is an existing shared borrow of (*c).f1
740            // Therefore we have read permission to *c, (*c).f1, and (*c).f2
741            // Then, we want to create a mutable borrow of (*c).f2
742            // This requires obtaining exclusive permission to (*c).f2
743            //
744            // We can upgrade capability of (*c).f2 from R to E by downgrading
745            // all other pre-and postfix places of (*c).f2 (in this case c will
746            // be downgraded to W and *c to None). In the example, (*c).f2 is
747            // actually the closest read ancestor, but this is not always the
748            // case (e.g. if we wanted to obtain (*c).f2.f3 instead)
749            //
750            // This also labels rps and adds placeholder projections
751            self.upgrade_closest_read_ancestor_to_exclusive_and_update_rps(place)?;
752            self.render_debug_graph(None, "after step 3");
753        }
754
755        // STEP 4
756        if obtain_cap.is_write() {
757            // If this place is a reference or contains references, reborrows of
758            // (postfixes of) place may have not yet expired, and therefore the borrowed
759            // caps are still in the PCG.
760            // This will remove them (since we're going to overwrite anyways)
761            self.pcg
762                .place_capabilities
763                .remove_all_strict_postfixes(place, self.ctxt);
764            self.render_debug_graph(None, "after step 4");
765        }
766
767        self.expand_to(place, obtain_type, self.ctxt)?;
768
769        if obtain_type == ObtainType::ForStorageDead
770            && let Some(cap @ (CapabilityKind::Exclusive | CapabilityKind::ShallowExclusive)) =
771                self.pcg.capability_of(place, self.ctxt)
772        {
773            // Temporary: mark as for_storage_dead until
774            // https://github.com/prusti/pcg/issues/137 is resolved.
775            self.record_and_apply_action(PcgAction::Owned(OwnedPcgAction::new(
776                RepackOp::weaken_for_storage_dead(place, cap, CapabilityKind::Write),
777                None,
778            )))?;
779        }
780
781        self.render_debug_graph(None, "after step 5");
782
783        Ok(())
784    }
785}
786
787impl<'pcg, 'a: 'pcg, 'tcx: 'a, Ctxt: DataflowCtxt<'a, 'tcx>> PlaceExpander<'a, 'tcx>
788    for PlaceObtainer<'pcg, 'a, 'tcx, Ctxt>
789{
790    fn contains_owned_expansion_to(&self, target: Place<'tcx>) -> bool {
791        self.pcg.owned[target.local]
792            .expansions()
793            .contains_expansion_to(target, self.ctxt)
794    }
795
796    fn borrows_graph(&self) -> &borrow_pcg::graph::BorrowsGraph<'tcx> {
797        self.pcg.borrow.graph
798    }
799
800    fn path_conditions(&self) -> borrow_pcg::validity_conditions::ValidityConditions {
801        self.pcg.borrow.validity_conditions.clone()
802    }
803
804    fn update_capabilities_for_borrow_expansion(
805        &mut self,
806        expansion: &BorrowPcgExpansion<'tcx>,
807        block_type: BlockType,
808        _ctxt: CompilerCtxt<'_, 'tcx>,
809    ) -> bool {
810        self.pcg
811            .place_capabilities
812            .update_for_expansion(expansion, block_type, self.ctxt)
813    }
814
815    fn location(&self) -> mir::Location {
816        self.location
817    }
818
819    fn update_capabilities_for_deref(
820        &mut self,
821        ref_place: Place<'tcx>,
822        capability: CapabilityKind,
823        _ctxt: CompilerCtxt<'_, 'tcx>,
824    ) -> bool {
825        self.pcg
826            .place_capabilities
827            .update_for_deref(ref_place, capability, self.ctxt)
828    }
829
830    fn capability_for_expand(
831        &self,
832        base_place: Place<'tcx>,
833        obtain_type: ObtainType,
834        ctxt: impl crate::utils::HasCompilerCtxt<'a, 'tcx>,
835    ) -> CapabilityKind {
836        let current_cap = if let Some(owned) = base_place.as_owned_place(ctxt) {
837            self.pcg.owned.owned_capability(owned).unwrap()
838        } else {
839            self.pcg.place_capabilities.get(base_place, ctxt).unwrap()
840        };
841        obtain_type.capability_for_expand(base_place, current_cap, ctxt)
842    }
